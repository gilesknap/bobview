1//! Symbol and picture widgets: SVG or PNG images from files or URLs.
2//!
3//! A symbol shows one of its `symbols`, picked by its PV's value, and may run
4//! its actions when clicked; a picture shows one `file`. A symbol translated
5//! from a DLS EDM symbol in an `.opi` has one image, a horizontal strip of
6//! frames `sub_image_width` wide, and shows the frame its rules pick with
7//! `image_index`. Images load on a
8//! thread and are kept, decoded, for every widget that names them; each
9//! widget keeps its own texture at the size it is drawn.
10

28/// Largest texture side we render an SVG at.
29const MAX_SIDE: f32 = 4096.0;

31/// A decoded image.
32#[derive(Clone)]
33enum Image {
34    /// Rendered again whenever the size it is drawn at changes.
35    Svg(Arc<usvg::Tree>),
36    Raster(Arc<ColorImage>),
37}

39impl Image {
40    fn size(&self) -> Vec2 {
41        match self {
42            Image::Svg(tree) => Vec2::new(tree.size().width(), tree.size().height()),
43            Image::Raster(img) => Vec2::new(img.size[0] as f32, img.size[1] as f32),
44        }
45    }
46}

48enum Entry {
49    /// Being read, for the windows that asked for it, each repainted when
50    /// it has been.
51    Loading(Vec<egui::ViewportId>),
52    Ready(Image),
53    Failed,
54}
55
56/// Images by where they come from, and the count of [`forget`] calls, so a
57/// load started before a reload does not store what it read.
58struct Images {
59    entries: HashMap<Source, Entry>,
60    generation: u64,
61}
62
63static IMAGES: LazyLock<Mutex<Images>> = LazyLock::new(|| {
64    Mutex::new(Images {
65        entries: HashMap::new(),
66        generation: 0,
67    })
68});
69
70/// Threads that load images; more images than this wait their turn.
71#[cfg(not(target_arch = "wasm32"))]
72const LOADERS: usize = 4;
73
74/// The queue the loader threads take images from, started on first use.
75#[cfg(not(target_arch = "wasm32"))]
76static QUEUE: LazyLock<Mutex<std::sync::mpsc::Sender<Job>>> = LazyLock::new(|| {
77    let (tx, rx) = std::sync::mpsc::channel::<Job>();
78    let rx = Arc::new(Mutex::new(rx));
79    for _ in 0..LOADERS {
80        let rx = rx.clone();
81        // An explicit stack: usvg recurses per element and the default is
82        // platform dependent.
83        let spawned = std::thread::Builder::new()
84            .name("image-loader".into())
85            .stack_size(16 * 1024 * 1024)
86            .spawn(move || {
87                loop {
88                    let job = rx.lock().recv();
89                    match job {
90                        Ok(job) => load(job),
91                        Err(_) => break,
92                    }
93                }
94            });
95        if let Err(e) = spawned {
96            log::error!("cannot start an image loader: {e}");
97        }
98    }
99    Mutex::new(tx)
100});
101
102struct Job {
103    source: Source,
104    generation: u64,
105    ctx: egui::Context,
106}

108/// Drop every image, loaded or loading, so they are read again on a reload.
109pub fn forget() {
110    let mut images = IMAGES.lock();
111    images.entries.clear();
112    images.generation += 1;
113}

115/// Whether any image is still loading.
116#[cfg(not(target_arch = "wasm32"))]
117pub fn loading() -> bool {
118    IMAGES
119        .lock()
120        .entries
121        .values()
122        .any(|e| matches!(e, Entry::Loading(_)))
123}

125/// The image at `source`: `Some(Ok)` when loaded, `Some(Err)` when it cannot
126/// be, and `None` while it loads (a repaint follows when it has).
127fn image(ctx: &egui::Context, source: &Source) -> Option<Result<Image, ()>> {
128    let mut images = IMAGES.lock();
129    // The window being drawn (each draws in a pass of its own).
130    let viewport = ctx.viewport_id();
131    match images.entries.get_mut(source) {
132        Some(Entry::Ready(img)) => return Some(Ok(img.clone())),
133        Some(Entry::Failed) => return Some(Err(())),
134        Some(Entry::Loading(waiting)) => {
135            if !waiting.contains(&viewport) {
136                waiting.push(viewport);
137            }
138            return None;
139        }
140        None => {}
141    }
142    images
143        .entries
144        .insert(source.clone(), Entry::Loading(vec![viewport]));
145    let job = Job {
146        source: source.clone(),
147        generation: images.generation,
148        ctx: ctx.clone(),
149    };
150    drop(images);
151    #[cfg(not(target_arch = "wasm32"))]
152    let _ = QUEUE.lock().send(job);
153    // A browser has one thread: fetch without blocking, then decode.
154    #[cfg(target_arch = "wasm32")]
155    wasm_bindgen_futures::spawn_local(async move {
156        let data = crate::fetch::bytes_async(&job.source).await;
157        finish(job, data);
158    });
159    None
160}

162/// Read and decode one image on a loader thread.
163#[cfg(not(target_arch = "wasm32"))]
164fn load(job: Job) {
165    let data = crate::fetch::bytes(&job.source);
166    finish(job, data);
167}

169/// Decode what was read for `job` and keep it.
170fn finish(job: Job, data: Result<Vec<u8>, bob::Error>) {
171    let source = &job.source;
172    let data = data.map_err(|e| e.to_string());
173    // The release profile uses `panic = "abort"`, so this cannot catch a
174    // decoder panic there: hostile input is instead rejected up front (image
175    // sizes are capped before anything is allocated). Where panics do unwind
176    // (debug and test builds) a decoder panic must not leave it loading.
177    let read = || match data {
178        Ok(data) => match decode(source, &data) {
179            Ok(img) => Entry::Ready(img),
180            Err(e) => {
181                log::warn!("cannot show {}: {e}", describe(source));
182                Entry::Failed
183            }
184        },
185        Err(e) => {
186            log::warn!("{e}");
187            Entry::Failed
188        }
189    };
190    let entry = std::panic::catch_unwind(read).unwrap_or(Entry::Failed);
191    let mut images = IMAGES.lock();
192    // Not `request_repaint`: on this thread that is whichever window is
193    // being drawn now.
194    let mut waiting = Vec::new();
195    if images.generation == job.generation
196        && let Some(Entry::Loading(w)) = images.entries.insert(job.source, entry)
197    {
198        waiting = w;
199    }
200    drop(images);
201    for viewport in waiting {
202        job.ctx.request_repaint_of(viewport);
203    }
204}

213fn decode(source: &Source, data: &[u8]) -> Result<Image, String> {
214    if data.starts_with(b"\x89PNG") {
215        return decode_png(data).map(|img| Image::Raster(Arc::new(img)));
216    }
217    let name = describe(source).to_ascii_lowercase();
218    let svg = name.ends_with(".svg")
219        || name.ends_with(".svgz")
220        || data.trim_ascii_start().starts_with(b"<");
221    if !svg {
222        return Err("only SVG and PNG images are supported".into());
223    }
224    // (usvg is built without `svgz`, so a gzipped file fails to parse.)
225    svg::parse(data).map(|tree| Image::Svg(Arc::new(tree)))
226}

228/// Most bytes the PNG decoder may allocate for itself, besides the image.
229const MAX_PNG_DECODER_BYTES: usize = 64 * 1024 * 1024;
230
231fn decode_png(data: &[u8]) -> Result<ColorImage, String> {
232    let mut decoder = png::Decoder::new(std::io::Cursor::new(data));
233    decoder.set_transformations(png::Transformations::normalize_to_color8());
234    decoder.set_limits(png::Limits {
235        bytes: MAX_PNG_DECODER_BYTES,
236    });
237    let mut reader = decoder.read_info().map_err(|e| e.to_string())?;
238    // `Limits` don't cover the output buffer we allocate next: a few bytes
239    // can declare gigabytes of pixels.
240    let (width, height) = reader.info().size();
241    if !svg::size_ok(width, height) {
242        return Err(format!("PNG of {width}x{height} pixels is too large"));
243    }
244    let mut buf = vec![0; reader.output_buffer_size().ok_or("PNG too large")?];
245    let info = reader.next_frame(&mut buf).map_err(|e| e.to_string())?;
246    let buf = &buf[..info.buffer_size()];
247    let size = [info.width as usize, info.height as usize];
248    let rgba: Vec<u8> = match info.color_type {
249        png::ColorType::Rgba => buf.to_vec(),
250        png::ColorType::Rgb => buf
251            .as_chunks::<3>()
252            .0
253            .iter()
254            .flat_map(|p| [p[0], p[1], p[2], 255])
255            .collect(),
256        png::ColorType::GrayscaleAlpha => buf
257            .as_chunks::<2>()
258            .0
259            .iter()
260            .flat_map(|p| [p[0], p[0], p[0], p[1]])
261            .collect(),
262        png::ColorType::Grayscale => buf.iter().flat_map(|&g| [g, g, g, 255]).collect(),
263        png::ColorType::Indexed => return Err("unexpected indexed PNG".into()),
264    };
265    Ok(ColorImage::from_rgba_unmultiplied(size, &rgba))
266}

276/// A widget's texture and what is on it: the image and its size in pixels.
277struct Texture {
278    key: (Source, [usize; 2]),
279    handle: TextureHandle,
280}
281
282/// The widget's texture showing `img` at `size` points, made or updated.
283fn texture<'a>(
284    ctx: &egui::Context,
285    slot: &'a mut Option<Texture>,
286    source: &Source,
287    img: &Image,
288    size: Vec2,
289) -> Option<&'a TextureHandle> {
290    let max_side = ctx.input(|i| i.max_texture_side) as f32;
291    let px = match img {
292        Image::Raster(r) if r.size.into_iter().any(|n| n as f32 > max_side) => {
293            log::warn!("cannot show {}: too large for a texture", describe(source));
294            return None;
295        }
296        Image::Raster(r) => r.size,
297        Image::Svg(_) => svg_pixels(size * ctx.pixels_per_point(), MAX_SIDE.min(max_side)),
298    };
299    let key = (source.clone(), px);
300    if slot.as_ref().is_none_or(|t| t.key != key) {
301        let pixels = match img {
302            Image::Raster(r) => (**r).clone(),
303            Image::Svg(tree) => render_svg(tree, px),
304        };
305        let options = TextureOptions::LINEAR;
306        match slot {
307            Some(t) => {
308                t.handle.set(pixels, options);
309                t.key = key;
310            }
311            None => {
312                *slot = Some(Texture {
313                    key,
314                    handle: ctx.load_texture("bobview-symbol", pixels, options),
315                });
316            }
317        }
318    }
319    slot.as_ref().map(|t| &t.handle)
320}

322/// The pixels to draw an SVG in to show it `size` pixels big, at most
323/// `max_side` on a side. The size goes up in steps of a fourth of an octave
324/// (19%), so that resizing it smoothly, as a pinch zoom does, draws it again
325/// a few times rather than on every frame; the texture is drawn scaled.
326fn svg_pixels(size: Vec2, max_side: f32) -> [usize; 2] {
327    let size = size.max(Vec2::splat(1.0));
328    let longest = size.max_elem();
329    let step = (longest.log2() * 4.0).ceil() / 4.0;
330    let scale = (step.exp2() / longest).min(max_side / longest);
331    let px = (size * scale).round().max(Vec2::splat(1.0));
332    [px.x as usize, px.y as usize]
333}

335fn render_svg(tree: &usvg::Tree, [w, h]: [usize; 2]) -> ColorImage {
336    let Some(mut pixmap) = tiny_skia::Pixmap::new(w as u32, h as u32) else {
337        return ColorImage::filled([w, h], Color32::TRANSPARENT);
338    };
339    let size = tree.size();
340    let scale = tiny_skia::Transform::from_scale(w as f32 / size.width(), h as f32 / size.height());
341    resvg::render(tree, scale, &mut pixmap.as_mut());
342    ColorImage::from_rgba_premultiplied([w, h], pixmap.data())
343}

345/// The largest size of aspect `native` that fits in `room`.
346fn fit(native: Vec2, room: Vec2) -> Vec2 {
347    if native.x <= 0.0 || native.y <= 0.0 {
348        return room;
349    }
350    let scale = (room.x / native.x).min(room.y / native.y);
351    native * scale
352}

354/// A rectangle of `size` centred on `center`, turned by `angle` about it:
355/// the `uv` part (default all) of `mesh`'s texture, or plain colour on an
356/// untextured mesh.
357fn quad(mesh: &mut Mesh, center: Pos2, size: Vec2, angle: f32, tint: Color32, uv: Option<Rect>) {
358    let start = mesh.vertices.len();
359    let uv = if mesh.texture_id == egui::TextureId::default() {
360        Rect::from_min_max(egui::epaint::WHITE_UV, egui::epaint::WHITE_UV)
361    } else {
362        uv.unwrap_or(Rect::from_min_max(Pos2::ZERO, Pos2::new(1.0, 1.0)))
363    };
364    mesh.add_rect_with_uv(Rect::from_center_size(center, size), uv, tint);
365    let rot = egui::emath::Rot2::from_angle(angle);
366    for v in &mut mesh.vertices[start..] {
367        v.pos = center + rot * (v.pos - center);
368    }
369}

371/// The symbol index a PV value selects, as Phoebus picks it; `None` for no
372/// usable value.
373fn value_index(w: &bob::Widget, s: &PvState) -> Option<i64> {
374    let at = |len: usize| w.i64("array_index").unwrap_or(0).clamp(0, len as i64 - 1) as usize;
375    match &s.value {
376        Value::Str(t) => t.trim().parse().ok(),
377        Value::DoubleArray(a) if !a.is_empty() => Some(a[at(a.len())] as i64),
378        Value::LongArray(a) if !a.is_empty() => Some(a[at(a.len())]),
379        Value::Double(_) | Value::Long(_) | Value::Enum(_) => s.value.as_i64(),
380        _ => None,
381    }
382}

384/// The width of a strip symbol's frames, if it is one.
385fn strip_width(w: &bob::Widget) -> Option<f32> {
386    w.f64("sub_image_width")
387        .filter(|sw| *sw > 0.0)
388        .map(|sw| sw as f32)
389}

391/// The part of a strip `width` pixels wide holding frame `index` of frames
392/// `frame` pixels wide; frame 0 for an index out of range.
393fn strip_uv(width: f32, frame: f32, index: i64) -> Rect {
394    let frames = (width / frame).floor().max(1.0) as i64;
395    let i = if (0..frames).contains(&index) {
396        index
397    } else {
398        0
399    };
400    let u = |n: i64| (n as f32 * frame / width).min(1.0);
401    Rect::from_min_max(Pos2::new(u(i), 0.0), Pos2::new(u(i + 1), 1.0))
402}

404/// The symbol index to show: for a strip, the frame its group link picks
405/// or else `image_index`; else the PV's value; without a usable one, the
406/// index last shown, else `initial_index`.
407fn index(node: &Node, s: Option<&PvState>) -> i64 {
408    if strip_width(&node.w).is_some() {
409        return crate::group_link::frame(node, s)
410            .unwrap_or_else(|| node.w.i64("image_index").unwrap_or(0));
411    }
412    let value = s
413        .filter(|s| s.connected)
414        .and_then(|s| value_index(&node.w, s));
415    value
416        .or(node.symbol.index)
417        .unwrap_or_else(|| node.w.i64("initial_index").unwrap_or(0))
418}

420pub fn draw_symbol(
421    ui: &mut egui::Ui,
422    node: &mut Node,
423    s: Option<&PvState>,
424    rect: Rect,
425    cx: &mut Cx,
426    id: Id,
427) -> Option<Response> {
428    let w = &node.w;
429    let painter = ui.painter().with_clip_rect(cx.clip);
430    let z = cx.view.zoom;
431    let angle = (w.f64("rotation").unwrap_or(0.0) as f32).to_radians();
432
433    // Clicks run every action, as Phoebus does when the symbol is set to.
434    let actions = w.actions();
435    let clickable = w.bool("run_actions_on_mouse_click") == Some(true)
436        && w.bool("enabled") != Some(false)
437        && !actions.is_empty();
438    let resp = clickable.then(|| {
439        ui.interact(rect.intersect(cx.clip), id, Sense::click())
440            .on_hover_cursor(egui::CursorIcon::PointingHand)
441    });
442
443    if w.bool("transparent") == Some(false) {
444        let bg = w
445            .color("background_color")
446            .map(|c| color(&c))
447            .unwrap_or(named("Background"));
448        let mut mesh = Mesh::default();
449        quad(&mut mesh, rect.center(), rect.size(), angle, bg, None);
450        painter.add(mesh);
451    }
452
453    let symbols = w.list("symbols", "symbol");
454    let idx = index(node, s);
455    node.symbol.index = Some(idx);
456    let strip = strip_width(w).map(|sw| (sw, idx));
457    let file = match usize::try_from(if strip.is_some() { 0 } else { idx }) {
458        Ok(i) => symbols
459            .get(i)
460            .cloned()
461            .or_else(|| w.text("fallback_symbol").map(str::to_string)),
462        Err(_) => None,
463    };
464    let file = file
465        .map(|f| node.expand(&f))
466        .filter(|f| !f.trim().is_empty());
467    let fallback = w
468        .text("fallback_symbol")
469        .map(|f| node.expand(f))
470        .filter(|f| !f.trim().is_empty());
471    // An empty fallback, rather than none, means draw nothing for a state
472    // without an image.
473    let blank = w.text("fallback_symbol").is_some() && fallback.is_none();
474
475    // Brighter under the pointer and darker while pressed.
476    let tint = match &resp {
477        Some(r) if r.is_pointer_button_down_on() => Color32::from_gray(180),
478        _ => Color32::WHITE,
479    };
480    let glow = resp
481        .as_ref()
482        .is_some_and(|r| r.hovered() && !r.is_pointer_button_down_on());
483
484    if let Some(file) = file {
485        let shown = show(
486            ui,
487            node,
488            &[
489                &file,
490                fallback.as_deref().filter(|f| *f != file).unwrap_or(""),
491            ],
492            rect,
493            angle,
494            tint,
495            glow,
496            strip,
497            &painter,
498        );
499        if !shown && !blank {
500            placeholder(&painter, rect, "symbol", z);
501        }
502    } else if idx >= 0 && !blank {
503        placeholder(&painter, rect, "symbol", z);
504    }
505
506    let w = &node.w;
507    if w.bool("show_index") == Some(true) && rect.width() >= 20.0 && rect.height() >= 20.0 {
508        let text = idx.to_string();
509        let font = FontId::proportional(16.0 * z);
510        let galley = painter.layout_no_wrap(text, font, Color32::WHITE);
511        let label = Rect::from_center_size(rect.center(), galley.size() + Vec2::splat(4.0 * z));
512        painter.rect_filled(label, CornerRadius::same(3), Color32::from_black_alpha(140));
513        painter.galley(
514            Align2::CENTER_CENTER
515                .align_size_within_rect(galley.size(), label)
516                .min,
517            galley,
518            Color32::WHITE,
519        );
520    }
521
522    // A PV that is not connected is marked with an overlay.
523    if s.is_some_and(|s| !s.connected) {
524        let overlay = w
525            .color("disconnect_overlay_color")
526            .map(|c| color(&c))
527            .unwrap_or_else(|| named("INVALID").gamma_multiply(0.5));
528        let mut mesh = Mesh::default();
529        quad(&mut mesh, rect.center(), rect.size(), angle, overlay, None);
530        painter.add(mesh);
531    }
532
533    if let Some(r) = &resp
534        && r.clicked()
535    {
536        for a in &actions {
537            run_action(node, a, cx);
538        }
539    }
540    resp
541}

543pub fn draw_picture(ui: &mut egui::Ui, node: &mut Node, rect: Rect, cx: &Cx) {
544    let painter = ui.painter().with_clip_rect(cx.clip);
545    let file = node.prop("file");
546    if file.trim().is_empty() {
547        placeholder(&painter, rect, "picture", cx.view.zoom);
548        return;
549    }
550    let angle = (node.w.f64("rotation").unwrap_or(0.0) as f32).to_radians();
551    let opacity = node.w.f64("opacity").unwrap_or(1.0).clamp(0.0, 1.0) as f32;
552    let tint = Color32::WHITE.gamma_multiply(opacity);
553    if !show(ui, node, &[&file], rect, angle, tint, false, None, &painter) {
554        placeholder(&painter, rect, "picture", cx.view.zoom);
555    }
556}

558/// Draw the first of `files` that loads, fitted to `rect` and turned by
559/// `angle`; of a `strip` (frame width, index), only that frame. False if
560/// none can be shown; true while one is still loading.
561#[allow(clippy::too_many_arguments)]
562fn show(
563    ui: &egui::Ui,
564    node: &mut Node,
565    files: &[&str],
566    rect: Rect,
567    angle: f32,
568    tint: Color32,
569    glow: bool,
570    strip: Option<(f32, i64)>,
571    painter: &egui::Painter,
572) -> bool {
573    let ctx = ui.ctx();
574    for file in files.iter().filter(|f| !f.trim().is_empty()) {
575        let source = node.resolve(file);
576        let img = match image(ctx, &source) {
577            None => return true,
578            Some(Err(())) => continue,
579            Some(Ok(img)) => img,
580        };
581        let (native, uv) = match strip {
582            Some((frame, index)) => (
583                Vec2::new(frame.min(img.size().x), img.size().y),
584                Some(strip_uv(img.size().x, frame, index)),
585            ),
586            None => (img.size(), None),
587        };
588        let size = picture_size(&node.w, native, rect.size(), angle);
589        let tex_size = match uv {
590            // An SVG strip is rendered whole, so its frame comes out at `size`.
591            Some(uv) => size / Vec2::new(uv.width().max(1e-3), 1.0),
592            None => size,
593        };
594        let Some(handle) = texture(ctx, &mut node.symbol.texture, &source, &img, tex_size) else {
595            continue;
596        };
597        let mut mesh = Mesh::with_texture(handle.id());
598        quad(&mut mesh, rect.center(), size, angle, tint, uv);
599        if glow {
600            // Drawn again additively (zero alpha): 30% brighter.
601            quad(
602                &mut mesh,
603                rect.center(),
604                size,
605                angle,
606                Color32::from_rgba_premultiplied(77, 77, 77, 0),
607                uv,
608            );
609        }
610        painter.add(mesh);
611        return true;
612    }
613    false
614}

616/// The size an image of `native` size is drawn at in a widget of `room`.
617/// A symbol fits the widget before turning; a picture is shrunk so that,
618/// turned, it still fits.
619fn picture_size(w: &bob::Widget, native: Vec2, room: Vec2, angle: f32) -> Vec2 {
620    if w.kind != "picture" {
621        return if w.bool("preserve_ratio") != Some(false) {
622            fit(native, room)
623        } else {
624            room
625        };
626    }
627    let size = if w.bool("stretch_image") == Some(true) {
628        room
629    } else {
630        fit(native, room)
631    };
632    let (sin, cos) = (angle.sin().abs(), angle.cos().abs());
633    let bounds = Vec2::new(size.x * cos + size.y * sin, size.x * sin + size.y * cos);
634    let scale = (room.x / bounds.x).min(room.y / bounds.y);
635    if scale < 1.0 { size * scale } else { size }
636}

668    fn crc32(bytes: &[u8]) -> u32 {
669        let mut crc = !0u32;
670        for &b in bytes {
671            crc ^= u32::from(b);
672            for _ in 0..8 {
673                crc = (crc >> 1) ^ (0xEDB8_8320 & (!(crc & 1)).wrapping_add(1));
674            }
675        }
676        !crc
677    }

679    /// A 1x1 RGBA PNG whose header claims `width` x `height`.
680    fn png_claiming(width: u32, height: u32) -> Vec<u8> {
681        let mut out = Vec::new();
682        let mut enc = png::Encoder::new(&mut out, 1, 1);
683        enc.set_color(png::ColorType::Rgba);
684        enc.set_depth(png::BitDepth::Eight);
685        let mut w = enc.write_header().unwrap();
686        w.write_image_data(&[1, 2, 3, 4]).unwrap();
687        w.finish().unwrap();
688        out[16..20].copy_from_slice(&width.to_be_bytes());
689        out[20..24].copy_from_slice(&height.to_be_bytes());
690        let crc = crc32(&out[12..29]);
691        out[29..33].copy_from_slice(&crc.to_be_bytes());
692        out
693    }
