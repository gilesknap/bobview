1//! Saved window layouts: which displays are open, in which windows, where.
2//!
3//! A layout is a JSON file in `$XDG_CONFIG_HOME/bobview/layouts` (by default
4//! `~/.config/bobview/layouts`, or `%APPDATA%\bobview\layouts` on Windows), named after the layout; `default.json` is
5//! restored at startup. Positions and sizes are in desktop points (egui's
6//! points times its zoom factor), so they don't depend on the controls' scale.
7

61/// One window of a layout: the main window, or a display window.
62#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
63pub struct WindowLayout {

97/// A number, or the string "fit".
98#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
99#[serde(try_from = "ZoomText", into = "ZoomText")]
100pub enum Zoom {
101    Factor(f32),
102    /// Fit the display to the window.
103    Fit,
104}
105
106/// How a zoom is written.
107#[derive(Clone, Serialize, Deserialize)]
108#[serde(untagged)]
109enum ZoomText {
110    Factor(f32),
111    Text(String),
112}
113
114impl TryFrom<ZoomText> for Zoom {
115    type Error = String;
116
117    fn try_from(z: ZoomText) -> Result<Self, String> {
118        match z {
119            ZoomText::Factor(f) => Ok(Self::Factor(f)),
120            ZoomText::Text(t) if t == "fit" => Ok(Self::Fit),
121            ZoomText::Text(t) => Err(format!("zoom {t:?}: expected a number or \"fit\"")),
122        }
123    }
124}
125
126impl From<Zoom> for ZoomText {
127    fn from(z: Zoom) -> Self {
128        match z {
129            Zoom::Factor(f) => Self::Factor(f),
130            Zoom::Fit => Self::Text("fit".into()),
131        }
132    }
133}
134
135impl Zoom {
136    pub fn new(fit: bool, zoom: f32) -> Self {
137        if fit { Self::Fit } else { Self::Factor(zoom) }
138    }
139
140    /// A fixed zoom, or None to fit to the window.
141    pub fn factor(&self) -> Option<f32> {
142        match self {
143            Self::Factor(f) => Some(f.clamp(0.2, 5.0)),
144            Self::Fit => None,
145        }
146    }
147}

149fn is_false(b: &bool) -> bool {
150    !b

153/// Where a window is and how big, in desktop points.
154#[derive(Clone, Copy, Debug, Default, PartialEq)]
155pub struct Geometry {
156    pub position: Option<Pos2>,
157    pub size: Option<Vec2>,
158}
159
160impl WindowLayout {
161    /// The layout of a window showing `loc`.
162    pub fn new(loc: &Location, geometry: Geometry) -> Self {
163        let mut macros = BTreeMap::new();
164        // Later definitions win, as in `Macros::get`.
165        for (k, v) in loc.macros.iter() {
166            macros.insert(k.to_string(), v.to_string());
167        }
168        Self {
169            main: false,
170            display: loc.source.to_string(),
171            project: match &loc.source {
172                Source::File(path) => project_of(path),
173                Source::Url(_) => None,
174            },
175            macros,
176            position: geometry.position.map(|p| [p.x, p.y]),
177            size: geometry.size.map(|s| [s.x, s.y]),
178            zoom: None,
179            toolbar: false,
180            pinned: None,
181            tree: false,
182        }
183    }
184
185    /// The display and macros to show.
186    pub fn location(&self) -> Location {
187        let mut source = crate::fetch::user_named(&self.display);
188        let mut project = self.project.clone();
189        // A DLS launcher's project is named after the module's version, so
190        // after the version changes take the launcher's project as it is now.
191        if let Some(dir) = &project
192            && bob::project::links(dir).is_none()
193            && let Some(script) = dir.parent()
194            && bob::project::through_launcher(dir)
195            && let Some(now) = bob::project::project_dir(script)
196        {
197            if let Source::File(path) = &source
198                && let Ok(rest) = path.strip_prefix(dir)
199            {
200                source = Source::File(now.join(rest));
201            }
202            project = Some(now);
203        }
204        // A real path saved with its project goes back through the project.
205        if let (Some(dir), Source::File(path)) = (&project, &source)
206            && !path.starts_with(dir)
207            && let Some(p) = bob::project::virtual_path(dir, path)
208        {
209            source = Source::File(p);
210        }
211        Location {
212            source,
213            macros: Macros::from_pairs(self.macros.clone()),
214        }
215    }
216
217    /// Where to put the window, dropping a position on none of `monitors`
218    /// (a laptop that was docked when the layout was saved); with no
219    /// monitors known, positions are kept.
220    pub fn geometry(&self, monitors: &[Rect]) -> Geometry {
221        Geometry::restore(self.position, self.size, monitors)
222    }
223}

225impl TrendLayout {
226    pub fn new(view: archiveview::View, geometry: Geometry) -> Self {
227        let span = match &view.range {
228            Range::Relative { secs } => *secs,
229            Range::Absolute { start, end } => {
230                match (
231                    archiveview::view::parse_time(start),
232                    archiveview::view::parse_time(end),
233                ) {
234                    (Some(a), Some(b)) if b > a => b - a,
235                    _ => 3600.0,
236                }
237            }
238        };
239        Self {
240            pvs: view.pvs.iter().map(|t| t.name.clone()).collect(),
241            span,
242            log: view.y.log,
243            view: Some(view),
244            position: geometry.position.map(|p| [p.x, p.y]),
245            size: geometry.size.map(|s| [s.x, s.y]),
246        }
247    }
248
249    /// What the plot shows: as saved, or for a layout from before trends
250    /// were archiveview plots, its PVs over the last `span` seconds (live),
251    /// on a log scale if it was.
252    pub fn view(&self) -> archiveview::View {
253        if let Some(view) = &self.view {
254            return view.clone();
255        }
256        let span = if self.span.is_finite() && self.span > 0.0 {
257            self.span
258        } else {
259            600.0
260        };
261        archiveview::View {
262            pvs: (self.pvs.iter())
263                .map(|name| TraceView {
264                    name: name.clone(),
265                    color: None,
266                    visible: true,
267                    axis: None,
268                })
269                .collect(),
270            range: Range::Relative { secs: span },
271            y: YAxis {
272                log: self.log,
273                ..YAxis::default()
274            },
275            ..archiveview::View::default()
276        }
277    }
278
279    /// Where to put the window, as `WindowLayout::geometry`.
280    pub fn geometry(&self, monitors: &[Rect]) -> Geometry {
281        Geometry::restore(self.position, self.size, monitors)
282    }
283}

285impl Geometry {
286    /// A saved position and size, dropping a position on none of
287    /// `monitors`, and either one that isn't usable.
288    fn restore(position: Option<[f32; 2]>, size: Option<[f32; 2]>, monitors: &[Rect]) -> Self {
289        let size = size
290            .map(|[w, h]| Vec2::new(w, h))
291            .filter(|s| s.x >= 1.0 && s.y >= 1.0 && s.x.is_finite() && s.y.is_finite());
292        let position = position
293            .map(|[x, y]| Pos2::new(x, y))
294            .filter(|p| p.x.is_finite() && p.y.is_finite())
295            .filter(|p| monitors.is_empty() || on_screen(*p, size, monitors));
296        Self { position, size }
297    }
298}
299
300/// Whether enough of a window's top edge (its title bar) at `pos` lies on a
301/// monitor to grab it.
302fn on_screen(pos: Pos2, size: Option<Vec2>, monitors: &[Rect]) -> bool {
303    let width = size.map_or(200.0, |s| s.x).max(1.0);
304    let top = Rect::from_min_size(pos, Vec2::new(width, 30.0));
305    monitors.iter().any(|m| {
306        let seen = m.intersect(top);
307        seen.is_positive() && seen.width() >= width.min(40.0) && seen.height() >= 10.0
308    })
309}

311/// The project directory that `path` reaches its file through: the nearest
312/// directory above it holding a `.project` (or the virtual project of a DLS
313/// launcher), when `path` itself doesn't exist.
314fn project_of(path: &Path) -> Option<PathBuf> {
315    if path.exists() {
316        return None;
317    }
318    path.ancestors()
319        .skip(1)
320        .find(|d| bob::project::is_project(d))
321        .map(Path::to_path_buf)
322}

342/// `$XDG_CONFIG_HOME/bobview`, by default `~/.config/bobview` (on Windows
343/// `%APPDATA%\bobview`, ahead of `~/.config/bobview`, when
344/// `XDG_CONFIG_HOME` is not set).
345pub fn config_dir() -> Option<PathBuf> {
346    let var = |name| std::env::var_os(name).filter(|d| !d.is_empty());
347    let base = var("XDG_CONFIG_HOME")
348        .map(PathBuf::from)
349        .or_else(|| var("APPDATA").filter(|_| cfg!(windows)).map(PathBuf::from))
350        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".config")))?;
351    Some(base.join("bobview"))
352}
353
354fn layouts_dir() -> Option<PathBuf> {
355    config_dir().map(|d| d.join("layouts"))
356}

358/// Whether `name` can name a layout file: not empty, no path in it.
359pub fn valid_name(name: &str) -> bool {
360    let name = name.trim();
361    !name.is_empty() && !name.starts_with('.') && !name.contains(['/', '\\'])
362}
363
364fn file(name: &str) -> Option<PathBuf> {
365    valid_name(name)
366        .then(|| layouts_dir().map(|d| d.join(format!("{}.json", name.trim()))))
367        .flatten()
368}

370/// The saved layouts' names, default first and the rest in order.
371pub fn list() -> Vec<String> {
372    let Some(dir) = layouts_dir() else {
373        return Vec::new();
374    };
375    let mut names: Vec<String> = std::fs::read_dir(dir)
376        .map(|rd| {
377            rd.filter_map(Result::ok)
378                .filter_map(|e| {
379                    let p = e.path();
380                    (p.extension()? == "json")
381                        .then(|| p.file_stem()?.to_str().map(str::to_string))
382                        .flatten()
383                })
384                .collect()
385        })
386        .unwrap_or_default();
387    names.sort_by(|a, b| (a != DEFAULT).cmp(&(b != DEFAULT)).then(a.cmp(b)));
388    names
389}

391pub fn exists(name: &str) -> bool {
392    file(name).is_some_and(|f| f.is_file())
393}

395pub fn load(name: &str) -> Result<Layout, String> {
396    let path = file(name).ok_or_else(|| format!("no layout {name:?}"))?;
397    let text = std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
398    let layout: Layout =
399        serde_json::from_str(&text).map_err(|e| format!("{}: {e}", path.display()))?;
400    if layout.version > VERSION {
401        return Err(format!(
402            "{}: layout version {} is newer than this bobview's {VERSION}",
403            path.display(),
404            layout.version
405        ));
406    }
407    Ok(layout)
408}

410/// The layout to start with: `named` (with `--layout`), which must load,
411/// else the default one when `default`, which bobview starts without, with
412/// a warning, if it doesn't load (written by a newer bobview, say).
413#[cfg_attr(
414    target_arch = "wasm32",
415    expect(dead_code, reason = "a browser starts from its address")
416)]
417pub fn startup(
418    named: Option<String>,
419    default: bool,
420    load: impl Fn(&str) -> Result<Layout, String>,
421) -> Result<Option<(Layout, String)>, String> {
422    if let Some(name) = named {
423        return load(&name)
424            .map(|l| Some((l, name.clone())))
425            .map_err(|e| format!("cannot restore layout {name}: {e}"));
426    }
427    if !default {
428        return Ok(None);
429    }
430    match load(DEFAULT) {
431        Ok(l) => Ok(Some((l, DEFAULT.to_string()))),
432        Err(e) => {
433            log::warn!("not restoring the default layout: {e}");
434            Ok(None)
435        }
436    }
437}

439pub fn save(name: &str, layout: &Layout) -> Result<(), String> {
440    let path = file(name).ok_or_else(|| format!("not a layout name: {name:?}"))?;
441    let write = || -> std::io::Result<()> {
442        let text = serde_json::to_string_pretty(layout).map_err(std::io::Error::other)?;
443        write_atomic(&path, &(text + "\n"))
444    };
445    write().map_err(|e| format!("{}: {e}", path.display()))?;
446    log::info!("saved layout {}", path.display());
447    Ok(())
448}

450/// Write `text` to `path` through a file beside it, so that a crash or a
451/// full disk leaves the old file whole rather than half written.
452fn write_atomic(path: &Path, text: &str) -> std::io::Result<()> {
453    let dir = path.parent().unwrap_or(Path::new("."));
454    std::fs::create_dir_all(dir)?;
455    let name = path.file_name().unwrap_or_default().to_string_lossy();
456    // Hidden and not .json, so never listed as a layout.
457    let tmp = dir.join(format!(".{name}.{}.tmp", std::process::id()));
458    std::fs::write(&tmp, text)
459        .and_then(|()| std::fs::rename(&tmp, path))
460        .inspect_err(|_| {
461            std::fs::remove_file(&tmp).ok();
462        })
463}
464
465pub fn delete(name: &str) -> Result<(), String> {
466    let path = file(name).ok_or_else(|| format!("no layout {name:?}"))?;
467    std::fs::remove_file(&path).map_err(|e| format!("{}: {e}", path.display()))
468}

470/// Settings kept between runs, beside the layouts.
471#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
472pub struct Settings {
473    /// Save the default layout when bobview exits.
474    #[serde(default)]
475    pub save_default_on_exit: bool,
476    /// The Archiver Appliance trends fetch history from: the last given
477    /// with `--archiver` or picked in a trend's archiver pill.
478    #[serde(default, skip_serializing_if = "Option::is_none")]
479    pub archiver_url: Option<String>,
480    /// The archivers used, newest first, which the pill offers.
481    #[serde(default, skip_serializing_if = "Vec::is_empty")]
482    pub recent_archivers: Vec<String>,
483}
484
485/// The most archivers [`Settings::recent_archivers`] keeps.
486const MAX_RECENT_ARCHIVERS: usize = 8;
487
488impl Settings {
489    /// Make `archiver` (a retrieval URL, or none) the one to use, and the
490    /// newest of the recent ones, with `outgoing` (the one used until now,
491    /// which may have come from the environment) next, so it can be picked
492    /// again. True if that changed anything.
493    pub fn remember_archiver(&mut self, archiver: Option<&str>, outgoing: Option<&str>) -> bool {
494        let before = (self.archiver_url.clone(), self.recent_archivers.clone());
495        self.archiver_url = archiver.map(str::to_string);
496        for url in [outgoing, archiver].into_iter().flatten() {
497            self.recent_archivers.retain(|u| u != url);
498            self.recent_archivers.insert(0, url.to_string());
499        }
500        self.recent_archivers.truncate(MAX_RECENT_ARCHIVERS);
501        before != (self.archiver_url.clone(), self.recent_archivers.clone())
502    }
503
504    pub fn load() -> Self {
505        config_dir()
506            .and_then(|d| std::fs::read_to_string(d.join("settings.json")).ok())
507            .and_then(|t| serde_json::from_str(&t).ok())
508            .unwrap_or_default()
509    }
510
511    pub fn save(&self) {
512        let Some(dir) = config_dir() else { return };
513        let written = serde_json::to_string_pretty(self)
514            .map_err(std::io::Error::other)
515            .and_then(|text| write_atomic(&dir.join("settings.json"), &(text + "\n")));
516        if let Err(e) = written {
517            log::warn!("cannot save settings in {}: {e}", dir.display());
518        }
519    }
520}
