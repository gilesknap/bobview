1//! PV access for bobview.
2//!
3//! Widgets ask a [`PvManager`] for a [`Pv`] by name and read its latest
4//! [`PvState`] each frame. The manager owns a tokio runtime on which one task
5//! per PV talks to the backend chosen by the name's prefix:
6//!
7//! | prefix    | backend                                 |
8//! |-----------|-----------------------------------------|
9//! | `ca://`   | Channel Access (the default)            |
10//! | `pva://`  | PV Access                               |
11//! | `loc://`  | local PVs held in this process          |
12//! | `sim://`  | simulated signals (sine, ramp, noise …) |
13//! | `=`, `eq://` | formulas over other PVs              |
14//!
15//! In a browser (wasm32) there is no runtime: `sim://` runs on the page's
16//! event loop, and `ca://` and `pva://` PVs come over a PV Web Socket
17//! ([`set_pvws`]).

87/// Settings and messages shared by every PV of one [`PvManager`].
88#[derive(Default)]
89pub(crate) struct Control {
90    /// Refuse every write except to `loc://` PVs (which only live here).
91    read_only: AtomicBool,
92    /// The latest message for the user (a refused or failed write), with a
93    /// number that changes with each one.
94    notice: Mutex<(u64, String)>,
95    /// The UI's repaint callback (see [`PvManager::set_notifier`]), so a
96    /// new notice is shown without waiting for some PV to change.
97    repaint: Arc<RwLock<Option<Notifier>>>,
98    /// Every tag made, called on a new notice too: each window shows it.
99    tags: Mutex<Vec<Weak<Tag>>>,
100}
101
102impl Control {
103    pub(crate) fn notify(&self, text: String) {
104        log::warn!("{text}");
105        {
106            let mut n = self.notice.lock();
107            *n = (n.0 + 1, text);
108        }
109        if let Some(n) = self.repaint.read().as_ref() {
110            n();
111        }
112        for tag in live(&self.tags) {
113            (tag.0)();
114        }
115    }
116}

118/// The tags in `tags` still held, the others dropped from it.
119fn live(tags: &Mutex<Vec<Weak<Tag>>>) -> Vec<Arc<Tag>> {
120    let mut tags = tags.lock();
121    tags.retain(|t| t.strong_count() > 0);
122    tags.iter().filter_map(Weak::upgrade).collect()
123}

136/// A write's last step, taken once the writer lock is released.
137enum After {
138    /// The backend task has stopped: the write is lost.
139    #[cfg(not(target_arch = "wasm32"))]
140    Stopped(Put),
141    /// The PV connected once and has since dropped: refused at once.
142    #[cfg(not(target_arch = "wasm32"))]
143    Dropped(Put),
144    /// [`WRITE_QUEUE`] writes are already waiting: refused.
145    Full(Put),
146    /// Set a local PV's value.
147    Local(Put),
148    /// The PV takes no writes.
149    Refused,

152thread_local! {
153    /// How many PVs' listeners this thread is calling, under their locks.
154    static CALLING: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
155    /// Work a listener left for when this thread holds no PV's lock.
156    static DEFERRED: std::cell::RefCell<std::collections::VecDeque<Box<dyn FnOnce()>>> =
157        const { std::cell::RefCell::new(std::collections::VecDeque::new()) };
158    /// Whether this thread is running that work, and how many times it has
159    /// begun to (see [`drain_number`]).
160    static DRAINING: std::cell::Cell<bool> = const { std::cell::Cell::new(false) };
161    static DRAINS: std::cell::Cell<u64> = const { std::cell::Cell::new(0) };
162}
163
164/// Call listeners in `f`, under a PV's lock; [`after_listeners`] once the
165/// lock is let go.
166fn calling_listeners<R>(f: impl FnOnce() -> R) -> R {
167    struct Calling;
168    impl Drop for Calling {
169        fn drop(&mut self) {
170            CALLING.with(|c| c.set(c.get() - 1));
171        }
172    }
173    CALLING.with(|c| c.set(c.get() + 1));
174    let _calling = Calling;
175    f()
176}
177
178/// Run the work listeners left, once this thread holds no PV's lock. Work
179/// that leaves more (and updates within it) joins this loop rather than
180/// nesting, so the stack stays flat however long a chain of formulas is.
181fn after_listeners() {
182    if CALLING.with(|c| c.get()) != 0 || DRAINING.with(|d| d.get()) {
183        return;
184    }
185    struct Draining;
186    impl Drop for Draining {
187        fn drop(&mut self) {
188            DRAINING.with(|d| d.set(false));
189        }
190    }
191    DRAINING.with(|d| d.set(true));
192    DRAINS.with(|n| n.set(n.get() + 1));
193    let _draining = Draining;
194    // One at a time, so a panic leaves the rest for the next time.
195    while let Some(work) = DEFERRED.with(|d| d.borrow_mut().pop_front()) {
196        work();
197    }
198}
199
200/// Which run of left work this thread is in: work it leaves for itself
201/// (one formula evaluating another) stays in the same one.
202pub(crate) fn drain_number() -> u64 {
203    DRAINS.with(|n| n.get())
204}
205
206/// Run `work` once this thread holds no PV's lock: now, unless it is in a
207/// listener. Work that locks a PV (subscribing, listening, publishing)
208/// could otherwise wait for one this thread holds, or take two in another
209/// thread's order.
210pub(crate) fn outside_listeners(work: Box<dyn FnOnce()>) {
211    DEFERRED.with(|d| d.borrow_mut().push_back(work));
212    after_listeners();
213}

215/// State shared between a [`Pv`] handle and the task serving it.
216pub(crate) struct Shared {
217    ctl: Arc<Control>,
218    name: String,
219    state: Mutex<PvState>,
220    /// How many times the state has changed.
221    version: AtomicU64,
222    writer: Mutex<Option<Writer>>,
223    /// Writes made before the backend task set the writer, sent once it does.
224    pending: Mutex<Vec<Put>>,
225    /// Whether a backend task sets a writer (`ca://`, `pva://`), so writes
226    /// made before then wait for it.
227    backend_writes: bool,
228    /// Writes accepted but not yet answered by [`Shared::write_result`]: the
229    /// task serving the PV outlives its last [`Pv`] until they are.
230    #[cfg(not(target_arch = "wasm32"))]
231    in_flight: AtomicUsize,
232    notify: Arc<RwLock<Option<Notifier>>>,
233    /// The tags of the handles it was subscribed through, called after each
234    /// change while they are held.
235    tags: Mutex<Vec<Weak<Tag>>>,
236    listeners: Mutex<Vec<Weak<ListenFn>>>,
237    /// Set (under the state lock) once the PV has first connected.
238    was_connected: AtomicBool,
239}

241impl Shared {
242    fn new(name: &PvName, notify: Arc<RwLock<Option<Notifier>>>, ctl: Arc<Control>) -> Arc<Self> {
243        Arc::new(Self {
244            ctl,
245            name: name.to_string(),
246            state: Mutex::new(PvState::default()),
247            version: AtomicU64::new(0),
248            writer: Mutex::new(None),
249            pending: Mutex::new(Vec::new()),
250            backend_writes: matches!(name.scheme, Scheme::Ca | Scheme::Pva),
251            #[cfg(not(target_arch = "wasm32"))]
252            in_flight: AtomicUsize::new(0),
253            notify,
254            tags: Mutex::new(Vec::new()),
255            listeners: Mutex::new(Vec::new()),
256            was_connected: AtomicBool::new(false),
257        })
258    }
259
260    /// Change the state and tell the UI about it.
261    pub(crate) fn update(&self, f: impl FnOnce(&mut PvState)) {
262        {
263            let mut state = self.state.lock();
264            f(&mut state);
265            self.version.fetch_add(1, Ordering::Relaxed);
266            if state.connected {
267                self.was_connected.store(true, Ordering::Relaxed);
268            }
269            // Under the state lock, so listeners see changes in order.
270            // A listener that panics is dropped, so it cannot stop the PV's
271            // task (or the UI thread writing a local PV) for good. Release
272            // builds abort on any panic instead (`panic = "abort"`).
273            let mut listeners = self.listeners.lock();
274            calling_listeners(|| {
275                listeners.retain(|l| match l.upgrade() {
276                    Some(l) => {
277                        let call = std::panic::AssertUnwindSafe(|| l(&state));
278                        let ok = std::panic::catch_unwind(call).is_ok();
279                        if !ok {
280                            log::error!("{}: a listener panicked and was removed", self.name);
281                        }
282                        ok
283                    }
284                    None => false,
285                })
286            });
287        }
288        after_listeners();
289        if let Some(n) = self.notify.read().as_ref() {
290            n();
291        }
292        for tag in live(&self.tags) {
293            (tag.0)();
294        }
295    }
296
297    /// Call `tag`'s notifier after each change, until it is dropped.
298    fn add_tag(&self, tag: &Arc<Tag>) {
299        let mut tags = self.tags.lock();
300        tags.retain(|t| t.strong_count() > 0);
301        if !tags
302            .iter()
303            .any(|t| std::ptr::eq(t.as_ptr(), Arc::as_ptr(tag)))
304        {
305            tags.push(Arc::downgrade(tag));
306        }
307    }

309    /// Whether the PV connected once and has since lost its connection;
310    /// not one that has yet to connect.
311    pub(crate) fn dropped(&self) -> bool {
312        let state = self.state.lock();
313        !state.connected && self.was_connected.load(Ordering::Relaxed)
314    }

316    pub(crate) fn snapshot(&self) -> PvState {
317        self.state.lock().clone()

320    /// A put failed (or worked, with `None`): remember why, and tell the user.
321    #[cfg(not(target_arch = "wasm32"))]
322    pub(crate) fn write_result(&self, text: &str, error: Option<String>) {
323        self.answered();
324        if let Some(e) = &error {
325            self.ctl
326                .notify(format!("{}: write '{text}' failed: {e}", self.name));
327        }
328        self.update(|s| s.write_error = error);
329    }
330
331    /// A write counted in flight is done with: sent, failed or refused.
332    #[cfg(not(target_arch = "wasm32"))]
333    fn answered(&self) {
334        let mut n = self.in_flight.load(Ordering::Acquire);
335        while n > 0 {
336            match self.in_flight.compare_exchange_weak(
337                n,
338                n - 1,
339                Ordering::AcqRel,
340                Ordering::Acquire,
341            ) {
342                Ok(_) => break,
343                Err(now) => n = now,
344            }
345        }
346    }
347
348    /// A put that the server does not answer was sent, or failed to be: a
349    /// failure is remembered as [`Shared::write_result`] does, but a send
350    /// confirms nothing, so it leaves an earlier failure shown.
351    #[cfg(not(target_arch = "wasm32"))]
352    pub(crate) fn write_sent(&self, text: &str, error: Option<String>) {
353        if error.is_some() {
354            self.write_result(text, error);
355        } else {
356            self.answered();
357        }
358    }

378    /// Set where writes go, and send any made before now.
379    pub(crate) fn set_writer(&self, writer: Writer) {
380        let mut slot = self.writer.lock();
381        *slot = Some(writer);
382        let pending = std::mem::take(&mut *self.pending.lock());
383        // Sent before the lock is released, so later writes follow them.
384        let mut failed: Vec<(Put, &str)> = Vec::new();
385        for put in pending {
386            match slot.as_ref() {
387                // Already counted in flight, so a failure is answered
388                // through `write_result`, which uncounts it.
389                #[cfg(not(target_arch = "wasm32"))]
390                Some(Writer::Channel(tx)) => match tx.try_send(put) {
391                    Ok(()) => {}
392                    Err(mpsc::error::TrySendError::Full(put)) => {
393                        failed.push((put, "too many writes waiting"))
394                    }
395                    Err(mpsc::error::TrySendError::Closed(put)) => {
396                        failed.push((put, "the PV's connection has stopped"))
397                    }
398                },
399                #[cfg(target_arch = "wasm32")]
400                Some(Writer::Pvws(name)) => {
401                    pvws::write(name, &self.snapshot(), put.text, put.long_string)
402                }
403                // A backend sets only a channel (or the socket).
404                _ => failed.push((put, "cannot be written")),
405            }
406        }
407        drop(slot);
408        for (put, why) in failed {
409            #[cfg(not(target_arch = "wasm32"))]
410            self.write_result(&put.text, Some(why.into()));
411            #[cfg(target_arch = "wasm32")]
412            self.ctl
413                .notify(format!("{}: write '{}' failed: {why}", self.name, put.text));
414        }
415    }

417    /// Send a new write through `writer`; with none yet, a `ca://` or
418    /// `pva://` PV (whose task is still starting) keeps it until there is
419    /// one. A write is counted in flight only once queued. Returns what is
420    /// left to do once the writer lock is released, so that it and the
421    /// state lock never nest.
422    fn send(&self, writer: Option<&Writer>, put: Put) -> Option<After> {
423        match writer {
424            #[cfg(not(target_arch = "wasm32"))]
425            Some(Writer::Channel(tx)) => {
426                // Counted before the task can answer it, and uncounted if
427                // it was not queued; a stopped task's write is answered (and
428                // uncounted) by `finish`.
429                self.in_flight.fetch_add(1, Ordering::AcqRel);
430                // The task serving the PV holds the receiver until it stops.
431                match tx.try_send(put) {
432                    Ok(()) => None,
433                    Err(mpsc::error::TrySendError::Full(put)) => {
434                        self.in_flight.fetch_sub(1, Ordering::AcqRel);
435                        Some(After::Full(put))
436                    }
437                    Err(mpsc::error::TrySendError::Closed(put)) => Some(After::Stopped(put)),
438                }
439            }
440            Some(Writer::Local) => Some(After::Local(put)),
441            #[cfg(target_arch = "wasm32")]
442            Some(Writer::Pvws(name)) => {
443                pvws::write(name, &self.snapshot(), put.text, put.long_string);
444                None
445            }
446            None if self.backend_writes => {
447                let mut pending = self.pending.lock();
448                if pending.len() >= WRITE_QUEUE {
449                    return Some(After::Full(put));
450                }
451                #[cfg(not(target_arch = "wasm32"))]
452                self.in_flight.fetch_add(1, Ordering::AcqRel);
453                pending.push(put);
454                None
455            }
456            None => Some(After::Refused),
457        }
458    }
459
460    fn finish(&self, after: After) {
461        match after {
462            #[cfg(not(target_arch = "wasm32"))]
463            After::Stopped(put) => {
464                self.write_result(&put.text, Some("the PV's connection has stopped".into()))
465            }
466            #[cfg(not(target_arch = "wasm32"))]
467            After::Dropped(put) => self.ctl.notify(format!(
468                "{}: not writing '{}', not connected",
469                self.name, put.text
470            )),
471            After::Full(put) => self.ctl.notify(format!(
472                "{}: not writing '{}', {WRITE_QUEUE} writes still waiting",
473                self.name, put.text
474            )),
475            After::Local(Put { text, long_string }) => self.update(|s| {
476                // Each write is a new sample, though its value be the same.
477                s.timestamp = Some(web_time::SystemTime::now());
478                s.value = if long_string && s.value.is_numeric_array() {
479                    Value::LongArray(
480                        long_string_bytes(&text, None)
481                            .iter()
482                            .map(|b| *b as i64)
483                            .collect(),
484                    )
485                } else {
486                    local::apply(&s.value, &s.labels, &text)
487                }
488            }),
489            After::Refused => self.ctl.notify(format!("{} cannot be written", self.name)),
490        }
491    }
492}

494/// What serves a PV: stopped when the last [`Pv`] is dropped.
495#[cfg(not(target_arch = "wasm32"))]
496type Task = tokio::task::JoinHandle<()>;
497#[cfg(target_arch = "wasm32")]
498type Task = pvws::Task;
499
500/// How long the task of a dropped PV may go on to send writes made on it,
501/// as a write to a PV that is not connected yet must first find it.
502#[cfg(not(target_arch = "wasm32"))]
503const WRITE_GRACE: std::time::Duration = std::time::Duration::from_secs(10);
504
505/// One subscription; the backend task stops when the last [`Pv`] is dropped,
506/// or once the writes made on it are sent.
507struct Entry {
508    name: PvName,
509    shared: Arc<Shared>,
510    task: Option<Task>,
511    /// The runtime the task runs on, to finish its writes after the drop.
512    #[cfg(not(target_arch = "wasm32"))]
513    rt: tokio::runtime::Handle,
514    /// A formula's inputs, followed for as long as it is held.
515    #[cfg_attr(not(test), allow(dead_code, reason = "only held"))]
516    formula: Option<Arc<formula::Formula>>,
517}
518
519impl Drop for Entry {
520    fn drop(&mut self) {
521        let Some(task) = self.task.take() else {
522            return;
523        };
524        // A write is often made on a PV subscribed just for it and dropped
525        // straight after: let the task send it (or give up) first.
526        #[cfg(not(target_arch = "wasm32"))]
527        if self.shared.in_flight.load(Ordering::Acquire) > 0 {
528            let shared = self.shared.clone();
529            self.rt.spawn(async move {
530                let start = tokio::time::Instant::now();
531                while shared.in_flight.load(Ordering::Acquire) > 0 && start.elapsed() < WRITE_GRACE
532                {
533                    tokio::time::sleep(std::time::Duration::from_millis(20)).await;
534                }
535                task.abort();
536            });
537            return;
538        }
539        task.abort();
540    }
541}

552    /// Whether the PV connected once and has since lost its connection. A

554    /// is queued.

559    /// Mark the PV connected or not, as its backend would; for tests of

562    pub fn set_connected_for_test(&self, connected: bool) {
563        self.0.shared.update(|s| s.connected = connected);

566    /// The latest value, alarm and metadata.
567    pub fn state(&self) -> PvState {
568        let mut state = self.0.shared.snapshot();
569        if self.refuses_writes() {
570            state.writable = false;
571            state.writable_known = true;
572        }
573        state
574    }

576    /// Whether it is connected, without copying its state.
577    pub fn connected(&self) -> bool {
578        self.0.shared.state.lock().connected
579    }

581    /// A number that changes whenever its state does (bar read-only mode),
582    /// so a caller can skip work on a PV that has not changed.
583    pub fn version(&self) -> u64 {
584        self.0.shared.version.load(Ordering::Relaxed)
585    }
586
587    /// Whether read-only mode (see [`PvManager::set_read_only`]) stops writes
588    /// to this PV: all but local ones, which cannot reach a control system.
589    fn refuses_writes(&self) -> bool {
590        let shared = &self.0.shared;
591        shared.ctl.read_only.load(Ordering::Relaxed)
592            && !matches!(*shared.writer.lock(), Some(Writer::Local))
593    }
594
595    /// Call `listener` with the current state now and with each change
596    /// after, until the caller drops it: unlike [`Pv::state`] read once a
597    /// frame, this sees every monitor update.
598    pub fn listen(&self, listener: &Listener) {
599        let shared = &self.0.shared;
600        {
601            let state = shared.state.lock();
602            calling_listeners(|| listener(&state));
603            let mut listeners = shared.listeners.lock();
604            // Drop dead ones here too: a PV that never changes is never updated.
605            listeners.retain(|l| l.strong_count() > 0);
606            listeners.push(Arc::downgrade(listener));
607        }
608        after_listeners();
609    }

611    /// Write a value given as text; the backend converts it to the PV's type.
612    pub fn write(&self, value: impl Into<String>) {
613        self.put(value.into(), false);
614    }
615
616    /// Write text as Phoebus does for a widget whose `format` is string: into
617    /// a char array as a long string (see [`long_string_bytes`]). Any other
618    /// PV is written as by [`Pv::write`].
619    ///
620    /// Over CA, as in Phoebus, a char array is a `DBF_CHAR` channel of more
621    /// than one element, and the text is cut to that element count with its
622    /// NUL kept. Over PVA and `loc://` it is any array of numbers. A PVA
623    /// array has no fixed element count to cut to, so an IOC that stores
624    /// only its first NELM elements may lose the NUL (and part of a
625    /// multi-byte character), as it does with Phoebus.
626    pub fn write_long_string(&self, value: impl Into<String>) {
627        self.put(value.into(), true);
628    }
629
630    fn put(&self, value: String, long_string: bool) {
631        let shared = &self.0.shared;
632        if self.refuses_writes() {
633            shared.ctl.notify(format!(
634                "{}: not writing '{value}', bobview is read-only",
635                shared.name
636            ));
637            return;
638        }
639        // Read first: the state and writer locks never nest.
640        #[cfg(not(target_arch = "wasm32"))]
641        let dropped = shared.dropped();
642        let put = Put {
643            text: value,
644            long_string,
645        };
646        let writer = shared.writer.lock();
647        let after = match writer.as_ref() {
648            // A write to a PV that has gone would wait for it to come back,
649            // and land late, perhaps long after the user moved on. One that
650            // has not connected yet is queued: it was most likely just
651            // subscribed, and the backend waits a while for it.
652            #[cfg(not(target_arch = "wasm32"))]
653            Some(Writer::Channel(_)) if dropped => Some(After::Dropped(put)),
654            w => shared.send(w, put),
655        };
656        drop(writer);
657        if let Some(after) = after {
658            shared.finish(after);
659        }
660    }
661}

663/// Creates and shares PV subscriptions.
664#[derive(Clone)]
665pub struct PvManager {
666    inner: Arc<Inner>,
667    /// What the PVs subscribed through this handle are tagged with (see
668    /// [`PvManager::tagged`]).
669    tag: Option<Arc<Tag>>,
670}
671
672struct Inner {
673    /// Always `Some` until drop, which shuts it down without blocking.
674    #[cfg(not(target_arch = "wasm32"))]
675    rt: Option<tokio::runtime::Runtime>,
676    entries: Mutex<Entries>,
677    locals: Mutex<HashMap<String, Arc<Shared>>>,
678    // Shared with the PV tasks, which must not hold `Inner` itself: the last
679    // reference could then be dropped on a runtime thread.
680    #[cfg(not(target_arch = "wasm32"))]
681    ca: Arc<tokio::sync::OnceCell<epics_ca_rs::client::CaClient>>,
682    #[cfg(not(target_arch = "wasm32"))]
683    pva: Arc<std::sync::OnceLock<epics_pva_rs::client::PvaClient>>,
684    notify: Arc<RwLock<Option<Notifier>>>,
685    ctl: Arc<Control>,
686}
687
688#[cfg(not(target_arch = "wasm32"))]
689impl Drop for Inner {
690    fn drop(&mut self) {
691        if let Some(rt) = self.rt.take() {
692            rt.shutdown_background();
693        }
694    }
695}
696
697/// Every subscription by key, including dropped ones until the next prune.
698#[derive(Default)]
699struct Entries {
700    map: HashMap<String, Weak<Entry>>,
701    /// Prune dropped entries once the map is this big, so a batch of
702    /// subscriptions doesn't scan the map for each one.
703    prune_at: usize,
704}
705
706impl Entries {
707    fn insert(&mut self, key: String, entry: Weak<Entry>) {
708        if self.map.len() >= self.prune_at {
709            self.map.retain(|_, w| w.strong_count() > 0);
710            self.prune_at = (2 * self.map.len()).max(64);
711        }
712        self.map.insert(key, entry);
713    }
714}
715
716impl Default for PvManager {
717    fn default() -> Self {
718        Self::new()
719    }
720}

722impl PvManager {
723    #[cfg(target_arch = "wasm32")]
724    pub fn new() -> Self {
725        let ctl = Arc::new(Control::default());
726        Self {
727            inner: Arc::new(Inner {
728                entries: Mutex::default(),
729                locals: Mutex::new(HashMap::new()),
730                notify: ctl.repaint.clone(),
731                ctl,
732            }),
733            tag: None,
734        }
735    }
736
737    #[cfg(not(target_arch = "wasm32"))]
738    pub fn new() -> Self {
739        let rt = tokio::runtime::Builder::new_multi_thread()
740            .worker_threads(2)
741            .thread_name("bobview-pv")
742            .enable_all()
743            .build()
744            .expect("failed to start the PV runtime");
745        let ctl = Arc::new(Control::default());
746        Self {
747            inner: Arc::new(Inner {
748                rt: Some(rt),
749                entries: Mutex::default(),
750                locals: Mutex::new(HashMap::new()),
751                ca: Arc::new(tokio::sync::OnceCell::new()),
752                pva: Arc::new(std::sync::OnceLock::new()),
753                notify: ctl.repaint.clone(),
754                ctl,
755            }),
756            tag: None,
757        }
758    }
759
760    /// Register a callback run after every PV change.
761    pub fn set_notifier(&self, notifier: Notifier) {
762        *self.inner.notify.write() = Some(notifier);
763    }
764
765    /// A handle to the same PVs that tags each PV subscribed through it with
766    /// `notifier`, called after each change of those PVs (and each new
767    /// notice) for as long as this handle or a clone of it is held. A UI
768    /// window gives its display one, to repaint only the windows showing a
769    /// PV that changed.
770    pub fn tagged(&self, notifier: Notifier) -> Self {
771        let tag = Arc::new(Tag(notifier));
772        let mut tags = self.inner.ctl.tags.lock();
773        tags.retain(|t| t.strong_count() > 0);
774        tags.push(Arc::downgrade(&tag));
775        drop(tags);
776        Self {
777            inner: self.inner.clone(),
778            tag: Some(tag),
779        }
780    }
781
782    /// A handle with this one's notifier under a new tag: PVs subscribed
783    /// through it are tagged afresh, and those subscribed only through this
784    /// one stop calling the notifier once it and its clones are dropped (a
785    /// window showing another display).
786    pub fn retagged(&self) -> Self {
787        match &self.tag {
788            Some(tag) => self.tagged(tag.0.clone()),
789            None => self.clone(),
790        }
791    }
792
793    /// Refuse (or allow again) writes to every PV but `loc://` ones; PVs then
794    /// report themselves not writable, so controls are disabled.
795    pub fn set_read_only(&self, read_only: bool) {
796        self.inner.ctl.read_only.store(read_only, Ordering::Relaxed);
797    }
798
799    pub fn read_only(&self) -> bool {
800        self.inner.ctl.read_only.load(Ordering::Relaxed)
801    }
802
803    /// The latest message for the user about a refused or failed write, and a
804    /// number that changes with each new one (starting at 0 for none).
805    pub fn notice(&self) -> (u64, String) {
806        self.inner.ctl.notice.lock().clone()
807    }
808
809    /// The tokio runtime the PV tasks run on.
810    #[cfg(not(target_arch = "wasm32"))]
811    pub fn runtime(&self) -> &tokio::runtime::Runtime {
812        self.inner.rt.as_ref().expect("runtime present until drop")
813    }
814
815    /// Get (or start) a subscription for `name`, e.g. `ca://X`, `sim://sine` or plain `X`.
816    pub fn subscribe(&self, name: &str) -> Pv {
817        let name = PvName::parse(name);
818        let key = name.key();
819        let mut entries = self.inner.entries.lock();
820        if let Some(entry) = entries.map.get(&key).and_then(Weak::upgrade) {
821            drop(entries);
822            if let Some(tag) = &self.tag {
823                entry.shared.add_tag(tag);
824            }
825            return Pv(entry);
826        }
827        // A macro left unexpanded makes a name that is not what was meant,
828        // and a local one would share its state with every other such PV.
829        if name.pv.contains("$(") || name.pv.contains("${") {
830            log::warn!("{}: PV name has an undefined macro", name.pv);
831        }
832
833        let shared = || Shared::new(&name, self.inner.notify.clone(), self.inner.ctl.clone());
834        let (shared, task, formula) = match name.scheme {
835            Scheme::Loc => (self.local(&name), None, None),
836            Scheme::Formula => {
837                let shared = shared();
838                let f = formula::Formula::new(&name, shared.clone(), Arc::downgrade(&self.inner));
839                (shared, None, Some(f))
840            }
841            Scheme::Unsupported => {
842                let shared = shared();
843                log::warn!("{name}: PV type not supported");
844                // New, so no one is listening or drawing it yet.
845                shared.state.lock().error = Some("PV type not supported".into());
846                (shared, None, None)
847            }
848            _ => {
849                let shared = shared();
850                let task = self.spawn_backend(&name, shared.clone());
851                (shared, task, None)
852            }
853        };
854        let entry = Arc::new(Entry {
855            name,
856            shared,
857            task,
858            #[cfg(not(target_arch = "wasm32"))]
859            rt: self.runtime().handle().clone(),
860            formula: formula.clone(),
861        });
862        entries.insert(key, Arc::downgrade(&entry));
863        // Its inputs are subscribed to with the entries unlocked.
864        drop(entries);
865        if let Some(tag) = &self.tag {
866            entry.shared.add_tag(tag);
867        }
868        if let Some(f) = formula {
869            f.start();
870        }
871        Pv(entry)
872    }
873
874    fn local(&self, name: &PvName) -> Arc<Shared> {
875        let mut locals = self.inner.locals.lock();
876        locals
877            .entry(name.key())
878            .or_insert_with(|| {
879                local::create(
880                    name,
881                    Shared::new(name, self.inner.notify.clone(), self.inner.ctl.clone()),
882                )
883            })
884            .clone()
885    }
886
887    #[cfg(target_arch = "wasm32")]
888    fn spawn_backend(&self, name: &PvName, shared: Arc<Shared>) -> Option<Task> {
889        Some(match name.scheme {
890            Scheme::Sim => pvws::spawn(sim::run(name.clone(), shared)),
891            Scheme::Ca | Scheme::Pva => pvws::subscribe(name, shared),
892            Scheme::Loc | Scheme::Formula | Scheme::Unsupported => {
893                unreachable!("local, formula and unsupported PVs have no task")
894            }
895        })
896    }
897
898    #[cfg(not(target_arch = "wasm32"))]
899    fn spawn_backend(&self, name: &PvName, shared: Arc<Shared>) -> Option<Task> {
900        let rt = self.runtime().handle();
901        match name.scheme {
902            Scheme::Ca => {
903                let ca = self.inner.ca.clone();
904                let pv = name.pv.clone();
905                Some(rt.spawn(async move {
906                    let client = ca.get_or_try_init(epics_ca_rs::client::CaClient::new).await;
907                    match client {
908                        Ok(client) => ca::run(client, &pv, shared).await,
909                        Err(e) => {
910                            log::error!("cannot start the CA client: {e}");
911                            shared.update(|s| s.error = Some(format!("CA client: {e}")));
912                        }
913                    }
914                }))
915            }
916            Scheme::Sim => Some(rt.spawn(sim::run(name.clone(), shared))),
917            Scheme::Pva => {
918                let pva = self.inner.pva.clone();
919                let pv = name.pv.clone();
920                Some(rt.spawn(async move {
921                    let client =
922                        pva.get_or_init(|| epics_pva_rs::client::PvaClient::builder().build());
923                    pva::run(client, &pv, shared).await
924                }))
925            }
926            Scheme::Loc | Scheme::Formula | Scheme::Unsupported => {
927                unreachable!("local, formula and unsupported PVs have no task")
928            }
929        }
930    }
