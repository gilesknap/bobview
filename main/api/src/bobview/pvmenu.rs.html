20/// Fill the context menu for a widget's PV. `trends` are the open trend
21/// windows, (id, title), that the PV can be added to.
22pub fn menu(
23    ui: &mut Ui,
24    node: &Node,
25    state: Option<&PvState>,
26    trends: &[(u64, String)],
27    requests: &mut Vec<Request>,
28) {
29    let name = node.pv_name.as_str();
30    let value = state.filter(|s| s.connected).map(|s| s.format(None, true));
31
32    // Also to the primary selection, as a middle-click on the widget does.
33    if ui.button("Copy PV name").clicked()
34        && let Some(names) = copy_names(node)
35    {
36        requests.push(Request::Copy(names));
37    }
38    if let (Some(state), Some(value)) = (state, &value) {
39        if ui.button("Copy value").clicked() {
40            ui.ctx().copy_text(value.clone());
41        }
42        if ui.button("Copy name and value").clicked() {
43            ui.ctx().copy_text(format!("{name} = {value}"));
44        }
45        if ui.button("Copy name, value, alarm and time").clicked() {
46            ui.ctx().copy_text(full_line(name, state));
47        }
48    }
49    if let Some(cmd) = cli_command(&PvName::parse(name))
50        && ui.button(format!("Copy \"{cmd}\"")).clicked()
51    {
52        ui.ctx().copy_text(cmd);
53    }
54    if let Some(pv) = &node.pv {
55        ui.separator();
56        if ui.button("Trend").clicked() {
57            requests.push(Request::Trend(vec![name.to_string()]));
58        }
59        let all = trend_names(node);
60        if all.len() > 1
61            && ui
62                .button(format!("Trend with rule PVs ({})", all.len()))
63                .clicked()
64        {
65            requests.push(Request::Trend(all.clone()));
66        }
67        add_to_trend(ui, "", &[name.to_string()], trends, requests);
68        if all.len() > 1 {
69            let with = format!(" with rule PVs ({})", all.len());
70            add_to_trend(ui, &with, &all, trends, requests);
71        }
72        if ui.button("PV details…").clicked() {
73            requests.push(Request::Details(Details {
74                pv: pv.clone(),
75                widget: widget_label(node),
76                file: node.base.clone(),
77            }));
78        }
79    }
80}

82/// "Add to trend" entries: a button for the one open trend window, or a
83/// submenu listing them when there are several.
84fn add_to_trend(
85    ui: &mut Ui,
86    what: &str,
87    names: &[String],
88    trends: &[(u64, String)],
89    requests: &mut Vec<Request>,
90) {
91    match trends {
92        [] => {}
93        [(id, title)] => {
94            if ui.button(format!("Add{what} to \"{title}\"")).clicked() {
95                requests.push(Request::TrendAdd(*id, names.to_vec()));
96            }
97        }
98        _ => {
99            ui.menu_button(format!("Add{what} to trend"), |ui| {
100                for (id, title) in trends {
101                    if ui.button(title).clicked() {
102                        requests.push(Request::TrendAdd(*id, names.to_vec()));
103                    }
104                }
105            });
106        }
107    }
108}

110/// The widget's PV and the other PVs its rules read, each once.
111fn trend_names(node: &Node) -> Vec<String> {
112    let mut names: Vec<String> = Vec::new();
113    let rules = node.rules.iter().flat_map(|r| r.pvs());
114    for pv in node.pv.iter().chain(rules) {
115        let name = pv.name().to_string();
116        if !names.contains(&name) {
117            names.push(name);
118        }
119    }
120    names
121}

123/// What a middle-click on the widget copies: its PV's name, or, for a plot
124/// with no PV of its own (an XY plot or strip chart), its traces' PVs,
125/// separated by spaces (see [`copy_name`]). Rule PVs are left out: they
126/// choose how the widget looks, and are not what it shows.
127pub fn copy_names(node: &Node) -> Option<String> {
128    let mut names: Vec<String> = Vec::new();
129    if !node.pv_name.is_empty() {
130        copy_name(&PvName::parse(&node.pv_name), &mut names);
131    } else {
132        for t in &node.traces {
133            for pv in t.y.iter().chain(&t.x) {
134                copy_name(pv.name(), &mut names);
135            }
136        }
137    }
138    (!names.is_empty()).then(|| names.join(" "))
139}

141/// Add a PV's name to `names` (unless it is there) as another EPICS tool
142/// takes it: a Channel Access name bare and whole, a PV Access, local or
143/// simulated one with its prefix (`loc://x`, without the type and initial
144/// value of `loc://x<VDouble>(1.5)`; a simulated PV keeps its parameters).
145/// A formula adds the PVs it reads instead, as its text has spaces.
146fn copy_name(name: &PvName, names: &mut Vec<String>) {
147    let name = match name.scheme {
148        Scheme::Ca => name.pv.clone(),
149        Scheme::Pva | Scheme::Loc | Scheme::Sim => name.key(),
150        Scheme::Unsupported => name.to_string(),
151        Scheme::Formula => {
152            for input in pv::formula_inputs(&name.pv) {
153                // An input is never a formula itself: `=` starts no name.
154                let input = PvName::parse(&input);
155                if input.scheme != Scheme::Formula {
156                    copy_name(&input, names);
157                }
158            }
159            return;
160        }
161    };
162    if !names.contains(&name) {
163        names.push(name);
164    }
165}

167/// The body of the PV details window.
168pub fn details_ui(ui: &mut Ui, details: &Details) {
169    let rows = details_rows(details, &details.pv.state());
170    egui::Grid::new("pv-details")
171        .num_columns(2)
172        .striped(true)
173        .show(ui, |ui| {
174            for (key, val) in &rows {
175                ui.label(*key);
176                ui.label(val);
177                ui.end_row();
178            }
179        });
180    ui.add_space(4.0);
181    if ui.button("Copy all").clicked() {
182        let text: Vec<String> = rows.iter().map(|(k, v)| format!("{k}: {v}")).collect();
183        ui.ctx().copy_text(text.join("\n"));
184    }
185}

187/// What the details window shows, as (label, value) rows.
188fn details_rows(details: &Details, s: &PvState) -> Vec<(&'static str, String)> {
189    let name = details.pv.name();
190    let mut rows = vec![
191        ("Name", name.to_string()),
192        ("Protocol", protocol(name.scheme).into()),
193        (
194            "Connection",
195            match (&s.connected, &s.error) {
196                (true, _) => "connected".into(),
197                (false, Some(e)) => format!("disconnected: {e}"),
198                (false, None) => "disconnected".into(),
199            },
200        ),
201    ];
202    if s.connected {
203        rows.push(("Value", s.format(None, true)));
204        rows.push(("Type", value_type(&s.value)));
205        rows.push((
206            "Alarm",
207            format!("{} (status {})", severity_name(s.severity), s.status),
208        ));
209        if let Some(t) = s.timestamp {
210            rows.push(("Time", format_time(t)));
211        }
212        if !s.units.is_empty() {
213            rows.push(("Units", s.units.clone()));
214        }
215        if let Some(p) = s.precision {
216            rows.push(("Precision", p.to_string()));
217        }
218        if let Some((lo, hi)) = s.display_range {
219            rows.push(("Display limits", format!("{lo} to {hi}")));
220        }
221        if let Some((lo, hi)) = s.control_range {
222            rows.push(("Control limits", format!("{lo} to {hi}")));
223        }
224        if !s.labels.is_empty() {
225            rows.push(("States", s.labels.join(", ")));
226        }
227        rows.push(("Writable", writable_text(s).into()));
228    }
229    // Kept while disconnected: a write may have failed for that very reason.
230    if let Some(e) = &s.write_error {
231        rows.push(("Last write failed", e.clone()));
232    }
233    rows.push(("Widget", details.widget.clone()));
234    if let Some(file) = &details.file {
235        rows.push(("Display", file.to_string()));
236    }
237    rows
238}

240/// Whether a write is allowed: PV Access gives no access rights, so there
241/// the answer is a guess until a put is refused.
242fn writable_text(s: &PvState) -> &'static str {
243    match (s.writable, s.writable_known) {
244        (true, true) => "yes",
245        (false, _) => "no",
246        (true, false) => "unknown (the server decides when you write)",
247    }
248}

250/// A one-line summary for bug reports and logbooks.
251fn full_line(name: &str, s: &PvState) -> String {
252    let mut line = format!(
253        "{name} = {} {}",
254        s.format(None, true),
255        severity_name(s.severity)
256    );
257    if let Some(t) = s.timestamp {
258        line.push(' ');
259        line.push_str(&format_time(t));
260    }
261    line
262}

264/// A shell command that reads the PV, for the protocols that have one.
265fn cli_command(name: &PvName) -> Option<String> {
266    match name.scheme {
267        Scheme::Ca => Some(format!("caget {}", shell_quote(&name.pv))),
268        Scheme::Pva => Some(format!("pvget {}", shell_quote(&name.pv))),
269        Scheme::Loc | Scheme::Sim | Scheme::Formula | Scheme::Unsupported => None,
270    }
271}
272
273/// Quote `s` as one shell word, so a pasted command can't run anything else.
274/// Names made only of the usual PV characters are left bare.
275fn shell_quote(s: &str) -> String {
276    let plain = |c: char| c.is_ascii_alphanumeric() || "_-:.+/=,@%".contains(c);
277    if !s.is_empty() && s.chars().all(plain) {
278        s.to_string()
279    } else {
280        format!("'{}'", s.replace('\'', r"'\''"))
281    }
282}

284fn widget_label(node: &Node) -> String {
285    match node.w.text("name") {
286        Some(name) if !name.is_empty() => format!("{} \"{name}\"", node.w.kind),
287        _ => node.w.kind.clone(),
288    }
289}

291fn protocol(scheme: Scheme) -> &'static str {
292    match scheme {
293        Scheme::Ca => "Channel Access",
294        Scheme::Pva => "PV Access",
295        Scheme::Loc => "local",
296        Scheme::Sim => "simulated",
297        Scheme::Formula => "formula",
298        Scheme::Unsupported => "not supported",
299    }
300}

302fn severity_name(sevr: Severity) -> &'static str {
303    match sevr {
304        Severity::NoAlarm => "NO_ALARM",
305        Severity::Minor => "MINOR",
306        Severity::Major => "MAJOR",
307        Severity::Invalid => "INVALID",
308    }
309}

311fn value_type(v: &Value) -> String {
312    match v {
313        Value::None => "none".into(),
314        Value::Double(_) => "double".into(),
315        Value::Long(_) => "integer".into(),
316        Value::Str(_) => "string".into(),
317        Value::Enum(_) => "enum".into(),
318        Value::DoubleArray(a) => format!("double[{}]", a.len()),
319        Value::LongArray(a) => format!("integer[{}]", a.len()),
320        Value::StrArray(a) => format!("string[{}]", a.len()),
321        Value::Image(img) => format!("image {}x{}", img.width, img.height),
322    }
323}

325/// ISO 8601 UTC with milliseconds, e.g. `2026-10-14T19:08:33.123Z`.
326fn format_time(t: SystemTime) -> String {
327    let Ok(d) = t.duration_since(UNIX_EPOCH) else {
328        return "before 1970".into();
329    };
330    let secs = d.as_secs();
331    let (days, rem) = (secs / 86_400, secs % 86_400);
332    let (y, m, day) = civil_from_days(days as i64);
333    format!(
334        "{y:04}-{m:02}-{day:02}T{:02}:{:02}:{:02}.{:03}Z",
335        rem / 3600,
336        rem / 60 % 60,
337        rem % 60,
338        d.subsec_millis()
339    )
340}

342/// Days since 1970-01-01 to (year, month, day), after Howard Hinnant's
343/// `civil_from_days`.
344fn civil_from_days(z: i64) -> (i64, u32, u32) {
345    let z = z + 719_468;
346    let era = z.div_euclid(146_097);
347    let doe = z.rem_euclid(146_097);
348    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
349    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
350    let mp = (5 * doy + 2) / 153;
351    let d = (doy - (153 * mp + 2) / 5 + 1) as u32;
352    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
353    let y = yoe + era * 400 + i64::from(m <= 2);
354    (y, m, d)
355}

394    #[test]
395    fn copied_names_are_as_other_tools_take_them() {
396        let name = |n| {
397            let mut names = Vec::new();
398            copy_name(&PvName::parse(n), &mut names);
399            names.join(" ")
400        };
401        assert_eq!(name("BL01T:X"), "BL01T:X");
402        assert_eq!(name("ca://BL01T:X"), "BL01T:X");
403        // EPICS record names may hold `<` and `(`.
404        assert_eq!(name("ca://T<1>(2)"), "T<1>(2)");
405        assert_eq!(name("pva://BL01T:IMG"), "pva://BL01T:IMG");
406        assert_eq!(name("loc://x<VDouble>(1.5)"), "loc://x");
407        assert_eq!(name("sim://sine(0, 10, 0.1)"), "sim://sine(0, 10, 0.1)");
408        // A formula's inputs, each once.
409        assert_eq!(
410            name("=`ca://A:B` + 'pva://C' * `A:B` + 'loc://m<VLong>(1)'"),
411            "A:B pva://C loc://m"
412        );
413        assert_eq!(name("=1 + 2"), "");
414    }
