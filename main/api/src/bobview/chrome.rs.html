1//! Sizing of bobview's own controls (toolbar, status line, menus)
2//! and of the native window.
3//!
4//! egui sizes everything in points and multiplies by the desktop's scale
5//! factor, which on a phone (2.5-3) makes the controls take a large share of
6//! a small screen. The display itself is unaffected: it is fitted to the space
7//! left over. So the controls get their own scale, set through egui's zoom
8//! factor: by default chosen from the screen size in physical pixels, or fixed
9//! with `--ui-scale` / `BOBVIEW_UI_SCALE`.
10
11use eframe::egui::{self, Rect, Vec2};
12
13/// The toolbar's share of the screen height it should not exceed.
14const TOOLBAR_FRACTION: f32 = 0.07;
15/// Smallest toolbar button, in physical pixels: a usable touch target.
16const MIN_BUTTON_PX: f32 = 40.0;
17/// Toolbar width, in points, below which it switches to the compact layout.
18pub const FULL_TOOLBAR_WIDTH: f32 = 600.0;
19/// Width of the compact toolbar in points; windows are never sized narrower.
20pub const COMPACT_TOOLBAR_WIDTH: f32 = 290.0;
21/// Largest share of the monitor an automatically sized window may take.
22const MAX_MONITOR_FRACTION: f32 = 0.9;
23/// The window size before a display is shown, in points at the desktop scale.
24pub const DEFAULT_WINDOW: Vec2 = Vec2::new(1024.0, 720.0);
25
26/// The controls' scale relative to the desktop's (egui's zoom factor).
27#[derive(Clone, Copy, Debug, PartialEq)]
28pub enum UiScale {
29    /// Chosen from the screen size.
30    Auto,
31    /// Fixed by the user: 1 is the desktop's own scale.
32    Fixed(f32),
33}
34
35impl UiScale {
36    /// Parse `auto` or a positive factor.
37    pub fn parse(text: &str) -> Option<Self> {
38        let text = text.trim();
39        if text.eq_ignore_ascii_case("auto") {
40            return Some(Self::Auto);
41        }
42        text.parse::<f32>()
43            .ok()
44            .filter(|f| f.is_finite() && *f > 0.0)
45            .map(|f| Self::Fixed(f.clamp(0.25, 4.0)))
46    }
47
48    /// From `BOBVIEW_UI_SCALE`, else automatic.
49    pub fn from_env() -> Self {
50        std::env::var("BOBVIEW_UI_SCALE")
51            .ok()
52            .and_then(|v| {
53                let scale = Self::parse(&v);
54                if scale.is_none() {
55                    log::warn!("ignoring BOBVIEW_UI_SCALE={v}: expected a number or auto");
56                }
57                scale
58            })
59            .unwrap_or(Self::Auto)
60    }
61
62    /// The zoom factor to give egui.
63    ///
64    /// `native` is the desktop's pixels per point, `screen_px` the height in
65    /// physical pixels of the screen (or window when the screen is unknown),
66    /// `toolbar` and `button` the toolbar and its buttons' heights in points.
67    pub fn zoom_factor(self, native: f32, screen_px: f32, toolbar: f32, button: f32) -> f32 {
68        match self {
69            Self::Fixed(f) => f,
70            Self::Auto => auto_pixels_per_point(native, screen_px, toolbar, button) / native,
71        }
72    }
73}
74
75/// Pixels per point for the controls: the desktop's own scale unless that
76/// makes the toolbar more than [`TOOLBAR_FRACTION`] of the screen, but never so
77/// small that a button is under [`MIN_BUTTON_PX`] (or that a point is under a
78/// pixel), and never larger than the desktop asks for.
79fn auto_pixels_per_point(native: f32, screen_px: f32, toolbar: f32, button: f32) -> f32 {
80    let native = native.max(0.1);
81    let cap = TOOLBAR_FRACTION * screen_px / toolbar.max(1.0);
82    let floor = (MIN_BUTTON_PX / button.max(1.0)).max(1.0).min(native);
83    cap.min(native).max(floor)
84}

86/// Whether the zoom factor has moved enough to be worth a relayout.
87pub fn changed(old: f32, new: f32) -> bool {
88    (old - new).abs() > 0.01 * old.abs().max(1e-3)
89}

91/// Inner window size, in points, that shows a display of `display` points
92/// (already multiplied by its zoom) under `chrome` points of controls,
93/// limited to `monitor` points when known.
94pub fn window_size(display: Vec2, chrome: f32, monitor: Option<Vec2>) -> Vec2 {
95    // On a monitor too small for the display at 100%, shrink the display
96    // keeping its shape, so Fit fills the window without a blank band.
97    let mut display = display;
98    if let Some(m) = monitor.filter(|m| m.x > 0.0 && m.y > 0.0) {
99        let room = m * MAX_MONITOR_FRACTION - Vec2::new(1.0, chrome.ceil() + 1.0);
100        let fit = (room.x / display.x).min(room.y / display.y);
101        if fit < 1.0 && fit > 0.0 {
102            display *= fit;
103        }
104    }
105    // A pixel either way keeps rounding from adding scroll bars.
106    Vec2::new(
107        display.x.ceil().max(COMPACT_TOOLBAR_WIDTH) + 1.0,
108        display.y.ceil() + chrome.ceil() + 1.0,
109    )
110}

112/// Inner window size, in points, for controls `by` points taller (or shorter
113/// when negative) to leave the display the room it has: only the height
114/// changes, and growing stops where the window, with `border` points of title
115/// bar and borders, is as tall as `monitor` points when known, or where it
116/// already is when it is taller than that.
117pub fn resized_for_controls(inner: Vec2, by: f32, monitor: Option<Vec2>, border: f32) -> Vec2 {
118    let tallest = monitor.map_or(f32::INFINITY, |m| (m.y - border).max(inner.y));
119    Vec2::new(inner.x, (inner.y + by).min(tallest).max(0.0))
120}

122/// Whether the window should follow the display's size: not when the user
123/// or the window manager has made it full screen or maximised, nor on a
124/// screen too small for the default window (a phone), where it fills the
125/// screen anyway.
126pub fn can_resize(info: &egui::ViewportInfo, zoom_factor: f32) -> bool {
127    info.fullscreen != Some(true)
128        && info.maximized != Some(true)
129        && !small_screen(info, zoom_factor)
130}
131
132/// Whether the window's monitor is too small for the default window (a
133/// phone).
134pub fn small_screen(info: &egui::ViewportInfo, zoom_factor: f32) -> bool {
135    // The monitor size is in points at the current zoom; the default window
136    // is in points at the desktop scale.
137    info.monitor_size
138        .map(|m| m * zoom_factor)
139        .is_some_and(|m| m.x < DEFAULT_WINDOW.x || m.y < DEFAULT_WINDOW.y)
140}

142/// Show `popup` as a menu of `add`, no taller than the window: egui keeps a
143/// popup in its window by moving it, so a taller one would be cut off. One
144/// that fits looks as it would without this: its scroll area takes no room.
145/// A context menu right-clicked again elsewhere moves there rather than
146/// closing, deliberately: egui's `CloseOnClick` closed it first.
147pub fn menu<R>(
148    popup: egui::Popup,
149    add: impl FnOnce(&mut egui::Ui) -> R,
150) -> Option<egui::InnerResponse<R>> {
151    // Closed by `menu_contents` on a click in it, which isn't on the bar.
152    popup
153        .close_behavior(egui::PopupCloseBehavior::CloseOnClickOutside)
154        .show(|ui| menu_contents(ui, add))
155}
156
157fn menu_contents<R>(ui: &mut egui::Ui, add: impl FnOnce(&mut egui::Ui) -> R) -> R {
158    let margin = egui::Frame::popup(ui.style()).total_margin().sum();
159    let window = ui.ctx().content_rect().size();
160    let room = (window.y - margin.y).max(0.0);
161    // A popup has last frame's size to draw in (at first, a default): it may
162    // be as tall as there is room for.
163    ui.set_max_height(room);
164    // Beside the items, not over them, and only there when there is more.
165    ui.spacing_mut().scroll = egui::style::ScrollStyle::solid();
166    let width = ui.spacing().scroll.allocated_width();
167    // While there is a bar the rows keep the width they were first laid out
168    // at, as egui would, the menu widening by it if the window has room (and
169    // a point, as a scroll area rounds its width).
170    let id = egui::containers::menu::find_menu_root(ui).id;
171    let seen = id.with("menu scrolls");
172    let (scrolled, mut rows): (bool, f32) = ui.data(|d| d.get_temp(seen)).unwrap_or_default();
173    let sizing = ui.is_sizing_pass();
174    let widened = rows + width + 1.0;
175    if scrolled && !sizing && widened + margin.x <= window.x {
176        ui.set_max_width(widened);
177    }
178    let out = egui::ScrollArea::vertical()
179        .max_height(room)
180        .min_scrolled_height(0.0)
181        .show(ui, add);
182    let inner = out.inner_rect;
183    let scrolls = out.content_size.y > inner.height();
184    if sizing {
185        rows = out.content_size.x;
186    }
187    ui.data_mut(|d| d.insert_temp(seen, (scrolls, rows)));
188    let bar = scrolls
189        .then(|| Rect::from_x_y_ranges(inner.right()..=inner.right() + width, inner.y_range()));
190    // As egui's `CloseOnClick`, but for that bar. Not the click that opened
191    // it, nor while a submenu is open, which then decides.
192    let shown = ui.ctx().read_response(id).map(|r| r.rect);
193    // Not the latest position: a touch's release has none.
194    let click = ui.input(|i| i.pointer.interact_pos().filter(|_| i.pointer.any_click()));
195    if let (Some(pos), Some(shown)) = (click, shown)
196        && shown.contains(pos)
197        && !bar.is_some_and(|bar| bar.contains(pos))
198        && egui::containers::menu::MenuState::is_deepest_open_sub_menu(ui.ctx(), id)
199    {
200        ui.close();
201    }
202    out.inner
203}
204
205/// A submenu of the menu `ui` is in, filled as [`menu_contents`] does.
206pub fn submenu<R>(ui: &mut egui::Ui, label: &str, add: impl FnOnce(&mut egui::Ui) -> R) {
207    ui.menu_button(label, |ui| menu_contents(ui, add));
208}

210/// The controls' scale and the window size, carried between frames.
211pub struct Chrome {
212    pub scale: UiScale,
213    /// Heights in points of the toolbar, one of its buttons, and all the
214    /// panels around the display, as last laid out.
215    pub toolbar: f32,
216    pub button: f32,
217    pub panels: f32,
218    /// Whether the zoom factor is where `scale` wants it.
219    settled: bool,
220    /// A display was shown: size the window to it if allowed.
221    resize: bool,
222    /// Size the window to every display shown, not only the first; the
223    /// toolbar's Auto-size toggle.
224    pub auto_size: bool,
225    /// The window has not been sized yet.
226    first: bool,
227    /// Keep the window's size for the next display shown: a layout gave it.
228    hold: bool,
229    /// Size the window to the display next frame, Auto-size or not.
230    fit: bool,
231    /// egui's zoom factor at the last pass.
232    last_zoom: Option<f32>,
233    /// The zoom factor the window system measured the monitor at, for every
234    /// pass of this frame.
235    monitor_zoom: Option<f32>,
236}

238impl Chrome {
239    pub fn new(scale: UiScale) -> Self {
240        Self {
241            scale,
242            toolbar: 32.0,
243            button: 26.0,
244            panels: 56.0,
245            settled: false,
246            resize: false,
247            auto_size: true,
248            first: true,
249            hold: false,
250            fit: false,
251            last_zoom: None,
252            monitor_zoom: None,
253        }
254    }
255
256    /// Whether the zoom factor is where the scale wants it.
257    pub fn settled(&self) -> bool {
258        self.settled
259    }
260
261    /// The window has been given its size (by a layout): keep it when the
262    /// next display is shown.
263    pub fn hold_size(&mut self) {
264        self.first = false;
265        self.hold = true;
266    }
267
268    /// Size the window to the display, as shown at its zoom, Auto-size or
269    /// not: the toolbar's zoom readout, which sets 100%.
270    pub fn fit_window(&mut self) {
271        self.resize = true;
272        self.hold = false;
273        self.fit = true;
274    }
275
276    /// A new display is being shown.
277    pub fn display_changed(&mut self) {
278        self.resize = true;
279    }
280
281    /// Set egui's zoom factor for this frame's controls (it applies from the
282    /// next frame).
283    pub fn begin(&mut self, ctx: &egui::Context) {
284        let native = ctx.native_pixels_per_point().unwrap_or(1.0);
285        let ppp = ctx.pixels_per_point();
286        let (window, monitor) = ctx.input(|i| (i.content_rect().size(), i.viewport().monitor_size));
287        // The window system measures the monitor in points at the zoom of
288        // the last frame, before the one just set (egui rescales only the
289        // window to that): taken at this one, a change of zoom would go
290        // round in a loop. A frame's later passes see the same measure.
291        let last = self.last_zoom.replace(ctx.zoom_factor());
292        if ctx.current_pass_index() == 0 {
293            self.monitor_zoom = Some(last.unwrap_or(ctx.zoom_factor()));
294        }
295        let measured = self.monitor_zoom.unwrap_or(ctx.zoom_factor());
296        let screen_px = match monitor.filter(|m| m.y > 0.0) {
297            Some(m) => m.y * native * measured,
298            None => window.y * ppp,
299        };
300        let zoom = self
301            .scale
302            .zoom_factor(native, screen_px, self.toolbar, self.button);
303        self.settled = !changed(ctx.zoom_factor(), zoom);
304        if !self.settled {
305            ctx.set_zoom_factor(zoom);
306        }
307    }
308
309    /// Size the window to a newly shown display, `display` points at the zoom
310    /// it will be shown at, once the controls have their final size.
311    pub fn end(&mut self, ctx: &egui::Context, display: Option<Vec2>) {
312        if !self.resize || !self.settled {
313            return;
314        }
315        // A display that failed to load still ends a layout's hold on
316        // the size.
317        let Some(display) = display else {
318            self.resize = false;
319            self.hold = false;
320            self.fit = false;
321            return;
322        };
323        let (info, inner) = ctx.input(|i| (i.viewport().clone(), i.content_rect().size()));
324        // Wait for the window to know its monitor, so as never to size a
325        // window larger than the screen.
326        if info.monitor_size.is_none() {
327            return;
328        }
329        self.resize = false;
330        if std::mem::take(&mut self.hold) {
331            return;
332        }
333        let fit = std::mem::take(&mut self.fit);
334        let follow = self.first || self.auto_size || fit;
335        self.first = false;
336        if !follow || !can_resize(&info, ctx.zoom_factor()) {
337            return;
338        }
339        let size = window_size(display, self.panels, info.monitor_size);
340        if (size - inner).length() >= 1.0 {
341            log::debug!("size the window from {inner:?} to {size:?} for the display");
342            ctx.send_viewport_cmd(egui::ViewportCommand::InnerSize(size));
343        }
344    }
345}

445    /// A window holding a button with a menu, drawn pass by pass.
446    struct MenuWindow {
447        ctx: egui::Context,
448        width: f32,
449        height: f32,
450        /// Drawn as egui would, not by `menu`.
451        plain: bool,
452        /// The menu's frame, as last drawn.
453        menu: Option<egui::Rect>,
454    }
455
456    impl MenuWindow {
457        fn new(height: f32) -> Self {
458            Self {
459                ctx: egui::Context::default(),
460                width: 800.0,
461                height,
462                plain: false,
463                menu: None,
464            }
465        }
466
467        fn pass(&mut self, events: Vec<egui::Event>, mut fill: impl FnMut(&mut egui::Ui)) {
468            let screen =
469                egui::Rect::from_min_size(egui::Pos2::ZERO, Vec2::new(self.width, self.height));
470            let input = egui::RawInput {
471                screen_rect: Some(screen),
472                events,
473                ..Default::default()
474            };
475            let plain = self.plain;
476            let mut shown_rect = None;
477            let mut out = self.ctx.run_ui(input, |ui| {
478                let button = ui.button("Menu");
479                let popup = egui::Popup::menu(&button);
480                let shown = if plain {
481                    popup.show(|ui| fill(ui))
482                } else {
483                    menu(popup, |ui| fill(ui))
484                };
485                shown_rect = shown.map(|r| r.response.rect);
486            });
487            out.textures_delta.clear();
488            self.menu = shown_rect;
489        }
490
491        /// Click the button, and draw the menu until it has settled.
492        fn open(&mut self, mut fill: impl FnMut(&mut egui::Ui)) -> egui::Rect {
493            let pos = egui::Pos2::new(15.0, 15.0);
494            self.pass(Vec::new(), &mut fill);
495            self.pass(vec![egui::Event::PointerMoved(pos)], &mut fill);
496            self.pass(vec![press(pos, true)], &mut fill);
497            self.pass(vec![press(pos, false)], &mut fill);
498            // The scroll bar slides in, and the menu widens by it.
499            for _ in 0..30 {
500                self.pass(Vec::new(), &mut fill);
501            }
502            self.menu.expect("the menu is open")
503        }
504    }
505
506    /// A menu of everything menus here hold: `rows` of buttons, a checkbox,
507    /// a separator and a submenu.
508    fn items(ui: &mut egui::Ui, rows: usize) {
509        for row in 0..rows {
510            ui.button(format!("Item {row}")).clicked();
511        }
512        ui.separator();
513        ui.checkbox(&mut true, "A checkbox with a long label");
514        submenu(ui, "A submenu", |ui| {
515            ui.button("In the submenu").clicked();
516        });
517    }

523            button: egui::PointerButton::Primary,

539    /// Press and release the primary button at `pos`, the pointer there
540    /// first.
541    fn click_at(win: &mut MenuWindow, pos: egui::Pos2, fill: &mut impl FnMut(&mut egui::Ui)) {
542        win.pass(vec![egui::Event::PointerMoved(pos)], &mut *fill);

628    /// A menu too tall for its window: the window, where its first item is,

768    /// The height of each row of a menu of `labels`, in a window `width` wide
769    /// and `height` tall, and the menu.
770    fn rows(width: f32, height: f32, labels: &[String]) -> (Vec<f32>, egui::Rect) {
771        use std::cell::RefCell;
772        let heights = RefCell::new(Vec::new());
773        let mut fill = |ui: &mut egui::Ui| {
774            heights.borrow_mut().clear();
775            for label in labels {
776                heights.borrow_mut().push(ui.button(label).rect.height());
777            }
778        };
779        let mut win = MenuWindow::new(height);
780        win.width = width;
781        let shown = win.open(&mut fill);
782        (heights.into_inner(), shown)
783    }

824    /// draws it or not: its frame, and the furthest right of its rows.
825    fn right_edges(width: f32, height: f32, labels: &[&str], plain: bool) -> (f32, f32) {

837        let shown = win.open(&mut fill);
