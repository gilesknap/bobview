1//! Render a display to a PNG without a window: egui runs headless and its
2//! triangles are rasterised in software. Used for checking layouts in CI and
3//! on machines without a display.
4

14/// The monitor an automatically sized snapshot window sits on, in points
15/// at the desktop scale.
16const MONITOR: Vec2 = Vec2::new(1920.0, 1080.0);
17
18/// Longest a snapshot waits, after `wait`, for displays and images to load
19/// and clicks to settle; after that it renders what it has.
20const DEADLINE: Duration = Duration::from_secs(60);
21
22/// What to render and how.
23pub struct Options {
24    /// Size in pixels, or `None` to start at the default window size and
25    /// follow the app's requests to resize the window.
26    pub size: Option<[u32; 2]>,
27    /// The desktop scale (pixels per point) to simulate.
28    pub scale_factor: f32,
29    pub ui_scale: UiScale,
30    pub zoom: Option<f32>,
31    /// Time for PVs to connect before anything else happens.
32    pub wait: Duration,
33    /// Pixels to click and the button to use, in order, once `wait` has passed.
34    pub clicks: Vec<(Pos2, egui::PointerButton)>,
35    /// Time after the last click before the snapshot, for what it opened
36    /// (such as a trend window) to fill.
37    pub settle: Duration,
38    pub read_only: bool,
39    /// Where trends fetch history from (not remembered).
40    pub archiver: Option<archiveview::ArchiverClient>,
41}
42
43pub fn run(start: Location, opts: &Options, out: &Path) -> Result<(), String> {
44    let ctx = egui::Context::default();
45    let mut app = BobApp::new(&ctx, Some(start), opts.zoom, opts.ui_scale);
46    app.set_read_only(opts.read_only);
47    app.set_archiver(opts.archiver.clone(), false);
48    // A --middle-click copies to the clipboard only: a snapshot leaves the
49    // desktop's primary selection alone.
50    app.set_primary(Box::new(crate::selection::NoPrimary));
51    let mut textures: HashMap<TextureId, ColorImage> = HashMap::new();
52    let begin = Instant::now();
53    let native = opts.scale_factor;
54    let mut px = match opts.size {
55        Some([w, h]) => Vec2::new(w as f32, h as f32),
56        None => chrome::DEFAULT_WINDOW * native,
57    };
58    let monitor = opts.size.is_none().then_some(MONITOR * native);
59
60    // Each click is a press frame then a release frame.
61    let mut pending: std::collections::VecDeque<Vec<egui::Event>> = opts
62        .clicks
63        .iter()
64        .flat_map(|&(pos, which)| {
65            let button = |pressed| egui::Event::PointerButton {
66                pos,
67                button: which,
68                pressed,
69                modifiers: Default::default(),
70            };
71            // A few idle frames after each click let windows appear.
72            [
73                vec![egui::Event::PointerMoved(pos), button(true)],
74                vec![button(false)],
75                vec![],
76                vec![],
77                vec![],
78                vec![],
79            ]
80        })
81        .collect();
82    let mut settle_until = None;
83
84    // Images that have not loaded this long after `wait` are left out, so a
85    // file that never finishes reading cannot hold the snapshot up.
86    let images_loading =
87        || crate::symbol::loading() && begin.elapsed() < opts.wait + Duration::from_secs(20);
88
89    // Run frames until PVs have had time to connect and clicks have settled.
90    let output = loop {
91        let started = begin.elapsed() >= opts.wait && !app.is_loading() && !images_loading();
92        let events = if started {
93            pending.pop_front().unwrap_or_default()
94        } else {
95            Vec::new()
96        };
97        // Everything egui sees is in points: pixels / (scale factor * zoom).
98        let ppp = native * ctx.zoom_factor();
99        let screen = Rect::from_min_size(Pos2::ZERO, px / ppp);
100        let events = events.into_iter().map(|e| to_points(e, ppp)).collect();
101        let mut input = egui::RawInput {
102            screen_rect: Some(screen),
103            time: Some(begin.elapsed().as_secs_f64()),
104            events,
105            ..Default::default()
106        };
107        let info = input.viewports.entry(egui::ViewportId::ROOT).or_default();
108        info.native_pixels_per_point = Some(native);
109        info.inner_rect = Some(screen);
110        info.monitor_size = monitor.map(|m| m / ppp);
111        let mut output = ctx.run_ui(input, |ui| app.draw(ui));
112        if let (Some(m), Some(vp)) = (monitor, output.viewport_output.get(&egui::ViewportId::ROOT))
113        {
114            for cmd in &vp.commands {
115                if let egui::ViewportCommand::InnerSize(size) = cmd {
116                    px = (*size * output.pixels_per_point).round().min(m);
117                }
118            }
119        }
120        for (id, deltas) in &output.textures_delta.set {
121            for delta in deltas {
122                apply_texture(&mut textures, *id, delta);
123            }
124        }
125        output.textures_delta.clear();
126        if begin.elapsed() > opts.wait + DEADLINE {
127            log::warn!(
128                "snapshot taken after {DEADLINE:?} with {}{}{} clicks not made",
129                if app.is_loading() {
130                    "a display still loading, "
131                } else {
132                    ""
133                },
134                if crate::symbol::loading() {
135                    "images still loading, "
136                } else {
137                    ""
138                },
139                pending.len().div_ceil(6),
140            );
141            break output;
142        }
143        if started && pending.is_empty() {
144            let until = *settle_until.get_or_insert_with(|| {
145                begin.elapsed()
146                    + if opts.clicks.is_empty() {
147                        Duration::ZERO
148                    } else {
149                        opts.settle
150                    }
151            });
152            if begin.elapsed() >= until && !app.is_loading() && !images_loading() {
153                break output;
154            }
155        }
156        std::thread::sleep(Duration::from_millis(if started { 50 } else { 100 }));
157    };
158
159    let ppp = output.pixels_per_point;
160    let prims = ctx.tessellate(output.shapes, ppp);
161    let mut canvas = Canvas::new(px.x as usize, px.y as usize);
162    for prim in prims {
163        if let epaint::Primitive::Mesh(mesh) = &prim.primitive {
164            canvas.draw_mesh(mesh, prim.clip_rect, ppp, textures.get(&mesh.texture_id));
165        }
166    }
167    canvas.save(out)
168}

170/// Convert an event's pixel position into points.
171fn to_points(event: egui::Event, ppp: f32) -> egui::Event {
172    let p = |pos: Pos2| (pos.to_vec2() / ppp).to_pos2();
173    match event {
174        egui::Event::PointerMoved(pos) => egui::Event::PointerMoved(p(pos)),
175        egui::Event::PointerButton {
176            pos,
177            button,
178            pressed,
179            modifiers,
180        } => egui::Event::PointerButton {
181            pos: p(pos),
182            button,
183            pressed,
184            modifiers,
185        },
186        other => other,
187    }
188}

190fn apply_texture(
191    textures: &mut HashMap<TextureId, ColorImage>,
192    id: TextureId,
193    delta: &epaint::ImageDelta,
194) {
195    let egui::ImageData::Color(image) = &delta.image;
196    match delta.pos {
197        None => {
198            textures.insert(id, (**image).clone());
199        }
200        Some([x0, y0]) => {
201            if let Some(tex) = textures.get_mut(&id) {
202                let [w, h] = image.size;
203                for y in 0..h {
204                    for x in 0..w {
205                        let (tx, ty) = (x0 + x, y0 + y);
206                        if tx < tex.size[0] && ty < tex.size[1] {
207                            tex.pixels[ty * tex.size[0] + tx] = image.pixels[y * w + x];
208                        }
209                    }
210                }
211            }
212        }
213    }
214}

216/// Premultiplied RGBA float framebuffer.
217struct Canvas {
218    w: usize,
219    h: usize,
220    px: Vec<[f32; 4]>,
221}
222
223impl Canvas {
224    fn new(w: usize, h: usize) -> Self {
225        Self {
226            w,
227            h,
228            px: vec![[0.0, 0.0, 0.0, 1.0]; w * h],
229        }
230    }
231
232    /// Draw a mesh in points at `ppp` pixels per point.
233    fn draw_mesh(&mut self, mesh: &epaint::Mesh, clip: Rect, ppp: f32, tex: Option<&ColorImage>) {
234        let clip = Rect::from_min_max(clip.min * ppp, clip.max * ppp);
235        let clip = clip.intersect(Rect::from_min_size(
236            Pos2::ZERO,
237            Vec2::new(self.w as f32, self.h as f32),
238        ));
239        if clip.is_negative() {
240            return;
241        }
242        for tri in mesh.indices.as_chunks::<3>().0 {
243            let v = tri.map(|i| {
244                let mut v = mesh.vertices[i as usize];
245                v.pos = (v.pos.to_vec2() * ppp).to_pos2();
246                v
247            });
248            self.triangle([&v[0], &v[1], &v[2]], clip, tex);
249        }
250    }
251
252    fn triangle(&mut self, v: [&epaint::Vertex; 3], clip: Rect, tex: Option<&ColorImage>) {
253        let (p0, p1, p2) = (v[0].pos, v[1].pos, v[2].pos);
254        let area = (p1.x - p0.x) * (p2.y - p0.y) - (p2.x - p0.x) * (p1.y - p0.y);
255        if area.abs() < 1e-6 {
256            return;
257        }
258        let min_x = p0.x.min(p1.x).min(p2.x).max(clip.min.x).floor() as i64;
259        let max_x = p0.x.max(p1.x).max(p2.x).min(clip.max.x).ceil() as i64;
260        let min_y = p0.y.min(p1.y).min(p2.y).max(clip.min.y).floor() as i64;
261        let max_y = p0.y.max(p1.y).max(p2.y).min(clip.max.y).ceil() as i64;
262        let col = |c: Color32| {
263            [c.r() as f32, c.g() as f32, c.b() as f32, c.a() as f32].map(|x| x / 255.0)
264        };
265        let cs = [col(v[0].color), col(v[1].color), col(v[2].color)];
266        for y in min_y.max(0)..max_y.min(self.h as i64) {
267            for x in min_x.max(0)..max_x.min(self.w as i64) {
268                let p = Pos2::new(x as f32 + 0.5, y as f32 + 0.5);
269                let w0 = ((p1.x - p.x) * (p2.y - p.y) - (p2.x - p.x) * (p1.y - p.y)) / area;
270                let w1 = ((p2.x - p.x) * (p0.y - p.y) - (p0.x - p.x) * (p2.y - p.y)) / area;
271                let w2 = 1.0 - w0 - w1;
272                if w0 < -1e-4 || w1 < -1e-4 || w2 < -1e-4 {
273                    continue;
274                }
275                let mut c = [0.0f32; 4];
276                for (i, ch) in c.iter_mut().enumerate() {
277                    *ch = cs[0][i] * w0 + cs[1][i] * w1 + cs[2][i] * w2;
278                }
279                if let Some(t) = tex {
280                    let u = v[0].uv.x * w0 + v[1].uv.x * w1 + v[2].uv.x * w2;
281                    let vv = v[0].uv.y * w0 + v[1].uv.y * w1 + v[2].uv.y * w2;
282                    let s = sample(t, u, vv);
283                    for (i, ch) in c.iter_mut().enumerate() {
284                        *ch *= s[i];
285                    }
286                }
287                let d = &mut self.px[y as usize * self.w + x as usize];
288                for i in 0..4 {
289                    d[i] = c[i] + d[i] * (1.0 - c[3]);
290                }
291            }
292        }
293    }
294
295    fn save(&self, path: &Path) -> Result<(), String> {
296        let file = std::fs::File::create(path).map_err(|e| format!("{}: {e}", path.display()))?;
297        let mut enc =
298            png::Encoder::new(std::io::BufWriter::new(file), self.w as u32, self.h as u32);
299        enc.set_color(png::ColorType::Rgb);
300        enc.set_depth(png::BitDepth::Eight);
301        let mut writer = enc.write_header().map_err(|e| e.to_string())?;
302        let data: Vec<u8> = self
303            .px
304            .iter()
305            .flat_map(|p| [p[0], p[1], p[2]].map(|c| (c.clamp(0.0, 1.0) * 255.0).round() as u8))
306            .collect();
307        writer.write_image_data(&data).map_err(|e| e.to_string())
308    }
309}

311/// Bilinear sample of a premultiplied texture, as 0..1 floats.
312fn sample(t: &ColorImage, u: f32, v: f32) -> [f32; 4] {
313    let [w, h] = t.size;
314    let x = (u * w as f32 - 0.5).clamp(0.0, (w - 1) as f32);
315    let y = (v * h as f32 - 0.5).clamp(0.0, (h - 1) as f32);
316    let (x0, y0) = (x.floor() as usize, y.floor() as usize);
317    let (x1, y1) = ((x0 + 1).min(w - 1), (y0 + 1).min(h - 1));
318    let (fx, fy) = (x - x0 as f32, y - y0 as f32);
319    let px = |x: usize, y: usize| {
320        let c = t.pixels[y * w + x];
321        [c.r(), c.g(), c.b(), c.a()].map(|v| v as f32 / 255.0)
322    };
323    let (a, b, c, d) = (px(x0, y0), px(x1, y0), px(x0, y1), px(x1, y1));
324    let mut out = [0.0; 4];
325    for i in 0..4 {
326        out[i] =
327            (a[i] * (1.0 - fx) + b[i] * fx) * (1.0 - fy) + (c[i] * (1.0 - fx) + d[i] * fx) * fy;
328    }
329    out
330}

336    #[test]
337    fn renders_every_test_display() {
338        let dir = std::env::temp_dir().join(format!("bobview-snap-{}", std::process::id()));
339        std::fs::create_dir_all(&dir).unwrap();
340        for f in crate::screen::display_set::files() {
341            let out = dir.join(f.file_name().unwrap()).with_extension("png");
342            let start = Location {
343                source: f.clone().into(),
344                macros: bob::Macros::default(),
345            };
346            let opts = Options {
347                size: Some([320, 240]),
348                scale_factor: 1.0,
349                ui_scale: UiScale::Auto,
350                zoom: Some(0.5),
351                wait: Duration::ZERO,
352                clicks: Vec::new(),
353                settle: Duration::ZERO,
354                read_only: false,
355                archiver: None,
356            };
357            run(start, &opts, &out).unwrap();
358            assert!(
359                std::fs::metadata(&out).unwrap().len() > 1000,
360                "{} rendered nothing",
361                f.display()
362            );
363        }
364        std::fs::remove_dir_all(&dir).ok();
365    }

367    fn png_size(path: &Path) -> (u32, u32) {
368        let decoder =
369            png::Decoder::new(std::io::BufReader::new(std::fs::File::open(path).unwrap()));
370        let info = decoder.read_info().unwrap();
371        (info.info().width, info.info().height)
372    }

374    #[test]
375    fn window_follows_a_small_display() {
376        let dir = std::env::temp_dir().join(format!("bobview-auto-{}", std::process::id()));
377        std::fs::create_dir_all(&dir).unwrap();
378        let bob = dir.join("small.bob");
379        std::fs::write(
380            &bob,
381            r#"<display version="2.0.0"><width>400</width><height>100</height></display>"#,
382        )
383        .unwrap();
384        for (scale_factor, size) in [(1.0, None), (2.0, None), (2.0, Some([640, 480]))] {
385            let out = dir.join(format!("small-{scale_factor}.png"));
386            let opts = Options {
387                size,
388                scale_factor,
389                ui_scale: UiScale::Auto,
390                zoom: None,
391                wait: Duration::ZERO,
392                clicks: Vec::new(),
393                settle: Duration::ZERO,
394                read_only: false,
395                archiver: None,
396            };
397            let start = Location {
398                source: bob::Source::File(bob.clone()),
399                macros: bob::Macros::default(),
400            };
401            run(start, &opts, &out).unwrap();
402            let (w, h) = png_size(&out);
403            if let Some([sw, sh]) = size {
404                assert_eq!((w, h), (sw, sh));
405            } else {
406                // The display at 100%, a pixel spare, and the controls below 80 points.
407                let s = scale_factor;
408                assert_eq!(w as f32, (401.0 * s).round(), "{s}");
409                assert!(h as f32 > 140.0 * s && (h as f32) < 180.0 * s, "{s}: {h}");
410            }
411        }
412        std::fs::remove_dir_all(&dir).ok();
413    }
