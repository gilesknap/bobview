1//! Trend windows: PVs plotted against time, archived history and live
2//! values, by archiveview's [`Plot`]. Opened from a widget's context menu,
3//! a BOY "Graph" button, or a Data Browser or StripTool launcher. The
4//! plot fetches history from the Archiver Appliance bobview was given
5//! (`--archiver`), or is live only without one.
6//!
7//! A middle-click on the plot adds the PVs the primary selection names,
8//! and one in its search box pastes the selection there: the plot reports
9//! the click, `BobApp` reads the selection (on a thread, see
10//! [`crate::selection::Reader`]) and hands the text back to
11//! [`Trend::selected`].
12
13use archiveview::plot::MiddleClick;
14use archiveview::{ArchiverClient, Plot, View};
15use eframe::egui;

17use web_time::{Duration, Instant};

21/// A trend window's plot, and where its window is.
22pub struct Trend {
23    /// Tells this window from the others.
24    pub id: u64,
25    plot: Plot,
26    /// Where to open the window, when a layout says.
27    pub place: Geometry,
28    /// Where the window is, for saving a layout.
29    pub geometry: Geometry,
30    /// The middle-click waiting for the primary selection: its number,
31    /// and when it was (not egui time, which each native window counts
32    /// from its own start).
33    reading: Option<(u64, Instant)>,
34    /// What a paste did, floating over the bottom of the plot for a while.
35    notice: Option<Notice>,
36    /// Text selected in the search box, and when, until it has been
37    /// steady for [`PRIMARY_DEBOUNCE`] to become the primary selection.
38    search_selection: Option<(String, Instant)>,
39}
40
41/// Text selected in the search box becomes the primary selection once it
42/// has stayed the same this long, as in archiveview: not at every
43/// shift+arrow, which over `ssh -X` costs a round trip each.
44const PRIMARY_DEBOUNCE: Duration = Duration::from_millis(150);
45
46/// How long a middle-click waits for the primary selection, as in
47/// archiveview: long enough for a slow owner (or `ssh -X`), short enough
48/// that a hung one (arboard waits up to 4 s per format) gets a "Could not
49/// read the selection" while the click is still fresh, rather than PVs
50/// turning up long after.
51pub const SELECTION_WAIT: Duration = Duration::from_secs(2);
52
53/// A short message over the plot.
54struct Notice {
55    text: String,
56    error: bool,
57    /// When it goes (egui time), from when it is first drawn.
58    until: Option<f64>,
59    secs: f64,
60}

62impl Trend {
63    /// A trend of the PVs `names`, each once, subscribed through `pvs`
64    /// (tagged to repaint its window) and with history from `archiver`.
65    pub fn new(
66        id: u64,
67        names: &[String],
68        pvs: &PvManager,
69        archiver: Option<ArchiverClient>,
70        recent: &[String],
71    ) -> Self {
72        let mut plot = Plot::with_id_salt(pvs, archiver, ("bobview-trend", id));
73        // The pill switches archivers, for every trend: see
74        // `BobApp::set_archiver`. Not in a browser, which can't reach one.
75        plot.set_archiver_switching(!cfg!(target_arch = "wasm32"));
76        plot.set_recent_archivers(recent);
77        plot.add_pvs(names);
78        Self {
79            id,
80            plot,
81            place: Geometry::default(),
82            geometry: Geometry::default(),
83            reading: None,
84            notice: None,
85            search_selection: None,
86        }
87    }
88
89    /// Whether a middle-click is waiting for the primary selection.
90    #[cfg(test)]
91    pub fn is_reading(&self) -> bool {
92        self.reading.is_some()
93    }

95    /// The notice's text, and whether it is an error.
96    #[cfg(test)]

98        self.notice.as_ref().map(|n| (n.text.as_str(), n.error))
99    }

134    /// Fetch history from `archiver` (or none), unless it does already.
135    pub fn set_archiver(&mut self, archiver: Option<&ArchiverClient>, recent: &[String]) {
136        if self.plot.archiver() != archiver {
137            self.plot.set_archiver(archiver.cloned());
138        }
139        self.plot.set_recent_archivers(recent);
140    }

142    /// Whether the user switched archivers in its pill since last asked:

146    }

148    /// Whether its plot is still fetching history, or waiting to draw

153    }

155    /// The window title: its PVs, or the first and how many more.
156    pub fn title(&self) -> String {
157        match self.plot.title() {
158            t if t.is_empty() => "Trend".into(),
159            t => format!("Trend: {t}"),
160        }
161    }

163    /// Where a middle-click on it since last asked landed, if there was
164    /// one: the caller reads the primary selection for it.
165    pub fn take_middle_click(&mut self) -> Option<MiddleClick> {
166        self.plot.take_middle_click()
167    }
168
169    /// Text selected in its search box that has been steady since before
170    /// `now` less [`PRIMARY_DEBOUNCE`], once, to make the primary
171    /// selection; and when to ask again, while some is waiting.
172    pub fn take_search_selection(&mut self, now: Instant) -> (Option<String>, Option<Duration>) {
173        if let Some(text) = self.plot.take_search_selection() {
174            self.search_selection = Some((text, now));
175        }
176        match &self.search_selection {
177            Some((_, since)) if now.duration_since(*since) >= PRIMARY_DEBOUNCE => {
178                (self.search_selection.take().map(|(t, _)| t), None)
179            }
180            Some((_, since)) => (None, Some(PRIMARY_DEBOUNCE - now.duration_since(*since))),
181            None => (None, None),
182        }
183    }
184
185    /// The middle-click numbered `n`, at `now`, is waiting for the
186    /// selection (an earlier one's answer is then ignored).
187    pub fn reading(&mut self, n: u64, now: Instant) {
188        self.reading = Some((n, now));
189    }
190
191    /// The primary selection's `text` for the middle-click numbered `n` on
192    /// `click`: add the PVs it names, or paste it into the search box. An
193    /// answer for an older click, or one that came too late, is dropped.
194    pub fn selected(&mut self, n: u64, click: MiddleClick, text: Option<String>) {
195        if self.reading.is_none_or(|(r, _)| r != n) {
196            return;
197        }
198        self.reading = None;
199        let text = text.unwrap_or_default();
200        match click {
201            // Says what it added in `ui`, from `take_pasted`.
202            MiddleClick::Plot => {
203                self.plot.paste_pvs(&text);
204            }
205            MiddleClick::Search => self.plot.paste_into_search(&text),
206        }
207    }
208
209    /// Give up on a middle-click that has waited [`SELECTION_WAIT`] by
210    /// `now` for the selection (its owner may have hung), and say so;
211    /// whether it did.
212    pub fn time_out_reading(&mut self, now: Instant) -> bool {
213        let late = self
214            .reading
215            .is_some_and(|(_, since)| now.duration_since(since) >= SELECTION_WAIT);
216        if late {
217            self.reading = None;
218            self.notify("Could not read the selection", true, 4.0);
219        }
220        late
221    }
222
223    fn notify(&mut self, text: impl Into<String>, error: bool, secs: f64) {
224        self.notice = Some(Notice {
225            text: text.into(),
226            error,
227            until: None,
228            secs,
229        });
230    }
231
232    /// Draw it in the space `ui` offers, and no more: in a window inside
233    /// the main one (an auto-sized `egui::Window`) the plot's legend
234    /// would otherwise push the window taller every frame.
235    pub fn ui(&mut self, ui: &mut egui::Ui) {
236        let rect = ui.available_rect_before_wrap();
237        let mut inner = ui.new_child(egui::UiBuilder::new().max_rect(rect));
238        self.plot.ui(&mut inner);
239        // A paste: a middle-click's, or Ctrl+V's.
240        if let Some(pasted) = self.plot.take_pasted() {
241            let error = pasted.added.is_empty() && pasted.present.is_empty();
242            self.notify(pasted.message(), error, 3.0);
243        }
244        self.toast(ui, rect);
245        ui.advance_cursor_after_rect(rect);
246    }
247
248    /// The notice, over the bottom middle of `rect`, until it expires.
249    fn toast(&mut self, ui: &mut egui::Ui, rect: egui::Rect) {
250        let Some(notice) = &mut self.notice else {
251            return;
252        };
253        let now = ui.input(|i| i.time);
254        let until = *notice.until.get_or_insert(now + notice.secs);
255        if now > until {
256            self.notice = None;
257            return;
258        }
259        ui.ctx()
260            .request_repaint_after(Duration::from_secs_f64(until - now));
261        let color = if notice.error {
262            ui.visuals().error_fg_color
263        } else {
264            ui.visuals().text_color()
265        };
266        let text = egui::RichText::new(&notice.text).color(color);
267        // In the trend's own layer, over the plot drawn before it: a trend
268        // window behind another (where windows are embedded) keeps its
269        // notice behind too.
270        let layout = egui::Layout::bottom_up(egui::Align::Center);
271        let mut over = ui.new_child(egui::UiBuilder::new().max_rect(rect).layout(layout));
272        over.add_space(56.0);
273        egui::Frame::popup(over.style())
274            .corner_radius(16.0)
275            .inner_margin(egui::Margin::symmetric(14, 8))
276            .show(&mut over, |ui| ui.label(text));
277    }

310    /// The longest line the trend draws, in points, and how wide it is,
311    /// once it has stopped loading: drawn headless, as a snapshot is.
312    fn longest_line(archiver: Option<ArchiverClient>) -> (usize, f32) {
313        use std::time::{Duration, Instant};
314        let pvs = PvManager::new();
315        // Archived by the mock, and served by no IOC: what it draws is
316        // history, not a live tail.
317        let mut t = Trend::new(0, &names(&["MOCK:SINE"]), &pvs, archiver, &[]);
318        let ctx = egui::Context::default();
319        let screen = egui::Rect::from_min_size(egui::Pos2::ZERO, egui::vec2(800.0, 500.0));
320        let begin = Instant::now();
321        let mut frames = 0;
322        let shapes = loop {
323            let input = egui::RawInput {
324                screen_rect: Some(screen),
325                time: Some(begin.elapsed().as_secs_f64()),
326                ..Default::default()
327            };
328            let mut out = ctx.run_ui(input, |ui| t.ui(ui));
329            out.textures_delta.clear();
330            frames += 1;
331            // A few frames after it has loaded, for the data to be drawn.
332            if frames > 3 && !t.is_loading() {
333                break out.shapes;
334            }
335            assert!(
336                begin.elapsed() < Duration::from_secs(30),
337                "the trend was still loading after 30s"
338            );
339            std::thread::sleep(Duration::from_millis(20));
340        };
341        fn walk(shape: &egui::Shape, best: &mut (usize, f32)) {
342            match shape {
343                egui::Shape::Vec(v) => v.iter().for_each(|s| walk(s, best)),
344                egui::Shape::Path(p) if p.points.len() > best.0 => {
345                    let (lo, hi) = p.points.iter().fold((f32::MAX, f32::MIN), |(lo, hi), q| {
346                        (lo.min(q.x), hi.max(q.x))
347                    });
348                    *best = (p.points.len(), hi - lo);
349                }
350                _ => {}
351            }
352        }
353        let mut best = (0, 0.0);
354        shapes.iter().for_each(|s| walk(&s.shape, &mut best));
355        best
356    }

359    fn shows_history_from_the_archiver() {
360        let mock = mockarchiver::MockArchiver::start_local().unwrap();
361        let archiver = ArchiverClient::new(&mock.url()).unwrap();
362        // MOCK:SINE has a 10 minute period, sampled every second: the

374        assert!(points < 50, "{points} points with no archiver");

377    #[test]
378    fn the_selection_answers_only_the_latest_middle_click_in_time() {
379        let pvs = PvManager::new();
380        let mut t = Trend::new(0, &names(&["A"]), &pvs, None, &[]);
381        let start = Instant::now();
382        t.reading(1, start);
383        t.reading(2, start);
384        // An earlier click's answer is dropped.
385        t.selected(1, MiddleClick::Plot, Some("OLD:PV".into()));
386        assert_eq!(t.names(), ["A"]);
387        assert!(t.is_reading());
388        t.selected(2, MiddleClick::Plot, Some("B:C".into()));
389        assert_eq!(t.names(), ["A", "B:C"]);
390        assert!(!t.is_reading());
391        // Too late: given up on, with a notice, and the answer dropped.
392        t.reading(3, start);
393        assert!(!t.time_out_reading(start + Duration::from_millis(500)));
394        assert!(t.time_out_reading(start + SELECTION_WAIT));
395        let (text, error) = t.notice().unwrap();
396        assert!(error && text.contains("Could not read"), "{text}");
397        t.selected(3, MiddleClick::Plot, Some("LATE:PV".into()));
398        assert_eq!(t.names(), ["A", "B:C"]);
399        // Into the search box: pasted there, not added.
400        t.reading(4, start);
401        t.selected(4, MiddleClick::Search, Some("D:E".into()));
402        assert_eq!(t.names(), ["A", "B:C"]);
403    }
