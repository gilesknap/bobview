1//! Trend windows: PVs plotted against time, archived history and live
2//! values, by archiveview's [`Plot`]. Opened from a widget's context menu,
3//! a BOY "Graph" button, or a Data Browser or StripTool launcher. The
4//! plot fetches history from the Archiver Appliance bobview was given
5//! (`--archiver`), or is live only without one.
6
7use archiveview::{ArchiverClient, Plot, View};
8use eframe::egui;
9use pv::PvManager;
10
11use crate::layout::Geometry;
12
13/// A trend window's plot, and where its window is.
14pub struct Trend {
15    /// Tells this window from the others.
16    pub id: u64,
17    plot: Plot,
18    /// Where to open the window, when a layout says.
19    pub place: Geometry,
20    /// Where the window is, for saving a layout.
21    pub geometry: Geometry,
22}
23
24impl Trend {
25    /// A trend of the PVs `names`, each once, subscribed through `pvs`
26    /// (tagged to repaint its window) and with history from `archiver`.
27    pub fn new(
28        id: u64,
29        names: &[String],
30        pvs: &PvManager,
31        archiver: Option<ArchiverClient>,
32        recent: &[String],
33    ) -> Self {
34        let mut plot = Plot::with_id_salt(pvs, archiver, ("bobview-trend", id));
35        // The pill switches archivers, for every trend: see
36        // `BobApp::set_archiver`. Not in a browser, which can't reach one.
37        plot.set_archiver_switching(!cfg!(target_arch = "wasm32"));
38        plot.set_recent_archivers(recent);
39        plot.add_pvs(names);
40        Self {
41            id,
42            plot,
43            place: Geometry::default(),
44            geometry: Geometry::default(),
45        }
46    }
47
48    /// The PVs' names, as added.
49    #[cfg(test)]
50    pub fn names(&self) -> Vec<String> {
51        self.plot.names()
52    }
53
54    /// What it shows, for a layout: its PVs, time range and value axes,
55    /// but not the archiver, which is bobview's.
56    pub fn view(&self) -> View {
57        View {
58            archiver: None,
59            ..self.plot.view()
60        }
61    }
62
63    /// Show `view` (from a layout), keeping bobview's archiver.
64    pub fn set_view(&mut self, view: &View) {
65        self.plot.set_view(&View {
66            archiver: None,
67            ..view.clone()
68        });
69    }
70
71    /// Add the PVs `names` that it doesn't have yet.
72    pub fn extend(&mut self, names: &[String]) {
73        self.plot.add_pvs(names);
74    }
75
76    /// The archiver it fetches history from.
77    pub fn archiver(&self) -> Option<&ArchiverClient> {
78        self.plot.archiver()
79    }
80
81    /// Fetch history from `archiver` (or none), unless it does already.
82    pub fn set_archiver(&mut self, archiver: Option<&ArchiverClient>, recent: &[String]) {
83        if self.plot.archiver() != archiver {
84            self.plot.set_archiver(archiver.cloned());
85        }
86        self.plot.set_recent_archivers(recent);
87    }
88
89    /// Whether the user switched archivers in its pill since last asked:
90    /// [`Trend::archiver`] is the new one.
91    pub fn take_archiver_change(&mut self) -> bool {
92        self.plot.take_archiver_change()
93    }
94
95    /// Whether its plot is still fetching history, or waiting to draw
96    /// a PV whose data has come (snapshots wait for this).
97    #[cfg(not(target_arch = "wasm32"))]
98    pub fn is_loading(&self) -> bool {
99        self.plot.is_loading()
100    }
101
102    /// The window title: its PVs, or the first and how many more.
103    pub fn title(&self) -> String {
104        match self.plot.title() {
105            t if t.is_empty() => "Trend".into(),
106            t => format!("Trend: {t}"),
107        }
108    }
109
110    /// Draw it in the space `ui` offers, and no more: in a window inside
111    /// the main one (an auto-sized `egui::Window`) the plot's legend
112    /// would otherwise push the window taller every frame.
113    pub fn ui(&mut self, ui: &mut egui::Ui) {
114        let rect = ui.available_rect_before_wrap();
115        let mut inner = ui.new_child(egui::UiBuilder::new().max_rect(rect));
116        self.plot.ui(&mut inner);
117        ui.advance_cursor_after_rect(rect);
118    }
119}

121#[cfg(test)]
122mod tests {
123    use super::*;
124
125    fn names(n: &[&str]) -> Vec<String> {
126        n.iter().map(|s| s.to_string()).collect()
127    }
128
129    #[test]
130    fn titles_name_the_pvs() {
131        let pvs = PvManager::new();
132        let t = Trend::new(0, &names(&["A:B", "ca://A:B", " "]), &pvs, None, &[]);
133        assert_eq!(t.names().len(), 1, "each PV once");
134        assert_eq!(t.title(), "Trend: A:B");
135        let t = Trend::new(1, &names(&["A", "B", "C"]), &pvs, None, &[]);
136        assert_eq!(t.title(), "Trend: A and 2 more");
137        assert_eq!(Trend::new(2, &[], &pvs, None, &[]).title(), "Trend");
138    }
139
140    #[test]
141    fn extending_adds_new_pvs_after_the_others() {
142        let pvs = PvManager::new();
143        let mut t = Trend::new(0, &names(&["A"]), &pvs, None, &[]);
144        t.extend(&names(&["ca://A", "B", "B", " ", "C"]));
145        // Each PV once, in the order added, so the new ones take the next
146        // colours as in a new trend of all three.
147        assert_eq!(t.names(), ["A", "B", "C"]);
148        assert_eq!(t.title(), "Trend: A and 2 more");
149    }
150
151    #[test]
152    fn views_leave_the_archiver_to_bobview() {
153        let pvs = PvManager::new();
154        let ours = ArchiverClient::new("http://ours:17668").unwrap();
155        let mut t = Trend::new(0, &names(&["A"]), &pvs, Some(ours.clone()), &[]);
156        assert_eq!(t.view().archiver, None, "not saved in layouts");
157        t.set_view(&View {
158            archiver: Some("http://theirs:17668/retrieval".into()),
159            ..t.view()
160        });
161        assert_eq!(t.archiver(), Some(&ours));
162        // Set again, it keeps the one it has (no refetch).
163        t.set_archiver(Some(&ours), &[]);
164        assert_eq!(t.archiver(), Some(&ours));
165        t.set_archiver(None, &[]);
166        assert_eq!(t.archiver(), None);
167    }
168}
