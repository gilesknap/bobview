88struct Cli {
89    /// display.bob, display.opi or http(s)://host/display.bob
90    #[arg(value_name = "DISPLAY")]
91    display: Option<String>,
92    /// Macros for the display
93    #[arg(short, long, value_name = "P=X,R=Y")]
94    macros: Option<String>,
95    /// Open a display file through the linked folders of the Eclipse project
96    /// in DIR (its .project, or for a BOY .opi screen the links of a DLS
97    /// runcss.sh in DIR), so that links between the project's modules work
98    #[arg(short, long, value_name = "DIR")]
99    project: Option<PathBuf>,
100    /// Fixed zoom (0.2 to 5) instead of fit-to-window
101    #[arg(short, long, value_name = "FACTOR", value_parser = parse_zoom)]
102    zoom: Option<f32>,
103    /// Start full screen
104    #[arg(short, long)]
105    fullscreen: bool,
106    /// No longer supported: text entries take the keyboard
107    #[arg(short, long, hide = true)]
108    keypad: bool,
109    /// No longer needed: pin mode replaced it
110    #[arg(long, hide = true)]
111    multi_window: bool,
112    /// Refuse every write to a PV (except loc:// ones, which stay in
113    /// bobview): controls are disabled and a write action says why it did
114    /// nothing
115    #[arg(long)]
116    read_only: bool,
117    /// Restore the saved window layout NAME
118    #[arg(long, value_name = "NAME")]
119    layout: Option<String>,
120    /// Don't restore the default layout
121    #[arg(long)]
122    no_layout: bool,
123    /// Size of the toolbar and menus relative to the desktop's scale
124    /// (default auto: smaller on a small high-DPI screen; also set by
125    /// BOBVIEW_UI_SCALE)
126    #[arg(long, value_name = "F|auto", value_parser = parse_ui_scale)]
127    ui_scale: Option<chrome::UiScale>,
128    /// Render to a PNG without a window and exit
129    #[arg(long, value_name = "OUT.png")]
130    snapshot: Option<PathBuf>,
131    /// Snapshot size in pixels; 'auto' starts at the window's default size
132    /// and resizes it as a window manager would, on a monitor of 1920x1080
133    /// points
134    #[arg(long, value_name = "WxH", default_value = "1280x800", value_parser = parse_size)]
135    size: Size,
136    /// Snapshot as on a screen with this desktop scale, e.g. 2.75 for a
137    /// phone; --click positions are in pixels
138    #[arg(long, value_name = "F", default_value_t = 1.0, value_parser = parse_scale_factor)]
139    scale_factor: f32,
140    /// Time for PVs to connect before a snapshot
141    #[arg(long, value_name = "SECONDS", default_value_t = 2.0, value_parser = parse_seconds)]
142    wait: f64,
143    /// Click here before the snapshot (repeatable)
144    #[arg(long, value_name = "X,Y", value_parser = parse_point)]

149    /// Time after the last click before the snapshot
150    #[arg(long, value_name = "SECONDS", default_value_t = 1.0, value_parser = parse_seconds)]

160impl Cli {
161    /// Parse the command line, exiting with a message if it is wrong.
162    fn from_env() -> (Self, Vec<(egui::Pos2, egui::PointerButton)>) {
163        use clap::{CommandFactory, FromArgMatches};
164        let matches = Self::command().get_matches();
165        let cli = Self::from_arg_matches(&matches).unwrap_or_else(|e| e.exit());
166        let clicks = cli.clicks(&matches);
167        (cli, clicks)
168    }
169
170    /// Both kinds of click, in the order given.
171    fn clicks(&self, matches: &clap::ArgMatches) -> Vec<(egui::Pos2, egui::PointerButton)> {
172        let at = |id| matches.indices_of(id).into_iter().flatten();
173        let mut clicks: Vec<_> = (at("click").zip(&self.click))
174            .map(|(i, p)| (i, *p, egui::PointerButton::Primary))
175            .chain(
176                at("right_click")
177                    .zip(&self.right_click)
178                    .map(|(i, p)| (i, *p, egui::PointerButton::Secondary)),
179            )
180            .collect();
181        clicks.sort_by_key(|(i, ..)| *i);
182        clicks.into_iter().map(|(_, p, b)| (p, b)).collect()
183    }
184}

186/// A zoom factor in the range the view allows.
187fn parse_zoom(text: &str) -> Result<f32, String> {
188    text.trim()
189        .parse::<f32>()
190        .ok()
191        .filter(|z| (0.2..=5.0).contains(z))
192        .ok_or_else(|| format!("expected a factor from 0.2 to 5, not {text:?}"))
193}

196fn parse_ui_scale(text: &str) -> Result<chrome::UiScale, String> {
197    chrome::UiScale::parse(text).ok_or_else(|| format!("expected a number or auto, not {text:?}"))
198}

200#[cfg(not(target_arch = "wasm32"))]
201fn parse_size(text: &str) -> Result<Size, String> {
202    if text.trim() == "auto" {
203        return Ok(Size(None));
204    }
205    text.split_once('x')
206        .and_then(|(w, h)| Some([w.trim().parse().ok()?, h.trim().parse().ok()?]))
207        .filter(|[w, h]: &[u32; 2]| *w > 0 && *h > 0)
208        .map(|wh| Size(Some(wh)))
209        .ok_or_else(|| format!("expected WIDTHxHEIGHT in pixels or auto, not {text:?}"))
210}
211
212#[cfg(not(target_arch = "wasm32"))]
213fn parse_scale_factor(text: &str) -> Result<f32, String> {
214    text.trim()
215        .parse::<f32>()
216        .ok()
217        .filter(|s| s.is_finite() && *s > 0.0)
218        .ok_or_else(|| format!("expected a scale above 0, not {text:?}"))
219}
220
221/// A time in seconds: a number, not negative (nor too big for a Duration).
222#[cfg(not(target_arch = "wasm32"))]
223fn parse_seconds(text: &str) -> Result<f64, String> {
224    text.trim()
225        .parse::<f64>()
226        .ok()
227        .filter(|s| (0.0..1e9).contains(s))
228        .ok_or_else(|| format!("expected a number of seconds, not {text:?}"))
229}
230
231#[cfg(not(target_arch = "wasm32"))]
232fn parse_point(text: &str) -> Result<egui::Pos2, String> {
233    text.split_once(',')
234        .and_then(|(x, y)| Some(egui::pos2(x.trim().parse().ok()?, y.trim().parse().ok()?)))
235        .filter(|p| p.x.is_finite() && p.y.is_finite())
236        .ok_or_else(|| format!("expected X,Y in pixels, not {text:?}"))
237}

277#[cfg(not(target_arch = "wasm32"))]
278fn run(cli: Cli, clicks: Vec<(egui::Pos2, egui::PointerButton)>) -> Result<(), Failure> {
279    if cli.keypad {
280        eprintln!("--keypad is no longer supported: ignored");
281    }
282    if cli.multi_window {
283        // Pin mode replaced multi-window mode: shift+click follows a link
284        // as it did.
285        eprintln!(
286            "--multi-window is no longer needed: links from the pinned main \
287             window open in another window, shift+click in a new one"
288        );
289    }
290    let mut file = cli.display.as_deref().map(fetch::user_named);
291    let mut macros = parse_macros(cli.macros.as_deref().unwrap_or_default());
292    let ui_scale = cli.ui_scale.unwrap_or_else(chrome::UiScale::from_env);
293    let is_opi = |p: &std::path::Path| p.extension().is_some_and(|x| x.eq_ignore_ascii_case("opi"));
294    if let (Some(dir), Some(Source::File(path))) = (&cli.project, &file) {
295        let Some(project) = bob::project::project_dir(dir) else {
296            return Err(Failure::usage(format!(
297                "{} has no .project, nor a {} with links",
298                dir.display(),
299                bob::project::LAUNCHER
300            )));
301        };
302        // A launcher's links are CS-Studio's, for BOY screens only.
303        if bob::project::through_launcher(&project) && !is_opi(path) {
304            return Err(Failure::usage(format!(
305                "{}: {} links apply only to BOY .opi screens",
306                path.display(),
307                bob::project::LAUNCHER
308            )));
309        }
310        match bob::project::virtual_path(&project, path) {
311            Some(p) => file = Some(Source::File(p)),
312            None => {
313                return Err(Failure::usage(format!(
314                    "{} is not in a linked folder of {}",
315                    path.display(),
316                    dir.display()
317                )));
318            }
319        }
320    } else if let Some(Source::File(path)) = &file
321        && is_opi(path)
322        && let Some(p) = bob::project::launcher_path(path)
323    {
324        // A DLS BOY screen beside its runcss.sh: through the project that
325        // launcher would give CS-Studio.
326        log::info!("{}: opening as {}", path.display(), p.display());
327        file = Some(Source::File(p));
328    }
329    // An .opi opened directly (in an Eclipse project) gets the macros its
330    // DLS launcher script would give it; those given here override them.
331    if let Some(Source::File(path)) = &file
332        && is_opi(path)
333        && let Some(defaults) = launch::defaults(path)
334    {
335        macros = defaults.with(&macros);
336    }
337    // The named layout, else the default one unless a display was named.
338    let layout = if cli.snapshot.is_some() {
339        None
340    } else {
341        let default = file.is_none() && !cli.no_layout && layout::exists(layout::DEFAULT);
342        layout::startup(cli.layout, default, layout::load).map_err(Failure::usage)?
343    };
344    // A layout's main window has a display of its own.
345    let layout_has_main = layout
346        .as_ref()
347        .is_some_and(|(l, _)| l.windows.iter().any(|w| w.main));
348    let start = file
349        .or_else(|| {
350            (!layout_has_main)
351                .then(default_display)
352                .flatten()
353                .map(Source::File)
354        })
355        .map(|source| app::Location { source, macros });
356
357    if let Some(out) = cli.snapshot {
358        let start = start.ok_or_else(|| Failure::usage("--snapshot needs a display file"))?;
359        let opts = snapshot::Options {
360            size: cli.size.0,
361            scale_factor: cli.scale_factor,
362            ui_scale,
363            zoom: cli.zoom,
364            wait: std::time::Duration::from_secs_f64(cli.wait),
365            clicks,
366            settle: std::time::Duration::from_secs_f64(cli.settle),
367            read_only: cli.read_only,
368        };
369        return snapshot::run(start, &opts, &out)
370            .map_err(|e| Failure::failed(format!("snapshot failed: {e}")));
371    }
372
373    let options = eframe::NativeOptions {
374        viewport: egui::ViewportBuilder::default()
375            .with_title("bobview")
376            .with_app_id("bobview")
377            .with_inner_size(chrome::DEFAULT_WINDOW)
378            .with_fullscreen(cli.fullscreen),
379        // No vsync: each window's buffer swap would wait for a screen
380        // refresh, and eframe draws the windows one after another on one
381        // thread, so windows repainting together would take a refresh each
382        // (~120 ms for 7, when they were all drawn in the main one's frame).
383        // We only repaint on change, so this doesn't spin.
384        glow_options: eframe::egui_glow::GlowConfiguration {
385            vsync: false,
386            ..Default::default()
387        },
388        ..Default::default()
389    };
390    let (zoom, read_only) = (cli.zoom, cli.read_only);
391    eframe::run_native(
392        "bobview",
393        options,
394        Box::new(move |cc| {
395            let mut app = app::BobApp::new(&cc.egui_ctx, start, zoom, ui_scale);
396            app.set_read_only(read_only);
397            if let Some((layout, name)) = layout {
398                app.restore_layout(layout, Some(name));
399            }
400            Ok(Box::new(app::App::new(app)))
401        }),
402    )
403    .map_err(|e| Failure::failed(e.to_string()))
404}

406/// Parse `P=X,R=Y` into macros; `\,` and `\\` in a value stand for `,`
407/// and `\`.
408fn parse_macros(text: &str) -> Macros {
409    // Split on unescaped commas, keeping escapes for now.
410    let mut pairs = vec![String::new()];
411    let mut chars = text.chars();
412    while let Some(c) = chars.next() {
413        match c {
414            ',' => pairs.push(String::new()),
415            '\\' => {
416                let last = pairs.last_mut().expect("a pair");
417                last.push(c);
418                last.extend(chars.next());
419            }
420            _ => pairs.last_mut().expect("a pair").push(c),
421        }
422    }
423    Macros::from_pairs(pairs.iter().filter_map(|kv| {
424        let (k, v) = kv.split_once('=')?;
425        Some((k.trim().to_string(), unescape_macro(v.trim())))
426    }))
427}
428
429fn unescape_macro(text: &str) -> String {
430    let mut out = String::with_capacity(text.len());
431    let mut chars = text.chars();
432    while let Some(c) = chars.next() {
433        match (c, chars.clone().next()) {
434            ('\\', Some(next @ (',' | '\\'))) => {
435                out.push(next);
436                chars.next();
437            }
438            _ => out.push(c),
439        }
440    }
441    out
442}

444/// The browser build: the page's query string names the display, its
445/// macros and the PV Web Socket, e.g.
446/// `?display=https://host/opi/index.bob&macros=P=X&pvws=pvws.host:443`
447/// (and `zoom=1.5`, `token=SECRET` for a server with `--write`).
448/// A relative display is found next to the page; the default is `index.bob`.
449/// With a token, displays come only from the page's server, and writes
450/// carry the token only to a PVWS there.
451#[cfg(target_arch = "wasm32")]
452fn main() {
453    use wasm_bindgen::JsCast;
454
455    init_logging();
456    let window = web_sys::window().expect("a browser window");
457    let location = window.location();
458    let query = web_sys::UrlSearchParams::new_with_str(&location.search().unwrap_or_default())
459        .expect("a query string");
460    let page = Source::Url(location.href().unwrap_or_default());
461    let display = query.get("display").unwrap_or_else(|| "index.bob".into());
462    let start = app::Location {
463        source: page.resolve(&display),
464        macros: parse_macros(&query.get("macros").unwrap_or_default()),
465    };
466    if let Some(pvws) = query.get("pvws") {
467        pv::set_pvws(&pvws);
468    }
469    if let Some(token) = query.get("token") {
470        // The token goes only to the page's own server, and only its
471        // displays may use it.
472        pv::set_write_token(&token);
473        fetch::same_origin_only();
474    }
475    let zoom = query.get("zoom").and_then(|z| {
476        parse_zoom(&z)
477            .inspect_err(|e| log::warn!("zoom: {e}; fitting to the window"))
478            .ok()
479    });
480
481    let canvas = window
482        .document()
483        .and_then(|d| d.get_element_by_id("bobview"))
484        .and_then(|c| c.dyn_into::<web_sys::HtmlCanvasElement>().ok())
485        .expect("a <canvas id=\"bobview\"> on the page");
486    wasm_bindgen_futures::spawn_local(async move {
487        let started = eframe::WebRunner::new()
488            .start(
489                canvas,
490                eframe::WebOptions::default(),
491                Box::new(move |cc| {
492                    let ui_scale = chrome::UiScale::from_env();
493                    let app = app::BobApp::new(&cc.egui_ctx, Some(start), zoom, ui_scale);
494                    Ok(Box::new(app::App::new(app)))
495                }),
496            )
497            .await;
498        if let Err(e) = started {
499            log::error!("cannot start: {e:?}");
500        }
501    });
502}

504/// The bundled index display, next to the binary or in ~/.local/share.
505#[cfg(not(target_arch = "wasm32"))]
506fn default_display() -> Option<PathBuf> {
507    let mut candidates = Vec::new();
508    if let Ok(exe) = std::env::current_exe()
509        && let Some(prefix) = exe.parent().and_then(|bin| bin.parent())
510    {
511        candidates.push(prefix.join("share/bobview/displays/index.bob"));
512        candidates.push(prefix.join("displays/index.bob"));
513    }
514    if let Some(home) = std::env::var_os("HOME") {
515        candidates.push(PathBuf::from(home).join(".local/share/bobview/displays/index.bob"));
516    }
517    candidates.into_iter().find(|p| p.exists())
518}

520struct Logger(log::LevelFilter);
521
522impl log::Log for Logger {
523    fn enabled(&self, m: &log::Metadata) -> bool {
524        // Library chatter (one line per CA channel connect) only when debugging.
525        let library = ["eframe", "egui", "epics", "winit", "glutin"]
526            .iter()
527            .any(|t| m.target().starts_with(t));
528        let limit = if library && self.0 < log::LevelFilter::Debug {
529            log::LevelFilter::Warn
530        } else {
531            self.0
532        };
533        m.level() <= limit
534    }
535
536    fn log(&self, r: &log::Record) {
537        if self.enabled(r.metadata()) {
538            #[cfg(not(target_arch = "wasm32"))]
539            eprintln!("{:<5} {}", r.level(), r.args());
540            #[cfg(target_arch = "wasm32")]
541            web_sys::console::log_1(&format!("{:<5} {}", r.level(), r.args()).into());
542        }
543    }
544
545    fn flush(&self) {}
546}
547
548fn init_logging() {
549    let level = match std::env::var("BOBVIEW_LOG").as_deref() {
550        Ok("debug") => log::LevelFilter::Debug,
551        Ok("trace") => log::LevelFilter::Trace,
552        Ok("warn") => log::LevelFilter::Warn,
553        Ok("error") => log::LevelFilter::Error,
554        _ => log::LevelFilter::Info,
555    };
556    let logger = Box::leak(Box::new(Logger(level)));
557    if log::set_logger(logger).is_ok() {
558        log::set_max_level(level);
559    }
560}

596        fn parse(args: &[&str]) -> Result<Cli, clap::Error> {
597            let args = std::iter::once("bobview").chain(args.iter().copied());
598            let matches = Cli::command().try_get_matches_from(args)?;
599            Cli::from_arg_matches(&matches)
600        }
601
602        #[test]
603        fn the_command_is_well_formed() {
604            Cli::command().debug_assert();
605        }
606
607        #[test]
608        fn defaults() {
609            let cli = parse(&["screen.bob"]).unwrap();
610            assert_eq!(cli.display.as_deref(), Some("screen.bob"));
611            assert_eq!(cli.size, Size(Some([1280, 800])));
612            assert_eq!((cli.scale_factor, cli.wait, cli.settle), (1.0, 2.0, 1.0));
613            assert_eq!(cli.zoom, None);
614        }
615
616        #[test]
617        fn reads_every_option() {
618            let cli = parse(&[
619                "-m",
620                "P=X",
621                "-z",
622                "1.5",
623                "-f",
624                "--read-only",
625                "--size",
626                "auto",
627                "--scale-factor",
628                "2.75",
629                "--wait",
630                "0",
631                "--settle",
632                "0.5",
633                "--ui-scale",
634                "auto",
635                "--snapshot",
636                "out.png",
637                "x.bob",
638            ])
639            .unwrap();
640            assert_eq!(cli.macros.as_deref(), Some("P=X"));
641            assert_eq!(cli.zoom, Some(1.5));
642            assert!(cli.fullscreen && cli.read_only);
643            assert_eq!(cli.size, Size(None));
644            assert_eq!((cli.scale_factor, cli.wait, cli.settle), (2.75, 0.0, 0.5));
645        }

647        #[test]
648        fn refuses_bad_values() {
649            for args in [
650                &["--zoom", "big"][..],
651                &["-z", "0"],
652                &["--size", "800"],
653                &["--size", "0x600"],
654                &["--size", "800x-1"],
655                &["--click", "10"],
656                &["--click", "a,b"],
657                &["--right-click", "1;2"],
658                &["--scale-factor", "0"],
659                &["--scale-factor", "x"],
660                &["--wait", "-1"],
661                &["--settle", "NaN"],
662                &["--ui-scale", "0"],
663                &["--no-such-option"],
664                &["a.bob", "b.bob"],
665            ] {
666                let e = parse(args).expect_err(&format!("{args:?}"));
667                assert_eq!(e.exit_code(), 2, "{args:?}");
668            }
669        }
670
671        #[test]
672        fn keeps_clicks_in_order() {
673            let args = ["--click", "1,2", "--right-click", "3, 4", "--click", "5,6"];
674            let argv = std::iter::once("bobview").chain(args);
675            let matches = Cli::command().try_get_matches_from(argv).unwrap();
676            let cli = Cli::from_arg_matches(&matches).unwrap();
677            assert_eq!(
678                cli.clicks(&matches),
679                [
680                    (pos2(1.0, 2.0), PointerButton::Primary),
681                    (pos2(3.0, 4.0), PointerButton::Secondary),
682                    (pos2(5.0, 6.0), PointerButton::Primary),
683                ]
684            );
685        }
686
687        #[test]
688        fn a_repeated_option_takes_the_last_value() {
689            let cli = parse(&["-z", "2", "-z", "3"]).unwrap();
690            assert_eq!(cli.zoom, Some(3.0));
691        }
