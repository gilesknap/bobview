83/// The smallest target, in points, a finger can hit reliably.
84pub const TOUCH_TARGET: f32 = 44.0;
85
86/// egui's own hit radius outside touch mode.
87const MOUSE_RADIUS: f32 = 5.0;
88
89/// How far outside a control a touch still hits it: enough to make a 20
90/// point control (a pvi slide button or checkbox at 100%) a full
91/// [`TOUCH_TARGET`].
92const TOUCH_RADIUS: f32 = (TOUCH_TARGET - 20.0) / 2.0;
93
94/// Grow the hit area of every control in touch mode, without
95/// changing how anything is drawn. egui hit-tests in two steps: a touch
96/// directly on a control always goes to that control (the topmost one), and
97/// only a touch on no control at all goes to the nearest control within
98/// `interact_radius`. So a small control gets a finger-sized target in the
99/// space around it, but never takes a touch from a neighbour that was hit
100/// directly, and the gap between two controls is shared by distance.
101pub fn touch_targets(ctx: &egui::Context, touch: bool) {
102    let radius = if touch { TOUCH_RADIUS } else { MOUSE_RADIUS };
103    if ctx.global_style().interaction.interact_radius != radius {
104        ctx.all_styles_mut(|s| s.interaction.interact_radius = radius);
105    }
106}

118/// Apply the outcomes of the node's rules to its properties, when they
119/// change. A rule on `x`, `y`, `width` or `height` moves or sizes the
120/// widget (and its children with it), and one on `pv_name` connects it to
121/// the PV named.
122pub(crate) fn update_rules(node: &mut Node, pvs: &pv::PvManager) {
123    if node.rules.is_empty() {
124        return;
125    }
126    // Rules depend only on their PVs: none changed, nothing to do.
127    let versions: Vec<u64> = node
128        .rules
129        .iter()
130        .flat_map(|r| r.pvs())
131        .map(pv::Pv::version)
132        .collect();
133    if node.rule_versions.as_ref() == Some(&versions) {
134        return;
135    }
136    node.rule_versions = Some(versions);
137    let outcomes: Vec<(String, Outcome)> = node
138        .rules
139        .iter()
140        .map(|r| (r.prop_id.clone(), r.evaluate()))
141        .collect();
142    if outcomes != node.rule_outcomes {
143        let orig = node.orig_props.get_or_insert_with(|| node.w.props.clone());
144        let mut props = orig.clone();
145        rules::apply(&mut props, &outcomes);
146        let before = crate::screen::geometry(&node.w);
147        node.w.props = props;
148        node.rule_outcomes = outcomes;
149        let after = crate::screen::geometry(&node.w);
150        if before != after {
151            node.reshape(before, after);
152        }
153        let pv_name = node.prop("pv_name");
154        if pv_name != node.pv_name {
155            node.pv = (!pv_name.is_empty()).then(|| pvs.subscribe(&pv_name));
156            node.pv_name = pv_name;
157        }
158    }
159}

161fn draw_node(ui: &mut Ui, node: &mut Node, cx: &mut Cx, id: Id) {
162    crate::group_link::update(node, cx.pvs);
163    update_rules(node, cx.pvs);
164    if node.w.bool("visible") == Some(false) {
165        return;
166    }
167    if node.w.kind == "label" && node.w.bool("auto_size") == Some(true) {
168        auto_size(ui.painter(), node, cx.view.zoom);
169    }
170    let rect = cx.view.rect(&node.rect);
171    if !rect.intersects(cx.clip) {
172        return;
173    }
174    let painter = ui.painter().with_clip_rect(cx.clip);
175    let z = cx.view.zoom;
176    let state = node.state();
177
178    let response = match node.w.kind.as_str() {
179        "label" => {
180            label(&painter, node, rect, z);
181            None
182        }
183        "textupdate" => {
184            text_update(&painter, node, state.as_ref(), rect, z);
185            None
186        }
187        "textentry" => Some(text_entry(ui, node, state.as_ref(), rect, cx, id)),
188        "led" => {
189            led(&painter, &node.w, state.as_ref(), rect, z);
190            None
191        }
192        "multi_state_led" => {
193            multi_state_led(&painter, &node.w, state.as_ref(), rect, z);
194            None
195        }
196        "byte_monitor" => {
197            byte_monitor(&painter, &node.w, state.as_ref(), rect, z);
198            None
199        }
200        "action_button" => Some(action_button(ui, node, rect, cx, id)),
201        "bool_button" => Some(bool_button(ui, node, state.as_ref(), rect, cx, id)),
202        "choice" | "radio" => Some(choice(ui, node, state.as_ref(), rect, cx, id)),
203        "combo" => Some(combo(ui, node, state.as_ref(), rect, cx, id)),
204        "checkbox" => Some(checkbox(ui, node, state.as_ref(), rect, cx, id)),
205        "slide_button" => Some(slide_button(ui, node, state.as_ref(), rect, cx, id)),
206        "rectangle" | "ellipse" | "arc" => {
207            shape(&painter, &node.w, rect, z);
208            None
209        }
210        "polyline" | "polygon" => {
211            poly(&painter, &node.w, cx.view, &node.rect);
212            None
213        }
214        "linearmeter" => {
215            crate::meter::linear_meter(&painter, &node.w, state.as_ref(), rect, z);
216            None
217        }
218        "xyplot" => {
219            crate::plot::draw(&painter, node, rect, z);
220            None
221        }
222        "stripchart" => {
223            crate::trend::draw_stripchart(&painter, node, rect, z);
224            None
225        }
226        "meter" => {
227            crate::meter::dial(&painter, &node.w, state.as_ref(), rect, z);
228            None
229        }
230        "progressbar" | "tank" | "thermometer" => {
231            bar(&painter, &node.w, state.as_ref(), rect, z);
232            None
233        }
234        "scaledslider" | "scrollbar" | "spinner" => {
235            Some(slider(ui, node, state.as_ref(), rect, cx, id))
236        }
237        "group" => {
238            group(ui, node, rect, cx, id);
239            None
240        }
241        "tabs" => {
242            tabs(ui, node, rect, cx, id);
243            None
244        }
245        "embedded" => {
246            embedded(ui, node, rect, cx, id);
247            None
248        }
249        "navtabs" => {
250            navtabs(ui, node, rect, cx, id);
251            None
252        }
253        "symbol" => crate::symbol::draw_symbol(ui, node, state.as_ref(), rect, cx, id),
254        "picture" => {
255            crate::symbol::draw_picture(ui, node, rect, cx);
256            None
257        }
258        "image" => Some(crate::image::draw(
259            ui,
260            node,
261            state.as_ref(),
262            rect,
263            cx.clip,
264            id,
265            z,
266        )),
267        other => {
268            let what = if node.pv_name.is_empty() {
269                other.to_string()
270            } else {
271                format!("{other}\n{}", node.pv_name)
272            };
273            placeholder(&painter, rect, &what, z);
274            None
275        }
276    };
277
278    if matches!(
279        node.w.kind.as_str(),
280        "label" | "textupdate" | "textentry" | "embedded"
281    ) {
282        border(&painter, &node.w, rect, z);
283    }
284    if let Some(state) = &state {
285        alarm_border(&painter, &node.w, state, rect, z);
286    }
287    // A shape or monitor (such as an EDM related-display hotspot from an
288    // `.opi`) that runs its actions when clicked.
289    let control = response.is_some();
290    let response = match response {
291        None if !is_container(&node.w.kind) => click_hook(ui, node, rect, cx, id),
292        r => r,
293    };
294
295    // A control that is disabled or whose PV is read-only is greyed out and
296    // shows a "not allowed" cursor; its writes are refused in `write`. The
297    // image and symbol widgets respond to the pointer but are not controls,
298    // nor is a monitor that only runs actions when clicked.
299    let disabled = control
300        && !matches!(node.w.kind.as_str(), "image" | "symbol")
301        && is_disabled(&node.w, state.as_ref());
302    // A slide button greys only its track: the rest of its box is see-through.
303    if disabled && node.w.kind != "slide_button" {
304        painter.rect_filled(rect, CornerRadius::ZERO, Color32::from_white_alpha(110));
305    }
306
307    // Hover shows the tooltip: the widget's own, or the PV name and value.
308    // Right-click (long-press on a touch screen) opens a menu to copy them or
309    // show the PV's details.
310    if !cx.clip.contains_rect(rect) {
311        return;
312    }
313    if !node.pv_name.is_empty() {
314        let resp = response.unwrap_or_else(|| ui.interact(rect, id.with("hover"), Sense::click()));
315        let resp = if disabled {
316            resp.on_hover_cursor(egui::CursorIcon::NotAllowed)
317        } else {
318            resp
319        };
320        // The context menu has its own popup id, so it does not fight a
321        // combo or action button's own menu (which uses the default id).
322        egui::Popup::context_menu(&resp)
323            .id(id.with("context-menu"))
324            .show(|ui| crate::pvmenu::menu(ui, node, state.as_ref(), cx.trends, cx.requests));
325        let text = tooltip(node, state.as_ref());
326        if !text.is_empty() {
327            resp.on_hover_text(text);
328        }
329    } else if node.w.get("tooltip").is_some() && !is_container(&node.w.kind) {
330        let text = tooltip(node, None);
331        if !text.is_empty() {
332            let resp =
333                response.unwrap_or_else(|| ui.interact(rect, id.with("hover"), Sense::hover()));
334            resp.on_hover_text(text);
335        }
336    }
337}

339/// The widget's `border_width` and `border_color`, drawn around it.
340fn border(painter: &egui::Painter, w: &Widget, rect: Rect, z: f32) {
341    let width = w.f64("border_width").unwrap_or(0.0) as f32 * z;
342    if width <= 0.0 {
343        return;
344    }
345    let c = w
346        .color("border_color")
347        .map(|c| color(&c))
348        .unwrap_or_else(|| named("Text"));
349    painter.rect_stroke(
350        rect,
351        CornerRadius::ZERO,
352        Stroke::new(width, c),
353        StrokeKind::Outside,
354    );
355}

357/// Run the actions of a widget with `run_actions_on_mouse_click` when it is
358/// clicked; its response, if it has any such actions.
359fn click_hook(ui: &mut Ui, node: &Node, rect: Rect, cx: &mut Cx, id: Id) -> Option<Response> {
360    let w = &node.w;
361    if w.bool("run_actions_on_mouse_click") != Some(true) || w.bool("enabled") == Some(false) {
362        return None;
363    }
364    let actions = w.actions();
365    if actions.is_empty() {
366        return None;
367    }
368    let resp = ui
369        .interact(rect.intersect(cx.clip), id.with("click"), Sense::click())
370        .on_hover_cursor(egui::CursorIcon::PointingHand);
371    if resp.clicked() {
372        for a in &actions {
373            run_action(node, a, cx);
374        }
375    }
376    Some(resp)
377}

384/// The hover text: the widget's `tooltip` with `$(pv_name)` and `$(pv_value)`
385/// filled in, or, without one, the PV name and value as Phoebus's default
386/// `$(pv_name)\n$(pv_value)` shows them (with the alarm severity).
387fn tooltip(node: &Node, s: Option<&PvState>) -> String {
388    let value = match s {
389        Some(s) if s.connected => s.format(None, true),
390        Some(s) => s.error.clone().unwrap_or_else(|| "disconnected".into()),
391        None => String::new(),
392    };
393    match node.w.text("tooltip") {
394        Some(t) => {
395            // `$(actions)` lists what the widget's actions do, as in Phoebus.
396            let actions = node.w.actions();
397            let t = if t.contains("$(actions)") {
398                let list: Vec<&str> = actions
399                    .iter()
400                    .map(|a| a.description())
401                    .filter(|d| !d.trim().is_empty())
402                    .collect();
403                t.replace("$(actions)", &list.join("\n"))
404            } else {
405                t.to_string()
406            };
407            tooltip_text(&t, &node.pv_name, &value, &|t| node.expand(t))
408        }
409        None => {
410            let name = plain_pv_name(&node.pv_name);
411            match s {
412                Some(s) if s.connected => format!("{name}\n{value} {:?}", s.severity),
413                Some(_) => format!("{name}\n{value}"),
414                None => name,
415            }
416        }
417    }
418}

420fn tooltip_text(
421    template: &str,
422    pv_name: &str,
423    value: &str,
424    expand: &dyn Fn(&str) -> String,
425) -> String {
426    let t = template
427        .replace("$(pv_name)", pv_name)
428        .replace("$(pv_value)", value);
429    expand(&t).trim().to_string()
430}

469/// Paint `text` in `rect` the way a JavaFX `Label` does: each row aligned on
470/// its own, only whole rows that fit the height, and overflow elided with `…`
471/// so the start of the text always stays visible.
472pub(crate) fn text_in(
473    painter: &egui::Painter,
474    rect: Rect,
475    text: &str,
476    font: FontId,
477    c: Color32,
478    align: Align2,
479    wrap: bool,
480) {
481    text_turned(painter, rect, text, font, c, align, wrap, 0);
482}

484/// [`text_in`], turned by Phoebus `rotation_step`: 1 is 90° anticlockwise
485/// (reading upwards), 2 is 180° and 3 is 90° clockwise.
486#[allow(clippy::too_many_arguments)]
487pub(crate) fn text_turned(
488    painter: &egui::Painter,
489    rect: Rect,
490    text: &str,
491    font: FontId,
492    c: Color32,
493    align: Align2,
494    wrap: bool,
495    step: i64,
496) {
497    use std::f32::consts::{FRAC_PI_2, PI};
498    let angle = match step {
499        1 => -FRAC_PI_2,
500        2 => PI,
501        3 => FRAC_PI_2,
502        _ => 0.0,
503    };
504    // Laid out in the box the text has before turning.
505    let size = if step == 1 || step == 3 {
506        Vec2::new(rect.height(), rect.width())
507    } else {
508        rect.size()
509    };
510    let area = Rect::from_center_size(rect.center(), size);
511    let row_height = painter.fonts_mut(|f| f.row_height(&font));
512    let job = text_job(text, font, c, size, row_height, align.x(), wrap);
513    let galley = painter.layout_job(job);
514    let x = match align.x() {
515        egui::Align::Min => area.left() + 1.0,
516        egui::Align::Center => area.center().x,
517        egui::Align::Max => area.right() - 1.0,
518    };
519    let height = galley.size().y;
520    let y = match align.y() {
521        egui::Align::Min => area.top(),
522        egui::Align::Center => area.center().y - height / 2.0,
523        egui::Align::Max => area.bottom() - height,
524    };
525    let painter = painter.with_clip_rect(rect.intersect(painter.clip_rect()));
526    if angle == 0.0 {
527        painter.galley(Pos2::new(x, y), galley, c);
528    } else {
529        let center = rect.center();
530        let pos = center + egui::emath::Rot2::from_angle(angle) * (Pos2::new(x, y) - center);
531        painter.add(egui::epaint::TextShape::new(pos, galley, c).with_angle(angle));
532    }
533}

535/// The layout for [`text_in`]: rows aligned on their own and limited to what
536/// fits in `size`, with the overflow elided.
537fn text_job(
538    text: &str,
539    font: FontId,
540    c: Color32,
541    size: Vec2,
542    row_height: f32,
543    halign: egui::Align,
544    wrap: bool,
545) -> egui::text::LayoutJob {
546    let mut job = egui::text::LayoutJob::simple(text.to_string(), font, c, f32::INFINITY);
547    job.halign = halign;
548    if !wrap && text.contains('\n') {
549        // like a JavaFX Label without wrapping: keep every explicit line,
550        // unwrapped, and let the widget's clip trim what doesn't fit
551        return job;
552    }
553    let rows = max_rows(size.y, row_height, wrap);
554    job.wrap = egui::text::TextWrapping {
555        // the full width: the widget clip hides the 1 px the text may be
556        // inset by, and any less elides text that fits at fractional zooms
557        max_rows: rows,
558        break_anywhere: rows == 1,
559        ..elide_wrap(size.x)
560    };
561    job
562}
563
564/// One row `width` wide, with what does not fit elided with `…`.
565fn elide_wrap(width: f32) -> egui::text::TextWrapping {
566    egui::text::TextWrapping {
567        max_width: width.max(1.0),
568        max_rows: 1,
569        break_anywhere: true,
570        overflow_character: Some('…'),
571    }
572}

574/// How many rows of text to lay out in a box `height` high: as many whole
575/// rows as fit when wrapping (allowing a little slack for font metrics), and
576/// always at least one.
577fn max_rows(height: f32, row_height: f32, wrap: bool) -> usize {
578    if !wrap || row_height <= 0.0 {
579        return 1;
580    }
581    (((height + 2.0) / row_height).floor() as usize).max(1)
582}

591/// Button-like 3D box.
592pub(crate) fn button_box(
593    painter: &egui::Painter,
594    rect: Rect,
595    fill_color: Color32,
596    pressed: bool,
597    z: f32,
598) {
599    // Like a JavaFX Button: rounded, with a grey outline on all four sides so
600    // it reads as a button on any background, a light top half for the
601    // gradient, and a darker fill while pressed.
602    let r = CornerRadius::same((3.0 * z).min(255.0) as u8);
603    let w = (1.0 * z).max(1.0);
604    let fill_color = if pressed {
605        fill_color.lerp_to_gamma(Color32::BLACK, 0.12)
606    } else {
607        fill_color
608    };
609    painter.rect_filled(rect, r, fill_color);
610    if !pressed {
611        let top = Rect::from_min_max(rect.min, Pos2::new(rect.right(), rect.center().y)).shrink(w);
612        let tr = CornerRadius {
613            sw: 0,
614            se: 0,
615            ..CornerRadius::same(r.nw.saturating_sub(1))
616        };
617        painter.rect_filled(top, tr, Color32::from_white_alpha(40));
618    }
619    painter.rect_stroke(
620        rect,
621        r,
622        Stroke::new(w, Color32::from_gray(150)),

641/// The `format` that shows (and writes) an array of numbers as a long
642/// string, as areaDetector's `FilePath` and other char waveforms are.
643const STRING_FORMAT: i64 = 6;
644
645/// Whether a control writes its text as a long string, as Phoebus does
646/// when its `format` is string: see [`pv::Pv::write_long_string`].
647fn writes_long_string(w: &Widget) -> bool {
648    w.i64("format") == Some(STRING_FORMAT)
649}

651/// The `bit` property: Phoebus defaults it to 0 for checkbox, slide_button and
652/// bool_button, and to -1 (the whole value) for LEDs.
653fn bit(w: &Widget) -> i64 {
654    let default = match w.kind.as_str() {
655        "checkbox" | "slide_button" | "bool_button" => 0,
656        _ => -1,
657    };
658    w.i64("bit").unwrap_or(default)
659}
660
661fn bit_on(v: i64, bit: i64) -> bool {
662    if bit < 0 { v != 0 } else { (v >> bit) & 1 == 1 }
663}
664
665fn is_on(w: &Widget, s: &PvState) -> bool {
666    bit_on(s.value.as_i64().unwrap_or(0), bit(w))
667}
668
669/// `value` with the widget's `bit` set to `on`, or 0/1 when `bit` is -1 (the
670/// whole value), as Phoebus's checkbox, bool and slide buttons write it.
671fn with_bit(bit: i64, value: i64, on: bool) -> i64 {
672    if !(0..63).contains(&bit) {
673        on as i64
674    } else if on {
675        value | (1 << bit)
676    } else {
677        value & !(1 << bit)
678    }
679}
680
681/// The value that toggles the widget's bit (or the whole value).
682fn toggled(w: &Widget, s: Option<&PvState>) -> i64 {
683    let bit = bit(w);
684    let on = s.is_some_and(|s| s.connected && is_on(w, s));
685    let v = s.and_then(|s| s.value.as_i64()).unwrap_or(0);
686    with_bit(bit, v, !on)
687}

689/// Phoebus disables a control whose `enabled` property is false or whose PV
690/// is connected without write access.
691fn is_disabled(w: &Widget, s: Option<&PvState>) -> bool {
692    w.bool("enabled") == Some(false) || s.is_some_and(|s| s.connected && !s.writable)
693}
694
695/// Whether a control may write its PV: enabled, connected and writable.
696fn can_write(w: &Widget, s: Option<&PvState>) -> bool {
697    !is_disabled(w, s) && s.is_some_and(|s| s.connected)
698}

881fn byte_monitor(painter: &egui::Painter, w: &Widget, s: Option<&PvState>, rect: Rect, z: f32) {
882    let n = w.i64("numBits").unwrap_or(8).clamp(1, 64) as usize;
883    let start = w.i64("startBit").unwrap_or(0).clamp(0, 63);
884    let reverse = w.bool("bitReverse").unwrap_or(false);
885    let horizontal = w.bool("horizontal").unwrap_or(true);
886    let on = w
887        .color("on_color")
888        .map(|c| color(&c))
889        .unwrap_or(Color32::from_rgb(60, 255, 60));
890    let off = w
891        .color("off_color")
892        .map(|c| color(&c))
893        .unwrap_or(Color32::from_rgb(60, 100, 60));
894    let value = s.filter(|s| s.connected).and_then(|s| s.value.as_i64());
895    let step = if horizontal {
896        rect.width() / n as f32
897    } else {
898        rect.height() / n as f32
899    };
900    for i in 0..n {
901        // Phoebus draws the highest bit first unless bitReverse is set.
902        let bit = if reverse {
903            start + i as i64
904        } else {
905            start + (n - 1 - i) as i64
906        };
907        let cell = if horizontal {
908            Rect::from_min_size(
909                rect.min + Vec2::new(step * i as f32, 0.0),
910                Vec2::new(step, rect.height()),
911            )
912        } else {
913            Rect::from_min_size(
914                rect.min + Vec2::new(0.0, step * i as f32),
915                Vec2::new(rect.width(), step),
916            )
917        };
918        let c = match value {
919            Some(v) if (v >> bit) & 1 == 1 => on,
920            Some(_) => off,
921            None => named("DISCONNECTED"),
922        };
923        draw_led(painter, w, cell.shrink(0.5 * z), c, "", z);
924    }
925}

927fn shape(painter: &egui::Painter, w: &Widget, rect: Rect, z: f32) {
928    let transparent = w.bool("transparent").unwrap_or(false);
929    let fill_c = if transparent {
930        Color32::TRANSPARENT
931    } else {
932        w.color("background_color")
933            .map(|c| color(&c))
934            .unwrap_or(Color32::from_rgb(30, 144, 255))
935    };
936    let lw = w.f64("line_width").unwrap_or(3.0) as f32 * z;
937    let line = w
938        .color("line_color")
939        .map(|c| color(&c))
940        .unwrap_or(Color32::from_rgb(0, 0, 255));
941    let stroke = if lw > 0.0 {
942        Stroke::new(lw, line)
943    } else {
944        Stroke::NONE
945    };
946    if w.kind == "arc" {
947        let start = w.f64("start_angle").unwrap_or(0.0) as f32;
948        let total = w.f64("total_angle").unwrap_or(90.0) as f32;
949        let inner = rect.shrink(lw / 2.0);
950        let points = arc_points(inner, start, total);
951        if !transparent && fill_c.a() > 0 {
952            // A pie wedge, as a fan from the centre (it may be concave).
953            let mut mesh = egui::Mesh::default();
954            mesh.colored_vertex(inner.center(), fill_c);
955            for p in &points {
956                mesh.colored_vertex(*p, fill_c);
957            }
958            for i in 1..points.len() as u32 {
959                mesh.add_triangle(0, i, i + 1);
960            }
961            painter.add(mesh);
962        }
963        if lw > 0.0 {
964            let mut line = points;
965            if !transparent {
966                line.insert(0, inner.center());
967                line.push(inner.center());
968            }
969            painter.add(Shape::line(line, stroke));
970        }
971    } else if w.kind == "rectangle" {
972        let cr = w
973            .f64("corner_width")
974            .unwrap_or(0.0)
975            .max(w.f64("corner_height").unwrap_or(0.0)) as f32
976            * z
977            / 2.0;
978        let r = CornerRadius::same(cr.min(255.0) as u8);
979        let style = w.i64("line_style").unwrap_or(0);
980        if style == 0 || cr > 0.0 || lw <= 0.0 {
981            painter.rect(rect, r, fill_c, stroke, StrokeKind::Inside);
982        } else {
983            painter.rect_filled(rect, r, fill_c);
984            let inner = rect.shrink(lw / 2.0);
985            let corners = [
986                inner.left_top(),
987                inner.right_top(),
988                inner.right_bottom(),
989                inner.left_bottom(),
990                inner.left_top(),
991            ];
992            crate::plot::line(painter, corners.to_vec(), stroke, style);
993        }
994    } else {
995        let inner = rect.shrink(lw / 2.0);
996        painter.add(Shape::ellipse_filled(
997            inner.center(),
998            inner.size() / 2.0,
999            fill_c,
1000        ));
1001        painter.add(Shape::ellipse_stroke(
1002            inner.center(),
1003            inner.size() / 2.0,
1004            stroke,
1005        ));
1006    }
1007}

1009/// Points along the arc of the ellipse in `rect` from `start` degrees over
1010/// `total` degrees, anticlockwise from three o'clock, as Phoebus draws arcs.
1011fn arc_points(rect: Rect, start: f32, total: f32) -> Vec<Pos2> {
1012    let n = ((total.abs() / 5.0).ceil() as usize).clamp(2, 73);
1013    let (c, r) = (rect.center(), rect.size() / 2.0);
1014    (0..=n)
1015        .map(|i| {
1016            let a = (start + total * i as f32 / n as f32).to_radians();
1017            Pos2::new(c.x + r.x * a.cos(), c.y - r.y * a.sin())
1018        })
1019        .collect()
1020}

1023    let display = w.points();

1025        .iter()
1026        .map(|(x, y)| view.pos(r[0] + x, r[1] + y))

1028    if points.len() < 2 {
1029        return;

1051fn bar(painter: &egui::Painter, w: &Widget, s: Option<&PvState>, rect: Rect, z: f32) {
1052    let (min, max) = s
1053        .filter(|_| w.bool("limits_from_pv").unwrap_or(true))

1055        .unwrap_or((

1057            w.f64("maximum").unwrap_or(100.0),
1058        ));
1059    let vertical = match w.kind.as_str() {
1060        "tank" | "thermometer" => true,
1061        "progressbar" => w.bool("horizontal") == Some(false),
1062        _ => false,
1063    };
1064    let fill_c = w
1065        .color("fill_color")
1066        .map(|c| color(&c))
1067        .unwrap_or(match w.kind.as_str() {
1068            "tank" => Color32::from_rgb(0, 0, 255),
1069            "thermometer" => Color32::from_rgb(255, 0, 0),
1070            _ => Color32::from_rgb(60, 255, 60),
1071        });
1072    fill(painter, rect, bg(w, "Read_Background"));
1073    outline(painter, rect, 1.0, Color32::from_gray(150));
1074    let Some(s) = s.filter(|s| s.connected) else {
1075        return;
1076    };
1077    let v = s.value.as_f64().unwrap_or(min);
1078    let frac = if max > min {
1079        ((v - min) / (max - min)).clamp(0.0, 1.0) as f32
1080    } else {
1081        0.0
1082    };
1083    let inner = rect.shrink(z.max(1.0));
1084    let filled = if vertical {
1085        Rect::from_min_max(
1086            Pos2::new(inner.left(), inner.bottom() - inner.height() * frac),
1087            inner.max,
1088        )
1089    } else {
1090        Rect::from_min_max(
1091            inner.min,
1092            Pos2::new(inner.left() + inner.width() * frac, inner.bottom()),
1093        )
1094    };
1095    fill(painter, filled, fill_c);
1096}

1100fn text_entry(
1101    ui: &mut Ui,
1102    node: &mut Node,
1103    state: Option<&PvState>,
1104    rect: Rect,
1105    cx: &mut Cx,
1106    id: Id,
1107) -> Response {
1108    let w = node.w.clone();
1109    let shown = match state {
1110        Some(s) if s.connected => format_value(&w, s),
1111        _ => format!("<{}>", node.pv_name),
1112    };
1113    let align = alignment(&w, 1);
1114    let enabled = can_write(&w, state);
1115    let mut buf = node.edit.clone().unwrap_or(shown);
1116    // Text that does not fit is elided with `…` like a text update's, except
1117    // while it is being edited, when every character must be there for the
1118    // cursor. (A click that focuses it lands in the elided layout, which
1119    // matches the full text up to the `…`.)
1120    let editing = ui.memory(|m| m.has_focus(id));
1121    let fnt = font(&w, cx.view.zoom);
1122    let color = fg(&w);
1123    let margin = Vec2::new(2.0, 0.0);
1124    let text_width = rect.width() - 2.0 * margin.x;
1125    let mut elided = |ui: &Ui, text: &dyn egui::TextBuffer, _wrap_width: f32| {
1126        let job = entry_job(
1127            text.as_str(),
1128            fnt.clone(),
1129            color,
1130            align.x(),
1131            text_width,
1132            editing,
1133        );
1134        ui.fonts_mut(|f| f.layout_job(job))
1135    };
1136    let edit = egui::TextEdit::singleline(&mut buf)
1137        .layouter(&mut elided)
1138        .id(id)
1139        .font(font(&w, cx.view.zoom))
1140        .text_color(fg(&w))
1141        .background_color(bg(&w, "Write_Background"))
1142        .horizontal_align(align.x())
1143        .vertical_align(egui::Align::Center)
1144        .margin(margin)
1145        .interactive(enabled)
1146        .min_size(rect.size());
1147    let resp = put_clipped(ui, id, rect, cx.clip, edit);
1148    if resp.has_focus() {
1149        node.edit = Some(buf);
1150    } else if resp.lost_focus() {
1151        if ui.input(|i| i.key_pressed(egui::Key::Enter)) {
1152            // Phoebus keeps the spaces of text written as a string.
1153            if writes_long_string(&node.w) {
1154                write_with(node, |pv| pv.write_long_string(buf));
1155            } else {
1156                // The edit starts from the text shown, units and all.
1157                write(node, crate::format::entry_value(&w, state, &buf));
1158            }
1159        }
1160        node.edit = None;
1161    }
1162    resp
1163}

1165/// The layout of a text entry's text: one row, as egui's own single-line
1166/// layout, but elided with `…` to `width` unless it is being edited.
1167fn entry_job(
1168    text: &str,
1169    font: FontId,
1170    c: Color32,
1171    halign: egui::Align,
1172    width: f32,
1173    editing: bool,
1174) -> egui::text::LayoutJob {
1175    let mut job = egui::text::LayoutJob::simple_singleline(text.to_string(), font, c);
1176    job.halign = halign;
1177    job.keep_trailing_whitespace = true;
1178    if !editing {
1179        job.wrap = elide_wrap(width);
1180    }
1181    job

1184/// Put a widget at `rect`, clipped. The scope starts at `rect`, not at the
1185/// cursor: one there would take in the space after the last widget each
1186/// time, and grow the display's scrolled size below its bottom. Its ids
1187/// derive from `id`, so a widget keeps its edit state (a drag, typing)
1188/// however many widgets came before it.
1189fn put_clipped(ui: &mut Ui, id: Id, rect: Rect, clip: Rect, widget: impl egui::Widget) -> Response {
1190    ui.scope_builder(
1191        egui::UiBuilder::new().max_rect(rect).id(id.with("clipped")),
1192        |ui| {
1193            ui.set_clip_rect(clip.intersect(ui.clip_rect()));
1194            ui.put(rect, widget)
1195        },
1196    )
1197    .inner
1198}

1203    let resp = ui.interact(rect.intersect(cx.clip), id, Sense::click());
1204    // A transparent button draws only its text, over what lies beneath.
1205    if w.bool("transparent") != Some(true) {
1206        button_box(
1207            ui.painter(),
1208            rect,
1209            bg(w, "Button_Background"),
1210            resp.is_pointer_button_down_on(),
1211            cx.view.zoom,
1212        );
1213    }
1214    let text = button_text(node, &actions);
1215    let painter = ui.painter().with_clip_rect(cx.clip);
1216    text_in(
1217        &painter,
1218        rect,
1219        &text,
1220        font(w, cx.view.zoom),
1221        fg(w),
1222        Align2::CENTER_CENTER,
1223        true,
1224    );
1225
1226    // A disabled button, or one whose PV is read-only, runs no actions.
1227    let allowed = !is_disabled(&node.w, node.state().as_ref());
1228    if allowed && actions.len() == 1 && resp.clicked() {
1229        run_action(node, &actions[0], cx);
1230    } else if allowed && actions.len() > 1 {
1231        egui::Popup::menu(&resp).show(|ui| {
1232            for a in &actions {
1233                if ui.button(node.expand(a.label())).clicked() {
1234                    run_action(node, a, cx);
1235                }
1236            }
1237        });
1238    }
1239    resp
1240}

1243/// exactly `$(actions)` (the default when there is none) names the one
1244/// action, or says how many there are; other text, empty too, is shown.
1245fn button_text(node: &Node, actions: &[Action]) -> String {

1258pub(crate) fn run_action(node: &Node, action: &Action, cx: &mut Cx) {
1259    match action {
1260        Action::OpenDisplay {
1261            file,
1262            macros,
1263            target,
1264            ..
1265        } => {
1266            // The file name may use the action's own macros, as in
1267            // `<file>$(IOC)/$(P).bob</file>` with IOC and P set by the action.
1268            let expanded =
1269                Macros::from_pairs(macros.iter().map(|(k, v)| (k.to_string(), node.expand(v))));
1270            let macros = node.macros.with(&expanded);
1271            let file = node.expand_with(&macros, file);
1272            cx.requests.push(Request::Open {
1273                source: node.resolve(&file),
1274                macros,
1275                target: target.clone(),
1276            });
1277        }
1278        Action::WritePv { pv_name, value, .. } => {
1279            let pv = action_pv(node, cx.pvs, &node.expand(pv_name));
1280            let value = node.expand(value);
1281            if write_needs_confirmation(node.base.as_ref()) && !cx.pvs.read_only() {
1282                cx.requests.push(Request::ConfirmWrite { pv, value });
1283            } else {
1284                // In read-only mode the PV refuses, and says so.
1285                pv.write(value);
1286            }
1287        }
1288        Action::OpenWebpage { url, .. } => {
1289            let url = node.expand(url);
1290            if webpage_allowed(node.base.as_ref(), &url) {
1291                cx.requests.push(Request::Webpage(url));
1292            } else {
1293                log::warn!("not opening '{url}': a display from the web may only open web pages");
1294            }
1295        }
1296        Action::Execute {
1297            text, description, ..
1298        } => {
1299            if let Some(names) = bob::databrowser_pvs(text) {
1300                trend(node, &names, cx);
1301            } else if is_toggle_script(text) {
1302                if let Some(s) = node.state() {
1303                    // As `(PVUtil.getDouble(pv) == 0) ? 1 : 0`: a value
1304                    // that is not a number is NaN there, so it writes 0.
1305                    let zero = s.value.as_f64() == Some(0.0);
1306                    let value = if zero { "1" } else { "0" };
1307                    match &node.pv {
1308                        Some(pv)
1309                            if write_needs_confirmation(node.base.as_ref())
1310                                && !cx.pvs.read_only()
1311                                && can_write(&node.w, Some(&s)) =>
1312                        {
1313                            cx.requests.push(Request::ConfirmWrite {
1314                                pv: pv.clone(),
1315                                value: value.into(),
1316                            });
1317                        }
1318                        _ => write(node, value),
1319                    }
1320                }
1321            } else {
1322                log::warn!("script '{description}' is not supported: bobview runs no scripts");
1323            }
1324        }
1325        Action::Command {
1326            command, directory, ..
1327        } => {
1328            let command = node.expand(command);
1329            // A StripTool launcher is shown as a trend, and not run.
1330            if let Some(names) = bob::strip_tool_pvs(&command) {
1331                cx.requests.push(Request::Trend(names));
1332                return;
1333            }
1334            let directory = node.expand(directory);
1335            let directory = (!directory.trim().is_empty()).then(|| node.resolve(&directory));
1336            // Only DLS launcher scripts of local displays, which open a
1337            // display; run on a thread, as a script may take seconds.
1338            cx.requests.push(Request::Command {
1339                base: node.base.clone(),
1340                directory,
1341                command,
1342            });
1343        }
1344        Action::CloseDisplay { .. } => cx.requests.push(Request::Close),
1345        Action::Trend { pv_names, .. } => trend(node, pv_names, cx),
1346        Action::Unsupported { kind, .. } => log::warn!("action '{kind}' is not supported"),
1347    }
1348}

1356/// Whether `script` is the toggle that techui-builder embeds in synoptic
1357/// symbols, which bobview runs natively since it runs no scripts:
1358///
1359/// ```js
1360/// pv = ScriptUtil.getPrimaryPV(widget)
1361/// var pv_value = PVUtil.getDouble(pv);
1362/// pv_value = (pv_value == 0) ? 1 : 0;
1363/// pv.setValue(pv_value);
1364/// ```
1365fn is_toggle_script(script: &str) -> bool {
1366    let code: String = script
1367        .lines()
1368        .filter(|l| !l.trim_start().starts_with("importClass") && !l.trim_start().starts_with("/*"))
1369        .flat_map(|l| l.chars())
1370        .filter(|c| !c.is_whitespace() && *c != ';')
1371        .collect();
1372    code == "pv=ScriptUtil.getPrimaryPV(widget)varpv_value=PVUtil.getDouble(pv)\
1373             pv_value=(pv_value==0)?1:0pv.setValue(pv_value)"
1374}

1376/// Whether a display at `base` may hand `url` to `xdg-open`. One fetched
1377/// over HTTP is not trusted with local files or other URL schemes (which
1378/// could open or run something on this machine), only with web pages; and
1379/// nothing may pass `xdg-open` an option.
1380fn webpage_allowed(base: Option<&Source>, url: &str) -> bool {
1381    let url = url.trim();
1382    if url.is_empty() || url.starts_with('-') {
1383        return false;
1384    }
1385    !base.is_some_and(Source::is_url) || Source::parse(url).is_url()
1386}

1388fn bool_button(
1389    ui: &mut Ui,
1390    node: &mut Node,
1391    s: Option<&PvState>,
1392    rect: Rect,
1393    cx: &mut Cx,
1394    id: Id,
1395) -> Response {
1396    let w = node.w.clone();
1397    let z = cx.view.zoom;
1398    let on = s.is_some_and(|s| s.connected && is_on(&w, s));
1399    let resp = ui.interact(rect.intersect(cx.clip), id, Sense::click());
1400    button_box(
1401        ui.painter(),
1402        rect,
1403        bg(&w, "Button_Background"),
1404        resp.is_pointer_button_down_on() || on,
1405        z,
1406    );
1407    let painter = ui.painter().with_clip_rect(cx.clip);
1408    let label = if on {
1409        w.text("on_label").unwrap_or("On")
1410    } else {
1411        w.text("off_label").unwrap_or("Off")
1412    };
1413    let mut text_rect = rect;
1414    if w.bool("show_led").unwrap_or(true) {
1415        let d = (rect.height() * 0.5).min(rect.width() * 0.3);
1416        let led_rect =
1417            Rect::from_center_size(Pos2::new(rect.left() + d, rect.center().y), Vec2::splat(d));
1418        draw_led(&painter, &w, led_rect, led_color(&w, s), "", z);
1419        text_rect.min.x += d * 1.6;
1420    }
1421    text_in(
1422        &painter,
1423        text_rect,
1424        label,
1425        font(&w, z),
1426        fg(&w),
1427        Align2::CENTER_CENTER,
1428        false,
1429    );
1430    // Mode 0 toggles; 1 and 2 push (2 inverted): the pressed value is written
1431    // when the button goes down and the released one when it comes up.
1432    let bit = bit(&w);
1433    match w.i64("mode").unwrap_or(0) {
1434        0 if resp.clicked() => write(node, toggled(&w, s).to_string()),
1435        m @ (1 | 2) => {
1436            let v = s.and_then(|s| s.value.as_i64()).unwrap_or(0);
1437            let value = |pressed: bool| with_bit(bit, v, pressed == (m == 1)).to_string();
1438            // Only the primary button pushes; a right-click opens the PV menu.
1439            let down = resp.is_pointer_button_down_on() && ui.input(|i| i.pointer.primary_down());
1440            let was_down = ui.data(|d| d.get_temp::<bool>(id).unwrap_or(false));
1441            if down != was_down {
1442                ui.data_mut(|d| d.insert_temp(id, down));
1443            }
1444            match push_edge(was_down, down, resp.clicked()) {
1445                Push::Press => write(node, value(true)),
1446                Push::Release => write(node, value(false)),
1447                Push::Both => {
1448                    write(node, value(true));
1449                    write(node, value(false));
1450                }
1451                Push::None => {}
1452            }
1453        }
1454        _ => {}
1455    }
1456    resp
1457}

1459#[derive(Debug, PartialEq)]
1460enum Push {
1461    None,
1462    Press,
1463    Release,
1464    /// Pressed and released within one frame.
1465    Both,
1466}
1467
1468/// What a push button writes, from whether it was down last frame, is down
1469/// now, and was clicked this frame.
1470fn push_edge(was_down: bool, down: bool, clicked: bool) -> Push {
1471    match (was_down, down) {
1472        (false, true) => Push::Press,
1473        (true, false) => Push::Release,
1474        (false, false) if clicked => Push::Both,
1475        _ => Push::None,
1476    }
1477}

1479/// The choices for a choice/radio/combo widget and the index of the current one.
1480fn choices(w: &Widget, s: Option<&PvState>) -> (Vec<String>, Option<usize>) {
1481    let from_pv = w.bool("items_from_pv").unwrap_or(true);
1482    let items = match s {
1483        Some(s) if from_pv && !s.labels.is_empty() => s.labels.clone(),
1484        _ => w.items(),
1485    };
1486    let current = s.filter(|s| s.connected).and_then(|s| match &s.value {
1487        Value::Enum(i) => Some(*i as usize),
1488        Value::Str(t) => items.iter().position(|i| i == t),
1489        v => v.as_i64().map(|i| i as usize),
1490    });
1491    (items, current)
1492}

1494fn choice(
1495    ui: &mut Ui,
1496    node: &mut Node,
1497    s: Option<&PvState>,
1498    rect: Rect,
1499    cx: &mut Cx,
1500    id: Id,

1504    let (items, current) = choices(&w, s);
1505    let horizontal = w.bool("horizontal").unwrap_or(true);

1516        (rect.width() / n as f32, 0.0)

1521        Vec2::new(dx, rect.height())
1522    } else {
1523        Vec2::new(rect.width(), dy)
1524    };

1527            Rect::from_min_size(rect.min + Vec2::new(dx * i as f32, dy * i as f32), size).shrink(z);
1528        let r = ui.interact(cell.intersect(cx.clip), id.with(i), Sense::click());
1529        let is_sel = current == Some(i);
1530        if w.kind == "radio" {
1531            let d = cell.height().min(14.0 * z);
1532            let c = Pos2::new(cell.left() + d / 2.0 + 2.0, cell.center().y);
1533            painter.circle(
1534                c,
1535                d / 2.0,
1536                Color32::WHITE,
1537                Stroke::new(1.0, Color32::DARK_GRAY),

1540                painter.circle_filled(c, d / 4.0, Color32::BLACK);
1541            }
1542            let tr = Rect::from_min_max(Pos2::new(cell.left() + d + 6.0, cell.top()), cell.max);
1543            text_in(
1544                &painter,
1545                tr,
1546                item,
1547                font(&w, z),
1548                fg(&w),
1549                Align2::LEFT_CENTER,
1550                false,
1551            );
1552        } else {
1553            let fill_c = if is_sel {
1554                selected
1555            } else {
1556                bg(&w, "Button_Background")
1557            };
1558            button_box(
1559                &painter,
1560                cell,
1561                fill_c,
1562                is_sel || r.is_pointer_button_down_on(),
1563                z,
1564            );
1565            text_in(
1566                &painter,
1567                cell,
1568                item,
1569                font(&w, z),
1570                fg(&w),
1571                Align2::CENTER_CENTER,
1572                false,
1573            );
1574        }
1575        if r.clicked() {
1576            write(node, item.clone());
1577        }
1578        resp = resp.union(r);
1579    }
1580    resp

1583fn combo(
1584    ui: &mut Ui,
1585    node: &mut Node,
1586    s: Option<&PvState>,
1587    rect: Rect,
1588    cx: &mut Cx,
1589    id: Id,
1590) -> Response {
1591    let w = node.w.clone();
1592    let (items, current) = choices(&w, s);
1593    let mux = w.menu_mux();
1594    // Like Phoebus, show the PV's value whatever the items are; the items
1595    // only fill the dropdown. A menu mux shows the item the value is for.
1596    let text = match s {
1597        Some(s) if s.connected => mux
1598            .as_ref()
1599            .and_then(|m| m.current(current, &s.format(None, false)))
1600            .and_then(|i| items.get(i).cloned())
1601            .unwrap_or_else(|| s.format(None, false)),
1602        _ => format!("<{}>", node.pv_name),
1603    };
1604    let resp = ui.interact(rect.intersect(cx.clip), id, Sense::click());
1605    let z = cx.view.zoom;
1606    let painter = ui.painter().with_clip_rect(cx.clip);
1607    button_box(&painter, rect, bg(&w, "Button_Background"), false, z);
1608    let arrow = rect.height().min(rect.width() / 4.0);
1609    let text_rect = Rect::from_min_max(
1610        rect.min + Vec2::new(4.0, 0.0),
1611        Pos2::new(rect.right() - arrow, rect.bottom()),
1612    );
1613    text_in(
1614        &painter,
1615        text_rect,
1616        &text,
1617        font(&w, z),
1618        fg(&w),
1619        Align2::LEFT_CENTER,
1620        false,
1621    );
1622    let c = Pos2::new(rect.right() - arrow / 2.0, rect.center().y);
1623    let a = arrow * 0.2;
1624    painter.add(Shape::convex_polygon(
1625        vec![
1626            c + Vec2::new(-a, -a / 2.0),
1627            c + Vec2::new(a, -a / 2.0),
1628            c + Vec2::new(0.0, a),
1629        ],
1630        fg(&w),
1631        Stroke::NONE,
1632    ));
1633    if !can_write(&w, s) {
1634        return resp;
1635    }
1636    // Like Phoebus, the list is at least as wide as the combo and in its
1637    // font (but never smaller than the rest of the UI, so it stays readable
1638    // when the display is zoomed out); a finger-sized row in touch mode.
1639    let item_font = combo_item_font(
1640        font(&w, z),
1641        &ui.style().text_styles[&egui::TextStyle::Button],
1642    );
1643    let row_h = if cx.touch { TOUCH_TARGET } else { 0.0 };
1644    egui::Popup::menu(&resp).width(rect.width()).show(|ui| {
1645        ui.set_min_width(rect.width() - ui.spacing().menu_margin.sum().x);
1646        for (i, item) in items.iter().enumerate() {
1647            let label = egui::RichText::new(item).font(item_font.clone());
1648            let button = egui::Button::selectable(*item == text, label)
1649                .min_size(Vec2::new(ui.available_width(), row_h));
1650            if ui.add(button).clicked() {
1651                match &mux {
1652                    Some(mux) => mux_pick(node, cx.pvs, mux, i),
1653                    None => write(node, item.clone()),
1654                }
1655            }
1656        }
1657    });
1658    resp
1659}

1863/// Where a slider's parts go, along its length and across it.
1864#[derive(Debug)]
1865struct SliderLayout {
1866    /// The band the thumb slides along.
1867    band: Rect,
1868    /// The thumb's length along the slider.
1869    thumb: f32,
1870    /// The thumb's centre at the minimum and at the maximum.
1871    from: f32,
1872    to: f32,
1873    /// The band on the far side of the thumb's for the scale's ticks, and
1874    /// whether its labels fit in it too.
1875    scale: Option<Rect>,
1876    labels: bool,
1877    /// The band on the near side for the alarm level markers.
1878    markers: Option<Rect>,
1879}

1881/// Lay out a slider in `rect`: the thumb's band, the scale on its far side
1882/// (below a horizontal slider, right of a vertical one) and the level
1883/// markers on its near side, all centred across it, as a JavaFX Slider
1884/// centres its track and tick marks. `scale` is the room the scale's
1885/// labels take across the slider and half of their length along it, and
1886/// `markers` the room the markers' labels take across it; `None` leaves
1887/// either out. What the box has no room for is left out, the markers
1888/// first, then the scale's labels and then its ticks, before the thumb
1889/// gets thin.
1890fn slider_layout(
1891    rect: Rect,
1892    horizontal: bool,
1893    z: f32,
1894    scale: Option<(f32, f32)>,
1895    markers: Option<f32>,
1896) -> SliderLayout {
1897    let (along, across) = if horizontal {
1898        (rect.x_range(), rect.y_range())
1899    } else {
1900        (rect.y_range(), rect.x_range())
1901    };
1902    let pad = 2.0 * z;
1903    let tick = 5.0 * z;
1904    let room = (across.span() - 2.0 * pad).max(1.0);
1905    let (full, thin) = (16.0 * z, 8.0 * z);
1906    let ticks = scale.is_some();
1907    let label = scale.map_or(0.0, |(l, _)| l);
1908    let (mark, depth, labels) = [
1909        (markers.map(|m| m + tick), ticks, ticks),
1910        (None, ticks, ticks),
1911        (None, ticks, false),
1912        (None, false, false),
1913    ]
1914    .into_iter()
1915    .map(|(m, t, l)| {
1916        let depth = if t { tick } else { 0.0 } + if l { label } else { 0.0 };
1917        (m.unwrap_or(0.0), depth, l)
1918    })
1919    .find(|(m, depth, _)| room - m - depth >= thin)
1920    .unwrap_or((0.0, 0.0, false));
1921    let thick = (room - mark - depth).clamp(1.0, full);
1922    let start = across.center() - (mark + thick + depth) / 2.0;
1923    let thumb = (thick * 0.6).max(1.0);
1924    let half_label = scale.filter(|_| labels).map_or(0.0, |(_, half)| half);
1925    let end = (thumb / 2.0).max(half_label) + pad;
1926    let (lo, hi) = if along.span() > 2.0 * end {
1927        (along.min + end, along.max - end)
1928    } else {
1929        (along.center(), along.center())
1930    };
1931    let band = (start + mark, start + mark + thick);
1932    let part = |(a, b): (f32, f32)| {
1933        if horizontal {
1934            Rect::from_x_y_ranges(along, a..=b)
1935        } else {
1936            Rect::from_x_y_ranges(a..=b, along)
1937        }
1938    };
1939    // A vertical slider has its minimum at the bottom.
1940    let (from, to) = if horizontal { (lo, hi) } else { (hi, lo) };
1941    SliderLayout {
1942        band: part(band),
1943        thumb,
1944        from,
1945        to,
1946        scale: (depth > 0.0).then(|| part((band.1, band.1 + depth))),
1947        labels,
1948        markers: (mark > 0.0).then(|| part((start, band.0))),
1949    }
1950}

1952/// The range a slider runs over, from its `minimum` and `maximum` or its
1953/// PV's: as Phoebus, 0 to 100 for one that is empty or inverted. A BOY
1954/// slider (the bobview-only `<boy_range>` from the `.opi` import) keeps
1955/// an inverted range, running from the maximum, and widens an empty one
1956/// to the minimum plus one, as BOY's scale did.
1957fn slider_range(w: &Widget, (min, max): (f64, f64)) -> (f64, f64) {
1958    if w.bool("boy_range") == Some(true) {
1959        return if min == max {
1960            (min, min + 1.0)
1961        } else {
1962            (min, max)
1963        };
1964    }
1965    if min < max { (min, max) } else { (0.0, 100.0) }
1966}

1968/// A scaled slider's alarm level markers, as Phoebus': each of LOLO, LOW,
1969/// HIGH and HIHI there is a level for, unless its `show_*` is false. The
1970/// levels are the PV's with `limits_from_pv` while it is connected, else
1971/// the widget's `level_*` (10, 20, 80 and 90 unless given). Room is kept
1972/// for them all; those outside the range are not drawn.
1973fn slider_markers(w: &Widget, s: Option<&PvState>) -> Vec<(f64, &'static str)> {
1974    let from_pv = w.bool("limits_from_pv").unwrap_or(true) && s.is_some_and(|s| s.connected);
1975    let mut levels = if from_pv {
1976        crate::meter::levels(w, s)
1977    } else {
1978        ["level_lolo", "level_low", "level_high", "level_hihi"]
1979            .map(|k| w.f64(k).filter(|v| v.is_finite()))
1980    };
1981    if !from_pv {
1982        for (level, default) in levels.iter_mut().zip([10.0, 20.0, 80.0, 90.0]) {
1983            level.get_or_insert(default);
1984        }
1985    }
1986    levels
1987        .into_iter()
1988        .zip([
1989            ("LOLO", "show_lolo"),
1990            ("LOW", "show_low"),
1991            ("HIGH", "show_high"),
1992            ("HIHI", "show_hihi"),
1993        ])
1994        .filter(|(_, (_, show))| w.bool(show).unwrap_or(true))
1995        .filter_map(|(level, (name, _))| Some((level?, name)))
1996        .collect()
1997}

1999/// Phoebus ScaledSlider and ScrollBar: a groove along the widget with a
2000/// thumb at the value, and for a scaled slider with `show_scale` labelled
2001/// ticks on one side and its alarm level markers on the other, as
2002/// Phoebus' JavaFX Slider; no value box. A click or drag moves the thumb,
2003/// and its value is written when the button is let go, over the range
2004/// [`slider_range`] gives.
2005fn slider(
2006    ui: &mut Ui,
2007    node: &mut Node,
2008    s: Option<&PvState>,
2009    rect: Rect,
2010    cx: &mut Cx,
2011    id: Id,
2012) -> Response {
2013    if node.w.kind == "spinner" {
2014        return spinner(ui, node, s, rect, cx, id);
2015    }
2016    let w = node.w.clone();
2017    let z = cx.view.zoom;
2018    let (min, max) = slider_range(&w, crate::meter::limits(&w, s));
2019    // Only a range too large to widen is still empty.
2020    let live = min != max;
2021    let horizontal = w.bool("horizontal").unwrap_or(true);
2022    let show_scale = w.kind == "scaledslider" && w.bool("show_scale").unwrap_or(true) && live;
2023    let painter = ui.painter().with_clip_rect(cx.clip.intersect(rect));
2024    if w.kind == "scaledslider" && w.bool("transparent") == Some(false) {
2025        fill(&painter, rect, bg(&w, "Background"));
2026    }
2027    let fnt = font(&w, z);
2028    let text = fg(&w);
2029    let size_of = |s: &str| {
2030        painter
2031            .layout_no_wrap(s.to_string(), fnt.clone(), text)
2032            .size()
2033    };
2034    let guess = (max - min).abs() / 10.0;
2035    let size = |v: f64| size_of(&crate::axis::label(v, guess));
2036    let (lo_size, hi_size) = (size(min), size(max));
2037    let widest = lo_size.x.max(hi_size.x);
2038    let line_h = lo_size.y;
2039    let room = if horizontal {
2040        (line_h, widest / 2.0)
2041    } else {
2042        (widest + 2.0 * z, line_h / 2.0)
2043    };
2044    let markers = if w.kind == "scaledslider" {
2045        slider_markers(&w, s)
2046    } else {
2047        Vec::new()
2048    };
2049    let mark_room = (!markers.is_empty() && live).then(|| {
2050        if horizontal {
2051            line_h
2052        } else {
2053            size_of("HIHI").x + 2.0 * z
2054        }
2055    });
2056    let lay = slider_layout(rect, horizontal, z, show_scale.then_some(room), mark_room);
2057    let scale = crate::axis::Scale {
2058        lo: min,
2059        hi: max,
2060        log: false,
2061        from: lay.from,
2062        to: lay.to,
2063    };
2064
2065    // Follow the pointer while it is down on an enabled slider, and write
2066    // where it was let go.
2067    let resp = ui.interact(rect.intersect(cx.clip), id, Sense::click_and_drag());
2068    let mut v = s
2069        .and_then(|s| s.value.as_f64())
2070        .filter(|v| v.is_finite())
2071        .unwrap_or(min);
2072    let enabled = live && !is_disabled(&w, s);
2073    if enabled && let Some(p) = resp.interact_pointer_pos() {
2074        let at = if horizontal { p.x } else { p.y };
2075        let t = if lay.to != lay.from {
2076            ((at - lay.from) / (lay.to - lay.from)).clamp(0.0, 1.0) as f64
2077        } else {
2078            0.0
2079        };
2080        v = min + (max - min) * t;
2081        ui.data_mut(|d| d.insert_temp(id, v));
2082    }
2083    if (resp.drag_stopped() || resp.clicked())
2084        && let Some(v) = ui.data_mut(|d| d.remove_temp::<f64>(id))
2085    {
2086        write(node, v.to_string());
2087    }
2088
2089    // The groove, then the scale, then the thumb over both.
2090    let band = lay.band;
2091    let groove_w = (band.height().min(band.width()) / 4.0).max(2.0 * z);
2092    let groove = if horizontal {
2093        Rect::from_center_size(
2094            Pos2::new((lay.from + lay.to) / 2.0, band.center().y),
2095            Vec2::new((lay.to - lay.from).abs() + lay.thumb, groove_w),
2096        )
2097    } else {
2098        Rect::from_center_size(
2099            Pos2::new(band.center().x, (lay.from + lay.to) / 2.0),
2100            Vec2::new(groove_w, (lay.to - lay.from).abs() + lay.thumb),
2101        )
2102    };
2103    painter.rect(
2104        groove,
2105        CornerRadius::same((groove_w / 2.0).min(255.0) as u8),
2106        Color32::from_gray(225),
2107        Stroke::new(1.0, Color32::from_gray(150)),
2108        StrokeKind::Inside,
2109    );
2110    // A tick at `p` along the slider, out from `edge` across it (`out` is
2111    // 1 to the far side, -1 to the near one), with its label beyond.
2112    let stroke = Stroke::new(1.0, text);
2113    let tick_at = |edge: f32, p: f32, out: f32, label: Option<String>| {
2114        let len = 5.0 * z * out;
2115        let (a, b) = if horizontal {
2116            (Pos2::new(p, edge), Pos2::new(p, edge + len))
2117        } else {
2118            (Pos2::new(edge, p), Pos2::new(edge + len, p))
2119        };
2120        painter.line_segment([a, b], stroke);
2121        if let Some(label) = label {
2122            let (align, gap) = match (horizontal, out > 0.0) {
2123                (true, true) => (Align2::CENTER_TOP, 0.0),
2124                (true, false) => (Align2::CENTER_BOTTOM, 0.0),
2125                (false, true) => (Align2::LEFT_CENTER, 2.0 * z),
2126                (false, false) => (Align2::RIGHT_CENTER, -2.0 * z),
2127            };
2128            painter.text(b + Vec2::new(gap, 0.0), align, label, fnt.clone(), text);
2129        }
2130    };
2131    if let Some(area) = lay.scale {
2132        // Major ticks at least `major_tick_step_hint` apart, and far
2133        // enough apart for their labels.
2134        let hint = w.f64("major_tick_step_hint").unwrap_or(40.0) as f32 * z;
2135        let label_len = if horizontal {
2136            widest + 8.0 * z
2137        } else {
2138            line_h * 1.5
2139        };
2140        let gaps = ((lay.to - lay.from).abs() / hint.max(label_len)).floor();
2141        let (ticks, step) = crate::axis::ticks(min, max, gaps as usize + 1);
2142        let edge = if horizontal { area.top() } else { area.left() };
2143        for t in ticks {
2144            let label = lay.labels.then(|| crate::axis::label(t, step));
2145            tick_at(edge, scale.pos(t), 1.0, label);
2146        }
2147    }
2148    if let Some(area) = lay.markers {
2149        let edge = if horizontal {
2150            area.bottom()
2151        } else {
2152            area.right()
2153        };
2154        let range = min.min(max)..=max.max(min);
2155        for (level, name) in markers.iter().filter(|(l, _)| range.contains(l)) {
2156            tick_at(edge, scale.pos(*level), -1.0, Some(name.to_string()));
2157        }
2158    }
2159    let at = if live {
2160        scale.pos(v.clamp(min.min(max), max.max(min)))
2161    } else {
2162        lay.from
2163    };
2164    let thumb = if horizontal {
2165        Rect::from_center_size(
2166            Pos2::new(at, band.center().y),
2167            Vec2::new(lay.thumb, band.height()),
2168        )
2169    } else {
2170        Rect::from_center_size(
2171            Pos2::new(band.center().x, at),
2172            Vec2::new(band.width(), lay.thumb),
2173        )
2174    };
2175    button_box(
2176        &painter,
2177        thumb,
2178        named("Button_Background"),
2179        enabled && resp.is_pointer_button_down_on(),
2180        z,
2181    );
2182    resp
2183}

2348/// How much an embedded display is scaled to fit a widget `size` (unzoomed):
2349/// resize mode 1 scales the content to fit; the others draw it 1:1, clipped
2350/// to the widget (4, Crop, included).
2351fn embed_scale(w: &Widget, screen: &Screen, size: Vec2) -> f32 {
2352    match w.i64("resize").unwrap_or(0) {
2353        1 if screen.width > 0.0 && screen.height > 0.0 => {
2354            (size.x / screen.width as f32).min(size.y / screen.height as f32)
2355        }
2356        _ => 1.0,
2357    }
2358}

2380fn navtabs(ui: &mut Ui, node: &mut Node, rect: Rect, cx: &mut Cx, id: Id) {
2381    let w = node.w.clone();
2382    let z = cx.view.zoom;
2383    let names: Vec<String> = tab_elements(&w)
2384        .iter()
2385        .map(|t| node.expand(t.child_text("name").unwrap_or("")))
2386        .collect();
2387    let vertical = w.i64("direction").unwrap_or(1) == 1;
2388    let tw = w.f64("tab_width").unwrap_or(100.0) as f32 * z;
2389    let th = w.f64("tab_height").unwrap_or(30.0) as f32 * z;
2390    let gap = w.f64("tab_spacing").unwrap_or(2.0) as f32 * z;
2391    let sel = w
2392        .color("selected_color")
2393        .map(|c| color(&c))
2394        .unwrap_or(Color32::from_rgb(236, 236, 236));
2395    let desel = w
2396        .color("deselected_color")
2397        .map(|c| color(&c))
2398        .unwrap_or(Color32::from_rgb(200, 200, 200));
2399    let painter = ui.painter().with_clip_rect(cx.clip);
2400    let mut clicked = None;
2401    for (i, name) in names.iter().enumerate() {
2402        let off = i as f32 * if vertical { th + gap } else { tw + gap };
2403        let pos = if vertical {
2404            Vec2::new(0.0, off)
2405        } else {
2406            Vec2::new(off, 0.0)
2407        };
2408        let tab = Rect::from_min_size(rect.min + pos, Vec2::new(tw, th));
2409        let r = ui.interact(tab.intersect(cx.clip), id.with(("nav", i)), Sense::click());
2410        let active = i == node.active_tab;
2411        button_box(&painter, tab, if active { sel } else { desel }, active, z);
2412        text_in(
2413            &painter,
2414            tab,
2415            name,
2416            font(&w, z),
2417            fg(&w),
2418            Align2::CENTER_CENTER,
2419            true,
2420        );
2421        if r.clicked() && !active {
2422            clicked = Some(i);
2423        }
2424    }
2425    if let Some(i) = clicked {
2426        Builder::ui(cx.pvs).pick_navtab(node, i, cx.depth);
2427    }
2428    if fetched(ui, node) {
2429        Builder::ui(cx.pvs).select_navtab(node, node.active_tab, cx.depth);
2430    }
2431    let body = if vertical {
2432        Rect::from_min_max(rect.min + Vec2::new(tw + gap, 0.0), rect.max)
2433    } else {
2434        Rect::from_min_max(rect.min + Vec2::new(0.0, th + gap), rect.max)
2435    };
2436    match node.embedded.as_mut() {
2437        Some(screen) => draw_embedded(ui, &w, screen, body, cx, id.with("body")),
2438        None => {
2439            if let Some(e) = &node.error {
2440                placeholder(&painter, body, e, z);
2441            }
2442        }
2443    }
2444}

2446pub(crate) fn alarm_border(painter: &egui::Painter, w: &Widget, s: &PvState, rect: Rect, z: f32) {
2447    if w.bool("border_alarm_sensitive") == Some(false) {
2448        return;
2449    }
2450    // A BOY widget (from an `.opi` file) drew INVALID and disconnected
2451    // solid white, DLS's BOY INVALID colour.
2452    if w.bool("boy_alarm_border") == Some(true)
2453        && (!s.connected || s.severity == pv::Severity::Invalid)
2454    {
2455        let width = (2.0 * z).max(1.0);
2456        outline(painter, rect.expand(width), width, Color32::WHITE);
2457        return;
2458    }
2459    let c = if s.connected {
2460        severity_color(s.severity)
2461    } else {
2462        Some(named("DISCONNECTED"))
2463    };
2464    if let Some(c) = c {
2465        let width = (2.0 * z).max(1.0);
2466        if s.connected {
2467            outline(painter, rect.expand(width), width, c);
2468        } else {
2469            // Dashed, as Phoebus marks disconnected PVs.
2470            let r = rect.expand(width / 2.0);
2471            let pts = [
2472                r.left_top(),
2473                r.right_top(),
2474                r.right_bottom(),
2475                r.left_bottom(),
2476                r.left_top(),
2477            ];
2478            painter.extend(Shape::dashed_line(
2479                &pts,
2480                Stroke::new(width, c),
2481                4.0 * z,
2482                3.0 * z,
2483            ));
2484        }
2485    }
