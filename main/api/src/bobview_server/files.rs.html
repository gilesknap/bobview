1//! Serve files from the served directories without following symbolic
2//! links out of them (the default; see --follow-symlinks).
3//!
4//! Files are streamed, with `Last-Modified` and `ETag` for revalidation
5//! (`If-None-Match`, `If-Modified-Since`) and single byte ranges, as
6//! tower-http's `ServeDir` does.

23const CHUNK: usize = 64 * 1024;

25/// The directories files are served from, in order, each with its real
26/// path (symbolic links resolved).
27pub struct Roots(Vec<(PathBuf, Option<PathBuf>)>);
28
29impl Roots {
30    pub fn new(dirs: &[PathBuf]) -> Self {
31        Self(
32            dirs.iter()
33                .map(|d| (d.clone(), std::fs::canonicalize(d).ok()))
34                .collect(),
35        )
36    }
37
38    /// Open the file a request path names, from the first root that has it
39    /// (`index.html` for a directory), unless it is really outside that
40    /// root: a symbolic link may lead anywhere on the machine. The file is
41    /// checked after it is opened, and served from what was opened, so a
42    /// link swapped in meanwhile is caught too.
43    fn open(&self, path: &str) -> Found {
44        let decoded = percent_encoding::percent_decode_str(path).decode_utf8_lossy();
45        let mut rel = PathBuf::new();
46        for segment in decoded.split('/') {
47            match segment {
48                "" | "." => {}
49                ".." => return Found::Missing,
50                s if s.contains('\\') || s.contains('\0') => return Found::Missing,
51                s => rel.push(s),
52            }
53        }
54        for (dir, real_dir) in &self.0 {
55            let inside = |real: &Path| real_dir.as_ref().is_some_and(|d| real.starts_with(d));
56            let mut file = dir.join(&rel);
57            match std::fs::metadata(&file) {
58                Ok(m) if m.is_dir() => {
59                    if !path.ends_with('/') {
60                        // Not redirected if it leads out, which would show
61                        // that the name is there.
62                        if !std::fs::canonicalize(&file).is_ok_and(|r| inside(&r)) {
63                            return Found::Refused;
64                        }
65                        // One path, not `//host/`, which a browser would
66                        // take for another server.
67                        return Found::Redirect(format!("/{}/", path.trim_start_matches('/')));
68                    }
69                    file.push("index.html");
70                }
71                Ok(m) if m.is_file() => {}
72                // A FIFO or device: opening one may block, or do something.
73                Ok(_) => return Found::Missing,
74                // Not here (or a dangling link): look in the next.
75                Err(_) => continue,
76            }
77            // Without blocking, should a FIFO be swapped in meanwhile; the
78            // check below refuses anything but a regular file.
79            let opened = {
80                use std::os::unix::fs::OpenOptionsExt;
81                std::fs::OpenOptions::new()
82                    .read(true)
83                    .custom_flags(libc::O_NONBLOCK)
84                    .open(&file)
85            };
86            let Ok(opened) = opened else {
87                continue;
88            };
89            let checked = || -> Option<Metadata> {
90                use std::os::unix::fs::MetadataExt;
91                let real = std::fs::canonicalize(&file).ok()?;
92                let (now, then) = (std::fs::metadata(&real).ok()?, opened.metadata().ok()?);
93                // The real path is inside, and is the file that was opened.
94                (inside(&real)
95                    && then.is_file()
96                    && (now.dev(), now.ino()) == (then.dev(), then.ino()))
97                    .then_some(then)
98            };
99            return match checked() {
100                Some(meta) => Found::File(opened, file, meta),
101                None => Found::Refused,
102            };
103        }
104        Found::Missing
105    }
106}

108/// What a request path names.
109enum Found {
110    File(File, PathBuf, Metadata),
111    /// A directory, named without its trailing `/`.
112    Redirect(String),
113    Missing,
114    /// Something outside the served directories.
115    Refused,
116}
117
118/// Serve a file from `roots`, refusing those reached through a symbolic
119/// link that leads out of them (see --follow-symlinks).
120pub async fn serve(roots: Arc<Roots>, request: Request) -> Response {
121    let head = match *request.method() {
122        Method::GET => false,
123        Method::HEAD => true,
124        _ => return StatusCode::METHOD_NOT_ALLOWED.into_response(),
125    };
126    let path = request.uri().path().to_string();
127    let found = tokio::task::spawn_blocking(move || roots.open(&path)).await;
128    match found {
129        Ok(Found::File(file, name, meta)) => respond(request.headers(), head, file, &name, &meta),
130        Ok(Found::Redirect(to)) => {
131            let to = match request.uri().query() {
132                Some(q) => format!("{to}?{q}"),
133                None => to,
134            };
135            match HeaderValue::from_str(&to) {
136                Ok(to) => (StatusCode::TEMPORARY_REDIRECT, [(LOCATION, to)]).into_response(),
137                Err(_) => StatusCode::NOT_FOUND.into_response(),
138            }
139        }
140        Ok(Found::Refused) => {
141            log::warn!(
142                "refused {}: a link out of the served directories (see --follow-symlinks)",
143                request.uri().path()
144            );
145            StatusCode::NOT_FOUND.into_response()
146        }
147        _ => StatusCode::NOT_FOUND.into_response(),
148    }
149}

151/// The validators of a file: its `ETag` and its modification time.
152struct Validators {
153    etag: String,
154    modified: Option<SystemTime>,
155}
156
157impl Validators {
158    fn of(meta: &Metadata) -> Self {
159        use std::os::unix::fs::MetadataExt;
160        let modified = meta.modified().ok();
161        let since = modified.and_then(|m| m.duration_since(UNIX_EPOCH).ok());
162        let nanos = since.unwrap_or_default().as_nanos();
163        Self {
164            etag: format!("\"{:x}-{:x}-{nanos:x}\"", meta.ino(), meta.len()),
165            modified,
166        }
167    }
168
169    fn last_modified(&self) -> Option<String> {
170        self.modified.map(httpdate::fmt_http_date)
171    }
172
173    /// Whether the client's copy is current: `If-None-Match` if given,
174    /// else `If-Modified-Since`, to the second.
175    fn not_modified(&self, headers: &HeaderMap) -> bool {
176        if let Some(tags) = headers.get(IF_NONE_MATCH) {
177            let Ok(tags) = tags.to_str() else {
178                return false;
179            };
180            // Compared weakly, as RFC 9110 has for If-None-Match.
181            return tags
182                .split(',')
183                .map(str::trim)
184                .any(|t| t == "*" || t.strip_prefix("W/").unwrap_or(t) == self.etag);
185        }
186        let since = headers
187            .get(IF_MODIFIED_SINCE)
188            .and_then(|v| v.to_str().ok())
189            .and_then(|v| httpdate::parse_http_date(v).ok());
190        match (since, self.modified) {
191            (Some(since), Some(modified)) => secs(modified) <= secs(since),
192            _ => false,
193        }
194    }
195
196    /// Whether `If-Range`, if given, names this version of the file.
197    fn same(&self, headers: &HeaderMap) -> bool {
198        match headers.get(IF_RANGE).map(HeaderValue::to_str) {
199            None => true,
200            Some(Ok(v)) if v.starts_with('"') => v == self.etag,
201            Some(Ok(v)) => self.last_modified().is_some_and(|lm| lm == v),
202            Some(Err(_)) => false,
203        }
204    }
205}
206
207fn secs(t: SystemTime) -> u64 {
208    t.duration_since(UNIX_EPOCH).unwrap_or_default().as_secs()
209}

211/// The part of a file of `len` bytes that a `Range` header asks for.
212#[derive(Debug, PartialEq)]
213enum Part {
214    Whole,
215    /// From the first byte to the last, inclusive.
216    Bytes(u64, u64),
217    Unsatisfiable,
218}
219
220/// Read a `Range` header. Only one range of bytes is served; anything
221/// else, several ranges or one that does not parse, gets the whole file.
222fn part(range: Option<&str>, len: u64) -> Part {
223    let Some(spec) = range.and_then(|r| r.trim().strip_prefix("bytes=")) else {
224        return Part::Whole;
225    };
226    let Some((first, last)) = spec.split_once('-') else {
227        return Part::Whole;
228    };
229    let (first, last) = (first.trim(), last.trim());
230    if spec.contains(',') {
231        return Part::Whole;
232    }
233    if first.is_empty() {
234        // The last `n` bytes.
235        return match last.parse::<u64>() {
236            Ok(0) => Part::Unsatisfiable,
237            Ok(_) if len == 0 => Part::Unsatisfiable,
238            Ok(n) => Part::Bytes(len.saturating_sub(n), len - 1),
239            Err(_) => Part::Whole,
240        };
241    }
242    let Ok(first) = first.parse::<u64>() else {
243        return Part::Whole;
244    };
245    let last = match last {
246        "" => u64::MAX,
247        l => match l.parse::<u64>() {
248            Ok(l) if l >= first => l,
249            _ => return Part::Whole,
250        },
251    };
252    if first >= len {
253        return Part::Unsatisfiable;
254    }
255    Part::Bytes(first, last.min(len - 1))
256}

258/// The response for an opened file: 304, 206, 416 or 200, streamed.
259fn respond(headers: &HeaderMap, head: bool, file: File, name: &Path, meta: &Metadata) -> Response {
260    let len = meta.len();
261    let valid = Validators::of(meta);
262    let mut out = HeaderMap::new();
263    let mime = mime_guess::from_path(name).first_or_octet_stream();
264    out.insert(
265        CONTENT_TYPE,
266        HeaderValue::from_str(mime.as_ref())
267            .unwrap_or(HeaderValue::from_static("application/octet-stream")),
268    );
269    out.insert(ACCEPT_RANGES, HeaderValue::from_static("bytes"));
270    if let Ok(etag) = HeaderValue::from_str(&valid.etag) {
271        out.insert(ETAG, etag);
272    }
273    if let Some(lm) = valid
274        .last_modified()
275        .and_then(|v| HeaderValue::from_str(&v).ok())
276    {
277        out.insert(LAST_MODIFIED, lm);
278    }
279    if valid.not_modified(headers) {
280        out.remove(CONTENT_TYPE);
281        return (StatusCode::NOT_MODIFIED, out).into_response();
282    }
283    let range = headers.get(RANGE).and_then(|r| r.to_str().ok());
284    let wanted = if valid.same(headers) {
285        part(range, len)
286    } else {
287        Part::Whole
288    };
289    let (status, first, count) = match wanted {
290        Part::Whole => (StatusCode::OK, 0, len),
291        Part::Bytes(first, last) => {
292            let range = format!("bytes {first}-{last}/{len}");
293            out.insert(CONTENT_RANGE, HeaderValue::from_str(&range).expect("ASCII"));
294            (StatusCode::PARTIAL_CONTENT, first, last - first + 1)
295        }
296        Part::Unsatisfiable => {
297            let range = format!("bytes */{len}");
298            out.insert(CONTENT_RANGE, HeaderValue::from_str(&range).expect("ASCII"));
299            return (StatusCode::RANGE_NOT_SATISFIABLE, out).into_response();
300        }
301    };
302    out.insert(CONTENT_LENGTH, HeaderValue::from(count));
303    if head {
304        return (status, out).into_response();
305    }
306    let body = match stream(file, first, count) {
307        Ok(body) => body,
308        Err(_) => return StatusCode::NOT_FOUND.into_response(),
309    };
310    (status, out, body).into_response()
311}

313/// `count` bytes of `file` from `first`, read a chunk at a time as the
314/// client takes them, so that memory does not grow with the file's size.
315fn stream(mut file: File, first: u64, count: u64) -> std::io::Result<Body> {
316    use std::io::{Seek, SeekFrom};
317    use tokio::io::AsyncReadExt;
318    if first > 0 {
319        file.seek(SeekFrom::Start(first))?;
320    }
321    let file = tokio::fs::File::from_std(file).take(count);
322    Ok(Body::from_stream(
323        tokio_util::io::ReaderStream::with_capacity(file, CHUNK),
324    ))
325}

332    #[test]
333    fn ranges() {
334        assert_eq!(part(None, 10), Part::Whole);
335        assert_eq!(part(Some("bytes=0-3"), 10), Part::Bytes(0, 3));
336        assert_eq!(part(Some("bytes=5-"), 10), Part::Bytes(5, 9));
337        assert_eq!(part(Some("bytes=5-100"), 10), Part::Bytes(5, 9));
338        assert_eq!(part(Some("bytes=-3"), 10), Part::Bytes(7, 9));
339        assert_eq!(part(Some("bytes=-30"), 10), Part::Bytes(0, 9));
340        assert_eq!(part(Some("bytes=10-"), 10), Part::Unsatisfiable);
341        assert_eq!(part(Some("bytes=-0"), 10), Part::Unsatisfiable);
342        assert_eq!(part(Some("bytes=-3"), 0), Part::Unsatisfiable);
343        // Served whole: several ranges, other units, nonsense.
344        assert_eq!(part(Some("bytes=0-1,4-5"), 10), Part::Whole);
345        assert_eq!(part(Some("items=0-1"), 10), Part::Whole);
346        assert_eq!(part(Some("bytes=3-1"), 10), Part::Whole);
347        assert_eq!(part(Some("bytes=x-"), 10), Part::Whole);
348    }

350    #[test]
351    fn links_out_of_the_root_are_refused() {
352        let base = std::env::temp_dir().join(format!("bobview-confine-{}", std::process::id()));
353        let opi = base.join("opi");
354        let web = base.join("web");
355        let outside = base.join("outside");
356        for d in [&opi, &web, &outside, &opi.join("sub")] {
357            std::fs::create_dir_all(d).unwrap();
358        }
359        std::fs::write(opi.join("a.bob"), "a").unwrap();
360        std::fs::write(opi.join("sub/b.bob"), "b").unwrap();
361        std::fs::write(web.join("index.html"), "w").unwrap();
362        std::fs::write(outside.join("secret"), "s").unwrap();
363        let link = |target: &PathBuf, name: &str| {
364            std::os::unix::fs::symlink(target, opi.join(name)).unwrap();
365        };
366        link(&outside.join("secret"), "secret.bob");
367        link(&outside, "out");
368        // Links within the root are fine.
369        link(&opi.join("sub"), "inside");
370
371        let roots = Roots::new(&[opi.clone(), web.clone()]);
372        let read = |path: &str| match roots.open(path) {
373            Found::File(mut f, ..) => {
374                let mut text = String::new();
375                std::io::Read::read_to_string(&mut f, &mut text).unwrap();
376                text
377            }
378            Found::Redirect(to) => format!("-> {to}"),
379            Found::Missing => "missing".into(),
380            Found::Refused => "refused".into(),
381        };
382        assert_eq!(read("/a.bob"), "a");
383        assert_eq!(read("/sub/b.bob"), "b");
384        assert_eq!(read("/inside/b.bob"), "b");
385        assert_eq!(read("/%73ub/b.bob"), "b");
386        assert_eq!(read("/index.html"), "w");
387        // The web build's index, as opi has none.
388        assert_eq!(read("/"), "w");
389        assert_eq!(read("/sub"), "-> /sub/");
390        assert_eq!(read("/missing.bob"), "missing");
391        assert_eq!(read("/../opi/a.bob"), "missing");
392        assert_eq!(read("/secret.bob"), "refused");
393        assert_eq!(read("/out/secret"), "refused");
394        assert_eq!(read("/%73ecret.bob"), "refused");
395        assert_eq!(read("/out%2Fsecret"), "refused");
396        // A FIFO, here or out of the root, is not opened: that would block.
397        let fifo = |p: &PathBuf| {
398            let ok = std::process::Command::new("mkfifo")
399                .arg(p)
400                .status()
401                .unwrap();
402            assert!(ok.success());
403        };
404        fifo(&opi.join("f.bob"));
405        fifo(&outside.join("f"));
406        link(&outside.join("f"), "outf.bob");
407        assert_eq!(read("/f.bob"), "missing");
408        assert_eq!(read("/outf.bob"), "missing");
409        // A directory out is refused before it would be redirected.
410        assert_eq!(read("/out"), "refused");
411        assert_eq!(read("//sub"), "-> /sub/");
412        std::fs::remove_dir_all(&base).unwrap();
413    }

415    #[tokio::test]
416    async fn files_are_revalidated_and_served_in_parts() {
417        use axum::http::header::*;
418        use axum::http::{HeaderMap, HeaderValue, StatusCode};
419        let dir = std::env::temp_dir().join(format!("bobview-files-{}", std::process::id()));
420        std::fs::create_dir_all(&dir).unwrap();
421        let path = dir.join("a.wasm");
422        std::fs::write(&path, "0123456789").unwrap();
423        let get = async |head: bool, pairs: &[(HeaderName, String)]| {
424            let mut headers = HeaderMap::new();
425            for (k, v) in pairs {
426                headers.insert(k, HeaderValue::from_str(v).unwrap());
427            }
428            let file = std::fs::File::open(&path).unwrap();
429            let meta = file.metadata().unwrap();
430            let r = super::respond(&headers, head, file, &path, &meta);
431            let (parts, body) = r.into_parts();
432            let body = axum::body::to_bytes(body, 1 << 20).await.unwrap();
433            let text = String::from_utf8(body.to_vec()).unwrap();
434            (parts.status, parts.headers, text)
435        };
436        let (status, h, body) = get(false, &[]).await;
437        assert_eq!((status, body.as_str()), (StatusCode::OK, "0123456789"));
438        assert_eq!(h[CONTENT_TYPE], "application/wasm");
439        assert_eq!(h[CONTENT_LENGTH], "10");
440        assert_eq!(h[ACCEPT_RANGES], "bytes");
441        let etag = h[ETAG].to_str().unwrap().to_string();
442        let modified = h[LAST_MODIFIED].to_str().unwrap().to_string();
443
444        let (status, _, body) = get(false, &[(IF_MODIFIED_SINCE, modified.clone())]).await;
445        assert_eq!((status, body.as_str()), (StatusCode::NOT_MODIFIED, ""));
446        let (status, _, _) = get(false, &[(IF_NONE_MATCH, etag.clone())]).await;
447        assert_eq!(status, StatusCode::NOT_MODIFIED);
448        let (status, _, _) = get(false, &[(IF_NONE_MATCH, format!("W/{etag}"))]).await;
449        assert_eq!(status, StatusCode::NOT_MODIFIED);
450        // A different tag wins over a date that would match.
451        let other = [
452            (IF_NONE_MATCH, "\"other\"".to_string()),
453            (IF_MODIFIED_SINCE, modified.clone()),
454        ];
455        assert_eq!(get(false, &other).await.0, StatusCode::OK);
456        let old = "Thu, 01 Jan 1970 00:00:00 GMT".to_string();
457        assert_eq!(
458            get(false, &[(IF_MODIFIED_SINCE, old)]).await.0,
459            StatusCode::OK
460        );
461
462        let (status, h, body) = get(false, &[(RANGE, "bytes=2-4".into())]).await;
463        assert_eq!(
464            (status, body.as_str()),
465            (StatusCode::PARTIAL_CONTENT, "234")
466        );
467        assert_eq!(h[CONTENT_RANGE], "bytes 2-4/10");
468        assert_eq!(h[CONTENT_LENGTH], "3");
469        let (_, _, body) = get(false, &[(RANGE, "bytes=-2".into())]).await;
470        assert_eq!(body, "89");
471        let (status, h, _) = get(false, &[(RANGE, "bytes=20-".into())]).await;
472        assert_eq!(status, StatusCode::RANGE_NOT_SATISFIABLE);
473        assert_eq!(h[CONTENT_RANGE], "bytes */10");
474        // If-Range: the range only if the file is the one named.
475        let same = [(RANGE, "bytes=0-0".into()), (IF_RANGE, etag.clone())];
476        assert_eq!(get(false, &same).await.2, "0");
477        let same = [(RANGE, "bytes=0-0".into()), (IF_RANGE, modified.clone())];
478        assert_eq!(get(false, &same).await.2, "0");
479        let changed = [(RANGE, "bytes=0-0".into()), (IF_RANGE, "\"old\"".into())];
480        assert_eq!(get(false, &changed).await.2, "0123456789");
481
482        // HEAD: the headers, with no body read.
483        let (status, h, body) = get(true, &[]).await;
484        assert_eq!((status, body.as_str()), (StatusCode::OK, ""));
485        assert_eq!(h[CONTENT_LENGTH], "10");
486        std::fs::remove_dir_all(&dir).unwrap();
487    }
