53/// A lean viewer for Phoebus .bob displays (and CS-Studio BOY .opi).
54#[cfg(not(target_arch = "wasm32"))]
55#[derive(clap::Parser, Debug)]
56#[command(
57    name = "bobview",
58    version,
59    after_help = AFTER_HELP,
60    args_override_self = true,
61    max_term_width = 100
62)]
63struct Cli {
64    /// display.bob, display.opi or http(s)://host/display.bob
65    #[arg(value_name = "DISPLAY")]
66    display: Option<String>,
67    /// Macros for the display
68    #[arg(short, long, value_name = "P=X,R=Y")]
69    macros: Option<String>,
70    /// Open a display file through the linked folders of the Eclipse project
71    /// in DIR (its .project, or for a BOY .opi screen the links of a DLS
72    /// runcss.sh in DIR), so that links between the project's modules work
73    #[arg(short, long, value_name = "DIR")]
74    project: Option<PathBuf>,
75    /// Fixed zoom (0.2 to 5) instead of fit-to-window
76    #[arg(short, long, value_name = "FACTOR", value_parser = parse_zoom)]
77    zoom: Option<f32>,
78    /// Start full screen
79    #[arg(short, long)]
80    fullscreen: bool,
81    /// No longer supported: text entries take the keyboard
82    #[arg(short, long, hide = true)]
83    keypad: bool,
84    /// No longer needed: pin mode replaced it
85    #[arg(long, hide = true)]
86    multi_window: bool,
87    /// Refuse every write to a PV (except loc:// ones, which stay in
88    /// bobview): controls are disabled and a write action says why it did
89    /// nothing
90    #[arg(long)]
91    read_only: bool,
92    /// Restore the saved window layout NAME
93    #[arg(long, value_name = "NAME")]
94    layout: Option<String>,
95    /// Don't restore the default layout
96    #[arg(long)]
97    no_layout: bool,
98    /// Size of the toolbar and menus relative to the desktop's scale
99    /// (default auto: smaller on a small high-DPI screen; also set by
100    /// BOBVIEW_UI_SCALE)
101    #[arg(long, value_name = "F|auto", value_parser = parse_ui_scale)]
102    ui_scale: Option<chrome::UiScale>,
103    /// Render to a PNG without a window and exit
104    #[arg(long, value_name = "OUT.png")]
105    snapshot: Option<PathBuf>,
106    /// Snapshot size in pixels; 'auto' starts at the window's default size
107    /// and resizes it as a window manager would, on a monitor of 1920x1080
108    /// points
109    #[arg(long, value_name = "WxH", default_value = "1280x800", value_parser = parse_size)]
110    size: Size,
111    /// Snapshot as on a screen with this desktop scale, e.g. 2.75 for a
112    /// phone; --click positions are in pixels
113    #[arg(long, value_name = "F", default_value_t = 1.0, value_parser = parse_scale_factor)]
114    scale_factor: f32,
115    /// Time for PVs to connect before a snapshot
116    #[arg(long, value_name = "SECONDS", default_value_t = 2.0, value_parser = parse_seconds)]
117    wait: f64,
118    /// Click here before the snapshot (repeatable)
119    #[arg(long, value_name = "X,Y", value_parser = parse_point)]
120    click: Vec<egui::Pos2>,
121    /// Right-click here before the snapshot (repeatable)
122    #[arg(long, value_name = "X,Y", value_parser = parse_point)]
123    right_click: Vec<egui::Pos2>,
124    /// Time after the last click before the snapshot
125    #[arg(long, value_name = "SECONDS", default_value_t = 1.0, value_parser = parse_seconds)]
126    settle: f64,
127}

134#[cfg(not(target_arch = "wasm32"))]
135impl Cli {
136    /// Parse the command line, exiting with a message if it is wrong.
137    fn from_env() -> (Self, Vec<(egui::Pos2, egui::PointerButton)>) {
138        use clap::{CommandFactory, FromArgMatches};
139        let matches = Self::command().get_matches();
140        let cli = Self::from_arg_matches(&matches).unwrap_or_else(|e| e.exit());
141        let clicks = cli.clicks(&matches);
142        (cli, clicks)
143    }
144
145    /// Both kinds of click, in the order given.
146    fn clicks(&self, matches: &clap::ArgMatches) -> Vec<(egui::Pos2, egui::PointerButton)> {
147        let at = |id| matches.indices_of(id).into_iter().flatten();
148        let mut clicks: Vec<_> = (at("click").zip(&self.click))
149            .map(|(i, p)| (i, *p, egui::PointerButton::Primary))
150            .chain(
151                at("right_click")
152                    .zip(&self.right_click)
153                    .map(|(i, p)| (i, *p, egui::PointerButton::Secondary)),
154            )
155            .collect();
156        clicks.sort_by_key(|(i, ..)| *i);
157        clicks.into_iter().map(|(_, p, b)| (p, b)).collect()
158    }
159}

161/// A zoom factor in the range the view allows.
162fn parse_zoom(text: &str) -> Result<f32, String> {
163    text.trim()
164        .parse::<f32>()
165        .ok()
166        .filter(|z| (0.2..=5.0).contains(z))
167        .ok_or_else(|| format!("expected a factor from 0.2 to 5, not {text:?}"))
168}

170#[cfg(not(target_arch = "wasm32"))]
171fn parse_ui_scale(text: &str) -> Result<chrome::UiScale, String> {
172    chrome::UiScale::parse(text).ok_or_else(|| format!("expected a number or auto, not {text:?}"))
173}

175#[cfg(not(target_arch = "wasm32"))]
176fn parse_size(text: &str) -> Result<Size, String> {
177    if text.trim() == "auto" {
178        return Ok(Size(None));
179    }
180    text.split_once('x')
181        .and_then(|(w, h)| Some([w.trim().parse().ok()?, h.trim().parse().ok()?]))
182        .filter(|[w, h]: &[u32; 2]| *w > 0 && *h > 0)
183        .map(|wh| Size(Some(wh)))
184        .ok_or_else(|| format!("expected WIDTHxHEIGHT in pixels or auto, not {text:?}"))
185}
186
187#[cfg(not(target_arch = "wasm32"))]
188fn parse_scale_factor(text: &str) -> Result<f32, String> {
189    text.trim()
190        .parse::<f32>()
191        .ok()
192        .filter(|s| s.is_finite() && *s > 0.0)
193        .ok_or_else(|| format!("expected a scale above 0, not {text:?}"))
194}

196/// A time in seconds: a number, not negative (nor too big for a Duration).
197#[cfg(not(target_arch = "wasm32"))]
198fn parse_seconds(text: &str) -> Result<f64, String> {
199    text.trim()
200        .parse::<f64>()
201        .ok()
202        .filter(|s| (0.0..1e9).contains(s))
203        .ok_or_else(|| format!("expected a number of seconds, not {text:?}"))
204}
205
206#[cfg(not(target_arch = "wasm32"))]
207fn parse_point(text: &str) -> Result<egui::Pos2, String> {
208    text.split_once(',')
209        .and_then(|(x, y)| Some(egui::pos2(x.trim().parse().ok()?, y.trim().parse().ok()?)))
210        .filter(|p| p.x.is_finite() && p.y.is_finite())
211        .ok_or_else(|| format!("expected X,Y in pixels, not {text:?}"))
212}

240fn main() -> std::process::ExitCode {

243    match run(cli, clicks) {
244        Ok(()) => std::process::ExitCode::SUCCESS,

252#[cfg(not(target_arch = "wasm32"))]
253fn run(cli: Cli, clicks: Vec<(egui::Pos2, egui::PointerButton)>) -> Result<(), Failure> {
254    if cli.keypad {
255        eprintln!("--keypad is no longer supported: ignored");
256    }
257    if cli.multi_window {
258        // Pin mode replaced multi-window mode: shift+click follows a link
259        // as it did.
260        eprintln!(
261            "--multi-window is no longer needed: links from the pinned main \
262             window open in another window, shift+click in a new one"
263        );
264    }
265    let mut file = cli.display.as_deref().map(fetch::user_named);
266    let mut macros = parse_macros(cli.macros.as_deref().unwrap_or_default());
267    let ui_scale = cli.ui_scale.unwrap_or_else(chrome::UiScale::from_env);
268    let is_opi = |p: &std::path::Path| p.extension().is_some_and(|x| x.eq_ignore_ascii_case("opi"));
269    if let (Some(dir), Some(Source::File(path))) = (&cli.project, &file) {
270        let Some(project) = bob::project::project_dir(dir) else {
271            return Err(Failure::usage(format!(
272                "{} has no .project, nor a {} with links",
273                dir.display(),
274                bob::project::LAUNCHER
275            )));
276        };
277        // A launcher's links are CS-Studio's, for BOY screens only.
278        if bob::project::through_launcher(&project) && !is_opi(path) {
279            return Err(Failure::usage(format!(
280                "{}: {} links apply only to BOY .opi screens",
281                path.display(),
282                bob::project::LAUNCHER
283            )));
284        }
285        match bob::project::virtual_path(&project, path) {
286            Some(p) => file = Some(Source::File(p)),
287            None => {
288                return Err(Failure::usage(format!(
289                    "{} is not in a linked folder of {}",
290                    path.display(),
291                    dir.display()
292                )));
293            }
294        }
295    } else if let Some(Source::File(path)) = &file
296        && is_opi(path)
297        && let Some(p) = bob::project::launcher_path(path)
298    {
299        // A DLS BOY screen beside its runcss.sh: through the project that
300        // launcher would give CS-Studio.
301        log::info!("{}: opening as {}", path.display(), p.display());
302        file = Some(Source::File(p));
303    }
304    // An .opi opened directly (in an Eclipse project) gets the macros its
305    // DLS launcher script would give it; those given here override them.
306    if let Some(Source::File(path)) = &file
307        && is_opi(path)
308        && let Some(defaults) = launch::defaults(path)
309    {
310        macros = defaults.with(&macros);
311    }
312    // The named layout, else the default one unless a display was named.
313    let layout = if cli.snapshot.is_some() {
314        None
315    } else {
316        let default = file.is_none() && !cli.no_layout && layout::exists(layout::DEFAULT);
317        layout::startup(cli.layout, default, layout::load).map_err(Failure::usage)?
318    };
319    // A layout's main window has a display of its own.
320    let layout_has_main = layout
321        .as_ref()
322        .is_some_and(|(l, _)| l.windows.iter().any(|w| w.main));
323    let start = file
324        .or_else(|| {
325            (!layout_has_main)
326                .then(default_display)
327                .flatten()
328                .map(Source::File)
329        })
330        .map(|source| app::Location { source, macros });
331
332    if let Some(out) = cli.snapshot {
333        let start = start.ok_or_else(|| Failure::usage("--snapshot needs a display file"))?;
334        let opts = snapshot::Options {
335            size: cli.size.0,
336            scale_factor: cli.scale_factor,
337            ui_scale,
338            zoom: cli.zoom,
339            wait: std::time::Duration::from_secs_f64(cli.wait),
340            clicks,
341            settle: std::time::Duration::from_secs_f64(cli.settle),
342            read_only: cli.read_only,
343        };
344        return snapshot::run(start, &opts, &out)
345            .map_err(|e| Failure::failed(format!("snapshot failed: {e}")));
346    }
347
348    let options = eframe::NativeOptions {
349        viewport: egui::ViewportBuilder::default()
350            .with_title("bobview")
351            .with_app_id("bobview")
352            .with_inner_size(chrome::DEFAULT_WINDOW)
353            .with_fullscreen(cli.fullscreen),
354        // No vsync: each window's buffer swap would wait for a screen
355        // refresh, and eframe draws the windows one after another on one
356        // thread, so windows repainting together would take a refresh each
357        // (~120 ms for 7, when they were all drawn in the main one's frame).
358        // We only repaint on change, so this doesn't spin.
359        glow_options: eframe::egui_glow::GlowConfiguration {
360            vsync: false,
361            ..Default::default()
362        },
363        ..Default::default()
364    };
365    let (zoom, read_only) = (cli.zoom, cli.read_only);
366    eframe::run_native(
367        "bobview",
368        options,
369        Box::new(move |cc| {
370            let mut app = app::BobApp::new(&cc.egui_ctx, start, zoom, ui_scale);
371            app.set_read_only(read_only);
372            if let Some((layout, name)) = layout {
373                app.restore_layout(layout, Some(name));
374            }
375            Ok(Box::new(app::App::new(app)))
376        }),
377    )
378    .map_err(|e| Failure::failed(e.to_string()))
379}

381/// Parse `P=X,R=Y` into macros; `\,` and `\\` in a value stand for `,`
382/// and `\`.
383fn parse_macros(text: &str) -> Macros {
384    // Split on unescaped commas, keeping escapes for now.
385    let mut pairs = vec![String::new()];
386    let mut chars = text.chars();
387    while let Some(c) = chars.next() {
388        match c {
389            ',' => pairs.push(String::new()),
390            '\\' => {
391                let last = pairs.last_mut().expect("a pair");
392                last.push(c);
393                last.extend(chars.next());
394            }
395            _ => pairs.last_mut().expect("a pair").push(c),
396        }
397    }
398    Macros::from_pairs(pairs.iter().filter_map(|kv| {
399        let (k, v) = kv.split_once('=')?;
400        Some((k.trim().to_string(), unescape_macro(v.trim())))
401    }))
402}
403
404fn unescape_macro(text: &str) -> String {
405    let mut out = String::with_capacity(text.len());
406    let mut chars = text.chars();
407    while let Some(c) = chars.next() {
408        match (c, chars.clone().next()) {
409            ('\\', Some(next @ (',' | '\\'))) => {
410                out.push(next);
411                chars.next();
412            }
413            _ => out.push(c),
414        }
415    }
416    out
417}

419/// The browser build: the page's query string names the display, its
420/// macros and the PV Web Socket, e.g.
421/// `?display=https://host/opi/index.bob&macros=P=X&pvws=pvws.host:443`
422/// (and `zoom=1.5`, `token=SECRET` for a server with `--write`).
423/// A relative display is found next to the page; the default is `index.bob`.
424/// With a token, displays come only from the page's server, and writes
425/// carry the token only to a PVWS there.
426#[cfg(target_arch = "wasm32")]
427fn main() {
428    use wasm_bindgen::JsCast;
429
430    init_logging();
431    let window = web_sys::window().expect("a browser window");
432    let location = window.location();
433    let query = web_sys::UrlSearchParams::new_with_str(&location.search().unwrap_or_default())
434        .expect("a query string");
435    let page = Source::Url(location.href().unwrap_or_default());
436    let display = query.get("display").unwrap_or_else(|| "index.bob".into());
437    let start = app::Location {
438        source: page.resolve(&display),
439        macros: parse_macros(&query.get("macros").unwrap_or_default()),
440    };
441    if let Some(pvws) = query.get("pvws") {
442        pv::set_pvws(&pvws);
443    }
444    if let Some(token) = query.get("token") {
445        // The token goes only to the page's own server, and only its
446        // displays may use it.
447        pv::set_write_token(&token);
448        fetch::same_origin_only();
449    }
450    let zoom = query.get("zoom").and_then(|z| {
451        parse_zoom(&z)
452            .inspect_err(|e| log::warn!("zoom: {e}; fitting to the window"))
453            .ok()
454    });
455
456    let canvas = window
457        .document()
458        .and_then(|d| d.get_element_by_id("bobview"))
459        .and_then(|c| c.dyn_into::<web_sys::HtmlCanvasElement>().ok())
460        .expect("a <canvas id=\"bobview\"> on the page");
461    wasm_bindgen_futures::spawn_local(async move {
462        let started = eframe::WebRunner::new()
463            .start(
464                canvas,
465                eframe::WebOptions::default(),
466                Box::new(move |cc| {
467                    let ui_scale = chrome::UiScale::from_env();
468                    let app = app::BobApp::new(&cc.egui_ctx, Some(start), zoom, ui_scale);
469                    Ok(Box::new(app::App::new(app)))
470                }),
471            )
472            .await;
473        if let Err(e) = started {
474            log::error!("cannot start: {e:?}");
475        }
476    });
477}

479/// The bundled index display, next to the binary or in ~/.local/share.
480#[cfg(not(target_arch = "wasm32"))]
481fn default_display() -> Option<PathBuf> {
482    let mut candidates = Vec::new();
483    if let Ok(exe) = std::env::current_exe()
484        && let Some(prefix) = exe.parent().and_then(|bin| bin.parent())
485    {
486        candidates.push(prefix.join("share/bobview/displays/index.bob"));
487        candidates.push(prefix.join("displays/index.bob"));
488    }
489    if let Some(home) = std::env::var_os("HOME") {
490        candidates.push(PathBuf::from(home).join(".local/share/bobview/displays/index.bob"));
491    }
492    candidates.into_iter().find(|p| p.exists())
493}

495struct Logger(log::LevelFilter);
496
497impl log::Log for Logger {
498    fn enabled(&self, m: &log::Metadata) -> bool {
499        // Library chatter (one line per CA channel connect) only when debugging.
500        let library = ["eframe", "egui", "epics", "winit", "glutin"]
501            .iter()
502            .any(|t| m.target().starts_with(t));
503        let limit = if library && self.0 < log::LevelFilter::Debug {
504            log::LevelFilter::Warn
505        } else {
506            self.0
507        };
508        m.level() <= limit
509    }
510
511    fn log(&self, r: &log::Record) {
512        if self.enabled(r.metadata()) {
513            #[cfg(not(target_arch = "wasm32"))]
514            eprintln!("{:<5} {}", r.level(), r.args());
515            #[cfg(target_arch = "wasm32")]
516            web_sys::console::log_1(&format!("{:<5} {}", r.level(), r.args()).into());
517        }
518    }
519
520    fn flush(&self) {}
521}
522
523fn init_logging() {
524    let level = match std::env::var("BOBVIEW_LOG").as_deref() {
525        Ok("debug") => log::LevelFilter::Debug,
526        Ok("trace") => log::LevelFilter::Trace,
527        Ok("warn") => log::LevelFilter::Warn,
528        Ok("error") => log::LevelFilter::Error,
529        _ => log::LevelFilter::Info,
530    };
531    let logger = Box::leak(Box::new(Logger(level)));
532    if log::set_logger(logger).is_ok() {
533        log::set_max_level(level);
534    }
535}
