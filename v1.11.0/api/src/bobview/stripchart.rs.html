21/// Samples kept per PV: at most this many (1.6 MB), and none older than a
22/// day.
23const MAX_SAMPLES: usize = 100_000;
24const MAX_AGE: f64 = 86_400.0;
25
26/// One PV's samples, in time order: (seconds since 1970, value), a NaN
27/// value marking where it disconnected.
28#[derive(Default)]
29pub struct Samples {
30    points: VecDeque<(f64, f64)>,
31    /// How many have ever been added, so a [`Thinned`] knows which are new.
32    pushed: u64,
33}
34
35impl Samples {
36    pub fn push(&mut self, t: f64, v: f64) {
37        // One gap is enough, and a trend starts with a value.
38        if v.is_nan() && self.points.back().is_none_or(|p| p.1.is_nan()) {
39            return;
40        }
41        // The clock may step back; the samples stay in order.
42        let t = self.points.back().map_or(t, |p| t.max(p.0));
43        self.points.push_back((t, v));
44        self.pushed += 1;
45        while self.points.len() > MAX_SAMPLES
46            || self.points.front().is_some_and(|p| p.0 < t - MAX_AGE)
47        {
48            self.points.pop_front();
49        }
50    }
51
52    /// The samples from `from` on, and the one before it, which holds at
53    /// `from`.
54    pub fn since(&self, from: f64) -> impl Iterator<Item = (f64, f64)> + '_ {
55        let i = self.points.partition_point(|p| p.0 < from);
56        self.points.range(i.saturating_sub(1)..).copied()
57    }
58
59    #[cfg(test)]
60    fn len(&self) -> usize {
61        self.points.len()
62    }
63}

65/// One PV's samples thinned for drawing: in each column of time `dt` wide
66/// (a pixel or less), the first, lowest, highest and last sample, which
67/// draw the same as all of them. Kept from frame to frame, taking in only
68/// the samples added since, so a day of fast samples is not copied out
69/// and stepped through on every frame.
70#[derive(Default)]
71struct Thinned {
72    dt: f64,
73    /// The start of the time it was built for.
74    from: f64,
75    /// [`Samples::pushed`] when it last took samples in.
76    seen: u64,
77    /// The finished columns' points, in time order, NaN marking gaps.
78    done: VecDeque<(f64, f64)>,
79    /// The column being filled.
80    open: Option<Column>,
81}
82
83/// A column's first, lowest, highest and last sample.
84#[derive(Clone, Copy)]
85struct Column {
86    index: i64,
87    first: (f64, f64),
88    low: (f64, f64),
89    high: (f64, f64),
90    last: (f64, f64),
91}
92
93impl Column {
94    /// Its points in time order, each once.
95    fn points(&self) -> impl Iterator<Item = (f64, f64)> {
96        let mut pts = [self.first, self.low, self.high, self.last];
97        pts.sort_by(|a, b| a.0.total_cmp(&b.0));
98        let mut n = 0;
99        for i in 0..pts.len() {
100            if n == 0 || pts[i] != pts[n - 1] {
101                pts[n] = pts[i];
102                n += 1;
103            }
104        }
105        pts.into_iter().take(n)
106    }
107}
108
109impl Thinned {
110    fn new(dt: f64, from: f64) -> Self {
111        Self {
112            dt,
113            from,
114            ..Default::default()
115        }
116    }
117
118    fn close(&mut self) {
119        if let Some(c) = self.open.take() {
120            self.done.extend(c.points());
121        }
122    }
123
124    fn add(&mut self, (t, v): (f64, f64)) {
125        if v.is_nan() {
126            // A gap ends the column's line.
127            self.close();
128            self.done.push_back((t, v));
129            return;
130        }
131        let index = (t / self.dt).floor() as i64;
132        match &mut self.open {
133            Some(c) if c.index == index => {
134                if v < c.low.1 {
135                    c.low = (t, v);
136                }
137                if v > c.high.1 {
138                    c.high = (t, v);
139                }
140                c.last = (t, v);
141            }
142            _ => {
143                self.close();
144                self.open = Some(Column {
145                    index,
146                    first: (t, v),
147                    low: (t, v),
148                    high: (t, v),
149                    last: (t, v),
150                });
151            }
152        }
153    }
154
155    /// Bring it up to date with `samples` for the time from `from` in
156    /// columns `dt` wide, rebuilding it only if the columns changed.
157    fn update(&mut self, samples: &Samples, from: f64, dt: f64) {
158        let new = samples.pushed - self.seen;
159        if self.dt != dt || from < self.from || new > samples.points.len() as u64 {
160            *self = Self::new(dt, from);
161            for p in samples.since(from) {
162                self.add(p);
163            }
164        } else {
165            let len = samples.points.len();
166            for p in samples.points.range(len - new as usize..) {
167                self.add(*p);
168            }
169        }
170        self.seen = samples.pushed;
171        // Forget what has scrolled off, but the point that holds at `from`.
172        let i = self.done.partition_point(|p| p.0 < from);
173        if i > 1 {
174            self.done.drain(..i - 1);
175        }
176    }
177
178    fn points(&self) -> impl Iterator<Item = (f64, f64)> + '_ {
179        self.done
180            .iter()
181            .copied()
182            .chain(self.open.iter().flat_map(Column::points))
183    }
184}

186/// The quickest a plot redraws for time moving on alone: a display redraws
187/// all of itself each frame, so a strip chart must not keep it at the
188/// 10 frames a second PV updates are held to. Trend windows drawn in the
189/// display's own viewport keep to it too (see [`crate::trend::Trend::ui`]).
190pub const MIN_SCROLL_REPAINT: f64 = 0.25;
191
192/// How long until a plot of `span` seconds over `width` points needs a
193/// redraw for time moving on: when it has scrolled a pixel, but not more
194/// often than [`MIN_SCROLL_REPAINT`] nor less than once a second.
195fn scroll_repaint(span: f64, width: f32) -> std::time::Duration {
196    let per_pixel = span / width.max(1.0) as f64;
197    std::time::Duration::from_secs_f64(per_pixel.clamp(MIN_SCROLL_REPAINT, 1.0))
198}

200/// The time now, in seconds since 1970.
201pub fn now() -> f64 {
202    SystemTime::now()
203        .duration_since(UNIX_EPOCH)
204        .map_or(0.0, |d| d.as_secs_f64())
205}

207/// What a PV update adds to its trend: its value, a gap if it disconnected,
208/// or nothing if it has no number (yet).
209fn sample(s: &PvState) -> Option<f64> {
210    if s.connected {
211        s.value.as_f64()
212    } else {
213        Some(f64::NAN)
214    }
215}

217/// A PV being trended: a sample on each update, at the time it arrived.
218pub struct Series {
219    pub pv: Pv,
220    /// Locked by the listener while the PV's state is locked, so never read
221    /// `pv.state()` while holding this lock.
222    samples: Arc<Mutex<Samples>>,
223    /// The samples as last drawn.
224    thinned: Mutex<Thinned>,
225    _listener: Listener,
226}
227
228impl Series {
229    pub fn new(pv: Pv) -> Self {
230        let samples = Arc::new(Mutex::new(Samples::default()));
231        let into = samples.clone();
232        let listener: Listener = Arc::new(move |s: &PvState| {
233            if let Some(v) = sample(s) {
234                into.lock().push(now(), v);
235            }
236        });
237        pv.listen(&listener);
238        Self {
239            pv,
240            samples,
241            thinned: Mutex::default(),
242            _listener: listener,
243        }
244    }
245
246    /// The points to draw from `from` to `to`, `dt` seconds a pixel: the
247    /// samples thinned to a few per pixel, with the last value held to `to`
248    /// while the PV is connected.
249    fn points(&self, from: f64, to: f64, dt: f64) -> Vec<(f64, f64)> {
250        // Copied out before the PV's state is read: an update holds that
251        // lock while it adds a sample.
252        let mut pts: Vec<(f64, f64)> = {
253            let mut thinned = self.thinned.lock();
254            thinned.update(&self.samples.lock(), from, dt);
255            thinned.points().collect()
256        };
257        if let Some(&(t, v)) = pts.last()
258            && v.is_finite()
259            && t < to
260            && self.pv.state().connected
261        {
262            pts.push((to, v));
263        }
264        pts
265    }
266}

288/// Plot `lines` over the `span` seconds up to `now`, with a time axis below
289/// and a value axis on the left.
290pub fn draw_plot(
291    painter: &egui::Painter,
292    rect: Rect,
293    lines: &[Line],
294    span: f64,
295    now: f64,
296    look: &Look,
297    z: f32,
298) {
299    let painter = &painter.with_clip_rect(painter.clip_rect().intersect(rect));
300    fill(painter, rect, look.back);
301    let (pad, tick) = (4.0 * z, 5.0 * z);
302    let from = now - span;
303    // Columns no wider than a pixel of the plot, which is narrower than `rect`.
304    let dt = span / rect.width().max(1.0) as f64;
305    let data: Vec<Vec<(f64, f64)>> = lines
306        .iter()
307        .map(|l| l.series.points(from, now, dt))
308        .collect();
309
310    let ok = |v: f64| v.is_finite() && (!look.log || v > 0.0);
311    let in_view = data
312        .iter()
313        .flat_map(|pts| pts.iter().filter(|p| p.0 >= from).map(|p| p.1));
314    let (lo, hi) = look
315        .range
316        .or_else(|| plot::autoscale(look.log, in_view, true))
317        .unwrap_or(if look.log { (1.0, 10.0) } else { (0.0, 1.0) });
318
319    let label_h = text_size(painter, "0", &look.font).y;
320    let bottom = rect.bottom() - pad - label_h - tick;
321    let top = rect.top() + pad + label_h / 2.0;
322    let ys = Scale {
323        lo,
324        hi,
325        log: look.log,
326        from: bottom,
327        to: top,
328    };
329    let (y_ticks, y_step) = ys.ticks(label_h * 2.5);
330    let y_labels: Vec<(f64, String)> = y_ticks
331        .iter()
332        .map(|v| (*v, axis::label(*v, y_step)))
333        .collect();
334    let label_w = y_labels
335        .iter()
336        .map(|(_, l)| text_size(painter, l, &look.font).x)
337        .fold(0.0, f32::max);
338    let title_w = if look.title.is_empty() {
339        0.0
340    } else {
341        text_size(painter, &look.title, &look.title_font).y + pad
342    };
343    let time_w = text_size(painter, "00:00:00", &look.font).x;
344    let left = rect.left() + pad + title_w + label_w + pad / 2.0 + tick;
345    let right = rect.right() - pad - time_w / 2.0;
346    let plot = Rect::from_min_max(Pos2::new(left, top), Pos2::new(right, bottom));
347    if plot.width() < 4.0 || plot.height() < 4.0 {
348        return;
349    }
350    let xs = Scale {
351        lo: from,
352        hi: now,
353        log: false,
354        from: plot.left(),
355        to: plot.right(),
356    };
357    let max_ticks = (plot.width() / (time_w + 20.0 * z)).floor().max(2.0) as usize;
358    let (t_ticks, t_step) = time_ticks(from, now, max_ticks, local_offset(now));
359
360    if look.grid {
361        let grid = Stroke::new(1.0, named("Grid").gamma_multiply(0.4));
362        for t in &t_ticks {
363            let x = xs.pos(*t);
364            painter.line_segment(
365                [Pos2::new(x, plot.top()), Pos2::new(x, plot.bottom())],
366                grid,
367            );
368        }
369        for (v, _) in &y_labels {
370            let y = ys.pos(*v);
371            painter.line_segment(
372                [Pos2::new(plot.left(), y), Pos2::new(plot.right(), y)],
373                grid,
374            );
375        }
376    }
377
378    let clipped = painter.with_clip_rect(painter.clip_rect().intersect(plot));
379    // Far-off points are pulled in so the maths stays finite.
380    let bounds = plot.expand(plot.width().max(plot.height()) * 4.0);
381    for (line, pts) in lines.iter().zip(&data) {
382        let screen = |(t, v): (f64, f64)| {
383            ok(v).then(|| {
384                Pos2::new(
385                    xs.pos(t).clamp(bounds.left(), bounds.right()),
386                    ys.pos(v).clamp(bounds.top(), bounds.bottom()),
387                )
388            })
389        };
390        let pts: Vec<Option<Pos2>> = pts.iter().map(|p| screen(*p)).collect();
391        // A value holds until the next sample, so the line steps; a gap
392        // (disconnected, or not a number) breaks it.
393        for run in pts.split(Option::is_none).filter(|r| !r.is_empty()) {
394            let mut steps: Vec<Pos2> = Vec::with_capacity(run.len() * 2);
395            for p in run.iter().flatten() {
396                if let Some(last) = steps.last().copied() {
397                    steps.push(Pos2::new(p.x, last.y));
398                }
399                steps.push(*p);
400            }
401            plot::line(
402                &clipped,
403                plot::decimate(steps),
404                Stroke::new(line.width, line.color),
405                0,
406            );
407        }
408    }
409
410    // The axes over the plot's edges.
411    let stroke = Stroke::new(z.max(1.0), look.fg);
412    painter.rect_stroke(plot, 0.0, stroke, egui::StrokeKind::Middle);
413    for t in &t_ticks {
414        let x = xs.pos(*t);
415        painter.line_segment(
416            [
417                Pos2::new(x, plot.bottom()),
418                Pos2::new(x, plot.bottom() + tick),
419            ],
420            stroke,
421        );
422        painter.text(
423            Pos2::new(x, plot.bottom() + tick),
424            Align2::CENTER_TOP,
425            time_label(*t, t_step),
426            look.font.clone(),
427            look.fg,
428        );
429    }
430    for (v, label) in &y_labels {
431        let y = ys.pos(*v);
432        painter.line_segment(
433            [Pos2::new(plot.left() - tick, y), Pos2::new(plot.left(), y)],
434            stroke,
435        );
436        painter.text(
437            Pos2::new(plot.left() - tick - pad / 2.0, y),
438            Align2::RIGHT_CENTER,
439            label,
440            look.font.clone(),
441            look.fg,
442        );
443    }
444    if !look.title.is_empty() {
445        let h = title_w - pad;
446        vertical_text(
447            painter,
448            Pos2::new(rect.left() + pad + h / 2.0, plot.center().y),
449            &look.title,
450            look.title_font.clone(),
451            look.fg,
452        );
453    }
454}

456/// Steps for the time axis, in seconds: whole seconds, minutes and hours.
457const TIME_STEPS: &[f64] = &[
458    1.0, 2.0, 5.0, 10.0, 15.0, 30.0, 60.0, 120.0, 300.0, 600.0, 900.0, 1800.0, 3600.0, 7200.0,
459    10_800.0, 21_600.0, 43_200.0, 86_400.0,
460];
461
462/// Ticks from `from` to `to` (seconds since 1970) at the smallest step
463/// giving at most `max_ticks`, on whole steps of local time (`offset`
464/// seconds ahead of UTC). Returns the ticks and the step.
465fn time_ticks(from: f64, to: f64, max_ticks: usize, offset: f64) -> (Vec<f64>, f64) {
466    let span = to - from;
467    if span <= 0.0 || !span.is_finite() {
468        return (Vec::new(), 0.0);
469    }
470    let step = TIME_STEPS
471        .iter()
472        .copied()
473        .find(|s| span / s <= max_ticks.max(1) as f64)
474        .unwrap_or_else(|| 86_400.0 * (span / 86_400.0 / max_ticks.max(1) as f64).ceil());
475    let first = ((from + offset) / step).ceil() * step - offset;
476    let ticks = (0..)
477        .map(|i| first + i as f64 * step)
478        .take_while(|t| *t <= to)
479        .take(1000)
480        .collect();
481    (ticks, step)
482}
483
484/// How far local time is ahead of UTC at `t`, in seconds.
485fn local_offset(t: f64) -> f64 {
486    Local
487        .timestamp_opt(t as i64, 0)
488        .single()
489        .map_or(0.0, |d| d.offset().local_minus_utc() as f64)
490}
491
492/// A time axis label in local time, as precise as the step.
493fn time_label(t: f64, step: f64) -> String {
494    let Some(d) = Local.timestamp_opt(t.round() as i64, 0).single() else {
495        return String::new();
496    };
497    let format = match step {
498        s if s < 60.0 => "%H:%M:%S",
499        s if s < 86_400.0 => "%H:%M",
500        _ => "%d %b",
501    };
502    d.format(format).to_string()
503}

507/// A strip chart's `time_range`, such as `1 minute` or `1 hour 30 min`, in
508/// seconds.
509pub fn parse_span(text: &str) -> Option<f64> {
510    let mut total = 0.0;
511    let mut words = text.split_whitespace().peekable();
512    let mut any = false;
513    while let Some(word) = words.next() {
514        // A number and its unit, apart (`10 min`) or together (`10min`).
515        let split = word
516            .find(|c: char| !(c.is_ascii_digit() || c == '.'))
517            .unwrap_or(word.len());
518        let (num, unit) = word.split_at(split);
519        let n: f64 = num.parse().ok()?;
520        let unit = if unit.is_empty() {
521            words.next().unwrap_or("s")
522        } else {
523            unit
524        };
525        let secs = match unit.to_ascii_lowercase().trim_end_matches('s') {
526            "" | "sec" | "second" => 1.0,
527            "m" | "min" | "minute" => 60.0,
528            "h" | "hour" => 3600.0,
529            "d" | "day" => 86_400.0,
530            "w" | "week" => 7.0 * 86_400.0,
531            _ => return None,
532        };
533        total += n * secs;
534        any = true;
535    }
536    (any && total > 0.0).then_some(total)
537}

539/// Draw a `.bob` strip chart: its traces' PVs over the last `time_range`,
540/// on its first value axis.
541pub fn draw_stripchart(painter: &egui::Painter, node: &Node, rect: Rect, z: f32) {
542    let w = &node.w;
543    let back = w
544        .color("background_color")
545        .map(|c| color(&c))
546        .unwrap_or(Color32::WHITE);
547    fill(painter, rect, back);
548    let fg = w
549        .color("foreground_color")
550        .map(|c| color(&c))
551        .unwrap_or(Color32::BLACK);
552    let pad = 4.0 * z;
553    let top = plot::title(painter, node, rect, fg, z);
554
555    let traces = plot::traces(node, z);
556    let shown: Vec<(&plot::Trace, &Series)> = traces
557        .iter()
558        .zip(&node.series)
559        .filter(|(t, _)| t.visible)
560        .filter_map(|(t, s)| Some((t, s.as_ref()?)))
561        .collect();
562    let legend_font = style::font_id(&w.font("legend_font").unwrap_or_default(), z);
563    let legend_h = if w.bool("show_legend").unwrap_or(true) && !shown.is_empty() {
564        text_size(painter, "Xg", &legend_font).y + pad
565    } else {
566        0.0
567    };
568    if legend_h > 0.0 {
569        let named: Vec<&plot::Trace> = shown.iter().map(|(t, _)| *t).collect();
570        plot::legend(painter, &named, rect, legend_h, pad, &legend_font, fg, z);
571    }
572
573    let axis = plot::axis_cfg(
574        w.get("y_axes").and_then(|e| e.child("y_axis")),
575        "Value",
576        node,
577        z,
578    );
579    let span = w
580        .text("time_range")
581        .map(|t| node.expand(t))
582        .and_then(|t| parse_span(&t))
583        .unwrap_or(60.0);
584    let look = Look {
585        log: axis.log,
586        range: (!axis.autoscale).then_some(axis.range),
587        title: if axis.visible {
588            axis.title.clone()
589        } else {
590            String::new()
591        },
592        grid: axis.grid,
593        fg: axis.color,
594        back,
595        font: axis.scale_font,
596        title_font: axis.title_font,
597    };
598    let lines: Vec<Line> = shown
599        .iter()
600        .map(|(t, s)| Line {
601            series: s,
602            color: t.color,
603            width: t.width,
604        })
605        .collect();
606    let area = Rect::from_min_max(
607        Pos2::new(rect.left(), top),
608        Pos2::new(rect.right(), rect.bottom() - legend_h),
609    );
610    draw_plot(painter, area, &lines, span, now(), &look, z);
611    // Keep time moving while the PVs are quiet, as `Trend::ui` does.
612    painter
613        .ctx()
614        .request_repaint_after(scroll_repaint(span, area.width()));
615
616    if let Some(s) = plot::worst_state(node) {
617        alarm_border(painter, w, &s, rect, z);
618    }
619}

696    #[test]
697    fn thinning_keeps_each_columns_ends_and_extremes() {
698        let mut s = Samples::default();
699        // A day of samples at 1 Hz, as a 1-day strip chart of 1000 pixels
700        // keeps: 86 seconds a pixel.
701        for i in 0..86_400 {
702            let t = i as f64;
703            s.push(
704                t,
705                (t * 0.01).sin() * 10.0 + if i == 5000 { 100.0 } else { 0.0 },
706            );
707        }
708        let dt = 86.4;
709        let pts = thinned(&s, 0.0, dt);
710        assert!(pts.len() <= 4 * 1001, "{} points", pts.len());
711        assert!(pts.windows(2).all(|w| w[0].0 <= w[1].0), "in time order");
712        // The spike and the first and last samples survive.
713        assert!(pts.contains(&(5000.0, s.since(5000.0).nth(1).unwrap().1)));
714        assert_eq!(pts.first(), s.since(0.0).next().as_ref());
715        assert_eq!(pts.last().map(|p| p.0), Some(86_399.0));
716        // Same extremes as all the samples.
717        let max = |p: &mut dyn Iterator<Item = (f64, f64)>| p.map(|p| p.1).fold(f64::MIN, f64::max);
718        assert_eq!(max(&mut pts.iter().copied()), max(&mut s.since(0.0)));
719    }

721    #[test]
722    fn thinning_takes_in_only_new_samples_and_keeps_gaps() {
723        let mut s = Samples::default();
724        let mut t = Thinned::default();
725        let push = |s: &mut Samples, t0: f64, n: usize| {
726            for i in 0..n {
727                let x = t0 + i as f64 * 0.05;
728                s.push(x, (x * 3.0).cos());
729            }
730        };
731        push(&mut s, 0.0, 500);
732        t.update(&s, 0.0, 0.5);
733        s.push(25.0, f64::NAN);
734        push(&mut s, 26.0, 500);
735        t.update(&s, 0.0, 0.5);
736        assert_eq!(t.seen, s.pushed);
737        // Taking in the new samples gives what thinning them all at once does.
738        let all: Vec<(f64, f64)> = t.points().collect();
739        assert_eq!(bits(&all), bits(&thinned(&s, 0.0, 0.5)));
740        assert_eq!(all.iter().filter(|p| p.1.is_nan()).count(), 1, "the gap");
741        // Time moving on forgets what has scrolled off, but the point that
742        // holds at the start.
743        t.update(&s, 40.0, 0.5);
744        let later: Vec<(f64, f64)> = t.points().collect();
745        assert!(later[0].0 < 40.0 && later[1].0 >= 40.0);
746        // A new width rebuilds it.
747        t.update(&s, 40.0, 0.25);
748        let rebuilt: Vec<(f64, f64)> = t.points().collect();
749        assert_eq!(bits(&rebuilt), bits(&thinned(&s, 40.0, 0.25)));
750    }
