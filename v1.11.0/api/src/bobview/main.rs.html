176#[cfg(not(target_arch = "wasm32"))]
177impl Cli {
178    /// Parse the command line, exiting with a message if it is wrong.
179    fn from_env() -> (Self, Vec<(egui::Pos2, egui::PointerButton)>) {
180        use clap::{CommandFactory, FromArgMatches};
181        let matches = Self::command().get_matches();
182        let cli = Self::from_arg_matches(&matches).unwrap_or_else(|e| e.exit());
183        let clicks = cli.clicks(&matches);
184        (cli, clicks)
185    }
186
187    /// Every kind of click, in the order given.
188    fn clicks(&self, matches: &clap::ArgMatches) -> Vec<(egui::Pos2, egui::PointerButton)> {
189        let at = |id| matches.indices_of(id).into_iter().flatten();
190        let mut clicks: Vec<_> = (at("click").zip(&self.click))
191            .map(|(i, p)| (i, *p, egui::PointerButton::Primary))
192            .chain(
193                at("right_click")
194                    .zip(&self.right_click)
195                    .map(|(i, p)| (i, *p, egui::PointerButton::Secondary)),
196            )
197            .chain(
198                at("middle_click")
199                    .zip(&self.middle_click)
200                    .map(|(i, p)| (i, *p, egui::PointerButton::Middle)),
201            )
202            .collect();
203        clicks.sort_by_key(|(i, ..)| *i);
204        clicks.into_iter().map(|(_, p, b)| (p, b)).collect()
205    }
206}

222/// The archiver trends use: `--archiver`, else none with `--no-archiver`,
223/// else `BOBVIEW_ARCHIVER` (`env`), else the one the settings remember
224/// (`saved`); and whether to remember it, as only `--archiver`'s is.
225#[cfg(not(target_arch = "wasm32"))]
226fn choose_archiver(
227    given: Option<archiveview::ArchiverClient>,
228    none: bool,
229    env: Option<&str>,
230    saved: Option<&str>,
231) -> (Option<archiveview::ArchiverClient>, bool) {
232    use archiveview::ArchiverClient;
233    if given.is_some() {
234        return (given, true);
235    }
236    if none {
237        return (None, false);
238    }
239    let from = |what: &str, url: &str| {
240        ArchiverClient::new(url)
241            .inspect_err(|e| log::warn!("{what}: {e}"))
242            .ok()
243    };
244    let env = env
245        .filter(|u| !u.trim().is_empty())
246        .and_then(|u| from("BOBVIEW_ARCHIVER", u));
247    let archiver = env.or_else(|| saved.and_then(|u| from("the saved archiver", u)));
248    (archiver, false)
249}

256#[cfg(not(target_arch = "wasm32"))]
257fn parse_size(text: &str) -> Result<Size, String> {
258    if text.trim() == "auto" {
259        return Ok(Size(None));
260    }
261    text.split_once('x')
262        .and_then(|(w, h)| Some([w.trim().parse().ok()?, h.trim().parse().ok()?]))
263        .filter(|[w, h]: &[u32; 2]| *w > 0 && *h > 0)
264        .map(|wh| Size(Some(wh)))
265        .ok_or_else(|| format!("expected WIDTHxHEIGHT in pixels or auto, not {text:?}"))
266}

268#[cfg(not(target_arch = "wasm32"))]
269fn parse_scale_factor(text: &str) -> Result<f32, String> {
270    text.trim()
271        .parse::<f32>()
272        .ok()
273        .filter(|s| s.is_finite() && *s > 0.0)
274        .ok_or_else(|| format!("expected a scale above 0, not {text:?}"))
275}

280    text.trim()
281        .parse::<f64>()
282        .ok()
283        .filter(|s| (0.0..1e9).contains(s))
284        .ok_or_else(|| format!("expected a number of seconds, not {text:?}"))

287#[cfg(not(target_arch = "wasm32"))]
288fn parse_point(text: &str) -> Result<egui::Pos2, String> {
289    text.split_once(',')
290        .and_then(|(x, y)| Some(egui::pos2(x.trim().parse().ok()?, y.trim().parse().ok()?)))
291        .filter(|p| p.x.is_finite() && p.y.is_finite())
292        .ok_or_else(|| format!("expected X,Y in pixels, not {text:?}"))

333#[cfg(not(target_arch = "wasm32"))]
334fn run(cli: Cli, clicks: Vec<(egui::Pos2, egui::PointerButton)>) -> Result<(), Failure> {
335    if cli.keypad {
336        eprintln!("--keypad is no longer supported: ignored");
337    }
338    if cli.multi_window {
339        // Pin mode replaced multi-window mode: shift+click follows a link
340        // as it did.
341        eprintln!(
342            "--multi-window is no longer needed: links from the pinned main \
343             window open in another window, shift+click in a new one"
344        );
345    }
346    let mut file = cli.display.as_deref().map(fetch::user_named);
347    let mut macros = parse_macros(cli.macros.as_deref().unwrap_or_default());
348    let ui_scale = cli.ui_scale.unwrap_or_else(chrome::UiScale::from_env);
349    let is_opi = |p: &std::path::Path| p.extension().is_some_and(|x| x.eq_ignore_ascii_case("opi"));
350    if let (Some(dir), Some(Source::File(path))) = (&cli.project, &file) {
351        let Some(project) = bob::project::project_dir(dir) else {
352            return Err(Failure::usage(format!(
353                "{} has no .project, nor a {} with links",
354                dir.display(),
355                bob::project::LAUNCHER
356            )));
357        };
358        // A launcher's links are CS-Studio's, for BOY screens only.
359        if bob::project::through_launcher(&project) && !is_opi(path) {
360            return Err(Failure::usage(format!(
361                "{}: {} links apply only to BOY .opi screens",
362                path.display(),
363                bob::project::LAUNCHER
364            )));
365        }
366        match bob::project::virtual_path(&project, path) {
367            Some(p) => file = Some(Source::File(p)),
368            None => {
369                return Err(Failure::usage(format!(
370                    "{} is not in a linked folder of {}",
371                    path.display(),
372                    dir.display()
373                )));
374            }
375        }
376    } else if let Some(Source::File(path)) = &file
377        && is_opi(path)
378        && let Some(p) = bob::project::launcher_path(path)
379    {
380        // A DLS BOY screen beside its runcss.sh: through the project that
381        // launcher would give CS-Studio.
382        log::info!("{}: opening as {}", path.display(), p.display());
383        file = Some(Source::File(p));
384    }
385    // An .opi opened directly (in an Eclipse project) gets the macros its
386    // DLS launcher script would give it; those given here override them.
387    if let Some(Source::File(path)) = &file
388        && is_opi(path)
389        && let Some(defaults) = launch::defaults(path)
390    {
391        macros = defaults.with(&macros);
392    }
393    // The named layout, else the default one unless a display was named.
394    let layout = if cli.snapshot.is_some() {
395        None
396    } else {
397        let default = file.is_none() && !cli.no_layout && layout::exists(layout::DEFAULT);
398        layout::startup(cli.layout.map(layout::named), default, layout::load)
399            .map_err(Failure::usage)?
400    };
401    // A layout's main window has a display of its own.
402    let layout_has_main = layout
403        .as_ref()
404        .is_some_and(|(l, _)| l.windows.iter().any(|w| w.main));
405    let start = file
406        .or_else(|| {
407            (!layout_has_main)
408                .then(default_display)
409                .flatten()
410                .map(Source::File)
411        })
412        .map(|source| app::Location { source, macros });
413
414    let (archiver, remember) = choose_archiver(
415        cli.archiver,
416        cli.no_archiver,
417        std::env::var("BOBVIEW_ARCHIVER").ok().as_deref(),
418        layout::Settings::load().archiver_url.as_deref(),
419    );
420    if let Some(a) = &archiver {
421        log::info!("trend history from {}", a.base());
422    }
423
424    if let Some(out) = cli.snapshot {
425        let start = start.ok_or_else(|| Failure::usage("--snapshot needs a display file"))?;
426        let opts = snapshot::Options {
427            size: cli.size.0,
428            scale_factor: cli.scale_factor,
429            ui_scale,
430            zoom: cli.zoom,
431            wait: std::time::Duration::from_secs_f64(cli.wait),
432            clicks,
433            settle: std::time::Duration::from_secs_f64(cli.settle),
434            read_only: cli.read_only,
435            archiver,
436        };
437        return snapshot::run(start, &opts, &out)
438            .map_err(|e| Failure::failed(format!("snapshot failed: {e}")));
439    }
440
441    let options = eframe::NativeOptions {
442        viewport: egui::ViewportBuilder::default()
443            .with_title("bobview")
444            .with_app_id("bobview")
445            .with_inner_size(chrome::DEFAULT_WINDOW)
446            .with_fullscreen(cli.fullscreen),
447        // No vsync: each window's buffer swap would wait for a screen
448        // refresh, and eframe draws the windows one after another on one
449        // thread, so windows repainting together would take a refresh each
450        // (~120 ms for 7, when they were all drawn in the main one's frame).
451        // We only repaint on change, so this doesn't spin.
452        glow_options: eframe::egui_glow::GlowConfiguration {
453            vsync: false,
454            ..Default::default()
455        },
456        ..Default::default()
457    };
458    let (zoom, read_only) = (cli.zoom, cli.read_only);
459    eframe::run_native(
460        "bobview",
461        options,
462        Box::new(move |cc| {
463            let mut app = app::BobApp::new(&cc.egui_ctx, start, zoom, ui_scale);
464            app.set_read_only(read_only);
465            app.set_archiver(archiver, remember);
466            if let Some((layout, name)) = layout {
467                app.restore_layout(layout, Some(name));
468            }
469            Ok(Box::new(app::App::new(app)))
470        }),
471    )
472    .map_err(|e| Failure::failed(e.to_string()))
473}

475/// Parse `P=X,R=Y` into macros; `\,` and `\\` in a value stand for `,`
476/// and `\`.
477fn parse_macros(text: &str) -> Macros {
478    // Split on unescaped commas, keeping escapes for now.
479    let mut pairs = vec![String::new()];
480    let mut chars = text.chars();
481    while let Some(c) = chars.next() {
482        match c {
483            ',' => pairs.push(String::new()),
484            '\\' => {
485                let last = pairs.last_mut().expect("a pair");
486                last.push(c);
487                last.extend(chars.next());
488            }
489            _ => pairs.last_mut().expect("a pair").push(c),
490        }
491    }
492    Macros::from_pairs(pairs.iter().filter_map(|kv| {
493        let (k, v) = kv.split_once('=')?;
494        Some((k.trim().to_string(), unescape_macro(v.trim())))
495    }))
496}
497
498fn unescape_macro(text: &str) -> String {
499    let mut out = String::with_capacity(text.len());
500    let mut chars = text.chars();
501    while let Some(c) = chars.next() {
502        match (c, chars.clone().next()) {
503            ('\\', Some(next @ (',' | '\\'))) => {
504                out.push(next);
505                chars.next();
506            }
507            _ => out.push(c),
508        }
509    }
510    out
511}

513/// The browser build: the page's query string names the display, its
514/// macros and the PV Web Socket, e.g.
515/// `?display=https://host/opi/index.bob&macros=P=X&pvws=pvws.host:443`
516/// (and `zoom=1.5`, `token=SECRET` for a server with `--write`).
517/// A relative display is found next to the page; the default is `index.bob`.
518/// With a token, displays come only from the page's server, and writes
519/// carry the token only to a PVWS there.
520#[cfg(target_arch = "wasm32")]
521fn main() {
522    use wasm_bindgen::JsCast;
523
524    init_logging();
525    let window = web_sys::window().expect("a browser window");
526    let location = window.location();
527    let query = web_sys::UrlSearchParams::new_with_str(&location.search().unwrap_or_default())
528        .expect("a query string");
529    let page = Source::Url(location.href().unwrap_or_default());
530    let display = query.get("display").unwrap_or_else(|| "index.bob".into());
531    let start = app::Location {
532        source: page.resolve(&display),
533        macros: parse_macros(&query.get("macros").unwrap_or_default()),
534    };
535    if let Some(pvws) = query.get("pvws") {
536        pv::set_pvws(&pvws);
537    }
538    if let Some(token) = query.get("token") {
539        // The token goes only to the page's own server, and only its
540        // displays may use it.
541        pv::set_write_token(&token);
542        fetch::same_origin_only();
543    }
544    let zoom = query.get("zoom").and_then(|z| {
545        parse_zoom(&z)
546            .inspect_err(|e| log::warn!("zoom: {e}; fitting to the window"))
547            .ok()
548    });
549
550    let canvas = window
551        .document()
552        .and_then(|d| d.get_element_by_id("bobview"))
553        .and_then(|c| c.dyn_into::<web_sys::HtmlCanvasElement>().ok())
554        .expect("a <canvas id=\"bobview\"> on the page");
555    wasm_bindgen_futures::spawn_local(async move {
556        let started = eframe::WebRunner::new()
557            .start(
558                canvas,
559                eframe::WebOptions::default(),
560                Box::new(move |cc| {
561                    let ui_scale = chrome::UiScale::from_env();
562                    let app = app::BobApp::new(&cc.egui_ctx, Some(start), zoom, ui_scale);
563                    Ok(Box::new(app::App::new(app)))
564                }),
565            )
566            .await;
567        if let Err(e) = started {
568            log::error!("cannot start: {e:?}");
569        }
570    });
571}

573/// The bundled index display, next to the binary or in ~/.local/share.
574#[cfg(not(target_arch = "wasm32"))]
575fn default_display() -> Option<PathBuf> {
576    let mut candidates = Vec::new();
577    if let Ok(exe) = std::env::current_exe()
578        && let Some(prefix) = exe.parent().and_then(|bin| bin.parent())
579    {
580        candidates.push(prefix.join("share/bobview/displays/index.bob"));
581        candidates.push(prefix.join("displays/index.bob"));
582    }
583    if let Some(home) = std::env::var_os("HOME") {
584        candidates.push(PathBuf::from(home).join(".local/share/bobview/displays/index.bob"));
585    }
586    candidates.into_iter().find(|p| p.exists())
587}

589struct Logger(log::LevelFilter);
590
591impl log::Log for Logger {
592    fn enabled(&self, m: &log::Metadata) -> bool {
593        // Library chatter (one line per CA channel connect) only when debugging.
594        let library = ["eframe", "egui", "epics", "winit", "glutin"]
595            .iter()
596            .any(|t| m.target().starts_with(t));
597        // arboard warns, at every window opened, that a Wayland compositor
598        // without data-control (GNOME's) has it fall back to XWayland.
599        let quiet = m.target().starts_with("arboard");
600        let limit = match self.0 < log::LevelFilter::Debug {
601            true if quiet => log::LevelFilter::Error,
602            true if library => log::LevelFilter::Warn,
603            _ => self.0,
604        };
605        m.level() <= limit
606    }
607
608    fn log(&self, r: &log::Record) {
609        if self.enabled(r.metadata()) {
610            #[cfg(not(target_arch = "wasm32"))]
611            eprintln!("{:<5} {}", r.level(), r.args());
612            #[cfg(target_arch = "wasm32")]
613            web_sys::console::log_1(&format!("{:<5} {}", r.level(), r.args()).into());
614        }
615    }
616
617    fn flush(&self) {}
618}
619
620fn init_logging() {
621    let level = match std::env::var("BOBVIEW_LOG").as_deref() {
622        Ok("debug") => log::LevelFilter::Debug,
623        Ok("trace") => log::LevelFilter::Trace,
624        Ok("warn") => log::LevelFilter::Warn,
625        Ok("error") => log::LevelFilter::Error,
626        _ => log::LevelFilter::Info,
627    };
628    let logger = Box::leak(Box::new(Logger(level)));
629    if log::set_logger(logger).is_ok() {
630        log::set_max_level(level);
631    }
632}
