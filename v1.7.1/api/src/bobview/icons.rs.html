1//! The toolbars' icons, drawn as strokes so they look the same on every
2//! platform (the fonts' symbols vary, and some are emoji).
3
4use eframe::egui::{self, Color32, Pos2, Rect, Shape, Stroke, StrokeKind, Vec2};
5
6#[derive(Clone, Copy, Debug, PartialEq)]
7pub enum Icon {
8    Back,
9    Forward,
10    Home,
11    Reload,
12    ZoomOut,
13    ZoomIn,
14    Navigator,
15    Open,
16    Pin,
17    Menu,
18}

90    /// Draw the icon `size` points square, centred in `rect`.
91    pub fn paint(self, painter: &egui::Painter, rect: Rect, size: f32, color: Color32) {
92        let scale = size / 16.0;
93        let origin = rect.center() - Vec2::splat(size / 2.0);
94        let stroke = Stroke::new((1.5 * scale).max(1.0), color);
95        for line in self.lines() {
96            let points = line
97                .iter()
98                .map(|[x, y]| origin + Vec2::new(x * scale, y * scale))
99                .collect::<Vec<Pos2>>();
100            painter.add(Shape::line(points, stroke));
101        }
102    }
103}

105/// The height of a toolbar button: a line of button text and its padding.
106pub fn height(ui: &egui::Ui) -> f32 {
107    ui.text_style_height(&egui::TextStyle::Button) + 2.0 * ui.spacing().button_padding.y
108}
109
110/// A button showing `icon`, as tall as a text button: highlighted when
111/// `selected`, inert unless `enabled`.
112pub fn button(ui: &mut egui::Ui, icon: Icon, selected: bool, enabled: bool) -> egui::Response {
113    let height = height(ui);
114    let size = Vec2::new(height * 1.15, height);
115    let sense = if enabled {
116        egui::Sense::click()
117    } else {
118        egui::Sense::hover()
119    };
120    let (rect, resp) = ui.allocate_exact_size(size, sense);
121    if ui.is_rect_visible(rect) {
122        // Disabled: as an idle button, only its icon faded.
123        let visuals = if enabled {
124            ui.style().interact_selectable(&resp, selected)
125        } else {
126            ui.visuals().widgets.inactive
127        };
128        let fill = if selected {
129            ui.visuals().selection.bg_fill
130        } else {
131            visuals.weak_bg_fill
132        };
133        let painter = ui.painter();
134        painter.rect(
135            rect,
136            visuals.corner_radius,
137            fill,
138            visuals.bg_stroke,
139            StrokeKind::Inside,
140        );
141        let color = if !enabled {
142            ui.visuals().weak_text_color()
143        } else if selected {
144            ui.visuals().selection.stroke.color
145        } else {
146            visuals.fg_stroke.color
147        };
148        icon.paint(painter, rect, height * 0.55, color);
149    }
150    resp
151}

153/// The height of a `group`: a button, its frame's margin and stroke.
154pub fn group_height(ui: &egui::Ui) -> f32 {
155    height(ui) + 2.0 * (GROUP_MARGIN + ui.visuals().widgets.noninteractive.bg_stroke.width)
156}
157
158/// Inside a group's frame, around its buttons.
159const GROUP_MARGIN: f32 = 1.0;
160
161/// Lay out controls that belong together, `framed` as one; unframed, they
162/// take the same room, so they line up with framed groups.
163pub fn group<R>(ui: &mut egui::Ui, framed: bool, add: impl FnOnce(&mut egui::Ui) -> R) -> R {
164    let mut stroke = ui.visuals().widgets.noninteractive.bg_stroke;
165    if !framed {
166        stroke.color = Color32::TRANSPARENT;
167    }
168    egui::Frame::new()
169        .stroke(stroke)
170        .corner_radius(5.0)
171        .inner_margin(GROUP_MARGIN)
172        .show(ui, |ui| {
173            ui.spacing_mut().item_spacing.x = if framed { 1.0 } else { 4.0 };
174            ui.horizontal(add).inner
175        })
176        .inner
177}
