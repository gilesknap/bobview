1//! Widgets that show the group a vacuum space's link field names, as the
2//! DLS `get_pv_name.py` script makes them (see [`bob::GroupLink`]).
3//!
4//! The link field is subscribed with the widget, and the widget is pointed
5//! at the group each time the field names a different one: its properties
6//! change as the script would set them, so its tooltip, actions and PV menu
7//! follow, and its PV becomes the group's `:STA`, dropping the one before.
8//! The script polls once a second; a link changes rarely, so this does
9//! nothing until its value does.
10
11use pv::{Pv, PvManager, PvState, Value};
12
13use crate::screen::Node;
14
15/// A widget's link field, and the group it last named.
16pub struct Link {
17    spec: bob::GroupLink,
18    pv: Pv,
19    /// The group the widget is pointed at, once the link has named one.
20    device: Option<String>,
21}
22
23impl Link {
24    /// The link of widget `w`, its field's name expanded with `expand`.
25    pub fn new(w: &bob::Widget, expand: &dyn Fn(&str) -> String, pvs: &PvManager) -> Option<Self> {
26        let spec = w.group_link()?;
27        let name = expand(&spec.pv_name);
28        if name.trim().is_empty() {
29            return None;
30        }
31        Some(Link {
32            pv: pvs.subscribe(&name),
33            spec,
34            device: None,
35        })
36    }
37}
38
39/// The group a link field's state names, if it is connected and names one.
40fn device(s: &PvState) -> Option<String> {
41    if !s.connected {
42        return None;
43    }
44    match &s.value {
45        Value::Str(t) => bob::group_device(t),
46        _ => None,
47    }
48}
49
50/// Point `node` at the group its link names now, if that is a new one.
51/// A link that is not connected leaves the widget as it was.
52pub fn update(node: &mut Node, pvs: &PvManager) {
53    let Some(link) = &node.link else { return };
54    let Some(device) = device(&link.pv.state()) else {
55        return;
56    };
57    if link.device.as_deref() == Some(device.as_str()) {
58        return;
59    }
60    log::debug!("{}: {} names {device}", node.w.name(), link.pv.name());
61    let spec = link.spec.clone();
62    spec.point_at(&mut node.w.props, &device);
63    // Rules start from the properties as they were, which now include these.
64    if let Some(orig) = &mut node.orig_props {
65        spec.point_at(orig, &device);
66    }
67    if spec.sets == bob::LinkSets::Pv {
68        node.pv_name = node.prop("pv_name");
69        node.pv = Some(pvs.subscribe(&node.pv_name));
70    }
71    if let Some(link) = &mut node.link {
72        link.device = Some(device);
73    }
74}
75
76/// The frame a linked symbol shows for its group's status `s`, once it is
77/// pointed at one; `None` leaves it to `image_index`.
78pub fn frame(node: &Node, s: Option<&PvState>) -> Option<i64> {
79    let link = node.link.as_ref()?;
80    let top = link.spec.frames?;
81    link.device.as_ref()?;
82    let status = s.filter(|s| s.connected).and_then(|s| s.value.as_i64());
83    Some(bob::group_frame(status, top))
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89    use crate::screen::Builder;
90    use bob::{Display, Macros};
91
92    #[test]
93    fn linked_widgets_follow_their_link() {
94        // As the DLS rack screens build them, with loc:// PVs: a symbol
95        // whose click opens the group's display, and a label naming it.
96        let d = Display::parse(
97            r#"<display><macros><space>loc://repoint_test</space></macros>
98            <widget type="symbol"><name></name><pv_name></pv_name>
99              <tooltip>$(pv_name)
100$(pv_value)</tooltip><sub_image_width>34</sub_image_width>
101              <actions><action type="open_display"><file>gionp.opi</file><macros><device>$(name)</device></macros></action></actions>
102              <group_link><pv_name>$(space)("")</pv_name><sets>pv</sets><frames>9</frames></group_link></widget>
103            <widget type="label"><text>SRXXC-VA-GXXX-YY</text>
104              <group_link><pv_name>$(space)</pv_name><sets>text</sets></group_link></widget>
105            </display>"#,
106        )
107        .unwrap();
108        let pvs = PvManager::new();
109        let mut s = Builder::new(&pvs).screen(d, &Macros::default(), None, 0);
110        let link = s.nodes[0].link.as_ref().unwrap().pv.clone();
111
112        // An empty link leaves both as they were loaded.
113        for n in &mut s.nodes {
114            update(n, &pvs);
115        }
116        assert!(s.nodes[0].pv.is_none());
117        assert_eq!(frame(&s.nodes[0], None), None);
118        assert_eq!(s.nodes[1].prop("text"), "SRXXC-VA-GXXX-YY");
119
120        link.write("loc://G1:STA NPP NMS");
121        for n in &mut s.nodes {
122            update(n, &pvs);
123        }
124        let sym = &s.nodes[0];
125        assert_eq!(sym.pv_name, "loc://G1:STA");
126        assert_eq!(sym.pv.as_ref().unwrap().name().to_string(), "loc://G1:STA");
127        assert_eq!(sym.expand("$(name)"), "loc://G1");
128        assert_eq!(s.nodes[1].prop("text"), "loc://G1");
129        let bob::Action::OpenDisplay { macros, .. } = &sym.w.actions()[0] else {
130            panic!("not an open display action")
131        };
132        assert_eq!(macros.get("device"), Some("loc://G1"));
133        // The group's status picks the frame: status + 1, within the strip.
134        let sta = sym.pv.as_ref().unwrap();
135        assert_eq!(frame(sym, sym.state().as_ref()), Some(1));
136        sta.write("3");
137        assert_eq!(frame(sym, sym.state().as_ref()), Some(4));
138        sta.write("9");
139        assert_eq!(frame(sym, sym.state().as_ref()), Some(0));
140
141        // A new group re-points the widget at its :STA.
142        link.write("loc://G2:CTL:SP NPP MS");
143        for n in &mut s.nodes {
144            update(n, &pvs);
145        }
146        let sym = &s.nodes[0];
147        assert_eq!(sym.pv_name, "loc://G2:STA");
148        assert_eq!(sym.expand("$(name)"), "loc://G2");
149        assert_eq!(s.nodes[1].prop("text"), "loc://G2");
150        // A write to the same group, through another field, keeps it.
151        link.write("loc://G2:STA NPP NMS");
152        update(&mut s.nodes[0], &pvs);
153        assert_eq!(s.nodes[0].pv_name, "loc://G2:STA");
154    }
155}
