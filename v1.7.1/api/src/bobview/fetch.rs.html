1//! Reading displays from files or over HTTP(S).
2//!
3//! Fetched displays are kept in memory, so going back and forward or
4//! reopening a screen doesn't fetch it again; [`forget`] clears them for a
5//! reload. Fetching blocks, so the app builds a fetched display on a thread.
6//!
7//! A browser cannot block, nor read files: there [`load`] only reads the
8//! cache, starting a fetch for what is missing and failing with [`LOADING`]
9//! until it arrives, and the app builds the display again once nothing is
10//! [`pending`].
11//!
12//! Natively, files that a display served from a public address names are
13//! never fetched from a private one (loopback, a private network, link-local
14//! and the like), so a display from the web cannot make bobview reach into
15//! the network it runs on; nor does a redirect lead from a public address to
16//! a private one, or from https to http. Files are found through
17//! [`resolve`], which keeps track of who named them. A browser applies its
18//! own rules (CORS, mixed content) instead.
19

29/// A fetched display: the URL it was finally served from (after any
30/// redirects, for resolving the files it names) and its text.
31#[derive(Clone)]
32struct Fetched {
33    url: String,
34    text: Arc<str>,
35}

37static CACHE: LazyLock<Mutex<HashMap<String, Fetched>>> = LazyLock::new(Default::default);

39/// Bumped by [`forget`], so a fetch started before a reload cannot put
40/// what it read back into the cache after it.
41#[cfg(not(target_arch = "wasm32"))]
42static GENERATION: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);

44/// Read and parse the display at `source`.
45pub fn load(source: &Source) -> Result<Display, Error> {
46    match source {
47        #[cfg(not(target_arch = "wasm32"))]
48        Source::File(path) => match native::bob_instead(path) {
49            Some(bob) => {
50                log::info!("{}: using {}", path.display(), bob.display());
51                Display::load(&bob)
52            }
53            None => Display::load(path),
54        },
55        #[cfg(target_arch = "wasm32")]
56        Source::File(path) => Err(no_files(path)),
57        Source::Url(url) => {
58            let fetched = get(url)?;
59            let mut display = Display::parse(&fetched.text)?;
60            display.source = Some(Source::Url(fetched.url));
61            Ok(display)
62        }
63    }
64}

66/// Drop every cached display, so the next load fetches it again. Which
67/// files displays from public addresses named is kept: fetches already
68/// queued are still checked, and a reload does not lift the check.
69pub fn forget() {
70    #[cfg(not(target_arch = "wasm32"))]
71    GENERATION.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
72    CACHE.lock().clear();
73    #[cfg(target_arch = "wasm32")]
74    web::FAILED.lock().clear();
75}

77/// The file or URL `text` that the user named (on the command line, in the
78/// Open panel, or in a saved layout). The user's own choice is fetched
79/// wherever it is, whether a display from a public address named it before
80/// or names it later.
81pub fn user_named(text: &str) -> Source {
82    let source = Source::parse(text);
83    #[cfg(not(target_arch = "wasm32"))]
84    if let Source::Url(url) = &source {
85        native::user_named(url);
86    }
87    source
88}

90/// Whether `source` was refused because a display from a public address
91/// named it, so it is not one to keep (in a layout, say) as the user's.
92pub fn refused(source: &Source) -> bool {
93    #[cfg(not(target_arch = "wasm32"))]
94    if let Source::Url(url) = source {
95        return native::refused(url);
96    }
97    let _ = source;
98    false
99}

101/// The file `file` named by the display at `base` (with no display, a plain
102/// [`Source::parse`]): relative names are found next to the display. One named by a
103/// display from a public address is fetched only from a public one.
104pub fn resolve(base: Option<&Source>, file: &str) -> Source {
105    let Some(base) = base else {
106        return Source::parse(file);
107    };
108    let source = base.resolve(file);
109    #[cfg(not(target_arch = "wasm32"))]
110    if let (Source::Url(base), Source::Url(url)) = (base, &source) {
111        native::named_by(base, url);
112    }
113    source
114}

187        if PUBLIC.lock().contains(base) && !USER.lock().contains(url) {
188            let mut from_public = FROM_PUBLIC.lock();
189            if !from_public.contains(url) {

207    /// GET `url`, following redirects, but not from https to http, nor from
208    /// a public address to a private one, nor to a private one at all for
209    /// a file a display from a public address named. Returns where it was
210    /// finally served from, and the response.
211    fn call(url: &str) -> Result<(String, Response<Body>), String> {
212        call_within(url, FETCH_TIME)
213    }

215    /// [`call`], all of it (redirects, and reading the body) within `time`.
216    pub(super) fn call_within(
217        url: &str,
218        time: Duration,
219    ) -> Result<(String, Response<Body>), String> {
220        let deadline = Instant::now() + time;
221        // Why a private address is refused, if it is.
222        let mut refuse = FROM_PUBLIC
223            .lock()
224            .contains(url)
225            .then_some("refused: a display from a public address named a private one");
226        let named = url;
227        let mut url = url.to_string();
228        for hop in 0..=MAX_REDIRECTS {
229            let addrs = addresses(&url);
230            if let Some(why) = refuse
231                && target_private(&addrs)
232            {
233                if hop == 0 {
234                    REFUSED.lock().insert(named.to_string());
235                }
236                return Err(format!("{why} ({url})"));
237            }
238            let left = deadline.saturating_duration_since(Instant::now());
239            if left.is_zero() {
240                return Err(format!("timed out after {}s", time.as_secs_f32()));
241            }
242            let resp = AGENT
243                .get(&url)
244                .config()
245                .timeout_global(Some(left))
246                .build()
247                .call()
248                .map_err(describe)?;
249            let status = resp.status();
250            if !status.is_redirection() {
251                return Ok((url, resp));
252            }
253            let Some(location) = resp
254                .headers()
255                .get(ureq::http::header::LOCATION)
256                .and_then(|l| l.to_str().ok())
257            else {
258                return Err(format!("HTTP {}", status.as_u16()));
259            };
260            if let Some(scheme) = scheme(location)
261                && !scheme.eq_ignore_ascii_case("http")
262                && !scheme.eq_ignore_ascii_case("https")
263            {
264                return Err(format!("refused a redirect to {location}"));
265            }
266            let next = Source::Url(url.clone()).resolve(location).to_string();
267            if downgrade(&url, &next) {
268                return Err(format!("refused a redirect from https to {next}"));
269            }
270            log::info!("{url} redirects to {next}");
271            if !origin_private(&addrs) {
272                refuse = Some("refused a redirect from a public address to a private one");
273            }
274            url = next;
275        }
276        Err("too many redirects".into())
277    }

279    /// The scheme of `url`, if it has one (else it is relative).
280    fn scheme(url: &str) -> Option<&str> {
281        let (scheme, _) = url.split_once(':')?;
282        let mut chars = scheme.chars();
283        let first = chars.next()?;
284        (first.is_ascii_alphabetic()
285            && chars.all(|c| c.is_ascii_alphanumeric() || "+-.".contains(c)))
286        .then_some(scheme)
287    }

289    /// Whether a redirect from `from` to `to` leaves https.
290    pub(super) fn downgrade(from: &str, to: &str) -> bool {
291        let https = |u: &str| {
292            u.get(..8)
293                .is_some_and(|s| s.eq_ignore_ascii_case("https://"))
294        };
295        https(from) && !https(to)
296    }

298    /// The addresses of the host of `url`, from the same lookup ureq
299    /// connects with; none if it cannot be looked up here (a proxy may reach
300    /// it).
301    pub(super) fn addresses(url: &str) -> Vec<IpAddr> {
302        let Ok(uri) = url.parse::<ureq::http::Uri>() else {
303            return Vec::new();
304        };
305        let Some(host) = uri.host() else {
306            return Vec::new();
307        };
308        let host = host.trim_start_matches('[').trim_end_matches(']');
309        let https = uri
310            .scheme_str()
311            .is_some_and(|s| s.eq_ignore_ascii_case("https"));
312        let port = uri.port_u16().unwrap_or(if https { 443 } else { 80 });
313        #[cfg(test)]
314        if host == "127.0.0.1"
315            && let Some(addrs) = PRETEND.lock().get(&port)
316        {
317            return addrs.clone();
318        }
319        match host.parse::<IpAddr>() {
320            Ok(ip) => vec![ip],
321            Err(_) => (host, port)
322                .to_socket_addrs()
323                .map(|addrs| addrs.map(|a| a.ip()).collect())
324                .unwrap_or_default(),
325        }
326    }

328    /// Whether a display served from a host with addresses `addrs` is on a
329    /// private network: only if all of them are private ([`private_ip`]),
330    /// since the display may have come from any of them.
331    pub(super) fn origin_private(addrs: &[IpAddr]) -> bool {
332        !addrs.is_empty() && addrs.iter().all(|&ip| private_ip(ip))
333    }

335    /// Whether a fetch from a host with addresses `addrs` could reach a
336    /// private network: if any of them is private ([`private_ip`]).
337    pub(super) fn target_private(addrs: &[IpAddr]) -> bool {
338        addrs.iter().any(|&ip| private_ip(ip))

341    /// Whether `ip` is not a public internet address: loopback, private
342    /// networks, link-local (cloud metadata), carrier-grade NAT,
343    /// unspecified, broadcast, IPv6 unique-local and site-local, and IPv4
344    /// in IPv6 (mapped, compatible, translated, NAT64 and 6to4), by the
345    /// IPv4 address within.
346    pub(super) fn private_ip(ip: IpAddr) -> bool {
347        match ip {
348            IpAddr::V4(v4) => {
349                let [a, b, c, _] = v4.octets();
350                v4.is_loopback()
351                    || v4.is_private()
352                    || v4.is_link_local()
353                    || v4.is_unspecified()
354                    || v4.is_broadcast()
355                    || a == 0
356                    || (a == 100 && (64..128).contains(&b))
357                    || (a == 192 && b == 0 && c == 0) // IETF protocol assignments
358                    || (a == 198 && b & 0xfe == 18) // benchmarking, used inside sites
359                    || a >= 240 // reserved
360            }
361            IpAddr::V6(v6) => {
362                let s = v6.segments();
363                let v4 = |hi: u16, lo: u16| {
364                    private_ip(IpAddr::V4(((u32::from(hi) << 16) | u32::from(lo)).into()))
365                };
366                match s {
367                    // IPv4-mapped ::ffff:a.b.c.d, and SIIT ::ffff:0:a.b.c.d.
368                    [0, 0, 0, 0, 0xffff, 0, hi, lo] | [0, 0, 0, 0, 0, 0xffff, hi, lo] => v4(hi, lo),
369                    // IPv4-compatible ::a.b.c.d (deprecated), :: and ::1.
370                    [0, 0, 0, 0, 0, 0, _, _] => true,
371                    // NAT64 64:ff9b::a.b.c.d; local-use NAT64 64:ff9b:1::/48
372                    // may embed any address anywhere.
373                    [0x64, 0xff9b, 0, 0, 0, 0, hi, lo] => v4(hi, lo),
374                    [0x64, 0xff9b, 1, ..] => true,
375                    // 6to4 2002:a.b.c.d::/48.
376                    [0x2002, hi, lo, ..] => v4(hi, lo),
377                    [first, ..] => {
378                        first & 0xfe00 == 0xfc00 // unique-local
379                            || first & 0xffc0 == 0xfe80 // link-local
380                            || first & 0xffc0 == 0xfec0 // site-local
381                    }
382                }
383            }
384        }
385    }

387    /// Read the file at `source` as bytes, such as an image a display names.
388    /// Nothing is cached here: the caller keeps what it decodes.
389    pub fn bytes(source: &Source) -> Result<Vec<u8>, Error> {
390        match source {
391            Source::File(path) => {
392                let mut data = Vec::new();
393                std::fs::File::open(bob::project::locate(path))
394                    .and_then(|f| f.take(MAX_SIZE + 1).read_to_end(&mut data))
395                    .map_err(|e| Error::Io(path.clone(), e))?;
396                if data.len() as u64 > MAX_SIZE {
397                    let why = format!("larger than {} MiB", MAX_SIZE >> 20);
398                    return Err(Error::Io(path.clone(), std::io::Error::other(why)));
399                }
400                Ok(data)
401            }
402            Source::Url(url) => {
403                log::info!("fetching {url}");
404                let fail = |why: String| Error::Fetch(url.to_string(), why);
405                call(url)
406                    .map_err(fail)?
407                    .1
408                    .body_mut()
409                    .with_config()
410                    .limit(MAX_SIZE)
411                    .read_to_vec()
412                    .map_err(|e| fail(describe(e)))
413            }
414        }
415    }

429    pub(super) fn get(url: &str) -> Result<Fetched, Error> {
430        if let Some(fetched) = CACHE.lock().get(url).cloned() {
431            return Ok(fetched);
432        }
433        let generation = GENERATION.load(std::sync::atomic::Ordering::SeqCst);
434        log::info!("fetching {url}");
435        let fail = |why: String| Error::Fetch(url.to_string(), why);
436        let (served, mut resp) = call(url).map_err(fail)?;
437        let text: std::sync::Arc<str> = resp
438            .body_mut()
439            .with_config()
440            .limit(MAX_SIZE)
441            .read_to_string()
442            .map_err(|e| fail(describe(e)))?
443            .into();
444        // Kept across reloads, so files named by a display already shown
445        // are still checked; updated as the display is fetched again.
446        let private = origin_private(&addresses(&served));
447        let mut public = PUBLIC.lock();
448        for u in [url, &served] {
449            if private {
450                public.remove(u);
451            } else {
452                public.insert(u.to_string());
453            }
454        }
455        drop(public);
456        let fetched = Fetched { url: served, text };
457        let mut cache = CACHE.lock();
458        // Forgotten while it was being fetched: it may be stale, so it is
459        // used this once but not kept.
460        if GENERATION.load(std::sync::atomic::Ordering::SeqCst) == generation {
461            cache.insert(url.to_string(), fetched.clone());
462        }
463        Ok(fetched)
464    }

466    fn describe(e: ureq::Error) -> String {

468            ureq::Error::StatusCode(code) => format!("HTTP {code}"),
469            ureq::Error::Io(e) => e.to_string(),
470            e => e.to_string(),
471        }

488    /// Why [`super::load`] failed while the display is still on its way.
489    pub const LOADING: &str = "loading";
490
491    /// Displays being fetched, and those that could not be, with why.
492    static PENDING: LazyLock<Mutex<HashSet<String>>> = LazyLock::new(Default::default);
493    pub(super) static FAILED: LazyLock<Mutex<HashMap<String, String>>> =
494        LazyLock::new(Default::default);
495
496    /// Whether any display is still being fetched.
497    pub fn pending() -> bool {
498        !PENDING.lock().is_empty()
499    }

501    /// Load displays only from the page's own server: a display from
502    /// another, opened by a link, could write PVs with the page's token.
503    static SAME_ORIGIN: AtomicBool = AtomicBool::new(false);
504
505    pub fn same_origin_only() {
506        SAME_ORIGIN.store(true, Ordering::Relaxed);
507    }
508
509    /// Why `url` may not be loaded, if it may not.
510    fn refused(url: &str) -> Option<String> {
511        if !SAME_ORIGIN.load(Ordering::Relaxed) {
512            return None;
513        }
514        // A page whose address cannot be read loads nothing, as the token
515        // is withheld from a PVWS whose origin cannot be judged.
516        let page = web_sys::window().and_then(|w| w.location().href().ok());
517        match page {
518            Some(page) if pv::same_origin(&page, url) => None,
519            _ => Some("not loading displays from other servers while writes are on".into()),
520        }
521    }
522
523    pub(super) fn get(url: &str) -> Result<Fetched, Error> {
524        if let Some(why) = refused(url) {
525            return Err(Error::Fetch(url.to_string(), why));
526        }
527        if let Some(fetched) = CACHE.lock().get(url).cloned() {
528            return Ok(fetched);
529        }
530        if let Some(why) = FAILED.lock().get(url) {
531            return Err(Error::Fetch(url.to_string(), why.clone()));
532        }
533        if PENDING.lock().insert(url.to_string()) {
534            let url = url.to_string();
535            wasm_bindgen_futures::spawn_local(async move {
536                // A redirect may lead elsewhere.
537                let result = fetch(&url)
538                    .await
539                    .and_then(|(served, data)| match refused(&served) {
540                        Some(why) => Err(why),
541                        None => Ok((served, data)),
542                    });
543                match result {
544                    Ok((served, data)) => {
545                        let text = String::from_utf8_lossy(&data).into();
546                        let fetched = Fetched { url: served, text };
547                        CACHE.lock().insert(url.clone(), fetched);
548                    }
549                    Err(why) => {
550                        log::warn!("cannot fetch {url}: {why}");
551                        FAILED.lock().insert(url.clone(), why);
552                    }
553                }
554                PENDING.lock().remove(&url);
555            });
556        }
557        Err(Error::Fetch(url.to_string(), LOADING.into()))
558    }

571    /// GET `url`, returning where it was served from and its body. The server
572    /// must allow this page's origin (CORS) unless it is the page's own.
573    async fn fetch(url: &str) -> Result<(String, Vec<u8>), String> {
574        log::info!("fetching {url}");
575        let js = |e: wasm_bindgen::JsValue| {
576            e.dyn_ref::<js_sys::Error>()
577                .map(|e| String::from(e.message()))
578                .unwrap_or_else(|| format!("{e:?}"))
579        };
580        let window = web_sys::window().ok_or("no window")?;
581        let resp: web_sys::Response = JsFuture::from(window.fetch_with_str(url))
582            .await
583            .map_err(js)?
584            .unchecked_into();
585        if !resp.ok() {
586            return Err(format!("HTTP {}", resp.status()));
587        }
588        let body = JsFuture::from(resp.array_buffer().map_err(js)?)
589            .await
590            .map_err(js)?;
591        let data = js_sys::Uint8Array::new(&body);
592        if data.length() as u64 > MAX_SIZE {
593            return Err(format!("larger than {} MiB", MAX_SIZE >> 20));
594        }
595        Ok((resp.url(), data.to_vec()))
596    }

616    /// [`serve`], with /away redirecting to `away`, /file to a file, and
617    /// /slow slowly to itself, while /stall answers after 3 s.
618    fn serve_to(away: &str) -> (String, Arc<Mutex<usize>>) {
619        let away = format!("302 Found\r\nLocation: {away}");
620        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
621        let base = format!("http://{}", listener.local_addr().unwrap());
622        let hits = Arc::new(Mutex::new(0));
623        let count = hits.clone();
624        std::thread::spawn(move || {
625            for stream in listener.incoming().flatten() {
626                let mut reader = BufReader::new(stream.try_clone().unwrap());
627                let mut first = String::new();
628                reader.read_line(&mut first).unwrap();
629                let mut line = String::new();
630                while reader.read_line(&mut line).unwrap() > 2 {
631                    line.clear();
632                }
633                *count.lock() += 1;
634                let (status, body) = if first.starts_with("GET /opi/a.bob ") {
635                    ("200 OK", DISPLAY)
636                } else if first.starts_with("GET /old/a.bob ") {
637                    ("301 Moved Permanently\r\nLocation: /opi/a.bob", "")
638                } else if first.starts_with("GET /away ") {
639                    (away.as_str(), "")
640                } else if first.starts_with("GET /file ") {
641                    ("302 Found\r\nLocation: file:///etc/passwd", "")
642                } else if first.starts_with("GET /slow ") {
643                    std::thread::sleep(std::time::Duration::from_millis(300));
644                    ("302 Found\r\nLocation: /slow", "")
645                } else if first.starts_with("GET /stall ") {
646                    std::thread::sleep(std::time::Duration::from_secs(3));
647                    ("200 OK", DISPLAY)
648                } else {
649                    ("404 Not Found", "")
650                };
651                let mut stream = stream;
652                // The client may have given up.
653                let _ = write!(
654                    stream,
655                    "HTTP/1.1 {status}\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
656                    body.len()
657                );
658            }
659        });
660        (base, hits)
661    }
