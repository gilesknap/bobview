63/// One PV's samples, in time order: (seconds since 1970, value), a NaN
64/// value marking where it disconnected.
65#[derive(Default)]
66pub struct Samples {
67    points: VecDeque<(f64, f64)>,
68    /// How many have ever been added, so a [`Thinned`] knows which are new.
69    pushed: u64,
70}
71
72impl Samples {
73    pub fn push(&mut self, t: f64, v: f64) {
74        // One gap is enough, and a trend starts with a value.
75        if v.is_nan() && self.points.back().is_none_or(|p| p.1.is_nan()) {
76            return;
77        }
78        // The clock may step back; the samples stay in order.
79        let t = self.points.back().map_or(t, |p| t.max(p.0));
80        self.points.push_back((t, v));
81        self.pushed += 1;
82        while self.points.len() > MAX_SAMPLES
83            || self.points.front().is_some_and(|p| p.0 < t - MAX_AGE)
84        {
85            self.points.pop_front();
86        }
87    }
88
89    /// The samples from `from` on, and the one before it, which holds at
90    /// `from`.
91    pub fn since(&self, from: f64) -> impl Iterator<Item = (f64, f64)> + '_ {
92        let i = self.points.partition_point(|p| p.0 < from);
93        self.points.range(i.saturating_sub(1)..).copied()
94    }
95
96    #[cfg(test)]
97    fn len(&self) -> usize {
98        self.points.len()
99    }
100}

102/// One PV's samples thinned for drawing: in each column of time `dt` wide
103/// (a pixel or less), the first, lowest, highest and last sample, which
104/// draw the same as all of them. Kept from frame to frame, taking in only
105/// the samples added since, so a day of fast samples is not copied out
106/// and stepped through on every frame.
107#[derive(Default)]
108struct Thinned {
109    dt: f64,
110    /// The start of the time it was built for.
111    from: f64,
112    /// [`Samples::pushed`] when it last took samples in.
113    seen: u64,
114    /// The finished columns' points, in time order, NaN marking gaps.
115    done: VecDeque<(f64, f64)>,
116    /// The column being filled.
117    open: Option<Column>,
118}
119
120/// A column's first, lowest, highest and last sample.
121#[derive(Clone, Copy)]
122struct Column {
123    index: i64,
124    first: (f64, f64),
125    low: (f64, f64),
126    high: (f64, f64),
127    last: (f64, f64),
128}
129
130impl Column {
131    /// Its points in time order, each once.
132    fn points(&self) -> impl Iterator<Item = (f64, f64)> {
133        let mut pts = [self.first, self.low, self.high, self.last];
134        pts.sort_by(|a, b| a.0.total_cmp(&b.0));
135        let mut n = 0;
136        for i in 0..pts.len() {
137            if n == 0 || pts[i] != pts[n - 1] {
138                pts[n] = pts[i];
139                n += 1;
140            }
141        }
142        pts.into_iter().take(n)
143    }
144}
145
146impl Thinned {
147    fn new(dt: f64, from: f64) -> Self {
148        Self {
149            dt,
150            from,
151            ..Default::default()
152        }
153    }
154
155    fn close(&mut self) {
156        if let Some(c) = self.open.take() {
157            self.done.extend(c.points());
158        }
159    }
160
161    fn add(&mut self, (t, v): (f64, f64)) {
162        if v.is_nan() {
163            // A gap ends the column's line.
164            self.close();
165            self.done.push_back((t, v));
166            return;
167        }
168        let index = (t / self.dt).floor() as i64;
169        match &mut self.open {
170            Some(c) if c.index == index => {
171                if v < c.low.1 {
172                    c.low = (t, v);
173                }
174                if v > c.high.1 {
175                    c.high = (t, v);
176                }
177                c.last = (t, v);
178            }
179            _ => {
180                self.close();
181                self.open = Some(Column {
182                    index,
183                    first: (t, v),
184                    low: (t, v),
185                    high: (t, v),
186                    last: (t, v),
187                });
188            }
189        }
190    }
191
192    /// Bring it up to date with `samples` for the time from `from` in
193    /// columns `dt` wide, rebuilding it only if the columns changed.
194    fn update(&mut self, samples: &Samples, from: f64, dt: f64) {
195        let new = samples.pushed - self.seen;
196        if self.dt != dt || from < self.from || new > samples.points.len() as u64 {
197            *self = Self::new(dt, from);
198            for p in samples.since(from) {
199                self.add(p);
200            }
201        } else {
202            let len = samples.points.len();
203            for p in samples.points.range(len - new as usize..) {
204                self.add(*p);
205            }
206        }
207        self.seen = samples.pushed;
208        // Forget what has scrolled off, but the point that holds at `from`.
209        let i = self.done.partition_point(|p| p.0 < from);
210        if i > 1 {
211            self.done.drain(..i - 1);
212        }
213    }
214
215    fn points(&self) -> impl Iterator<Item = (f64, f64)> + '_ {
216        self.done
217            .iter()
218            .copied()
219            .chain(self.open.iter().flat_map(Column::points))
220    }
221}

223/// The quickest a plot redraws for time moving on alone: a display redraws
224/// all of itself each frame, so a strip chart must not keep it at the
225/// 10 frames a second PV updates are held to.
226const MIN_SCROLL_REPAINT: f64 = 0.25;
227
228/// How long until a plot of `span` seconds over `width` points needs a
229/// redraw for time moving on: when it has scrolled a pixel, but not more
230/// often than [`MIN_SCROLL_REPAINT`] nor less than once a second.
231fn scroll_repaint(span: f64, width: f32) -> std::time::Duration {
232    let per_pixel = span / width.max(1.0) as f64;
233    std::time::Duration::from_secs_f64(per_pixel.clamp(MIN_SCROLL_REPAINT, 1.0))
234}

236/// The time now, in seconds since 1970.
237pub fn now() -> f64 {
238    SystemTime::now()
239        .duration_since(UNIX_EPOCH)
240        .map_or(0.0, |d| d.as_secs_f64())
241}

243/// What a PV update adds to its trend: its value, a gap if it disconnected,
244/// or nothing if it has no number (yet).
245fn sample(s: &PvState) -> Option<f64> {
246    if s.connected {
247        s.value.as_f64()
248    } else {
249        Some(f64::NAN)
250    }

253/// A PV being trended: a sample on each update, at the time it arrived.
254pub struct Series {
255    pub pv: Pv,
256    /// Locked by the listener while the PV's state is locked, so never read
257    /// `pv.state()` while holding this lock.
258    samples: Arc<Mutex<Samples>>,
259    /// The samples as last drawn.
260    thinned: Mutex<Thinned>,
261    _listener: Listener,
262}
263
264impl Series {
265    pub fn new(pv: Pv) -> Self {
266        let samples = Arc::new(Mutex::new(Samples::default()));
267        let into = samples.clone();
268        let listener: Listener = Arc::new(move |s: &PvState| {
269            if let Some(v) = sample(s) {
270                into.lock().push(now(), v);
271            }
272        });
273        pv.listen(&listener);
274        Self {
275            pv,
276            samples,
277            thinned: Mutex::default(),
278            _listener: listener,
279        }
280    }
281
282    /// The points to draw from `from` to `to`, `dt` seconds a pixel: the
283    /// samples thinned to a few per pixel, with the last value held to `to`
284    /// while the PV is connected.
285    fn points(&self, from: f64, to: f64, dt: f64) -> Vec<(f64, f64)> {
286        // Copied out before the PV's state is read: an update holds that
287        // lock while it adds a sample.
288        let mut pts: Vec<(f64, f64)> = {
289            let mut thinned = self.thinned.lock();
290            thinned.update(&self.samples.lock(), from, dt);
291            thinned.points().collect()
292        };
293        if let Some(&(t, v)) = pts.last()
294            && v.is_finite()
295            && t < to
296            && self.pv.state().connected
297        {
298            pts.push((to, v));
299        }
300        pts
301    }
302}

324/// Plot `lines` over the `span` seconds up to `now`, with a time axis below
325/// and a value axis on the left.
326pub fn draw_plot(
327    painter: &egui::Painter,
328    rect: Rect,
329    lines: &[Line],
330    span: f64,
331    now: f64,
332    look: &Look,
333    z: f32,
334) {
335    let painter = &painter.with_clip_rect(painter.clip_rect().intersect(rect));
336    fill(painter, rect, look.back);
337    let (pad, tick) = (4.0 * z, 5.0 * z);
338    let from = now - span;
339    // Columns no wider than a pixel of the plot, which is narrower than `rect`.
340    let dt = span / rect.width().max(1.0) as f64;
341    let data: Vec<Vec<(f64, f64)>> = lines
342        .iter()
343        .map(|l| l.series.points(from, now, dt))
344        .collect();
345
346    let ok = |v: f64| v.is_finite() && (!look.log || v > 0.0);
347    let in_view = data
348        .iter()
349        .flat_map(|pts| pts.iter().filter(|p| p.0 >= from).map(|p| p.1));
350    let (lo, hi) = look
351        .range
352        .or_else(|| plot::autoscale(look.log, in_view, true))
353        .unwrap_or(if look.log { (1.0, 10.0) } else { (0.0, 1.0) });
354
355    let label_h = text_size(painter, "0", &look.font).y;
356    let bottom = rect.bottom() - pad - label_h - tick;
357    let top = rect.top() + pad + label_h / 2.0;
358    let ys = Scale {
359        lo,
360        hi,
361        log: look.log,
362        from: bottom,
363        to: top,
364    };
365    let (y_ticks, y_step) = ys.ticks(label_h * 2.5);
366    let y_labels: Vec<(f64, String)> = y_ticks
367        .iter()
368        .map(|v| (*v, axis::label(*v, y_step)))
369        .collect();
370    let label_w = y_labels
371        .iter()
372        .map(|(_, l)| text_size(painter, l, &look.font).x)
373        .fold(0.0, f32::max);
374    let title_w = if look.title.is_empty() {
375        0.0
376    } else {
377        text_size(painter, &look.title, &look.title_font).y + pad
378    };
379    let time_w = text_size(painter, "00:00:00", &look.font).x;
380    let left = rect.left() + pad + title_w + label_w + pad / 2.0 + tick;
381    let right = rect.right() - pad - time_w / 2.0;
382    let plot = Rect::from_min_max(Pos2::new(left, top), Pos2::new(right, bottom));
383    if plot.width() < 4.0 || plot.height() < 4.0 {
384        return;
385    }
386    let xs = Scale {
387        lo: from,
388        hi: now,
389        log: false,
390        from: plot.left(),
391        to: plot.right(),
392    };
393    let max_ticks = (plot.width() / (time_w + 20.0 * z)).floor().max(2.0) as usize;
394    let (t_ticks, t_step) = time_ticks(from, now, max_ticks, local_offset(now));
395
396    if look.grid {
397        let grid = Stroke::new(1.0, named("Grid").gamma_multiply(0.4));
398        for t in &t_ticks {
399            let x = xs.pos(*t);
400            painter.line_segment(
401                [Pos2::new(x, plot.top()), Pos2::new(x, plot.bottom())],
402                grid,
403            );
404        }
405        for (v, _) in &y_labels {
406            let y = ys.pos(*v);
407            painter.line_segment(
408                [Pos2::new(plot.left(), y), Pos2::new(plot.right(), y)],
409                grid,
410            );
411        }
412    }
413
414    let clipped = painter.with_clip_rect(painter.clip_rect().intersect(plot));
415    // Far-off points are pulled in so the maths stays finite.
416    let bounds = plot.expand(plot.width().max(plot.height()) * 4.0);
417    for (line, pts) in lines.iter().zip(&data) {
418        let screen = |(t, v): (f64, f64)| {
419            ok(v).then(|| {
420                Pos2::new(
421                    xs.pos(t).clamp(bounds.left(), bounds.right()),
422                    ys.pos(v).clamp(bounds.top(), bounds.bottom()),
423                )
424            })
425        };
426        let pts: Vec<Option<Pos2>> = pts.iter().map(|p| screen(*p)).collect();
427        // A value holds until the next sample, so the line steps; a gap
428        // (disconnected, or not a number) breaks it.
429        for run in pts.split(Option::is_none).filter(|r| !r.is_empty()) {
430            let mut steps: Vec<Pos2> = Vec::with_capacity(run.len() * 2);
431            for p in run.iter().flatten() {
432                if let Some(last) = steps.last().copied() {
433                    steps.push(Pos2::new(p.x, last.y));
434                }
435                steps.push(*p);
436            }
437            plot::line(
438                &clipped,
439                plot::decimate(steps),
440                Stroke::new(line.width, line.color),
441                0,
442            );
443        }
444    }
445
446    // The axes over the plot's edges.
447    let stroke = Stroke::new(z.max(1.0), look.fg);
448    painter.rect_stroke(plot, 0.0, stroke, egui::StrokeKind::Middle);
449    for t in &t_ticks {
450        let x = xs.pos(*t);
451        painter.line_segment(
452            [
453                Pos2::new(x, plot.bottom()),
454                Pos2::new(x, plot.bottom() + tick),
455            ],
456            stroke,
457        );
458        painter.text(
459            Pos2::new(x, plot.bottom() + tick),
460            Align2::CENTER_TOP,
461            time_label(*t, t_step),
462            look.font.clone(),
463            look.fg,
464        );
465    }
466    for (v, label) in &y_labels {
467        let y = ys.pos(*v);
468        painter.line_segment(
469            [Pos2::new(plot.left() - tick, y), Pos2::new(plot.left(), y)],
470            stroke,
471        );
472        painter.text(
473            Pos2::new(plot.left() - tick - pad / 2.0, y),
474            Align2::RIGHT_CENTER,
475            label,
476            look.font.clone(),
477            look.fg,
478        );
479    }
480    if !look.title.is_empty() {
481        let h = title_w - pad;
482        vertical_text(
483            painter,
484            Pos2::new(rect.left() + pad + h / 2.0, plot.center().y),
485            &look.title,
486            look.title_font.clone(),
487            look.fg,
488        );
489    }
490}

492/// Steps for the time axis, in seconds: whole seconds, minutes and hours.
493const TIME_STEPS: &[f64] = &[
494    1.0, 2.0, 5.0, 10.0, 15.0, 30.0, 60.0, 120.0, 300.0, 600.0, 900.0, 1800.0, 3600.0, 7200.0,
495    10_800.0, 21_600.0, 43_200.0, 86_400.0,
496];
497
498/// Ticks from `from` to `to` (seconds since 1970) at the smallest step
499/// giving at most `max_ticks`, on whole steps of local time (`offset`
500/// seconds ahead of UTC). Returns the ticks and the step.
501fn time_ticks(from: f64, to: f64, max_ticks: usize, offset: f64) -> (Vec<f64>, f64) {
502    let span = to - from;
503    if span <= 0.0 || !span.is_finite() {
504        return (Vec::new(), 0.0);
505    }
506    let step = TIME_STEPS
507        .iter()
508        .copied()
509        .find(|s| span / s <= max_ticks.max(1) as f64)
510        .unwrap_or_else(|| 86_400.0 * (span / 86_400.0 / max_ticks.max(1) as f64).ceil());
511    let first = ((from + offset) / step).ceil() * step - offset;
512    let ticks = (0..)
513        .map(|i| first + i as f64 * step)
514        .take_while(|t| *t <= to)
515        .take(1000)
516        .collect();
517    (ticks, step)
518}
519
520/// How far local time is ahead of UTC at `t`, in seconds.
521fn local_offset(t: f64) -> f64 {
522    Local
523        .timestamp_opt(t as i64, 0)
524        .single()
525        .map_or(0.0, |d| d.offset().local_minus_utc() as f64)
526}
527
528/// A time axis label in local time, as precise as the step.
529fn time_label(t: f64, step: f64) -> String {
530    let Some(d) = Local.timestamp_opt(t.round() as i64, 0).single() else {
531        return String::new();
532    };
533    let format = match step {
534        s if s < 60.0 => "%H:%M:%S",
535        s if s < 86_400.0 => "%H:%M",
536        _ => "%d %b",
537    };
538    d.format(format).to_string()
539}

594        }

598    /// Add the PVs `names` that it doesn't have yet, each taking the next
599    /// line colour as a new trend's PVs do.

603        }

606    fn add(&mut self, name: &str) {
607        let name = name.trim();
608        let key = pv::PvName::parse(name).key();
609        if !name.is_empty() && !self.series.iter().any(|s| s.pv.name().key() == key) {
610            self.series.push(Series::new(self.pvs.subscribe(name)));
611        }
612    }

616        let names: Vec<String> = self
617            .series
618            .iter()
619            .map(|s| crate::app::plain_name(s.pv.name()))
620            .collect();
621        match names.as_slice() {
622            [] => "Trend".into(),
623            [one] => format!("Trend: {one}"),
624            [a, b] => format!("Trend: {a}, {b}"),
625            [first, rest @ ..] => format!("Trend: {first} and {} more", rest.len()),
626        }
627    }

640            let edit = ui.add(
641                egui::TextEdit::singleline(&mut self.adding)

643                    .desired_width(160.0),
644            );
645            if edit.lost_focus() && ui.input(|i| i.key_pressed(egui::Key::Enter)) {
646                let name = std::mem::take(&mut self.adding);
647                self.add(&name);
648            }
649        });
650
651        // The legend under the plot: a row per PV.
652        let row = ui.spacing().interact_size.y + ui.spacing().item_spacing.y;
653        let legend_h = row * self.series.len() as f32 + ui.spacing().item_spacing.y;
654        let size = Vec2::new(
655            ui.available_width(),
656            (ui.available_height() - legend_h).max(80.0),
657        );
658        let (rect, _) = ui.allocate_exact_size(size, egui::Sense::hover());
659        let lines: Vec<Line> = self
660            .series
661            .iter()
662            .enumerate()
663            .map(|(i, s)| Line {
664                series: s,
665                color: line_color(i),
666                width: 1.5,
667            })
668            .collect();
669        let font = egui::TextStyle::Small.resolve(ui.style());
670        let look = Look {
671            log: self.log,
672            range: None,
673            title: common_units(&self.series),
674            grid: true,
675            fg: ui.visuals().text_color(),
676            back: ui.visuals().extreme_bg_color,
677            title_font: font.clone(),
678            font,
679        };
680        draw_plot(ui.painter(), rect, &lines, self.span, now(), &look, 1.0);
681
682        let mut remove = None;
683        egui::Grid::new(("trend-legend", self.id))
684            .num_columns(4)
685            .show(ui, |ui| {
686                for (i, s) in self.series.iter().enumerate() {
687                    let state = s.pv.state();
688                    let (swatch, _) =
689                        ui.allocate_exact_size(Vec2::new(24.0, 12.0), egui::Sense::hover());
690                    ui.painter().line_segment(
691                        [swatch.left_center(), swatch.right_center()],
692                        Stroke::new(3.0, line_color(i)),
693                    );
694                    ui.label(crate::app::plain_name(s.pv.name()));
695                    if state.connected {
696                        let text = egui::RichText::new(state.format(None, true));
697                        ui.label(match style::severity_color(state.severity) {
698                            Some(c) => text.color(c),
699                            None => text,
700                        });
701                    } else {
702                        ui.label(egui::RichText::new("Disconnected").color(Color32::GRAY));
703                    }
704                    if ui.small_button("✖").on_hover_text("Remove").clicked() {
705                        remove = Some(i);
706                    }
707                    ui.end_row();
708                }
709            });
710        if let Some(i) = remove {
711            self.series.remove(i);
712        }
713
714        // Keep time moving while the PVs are quiet.
715        ui.ctx()
716            .request_repaint_after(scroll_repaint(self.span, rect.width()));
717    }
718}

720fn span_name(span: f64) -> String {
721    SPANS
722        .iter()
723        .find(|(s, _)| *s == span)
724        .map_or_else(|| format!("{span} s"), |(_, n)| n.to_string())

727/// The PVs' units, when they all have the same.
728fn common_units(series: &[Series]) -> String {
729    let units: Vec<String> = series.iter().map(|s| s.pv.state().units).collect();
730    match units.split_first() {
731        Some((first, rest)) if rest.iter().all(|u| u == first) => first.clone(),
732        _ => String::new(),
733    }
734}

738/// A strip chart's `time_range`, such as `1 minute` or `1 hour 30 min`, in
739/// seconds.
740pub fn parse_span(text: &str) -> Option<f64> {
741    let mut total = 0.0;
742    let mut words = text.split_whitespace().peekable();
743    let mut any = false;
744    while let Some(word) = words.next() {
745        // A number and its unit, apart (`10 min`) or together (`10min`).
746        let split = word
747            .find(|c: char| !(c.is_ascii_digit() || c == '.'))
748            .unwrap_or(word.len());
749        let (num, unit) = word.split_at(split);
750        let n: f64 = num.parse().ok()?;
751        let unit = if unit.is_empty() {
752            words.next().unwrap_or("s")
753        } else {
754            unit
755        };
756        let secs = match unit.to_ascii_lowercase().trim_end_matches('s') {
757            "" | "sec" | "second" => 1.0,
758            "m" | "min" | "minute" => 60.0,
759            "h" | "hour" => 3600.0,
760            "d" | "day" => 86_400.0,
761            "w" | "week" => 7.0 * 86_400.0,
762            _ => return None,
763        };
764        total += n * secs;
765        any = true;
766    }
767    (any && total > 0.0).then_some(total)
768}

770/// Draw a `.bob` strip chart: its traces' PVs over the last `time_range`,
771/// on its first value axis.
772pub fn draw_stripchart(painter: &egui::Painter, node: &Node, rect: Rect, z: f32) {
773    let w = &node.w;
774    let back = w
775        .color("background_color")
776        .map(|c| color(&c))
777        .unwrap_or(Color32::WHITE);
778    fill(painter, rect, back);
779    let fg = w
780        .color("foreground_color")
781        .map(|c| color(&c))
782        .unwrap_or(Color32::BLACK);
783    let pad = 4.0 * z;
784    let top = plot::title(painter, node, rect, fg, z);
785
786    let traces = plot::traces(node, z);
787    let shown: Vec<(&plot::Trace, &Series)> = traces
788        .iter()
789        .zip(&node.series)
790        .filter(|(t, _)| t.visible)
791        .filter_map(|(t, s)| Some((t, s.as_ref()?)))
792        .collect();
793    let legend_font = style::font_id(&w.font("legend_font").unwrap_or_default(), z);
794    let legend_h = if w.bool("show_legend").unwrap_or(true) && !shown.is_empty() {
795        text_size(painter, "Xg", &legend_font).y + pad
796    } else {
797        0.0
798    };
799    if legend_h > 0.0 {
800        let named: Vec<&plot::Trace> = shown.iter().map(|(t, _)| *t).collect();
801        plot::legend(painter, &named, rect, legend_h, pad, &legend_font, fg, z);
802    }
803
804    let axis = plot::axis_cfg(
805        w.get("y_axes").and_then(|e| e.child("y_axis")),
806        "Value",
807        node,
808        z,
809    );
810    let span = w
811        .text("time_range")
812        .map(|t| node.expand(t))
813        .and_then(|t| parse_span(&t))
814        .unwrap_or(60.0);
815    let look = Look {
816        log: axis.log,
817        range: (!axis.autoscale).then_some(axis.range),
818        title: if axis.visible {
819            axis.title.clone()
820        } else {
821            String::new()
822        },
823        grid: axis.grid,
824        fg: axis.color,
825        back,
826        font: axis.scale_font,
827        title_font: axis.title_font,
828    };
829    let lines: Vec<Line> = shown
830        .iter()
831        .map(|(t, s)| Line {
832            series: s,
833            color: t.color,
834            width: t.width,
835        })
836        .collect();
837    let area = Rect::from_min_max(
838        Pos2::new(rect.left(), top),
839        Pos2::new(rect.right(), rect.bottom() - legend_h),
840    );
841    draw_plot(painter, area, &lines, span, now(), &look, z);
842    // Keep time moving while the PVs are quiet, as `Trend::ui` does.
843    painter
844        .ctx()
845        .request_repaint_after(scroll_repaint(span, area.width()));
846
847    if let Some(s) = plot::worst_state(node) {
848        alarm_border(painter, w, &s, rect, z);
849    }
850}

952    #[test]
953    fn thinning_takes_in_only_new_samples_and_keeps_gaps() {
954        let mut s = Samples::default();
955        let mut t = Thinned::default();
956        let push = |s: &mut Samples, t0: f64, n: usize| {
957            for i in 0..n {
958                let x = t0 + i as f64 * 0.05;
959                s.push(x, (x * 3.0).cos());
960            }
961        };
962        push(&mut s, 0.0, 500);
963        t.update(&s, 0.0, 0.5);
964        s.push(25.0, f64::NAN);
965        push(&mut s, 26.0, 500);
966        t.update(&s, 0.0, 0.5);
967        assert_eq!(t.seen, s.pushed);
968        // Taking in the new samples gives what thinning them all at once does.
969        let all: Vec<(f64, f64)> = t.points().collect();
970        assert_eq!(bits(&all), bits(&thinned(&s, 0.0, 0.5)));
971        assert_eq!(all.iter().filter(|p| p.1.is_nan()).count(), 1, "the gap");
972        // Time moving on forgets what has scrolled off, but the point that
973        // holds at the start.
974        t.update(&s, 40.0, 0.5);
975        let later: Vec<(f64, f64)> = t.points().collect();
976        assert!(later[0].0 < 40.0 && later[1].0 >= 40.0);
977        // A new width rebuilds it.
978        t.update(&s, 40.0, 0.25);
979        let rebuilt: Vec<(f64, f64)> = t.points().collect();
980        assert_eq!(bits(&rebuilt), bits(&thinned(&s, 40.0, 0.25)));
