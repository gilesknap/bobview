1//! Filling polygons that may be concave (L shapes, arrows, synoptic
2//! outlines), which egui's `convex_polygon` fills wrongly.
3
4use std::collections::HashMap;
5use std::hash::{Hash, Hasher};
6use std::sync::Arc;
7
8use eframe::egui::{self, Color32, Mesh, Pos2, Shape, Stroke};
9
10/// Outlines with more points than this are filled as a fan, as before:
11/// ear clipping is quadratic or worse in the number of points.
12const MAX_POINTS: usize = 1000;
13
14/// Fill the polygon `screen`, the points `display` placed on the screen.
15/// A convex one is filled by egui, with anti-aliased edges; a concave one
16/// is cut into triangles, worked out once for its display points (moving
17/// or zooming it does not change them) and kept in a bounded cache in
18/// egui's memory.
19pub fn fill(painter: &egui::Painter, display: &[(f64, f64)], screen: Vec<Pos2>, color: Color32) {
20    if screen.len() < 3 || color.a() == 0 {
21        return;
22    }
23    let mut hasher = std::collections::hash_map::DefaultHasher::new();
24    for (x, y) in display {
25        x.to_bits().hash(&mut hasher);
26        y.to_bits().hash(&mut hasher);
27    }
28    let key = hasher.finish();
29    let id = egui::Id::new("bobview-polygons");
30    let triangles = painter.ctx().memory_mut(|m| {
31        m.data
32            .get_temp_mut_or_default::<Cache>(id)
33            .get_or_insert(key, || {
34                let pts: Vec<Pos2> = display
35                    .iter()
36                    .map(|(x, y)| Pos2::new(*x as f32, *y as f32))
37                    .collect();
38                triangulate(&pts)
39            })
40    });
41    match triangles.as_ref() {
42        Some(indices) => {
43            let mut mesh = Mesh::default();
44            for p in &screen {
45                mesh.colored_vertex(*p, color);
46            }
47            mesh.indices = indices.clone();
48            painter.add(Shape::mesh(mesh));
49        }
50        _ => {
51            painter.add(Shape::convex_polygon(screen, color, Stroke::NONE));
52        }
53    }
54}
55
56/// Triangles for each outline drawn, by a hash of its display points.
57type Triangles = Arc<Option<Vec<u32>>>;
58
59/// How many outlines the cache takes before it starts a new generation.
60const CACHE_SIZE: usize = 4096;
61
62/// The triangles worked out for outlines, in two generations so that it
63/// stays bounded when a rule keeps changing a polygon's points: once the
64/// newer is full it becomes the older and the older is dropped. An outline
65/// still in use is found in the older and moved to the newer, so it is
66/// not cut again.
67#[derive(Clone, Default)]
68struct Cache {
69    newer: HashMap<u64, Triangles>,
70    older: HashMap<u64, Triangles>,
71}
72
73impl Cache {
74    fn get_or_insert(&mut self, key: u64, make: impl FnOnce() -> Option<Vec<u32>>) -> Triangles {
75        if let Some(t) = self.newer.get(&key) {
76            return t.clone();
77        }
78        let t = self.older.remove(&key).unwrap_or_else(|| Arc::new(make()));
79        if self.newer.len() >= CACHE_SIZE {
80            self.older = std::mem::take(&mut self.newer);
81        }
82        self.newer.insert(key, t.clone());
83        t
84    }
85}

87/// Twice the signed area of `pts`: positive when they go anticlockwise in
88/// y-up coordinates (clockwise on a y-down screen).
89fn area2(pts: &[Pos2]) -> f32 {
90    let n = pts.len();
91    (0..n)
92        .map(|i| {
93            let (a, b) = (pts[i], pts[(i + 1) % n]);
94            a.x * b.y - b.x * a.y
95        })
96        .sum()
97}
98
99fn cross(o: Pos2, a: Pos2, b: Pos2) -> f32 {
100    (a - o).x * (b - o).y - (a - o).y * (b - o).x
101}
102
103/// Whether `p` is inside or on the triangle `a b c`, which turns the way
104/// `sign` says.
105fn in_triangle(p: Pos2, a: Pos2, b: Pos2, c: Pos2, sign: f32) -> bool {
106    cross(a, b, p) * sign >= 0.0 && cross(b, c, p) * sign >= 0.0 && cross(c, a, p) * sign >= 0.0
107}

109/// The triangles filling the outline `pts`, three indices into it each, by
110/// ear clipping; `None` for a convex outline (egui fills those itself) or
111/// one too big to cut. An outline that crosses itself runs out of ears at
112/// some point, and what is left is filled as a fan.
113pub fn triangulate(pts: &[Pos2]) -> Option<Vec<u32>> {
114    let n = pts.len();
115    if !(4..=MAX_POINTS).contains(&n) {
116        return None;
117    }
118    let area = area2(pts);
119    if area == 0.0 {
120        return None;
121    }
122    let sign = area.signum();
123    // Convex: every corner turns the same way as the whole outline.
124    let turns = |i: usize| cross(pts[(i + n - 1) % n], pts[i], pts[(i + 1) % n]) * sign;
125    if (0..n).all(|i| turns(i) >= 0.0) {
126        return None;
127    }
128
129    let mut left: Vec<usize> = (0..n).collect();
130    let mut out = Vec::with_capacity(3 * (n - 2));
131    while left.len() > 3 {
132        let m = left.len();
133        let mut cut = None;
134        for i in 0..m {
135            let (a, b, c) = (left[(i + m - 1) % m], left[i], left[(i + 1) % m]);
136            let turn = cross(pts[a], pts[b], pts[c]) * sign;
137            if turn == 0.0 {
138                // A point on a straight edge (or repeated) adds nothing.
139                cut = Some((i, None));
140                break;
141            }
142            if turn < 0.0 {
143                continue;
144            }
145            // An ear: no other point of the outline in it.
146            let blocked = left.iter().any(|&k| {
147                k != a
148                    && k != b
149                    && k != c
150                    && pts[k] != pts[a]
151                    && pts[k] != pts[b]
152                    && pts[k] != pts[c]
153                    && in_triangle(pts[k], pts[a], pts[b], pts[c], sign)
154            });
155            if !blocked {
156                cut = Some((i, Some([a, b, c])));
157                break;
158            }
159        }
160        match cut {
161            Some((i, tri)) => {
162                out.extend(tri.into_iter().flatten().map(|k| k as u32));
163                left.remove(i);
164            }
165            None => {
166                // No ear: the outline crosses itself. Fan out the rest.
167                for k in 1..left.len() - 1 {
168                    out.extend([left[0], left[k], left[k + 1]].map(|k| k as u32));
169                }
170                return Some(out);
171            }
172        }
173    }
174    out.extend(left.iter().map(|&k| k as u32));
175    Some(out)
176}

182    #[test]
183    fn the_cache_stays_bounded_and_keeps_outlines_in_use() {
184        let mut cache = Cache::default();
185        let mut cuts = 0;
186        let mut get = |cache: &mut Cache, key| {
187            cache.get_or_insert(key, || {
188                cuts += 1;
189                None
190            });
191        };
192        // A rule moving one polygon through ever new shapes, while another
193        // stays as it is.
194        for key in 1..=10 * CACHE_SIZE as u64 {
195            get(&mut cache, 0);
196            get(&mut cache, key);
197            assert!(cache.newer.len() + cache.older.len() <= 2 * CACHE_SIZE);
198        }
199        drop(get);
200        assert_eq!(cuts, 1 + 10 * CACHE_SIZE, "the steady one is cut once");
201    }
202
203    fn pts(p: &[(f32, f32)]) -> Vec<Pos2> {
204        p.iter().map(|&(x, y)| Pos2::new(x, y)).collect()
205    }
206
207    /// The area the triangles cover, which for a correct cut is the
208    /// polygon's own.
209    fn covered(p: &[Pos2], tris: &[u32]) -> f32 {
210        tris.chunks(3)
211            .map(|t| cross(p[t[0] as usize], p[t[1] as usize], p[t[2] as usize]).abs() / 2.0)
212            .sum()
213    }
214
215    /// Whether `q` is inside any of the triangles.
216    fn filled(p: &[Pos2], tris: &[u32], q: Pos2) -> bool {
217        tris.chunks(3).any(|t| {
218            let (a, b, c) = (p[t[0] as usize], p[t[1] as usize], p[t[2] as usize]);
219            in_triangle(q, a, b, c, cross(a, b, c).signum())
220        })
221    }

223    #[test]
224    fn convex_polygons_are_left_to_egui() {
225        assert_eq!(
226            triangulate(&pts(&[(0., 0.), (10., 0.), (10., 10.), (0., 10.)])),
227            None
228        );
229        assert_eq!(triangulate(&pts(&[(0., 0.), (10., 0.), (5., 10.)])), None);
230    }

232    #[test]
233    fn an_l_shape_is_filled_but_not_its_notch() {
234        // An L, both ways round: a convex fill would cover the notch.
235        let l = pts(&[
236            (0., 0.),
237            (10., 0.),
238            (10., 4.),
239            (4., 4.),
240            (4., 10.),
241            (0., 10.),
242        ]);
243        for outline in [l.clone(), l.into_iter().rev().collect()] {
244            let tris = triangulate(&outline).expect("concave");
245            assert_eq!(tris.len(), 3 * (outline.len() - 2));
246            assert_eq!(covered(&outline, &tris), 64.0);
247            assert!(filled(&outline, &tris, Pos2::new(2.0, 8.0)));
248            assert!(filled(&outline, &tris, Pos2::new(8.0, 2.0)));
249            assert!(!filled(&outline, &tris, Pos2::new(8.0, 8.0)), "the notch");
250        }
251    }
