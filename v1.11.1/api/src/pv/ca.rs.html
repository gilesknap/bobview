14/// The `DBR_CHAR` request type, which a long string is written as, as
15/// `caput -S` does; the server converts it to the field's own type.
16const DBR_CHAR: u16 = 4;
17
18/// Whether to write `put` as a long string to a PV whose native type and
19/// element count are `native` and `count`, as Phoebus's CA layer decides
20/// it for text: when the PV is a char array (`DBF_CHAR`, more than one
21/// element). That depends on the channel, not on the value last read, which
22/// for an array holding one element can arrive as a scalar.
23///
24/// With a string `format` all text is written so. Otherwise text that is a
25/// list of numbers is written as numbers, as Phoebus parses a text entry's
26/// text for an array; its list splits on commas only, and its numbers are
27/// finite, so `1 2`, `nan` or `inf` are text, as in Phoebus.
28fn as_long_string(put: &Put, native: Option<DbFieldType>, count: Option<usize>) -> bool {
29    let char_array = matches!(native, Some(DbFieldType::Char | DbFieldType::UChar))
30        && count.is_some_and(|n| n > 1);
31    char_array && (put.long_string || !is_number_list(&put.text))
32}
33
34/// Whether `text` is a list of finite numbers, `[1, 2]` or `1,2`, as
35/// Phoebus's `FormatOptionHandler.parse` reads one for an array.
36fn is_number_list(text: &str) -> bool {
37    let t = text.trim();
38    let t = t.strip_prefix('[').unwrap_or(t);
39    let t = t.strip_suffix(']').unwrap_or(t);
40    t.split(',')
41        .all(|x| x.trim().parse::<f64>().is_ok_and(f64::is_finite))
42}

44/// How long a write to a PV that has yet to connect waits for it.
45const WRITE_CONNECT: std::time::Duration = std::time::Duration::from_secs(5);
46
47pub(crate) async fn run(client: &CaClient, pv: &str, shared: Arc<Shared>) {
48    let channel = client.create_channel(pv);
49    let mut events = channel.connection_events();
50    let (tx, mut writes) = crate::write_queue();
51    shared.set_writer(Writer::Channel(tx));
52
53    let mut retry = std::time::Duration::from_millis(500);
54    let mut monitor = loop {
55        // Value and alarm changes, as Phoebus asks for: not DBE_LOG, which
56        // archive deadbands (ADEL) send, and the display does not need.
57        match channel
58            .subscribe_with_mask(0.0, DBE_VALUE | DBE_ALARM)
59            .await
60        {
61            Ok(m) => break m,
62            Err(e) => {
63                log::warn!("{pv}: cannot subscribe, retrying in {retry:?}: {e}");
64                shared.update(|s| s.error = Some(e.to_string()));
65                tokio::time::sleep(retry).await;
66                retry = (retry * 2).min(std::time::Duration::from_secs(30));
67            }
68        }
69    };
70
71    // When to read the value if no monitor update has come since the channel
72    // connected; see `MONITOR_GRACE`.
73    let mut fallback: Option<tokio::time::Instant> = None;
74    // Whether the channel is connected, though its value may not have come.
75    let mut linked = false;
76    // A write to a PV that has yet to connect, as a `write_pv` action to a
77    // PV not connected in advance makes, and when it gives up waiting. The
78    // writes behind it stay queued meanwhile, so they keep their order, and
79    // the PV's events are still served.
80    let mut waiting: Option<(Put, tokio::time::Instant)> = None;
81    loop {
82        if let Some((put, _)) = waiting.take_if(|_| shared.connected())
83            && shared.can_write(&put.text)
84        {
85            write(&channel, &shared, &put).await;
86        }
87        tokio::select! {
88            event = events.recv() => match event {
89                Ok(ConnectionEvent::Connected) => {
90                    linked = true;
91                    fetch_metadata(&channel, &shared).await;
92                    fallback = Some(tokio::time::Instant::now() + MONITOR_GRACE);
93                }
94                Ok(ConnectionEvent::Disconnected) => {
95                    linked = false;
96                    fallback = None;
97                    shared.update(|s| s.connected = false);
98                }
99                Ok(ConnectionEvent::AccessRightsChanged { write, .. }) => {
100                    shared.update(|s| { s.writable = write; s.writable_known = true; });
101                }
102                Ok(_) | Err(broadcast::error::RecvError::Lagged(_)) => {}
103                Err(broadcast::error::RecvError::Closed) => return,
104            },
105            _ = tokio::time::sleep_until(fallback.unwrap_or_else(tokio::time::Instant::now)),
106                if fallback.is_some() =>
107            {
108                fallback = None;
109                if !shared.connected() {
110                    read_value(&channel, &shared).await;
111                }
112            }
113            update = monitor.recv() => match update {
114                Some(Ok(snap)) => {
115                    fallback = None;
116                    apply(&shared, &channel, &snap, false);
117                }
118                Some(Err(e)) => {
119                    log::debug!("{pv}: {e}");
120                    shared.update(|s| s.connected = false);
121                }
122                None => return,
123            },
124            Some(put) = writes.recv(), if waiting.is_none() => {
125                // Refused at once if the PV has dropped.
126                if !shared.can_write(&put.text) {
127                    continue;
128                }
129                if shared.connected() {
130                    write(&channel, &shared, &put).await;
131                } else {
132                    waiting = Some((put, tokio::time::Instant::now() + WRITE_CONNECT));
133                }
134            }
135            _ = tokio::time::sleep_until(
136                waiting.as_ref().map_or_else(tokio::time::Instant::now, |w| w.1)
137            ), if waiting.is_some() => {
138                let Some((put, _)) = waiting.take() else { continue };
139                if linked {
140                    // Connected, but no value has come yet: send it as text.
141                    write(&channel, &shared, &put).await;
142                    continue;
143                }
144                // Those queued behind it fail too, rather than each waiting
145                // in turn.
146                let why = format!("not connected after {}s", WRITE_CONNECT.as_secs());
147                shared.write_result(&put.text, Some(why.clone()));
148                while let Ok(put) = writes.try_recv() {
149                    shared.write_result(&put.text, Some(why.clone()));
150                }
151            }
152        }
153    }
154}

156/// Send one write to a connected channel and report how it went.
157async fn write(channel: &CaChannel, shared: &Shared, put: &Put) {
158    let text = &put.text;
159    let value = shared.snapshot().value;
160    let native = channel.native_field_type().ok();
161    let count = channel.element_count().ok().map(|n| n as usize);
162    // Plain CA_PROTO_WRITE, as Phoebus does: a put-callback on a busy
163    // record (areaDetector Acquire) completes only when it returns to 0,
164    // and the IOC queues any later put-callback on that record behind it,
165    // so Stop could never reach it. Sent from the PV's task, not a spawned
166    // one, so writes keep their order.
167    let result = if as_long_string(put, native, count) {
168        let bytes = crate::long_string_bytes(text, count);
169        channel
170            .put_as_dbr_nowait(DBR_CHAR, &EpicsValue::CharArray(bytes.into()))
171            .await
172    } else {
173        // Arrays are typed as they are shown, `[1, 2]` or `["a", "b"]`; the
174        // server converts each string element to the field type.
175        let items: Option<Vec<String>> = match value {
176            Value::DoubleArray(_) | Value::LongArray(_) => Some(
177                crate::split_array(text)
178                    .into_iter()
179                    .map(String::from)
180                    .collect(),
181            ),
182            Value::StrArray(_) => Some(crate::split_strings(text)),
183            _ => None,
184        };
185        if let Some(items) = items {
186            channel.put_string_array_nowait(&items).await
187        } else {
188            channel.put_string_nowait(text).await
189        }
190    };
191    // No reply comes to a plain write: `Ok` only says it was sent.
192    shared.write_sent(text, result.err().map(|e| e.to_string()));
193}

195/// How long after a channel connects its monitor has to send the first
196/// update before the value is read instead. The server sends one at once,
197/// but epics-ca-rs leaves a subscription uninstalled when it cannot re-send
198/// it on a reconnect (as C's `nciu::resubscribe` does), and the PV would
199/// then never show as connected.
200const MONITOR_GRACE: std::time::Duration = std::time::Duration::from_secs(5);
201
202/// Read the value, alarm and time stamp, for a monitor that has sent
203/// nothing since the channel connected.
204async fn read_value(channel: &CaChannel, shared: &Shared) {
205    match channel.get_with_metadata(DbrClass::Time).await {
206        Ok(snap) => {
207            log::warn!(
208                "{}: no monitor update since connecting; showing a read value",
209                shared.name
210            );
211            apply(shared, channel, &snap, false);
212        }
213        Err(e) => log::debug!("{}: value read failed: {e}", shared.name),
214    }
215}

217/// Read units, precision, limits and enum labels once per connection.
218async fn fetch_metadata(channel: &CaChannel, shared: &Shared) {
219    match channel.get_with_metadata(DbrClass::Ctrl).await {
220        Ok(snap) => apply(shared, channel, &snap, true),
221        Err(e) => {
222            log::debug!("metadata read failed: {e}");
223            shared.update(|s| s.connected = true);
224        }
225    }
226}

228fn apply(shared: &Shared, channel: &CaChannel, snap: &Snapshot, metadata: bool) {
229    let value = (!metadata).then(|| {
230        let value = convert(&snap.value);
231        // An array holding one element arrives as a scalar; keep it an
232        // array, as Phoebus does for a channel of more than one element, so
233        // that it is shown and written as one.
234        if channel.element_count().is_ok_and(|n| n > 1) {
235            into_array(value)
236        } else {
237            value
238        }
239    });
240    shared.update(|s| apply_to(s, snap, value, metadata));
241}
242
243/// Apply a monitor update, or the metadata read once per connection. A
244/// DBR_CTRL read carries no time stamp (it decodes as 1970), and may land
245/// after the first monitor update, so it sets only the metadata: units,
246/// precision, limits and labels.
247fn apply_to(s: &mut PvState, snap: &Snapshot, value: Option<Value>, metadata: bool) {
248    s.error = None;
249    if let Some(value) = value {
250        s.connected = true;
251        s.value = value;
252        s.severity = Severity::from_epics(snap.alarm.severity);
253        s.status = snap.alarm.status;
254        s.timestamp = Some(snap.timestamp.into());
255    }
256    if !metadata {
257        return;
258    }
259    if let Some(d) = &snap.display {
260        s.units = d.units.as_str_lossy().into_owned();
261        s.precision = Some(d.precision as i32);
262        s.display_range = Some((d.lower_disp_limit, d.upper_disp_limit));
263        s.alarm_range = Some((d.lower_alarm_limit, d.upper_alarm_limit));
264        s.warning_range = Some((d.lower_warning_limit, d.upper_warning_limit));
265    }
266    if let Some(c) = &snap.control {
267        s.control_range = Some((c.lower_ctrl_limit, c.upper_ctrl_limit));
268    }
269    if let Some(e) = &snap.enums {
270        s.labels = e
271            .strings
272            .iter()
273            .map(|l| l.as_str_lossy().into_owned())
274            .collect();
275    }
276}

278/// A scalar as an array of its one element; any other value as it is.
279fn into_array(value: Value) -> Value {
280    match value {
281        Value::Double(x) => Value::DoubleArray(Arc::from([x])),
282        Value::Long(x) => Value::LongArray(Arc::from([x])),
283        Value::Enum(x) => Value::LongArray(Arc::from([x as i64])),
284        Value::Str(x) => Value::StrArray(Arc::from([x])),
285        other => other,
286    }
287}

289fn convert(v: &EpicsValue) -> Value {
290    use EpicsValue as E;
291    fn longs<T: Copy + Into<i64>>(a: &[T]) -> Value {
292        Value::LongArray(a.iter().map(|x| (*x).into()).collect())
293    }
294    match v {
295        E::String(s) => Value::Str(s.as_str_lossy().into_owned()),
296        E::Short(x) => Value::Long(*x as i64),
297        E::Float(x) => Value::Double(*x as f64),
298        E::Enum(x) => Value::Enum(*x),
299        E::EnumWithChoices { index, .. } => Value::Enum(*index),
300        E::Char(x) => Value::Long(*x as i64),
301        E::Long(x) => Value::Long(*x as i64),
302        E::Double(x) => Value::Double(*x),
303        E::Int64(x) => Value::Long(*x),
304        E::UInt64(x) => Value::from_u64(*x),
305        E::UShort(x) => Value::Long(*x as i64),
306        E::ULong(x) => Value::Long(*x as i64),
307        E::UChar(x) => Value::Long(*x as i64),
308        E::ShortArray(a) => longs(a),
309        E::FloatArray(a) => Value::DoubleArray(a.iter().map(|x| *x as f64).collect()),
310        E::EnumArray(a) => longs(a),
311        E::DoubleArray(a) => Value::DoubleArray(Arc::from(a.as_slice())),
312        E::LongArray(a) => longs(a),
313        E::CharArray(a) => longs(a),
314        E::Int64Array(a) => longs(a),
315        E::UInt64Array(a) => Value::from_u64s(a),
316        E::UShortArray(a) => longs(a),
317        E::ULongArray(a) => longs(a),
318        E::UCharArray(a) => longs(a),
319        E::StringArray(a) => {
320            Value::StrArray(a.iter().map(|s| s.as_str_lossy().into_owned()).collect())
321        }
322    }
323}

331    #[test]
332    fn metadata_after_a_monitor_update_keeps_its_value_and_time() {
333        let stamp = UNIX_EPOCH + Duration::from_secs(1_700_000_000);
334        let monitor = Snapshot::new(EpicsValue::Double(5.0), 3, 1, stamp);
335        // DBR_CTRL has no time: it decodes as 1970. Its value and alarm may
336        // be older than the monitor's on a reconnect.
337        let mut ctrl = Snapshot::new(EpicsValue::Double(0.0), 0, 0, UNIX_EPOCH);
338        ctrl.display = Some(DisplayInfo {
339            units: "mm".into(),
340            precision: 2,
341            upper_disp_limit: 10.0,
342            ..Default::default()
343        });
344
345        let mut s = PvState::default();
346        apply_to(&mut s, &monitor, Some(convert(&monitor.value)), false);
347        apply_to(&mut s, &ctrl, None, true);
348        assert!(s.connected);
349        assert_eq!(s.value, Value::Double(5.0));
350        assert_eq!(s.severity, Severity::Minor);
351        assert_eq!(s.status, 3);
352        assert_eq!(s.timestamp, Some(stamp));
353        assert_eq!(s.units, "mm");
354        assert_eq!(s.precision, Some(2));
355        assert_eq!(s.display_range, Some((0.0, 10.0)));
356    }

396    #[test]
397    fn chooses_long_string_writes_as_phoebus_does() {
398        let char_wf = (Some(DbFieldType::Char), Some(256));
399        let long_string = |text, string_format, (native, count)| {
400            as_long_string(&put(text, string_format), native, count)
401        };
402        // A string format writes all text into a char array, whatever
403        // value was last read (one element arrives as a scalar).
404        assert!(long_string("/data", true, char_wf));
405        assert!(long_string("1, 2", true, char_wf));
406        assert!(long_string("", true, char_wf));
407        // ... but not into any other PV, which is written the usual way,
408        // as Phoebus writes text to it.
409        assert!(!long_string(
410            "/data",
411            true,
412            (Some(DbFieldType::Long), Some(256))
413        ));
414        assert!(!long_string(
415            "/data",
416            true,
417            (Some(DbFieldType::String), Some(1))
418        ));
419        assert!(!long_string("x", true, (Some(DbFieldType::Char), Some(1))));
420        assert!(!long_string("x", true, (None, None)));
421        // Otherwise text that is not a list of numbers, to a char array.
422        assert!(long_string("/data", false, char_wf));
423        assert!(long_string("", false, char_wf));
424        assert!(!long_string("[1, 2]", false, char_wf));
425        assert!(!long_string("5", false, char_wf));
426        // Phoebus splits on commas only, and its numbers are finite.
427        assert!(long_string("1 2", false, char_wf));
428        assert!(long_string("nan", false, char_wf));
429        assert!(long_string("inf", false, char_wf));
430        assert!(!long_string(
431            "/data",
432            false,
433            (Some(DbFieldType::Long), Some(256))
434        ));
435        assert!(!long_string("x", false, (Some(DbFieldType::Char), Some(1))));
436    }
