1//! Saved window layouts: which displays are open, in which windows, where.
2//!
3//! A layout is a JSON file in `$XDG_CONFIG_HOME/bobview/layouts` (by default
4//! `~/.config/bobview/layouts`, or `%APPDATA%\bobview\layouts` on Windows), named after the layout; `default.json` is
5//! restored at startup. `--layout` also takes a layout file's path, which is
6//! then the layout's name: loaded from and saved to that file. Positions and
7//! sizes are in desktop points (egui's points times its zoom factor), so they
8//! don't depend on the controls' scale.
9

99/// A number, or the string "fit".
100#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
101#[serde(try_from = "ZoomText", into = "ZoomText")]
102pub enum Zoom {
103    Factor(f32),
104    /// Fit the display to the window.
105    Fit,
106}
107
108/// How a zoom is written.
109#[derive(Clone, Serialize, Deserialize)]
110#[serde(untagged)]
111enum ZoomText {
112    Factor(f32),
113    Text(String),
114}
115
116impl TryFrom<ZoomText> for Zoom {
117    type Error = String;
118
119    fn try_from(z: ZoomText) -> Result<Self, String> {
120        match z {
121            ZoomText::Factor(f) => Ok(Self::Factor(f)),
122            ZoomText::Text(t) if t == "fit" => Ok(Self::Fit),
123            ZoomText::Text(t) => Err(format!("zoom {t:?}: expected a number or \"fit\"")),
124        }
125    }
126}
127
128impl From<Zoom> for ZoomText {
129    fn from(z: Zoom) -> Self {
130        match z {
131            Zoom::Factor(f) => Self::Factor(f),
132            Zoom::Fit => Self::Text("fit".into()),
133        }
134    }
135}
136
137impl Zoom {
138    pub fn new(fit: bool, zoom: f32) -> Self {
139        if fit { Self::Fit } else { Self::Factor(zoom) }
140    }
141
142    /// A fixed zoom, or None to fit to the window.
143    pub fn factor(&self) -> Option<f32> {
144        match self {
145            Self::Factor(f) => Some(f.clamp(0.2, 5.0)),
146            Self::Fit => None,
147        }
148    }
149}

155/// Where a window is and how big, in desktop points.
156#[derive(Clone, Copy, Debug, Default, PartialEq)]
157pub struct Geometry {
158    pub position: Option<Pos2>,
159    pub size: Option<Vec2>,
160}

162impl WindowLayout {
163    /// The layout of a window showing `loc`.
164    pub fn new(loc: &Location, geometry: Geometry) -> Self {
165        let mut macros = BTreeMap::new();
166        // Later definitions win, as in `Macros::get`.
167        for (k, v) in loc.macros.iter() {
168            macros.insert(k.to_string(), v.to_string());
169        }
170        Self {
171            main: false,
172            display: loc.source.to_string(),
173            project: match &loc.source {
174                Source::File(path) => project_of(path),
175                Source::Url(_) => None,
176            },
177            macros,
178            position: geometry.position.map(|p| [p.x, p.y]),
179            size: geometry.size.map(|s| [s.x, s.y]),
180            zoom: None,
181            toolbar: false,
182            pinned: None,
183            tree: false,
184        }
185    }
186
187    /// The display and macros to show.
188    pub fn location(&self) -> Location {
189        let mut source = crate::fetch::user_named(&self.display);
190        let mut project = self.project.clone();
191        // A DLS launcher's project is named after the module's version, so
192        // after the version changes take the launcher's project as it is now.
193        if let Some(dir) = &project
194            && bob::project::links(dir).is_none()
195            && let Some(script) = dir.parent()
196            && bob::project::through_launcher(dir)
197            && let Some(now) = bob::project::project_dir(script)
198        {
199            if let Source::File(path) = &source
200                && let Ok(rest) = path.strip_prefix(dir)
201            {
202                source = Source::File(now.join(rest));
203            }
204            project = Some(now);
205        }
206        // A real path saved with its project goes back through the project.
207        if let (Some(dir), Source::File(path)) = (&project, &source)
208            && !path.starts_with(dir)
209            && let Some(p) = bob::project::virtual_path(dir, path)
210        {
211            source = Source::File(p);
212        }
213        Location {
214            source,
215            macros: Macros::from_pairs(self.macros.clone()),
216        }
217    }
218
219    /// Where to put the window, dropping a position on none of `monitors`
220    /// (a laptop that was docked when the layout was saved); with no
221    /// monitors known, positions are kept.
222    pub fn geometry(&self, monitors: &[Rect]) -> Geometry {
223        Geometry::restore(self.position, self.size, monitors)
224    }
225}

227impl TrendLayout {
228    pub fn new(view: archiveview::View, geometry: Geometry) -> Self {
229        let span = match &view.range {
230            Range::Relative { secs } => *secs,
231            Range::Absolute { start, end } => {
232                match (
233                    archiveview::view::parse_time(start),
234                    archiveview::view::parse_time(end),
235                ) {
236                    (Some(a), Some(b)) if b > a => b - a,
237                    _ => 3600.0,
238                }
239            }
240        };
241        Self {
242            pvs: view.pvs.iter().map(|t| t.name.clone()).collect(),
243            span,
244            log: view.y.log,
245            view: Some(view),
246            position: geometry.position.map(|p| [p.x, p.y]),
247            size: geometry.size.map(|s| [s.x, s.y]),
248        }
249    }
250
251    /// What the plot shows: as saved, or for a layout from before trends
252    /// were archiveview plots, its PVs over the last `span` seconds (live),
253    /// on a log scale if it was.
254    pub fn view(&self) -> archiveview::View {
255        if let Some(view) = &self.view {
256            return view.clone();
257        }
258        let span = if self.span.is_finite() && self.span > 0.0 {
259            self.span
260        } else {
261            600.0
262        };
263        archiveview::View {
264            pvs: (self.pvs.iter())
265                .map(|name| TraceView {
266                    name: name.clone(),
267                    color: None,
268                    visible: true,
269                    axis: None,
270                })
271                .collect(),
272            range: Range::Relative { secs: span },
273            y: YAxis {
274                log: self.log,
275                ..YAxis::default()
276            },
277            ..archiveview::View::default()
278        }
279    }
280
281    /// Where to put the window, as `WindowLayout::geometry`.
282    pub fn geometry(&self, monitors: &[Rect]) -> Geometry {
283        Geometry::restore(self.position, self.size, monitors)
284    }
285}

287impl Geometry {
288    /// A saved position and size, dropping a position on none of
289    /// `monitors`, and either one that isn't usable.
290    fn restore(position: Option<[f32; 2]>, size: Option<[f32; 2]>, monitors: &[Rect]) -> Self {
291        let size = size
292            .map(|[w, h]| Vec2::new(w, h))
293            .filter(|s| s.x >= 1.0 && s.y >= 1.0 && s.x.is_finite() && s.y.is_finite());
294        let position = position
295            .map(|[x, y]| Pos2::new(x, y))
296            .filter(|p| p.x.is_finite() && p.y.is_finite())
297            .filter(|p| monitors.is_empty() || on_screen(*p, size, monitors));
298        Self { position, size }
299    }
300}
301
302/// Whether enough of a window's top edge (its title bar) at `pos` lies on a
303/// monitor to grab it.
304fn on_screen(pos: Pos2, size: Option<Vec2>, monitors: &[Rect]) -> bool {
305    let width = size.map_or(200.0, |s| s.x).max(1.0);
306    let top = Rect::from_min_size(pos, Vec2::new(width, 30.0));
307    monitors.iter().any(|m| {
308        let seen = m.intersect(top);
309        seen.is_positive() && seen.width() >= width.min(40.0) && seen.height() >= 10.0
310    })
311}

313/// The project directory that `path` reaches its file through: the nearest
314/// directory above it holding a `.project` (or the virtual project of a DLS
315/// launcher), when `path` itself doesn't exist.
316fn project_of(path: &Path) -> Option<PathBuf> {
317    if path.exists() {
318        return None;
319    }
320    path.ancestors()
321        .skip(1)
322        .find(|d| bob::project::is_project(d))
323        .map(Path::to_path_buf)
324}

344/// `$XDG_CONFIG_HOME/bobview`, by default `~/.config/bobview` (on Windows
345/// `%APPDATA%\bobview`, ahead of `~/.config/bobview`, when
346/// `XDG_CONFIG_HOME` is not set).
347pub fn config_dir() -> Option<PathBuf> {
348    let var = |name| std::env::var_os(name).filter(|d| !d.is_empty());
349    let base = var("XDG_CONFIG_HOME")
350        .map(PathBuf::from)
351        .or_else(|| var("APPDATA").filter(|_| cfg!(windows)).map(PathBuf::from))
352        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".config")))?;
353    Some(base.join("bobview"))
354}
355
356fn layouts_dir() -> Option<PathBuf> {
357    config_dir().map(|d| d.join("layouts"))
358}

360/// Whether `name` can name a layout file: not empty, no path in it.
361pub fn valid_name(name: &str) -> bool {
362    let name = name.trim();
363    !name.is_empty() && !name.starts_with('.') && !name.contains(['/', '\\'])
364}

366/// `--layout`'s argument as a layout's name: a JSON file's absolute path
367/// when it names a file (has a directory in it or ends in .json), so that
368/// saving the layout writes back to that file, else a saved layout's name.
369#[cfg_attr(
370    target_arch = "wasm32",
371    expect(dead_code, reason = "a browser starts from its address")
372)]
373pub fn named(arg: String) -> String {
374    let path = Path::new(&arg);
375    if path.components().count() < 2 && path.extension().is_none_or(|e| e != "json") {
376        return arg;
377    }
378    std::path::absolute(path).map_or(arg, |p| p.display().to_string())
379}

381/// How the toolbar shows layout `name`: a layout file by its name, without
382/// .json.
383pub fn label(name: &str) -> &str {
384    let path = Path::new(name);
385    match path.file_stem().and_then(|s| s.to_str()) {
386        Some(stem) if path.is_absolute() => stem,
387        _ => name,
388    }
389}

391fn file(name: &str) -> Option<PathBuf> {
392    let path = Path::new(name);
393    if path.is_absolute() {
394        return Some(path.to_path_buf());
395    }
396    valid_name(name)
397        .then(|| layouts_dir().map(|d| d.join(format!("{}.json", name.trim()))))
398        .flatten()
399}

401/// The saved layouts' names, default first and the rest in order.
402pub fn list() -> Vec<String> {
403    let Some(dir) = layouts_dir() else {
404        return Vec::new();
405    };
406    let mut names: Vec<String> = std::fs::read_dir(dir)
407        .map(|rd| {
408            rd.filter_map(Result::ok)
409                .filter_map(|e| {
410                    let p = e.path();
411                    (p.extension()? == "json")
412                        .then(|| p.file_stem()?.to_str().map(str::to_string))
413                        .flatten()
414                })
415                .collect()
416        })
417        .unwrap_or_default();
418    names.sort_by(|a, b| (a != DEFAULT).cmp(&(b != DEFAULT)).then(a.cmp(b)));
419    names
420}

426pub fn load(name: &str) -> Result<Layout, String> {
427    let path = file(name).ok_or_else(|| format!("no layout {name:?}"))?;
428    let text = std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
429    let layout: Layout =
430        serde_json::from_str(&text).map_err(|e| format!("{}: {e}", path.display()))?;
431    if layout.version > VERSION {
432        return Err(format!(
433            "{}: layout version {} is newer than this bobview's {VERSION}",
434            path.display(),
435            layout.version
436        ));
437    }
438    Ok(layout)
439}

441/// The layout to start with: `named` (with `--layout`), which must load,
442/// else the default one when `default`, which bobview starts without, with
443/// a warning, if it doesn't load (written by a newer bobview, say).
444#[cfg_attr(
445    target_arch = "wasm32",
446    expect(dead_code, reason = "a browser starts from its address")
447)]
448pub fn startup(
449    named: Option<String>,
450    default: bool,
451    load: impl Fn(&str) -> Result<Layout, String>,
452) -> Result<Option<(Layout, String)>, String> {
453    if let Some(name) = named {
454        return load(&name)
455            .map(|l| Some((l, name.clone())))
456            .map_err(|e| format!("cannot restore layout {name}: {e}"));
457    }
458    if !default {
459        return Ok(None);
460    }
461    match load(DEFAULT) {
462        Ok(l) => Ok(Some((l, DEFAULT.to_string()))),
463        Err(e) => {
464            log::warn!("not restoring the default layout: {e}");
465            Ok(None)
466        }
467    }
468}

470pub fn save(name: &str, layout: &Layout) -> Result<(), String> {
471    let path = file(name).ok_or_else(|| format!("not a layout name: {name:?}"))?;
472    let write = || -> std::io::Result<()> {
473        let text = serde_json::to_string_pretty(layout).map_err(std::io::Error::other)?;
474        write_atomic(&path, &(text + "\n"))
475    };
476    write().map_err(|e| format!("{}: {e}", path.display()))?;
477    log::info!("saved layout {}", path.display());
478    Ok(())
479}

481/// Write `text` to `path` through a file beside it, so that a crash or a
482/// full disk leaves the old file whole rather than half written.
483fn write_atomic(path: &Path, text: &str) -> std::io::Result<()> {
484    let dir = path.parent().unwrap_or(Path::new("."));
485    std::fs::create_dir_all(dir)?;
486    let name = path.file_name().unwrap_or_default().to_string_lossy();
487    // Hidden and not .json, so never listed as a layout.
488    let tmp = dir.join(format!(".{name}.{}.tmp", std::process::id()));
489    std::fs::write(&tmp, text)
490        .and_then(|()| std::fs::rename(&tmp, path))
491        .inspect_err(|_| {
492            std::fs::remove_file(&tmp).ok();
493        })
494}
495
496pub fn delete(name: &str) -> Result<(), String> {
497    let path = file(name).ok_or_else(|| format!("no layout {name:?}"))?;
498    std::fs::remove_file(&path).map_err(|e| format!("{}: {e}", path.display()))
499}

501/// Settings kept between runs, beside the layouts.
502#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
503pub struct Settings {
504    /// Save the default layout when bobview exits.
505    #[serde(default)]
506    pub save_default_on_exit: bool,
507    /// The Archiver Appliance trends fetch history from: the last given
508    /// with `--archiver` or picked in a trend's archiver pill.
509    #[serde(default, skip_serializing_if = "Option::is_none")]
510    pub archiver_url: Option<String>,
511    /// The archivers used, newest first, which the pill offers.
512    #[serde(default, skip_serializing_if = "Vec::is_empty")]
513    pub recent_archivers: Vec<String>,
514}
515
516/// The most archivers [`Settings::recent_archivers`] keeps.
517const MAX_RECENT_ARCHIVERS: usize = 8;
518
519impl Settings {
520    /// Make `archiver` (a retrieval URL, or none) the one to use, and the
521    /// newest of the recent ones, with `outgoing` (the one used until now,
522    /// which may have come from the environment) next, so it can be picked
523    /// again. True if that changed anything.
524    pub fn remember_archiver(&mut self, archiver: Option<&str>, outgoing: Option<&str>) -> bool {
525        let before = (self.archiver_url.clone(), self.recent_archivers.clone());
526        self.archiver_url = archiver.map(str::to_string);
527        for url in [outgoing, archiver].into_iter().flatten() {
528            self.recent_archivers.retain(|u| u != url);
529            self.recent_archivers.insert(0, url.to_string());
530        }
531        self.recent_archivers.truncate(MAX_RECENT_ARCHIVERS);
532        before != (self.archiver_url.clone(), self.recent_archivers.clone())
533    }
534
535    pub fn load() -> Self {
536        config_dir()
537            .and_then(|d| std::fs::read_to_string(d.join("settings.json")).ok())
538            .and_then(|t| serde_json::from_str(&t).ok())
539            .unwrap_or_default()
540    }
541
542    pub fn save(&self) {
543        let Some(dir) = config_dir() else { return };
544        let written = serde_json::to_string_pretty(self)
545            .map_err(std::io::Error::other)
546            .and_then(|text| write_atomic(&dir.join("settings.json"), &(text + "\n")));
547        if let Err(e) = written {
548            log::warn!("cannot save settings in {}: {e}", dir.display());
549        }
550    }
551}
