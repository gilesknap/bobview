103/// Embedded displays and widgets that a load may still make. It belongs to
104/// the root screen and is shared with its nodes, so that displays embedded
105/// later (as the web build fetches them) draw on the same budget.
106struct Budget {
107    screens: AtomicU32,
108    widgets: AtomicU32,
109    /// Bytes of expanded macro values and PV names.
110    macros: AtomicUsize,
111    /// Bytes of embedded display files.
112    files: AtomicUsize,
113}
114
115impl Budget {
116    /// The budget of a new load.
117    fn new() -> Arc<Self> {
118        Arc::new(Budget {
119            screens: AtomicU32::new(MAX_SCREENS),
120            widgets: AtomicU32::new(MAX_WIDGETS),
121            macros: AtomicUsize::new(bob::MAX_LOAD_BYTES),
122            files: AtomicUsize::new(MAX_EMBEDDED_BYTES),
123        })
124    }
125
126    /// Take `n` from `left`, or fail, taking nothing, if less is left. Build
127    /// threads embedding at once can't together take more than there is.
128    fn take_bytes(left: &AtomicUsize, n: usize) -> bool {
129        let mut l = left.load(Ordering::Relaxed);
130        loop {
131            let Some(rest) = l.checked_sub(n) else {
132                return false;
133            };
134            match left.compare_exchange(l, rest, Ordering::Relaxed, Ordering::Relaxed) {
135                Ok(_) => return true,
136                Err(now) => l = now,
137            }
138        }
139    }
140
141    /// Take one from `left`, or fail if none is left.
142    fn take(left: &AtomicU32) -> bool {
143        loop {
144            let n = left.load(Ordering::Relaxed);
145            if n == 0 {
146                return false;
147            }
148            if left
149                .compare_exchange(n, n - 1, Ordering::Relaxed, Ordering::Relaxed)
150                .is_ok()
151            {
152                return true;
153            }
154        }
155    }
156}

197/// What a display asks for that bobview does not do.
198#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
199pub struct Compat {
200    /// Widgets with Jython/JavaScript scripts (not scripts), which are ignored.
201    pub scripts: usize,
202    /// Widgets of a type bobview does not draw: a placeholder is shown.
203    pub placeholders: usize,
204}
205
206impl Compat {
207    /// Count over `nodes`, including groups, tabs and embedded displays.
208    pub fn of(nodes: &[Node]) -> Self {
209        let mut c = Self::default();
210        for n in nodes {
211            // A script bobview recognises (the vacuum group link) is run.
212            if n.w.has_scripts() && n.link.is_none() {
213                c.scripts += 1;
214            }
215            if !DRAWN.contains(&n.w.kind.as_str()) {
216                c.placeholders += 1;
217            }
218            let mut add = |o: Self| {
219                c.scripts += o.scripts;
220                c.placeholders += o.placeholders;
221            };
222            add(Self::of(&n.children));
223            n.tabs.iter().for_each(|t| add(Self::of(t)));
224            if let Some(e) = &n.embedded {
225                add(Self::of(&e.nodes));
226            }
227        }
228        c
229    }
230
231    /// The status line text, or none when there is nothing to say.
232    pub fn text(&self) -> Option<String> {
233        let mut parts = Vec::new();
234        if self.scripts > 0 {
235            let s = if self.scripts == 1 {
236                "widget"
237            } else {
238                "widgets"
239            };
240            parts.push(format!("{} {s} with scripts ignored", self.scripts));
241        }
242        if self.placeholders > 0 {
243            let s = if self.placeholders == 1 {
244                "widget"
245            } else {
246                "widgets"
247            };
248            parts.push(format!("{} unsupported {s}", self.placeholders));
249        }
250        (!parts.is_empty()).then(|| parts.join(", "))
251    }
252}

266/// Counts an embedded display in [`IN_FLIGHT`] from when its thread starts
267/// until its result is shown or thrown away (or the thread panics).
268#[cfg(not(target_arch = "wasm32"))]
269struct InFlight;
270
271#[cfg(not(target_arch = "wasm32"))]
272impl InFlight {
273    fn new() -> Self {
274        IN_FLIGHT.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
275        InFlight
276    }
277}

279#[cfg(not(target_arch = "wasm32"))]
280impl Drop for InFlight {
281    fn drop(&mut self) {
282        IN_FLIGHT.fetch_sub(1, std::sync::atomic::Ordering::SeqCst);
283    }

286/// Show every embedded display under `nodes` whose thread has built it,
287/// drawn or not (a hidden tab's too), so none stays counted by
288/// [`building`]; true while any is still being built.
289#[cfg(not(target_arch = "wasm32"))]
290pub fn poll_built(nodes: &mut [Node]) -> bool {
291    let mut any = false;
292    for n in nodes {
293        any |= n.poll_building(Duration::ZERO);
294        any |= poll_built(&mut n.children);
295        for tab in &mut n.tabs {
296            any |= poll_built(tab);
297        }
298        if let Some(s) = &mut n.embedded {
299            any |= poll_built(&mut s.nodes);
300        }
301    }
302    any
303}

395    /// Move and size the widget from geometry `before` (x, y, width,
396    /// height as its properties gave them) to `after`, as a rule does; its
397    /// children move with it. Changes, not the values, are applied, so a
398    /// size set from the content (an embedded display's resize mode 2) or
399    /// the parent's offset is kept.
400    pub fn reshape(&mut self, before: [f64; 4], after: [f64; 4]) {
401        for (r, (b, a)) in self.rect.iter_mut().zip(before.iter().zip(after)) {
402            *r += a - b;
403        }

405        if dx != 0.0 || dy != 0.0 {
406            self.children
407                .iter_mut()
408                .chain(self.tabs.iter_mut().flatten())
409                .for_each(|n| n.shift(dx, dy));
410        }
411    }

416        self.rect[1] += dy;
417        self.children
418            .iter_mut()
419            .chain(self.tabs.iter_mut().flatten())

428    /// Show an embedded display once built from `source`, or why it was not.

431            Ok(screen) => {

433                if self.w.i64("resize") == Some(2) {

451                    e => e.to_string(),
452                });
453            }
454        }

457    /// Show the embedded display being built on a thread if it now is,
458    /// waiting up to `wait` for it; true while it is still being built.
459    #[cfg(not(target_arch = "wasm32"))]
460    pub fn poll_building(&mut self, wait: Duration) -> bool {
461        let Some(b) = &self.building else {
462            return false;
463        };
464        let built = match b.rx.recv_timeout(wait) {
465            // Shown below, so it stops counting as in flight after this.
466            Ok((built, _in_flight)) => built,
467            Err(std::sync::mpsc::RecvTimeoutError::Timeout) => return true,
468            Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => {
469                // The thread panicked building it.
470                Err(build_failed(&b.source))
471            }
472        };
473        let source = b.source.clone();
474        self.building = None;
475        self.place(&source, built);
476        false
477    }

480/// Expand macros in `text` of widget `w`, from a display loaded from
481/// `base`. As Phoebus's `MacroOrPropertyProvider`, a name that is neither a
482/// macro nor a property falls back to an environment variable, but only in
483/// a display from a file: one from the web must not read them (it could
484/// send them out in a PV name or URL).
485pub(crate) fn expand_for(w: &Widget, macros: &Macros, base: Option<&Source>, text: &str) -> String {
486    let env = matches!(base, Some(Source::File(_)));
487    expand_props(w, macros, env, text, &Default::default())
488}
489
490/// Deepest chain of properties naming one another that is expanded.
491const MAX_PROP_DEPTH: usize = 16;
492
493/// The properties being expanded, and those already expanded.
494type Expanding = std::cell::RefCell<(
495    Vec<String>,
496    std::collections::HashMap<String, Option<String>>,
497)>;
498
499/// [`expand_for`], with the properties being expanded and their values so
500/// far in `state`. A widget property may name others, as BOY's rule PV
501/// `$(pv_name)` does with `$(name):STA` and the name
502/// `$(prefix)$(cell)S-VA-PIRG-01`; one that names itself, directly or not,
503/// or anything unknown, too long or naming others too deeply is left as
504/// written. Each is expanded once, so properties that name others several
505/// times don't take exponential time.
506fn expand_props(w: &Widget, macros: &Macros, env: bool, text: &str, state: &Expanding) -> String {
507    bob::expand(text, &|name| {
508        macros.get(name).map(str::to_string).or_else(|| {
509            if let Some(value) = state.borrow().1.get(name) {
510                return value.clone();
511            }
512            // Each property named recurses, so a long chain of them must not
513            // exhaust the stack.
514            let expanding = state.borrow();
515            if expanding.0.len() >= MAX_PROP_DEPTH || expanding.0.iter().any(|a| a == name) {
516                return None;
517            }
518            drop(expanding);
519            let Some(prop) = w.text(name) else {
520                return env.then(|| env_var(name)).flatten();
521            };
522            state.borrow_mut().0.push(name.to_string());
523            let value = expand_props(w, macros, env, prop, state);
524            let value = Some(value).filter(|v| !v.contains('$') && v.len() <= 1 << 16);
525            let mut state = state.borrow_mut();
526            state.0.pop();
527            state.1.insert(name.to_string(), value.clone());
528            value
529        })
530    })
531}

555/// The display's name with `macros` expanded; a display with no name is
556/// called after its file, as in BOY and Phoebus.
557pub(crate) fn title(display: &Display, macros: &Macros) -> String {
558    let name = bob::expand(&display.name, &|n| display_macro(display, macros, n));
559    if !name.trim().is_empty() {
560        return name;
561    }
562    let file = match &display.source {
563        Some(Source::File(p)) => p.to_string_lossy().into_owned(),
564        Some(Source::Url(u)) => u.split(['?', '#']).next().unwrap_or("").to_string(),
565        None => String::new(),
566    };
567    let base = file.rsplit(std::path::is_separator).next().unwrap_or("");
568    base.rsplit_once('.')
569        .map_or(base, |(stem, _)| stem)
570        .to_string()
571}

626/// A fresh `DID_<n>`, BOY's id for an opened display.
627fn display_id() -> String {
628    static NEXT: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(1);
629    let n = NEXT.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
630    format!("DID_{n}")
631}

634    /// [`expand_for`], taking the size of the text and result from the load's
635    /// macro budget; once it is spent, text is left as written.
636    fn expand_charged(
637        &self,
638        w: &Widget,
639        macros: &Macros,
640        base: Option<&Source>,
641        text: &str,
642    ) -> String {
643        if self.budget.macros.load(Ordering::Relaxed) == 0 {
644            bob::warn_spent();
645            return text.to_string();
646        }
647        let out = expand_for(w, macros, base, text);
648        bob::spend(&self.budget.macros, text.len() + out.len());
649        out
650    }

652    pub fn screen(
653        &self,
654        display: Display,
655        macros: &Macros,
656        group: Option<&str>,
657        depth: u32,
658    ) -> Screen {
659        // Macros passed in override the display's own defaults.
660        let mut macros = Macros::over(&display.macros, macros, &self.budget.macros);
661        // BOY gives each display it opens a `$(DID)`, which screens use to
662        // keep their `loc://$(DID)…` PVs apart. Widgets of an embedded
663        // display become the host's, so they share the host's id.
664        if display.legacy && (depth == 0 || macros.get("DID").is_none()) {
665            macros = macros.with(&Macros::from_pairs([("DID", display_id())]));
666        }
667        let base = display.source.clone();
668        let (widgets, width, height) = match group.filter(|g| !g.is_empty()) {
669            Some(name) => match find_group(&display.widgets, name) {
670                Some(g) => {
671                    let (w, h) = (
672                        g.f64("width").unwrap_or(300.0),
673                        g.f64("height").unwrap_or(200.0),
674                    );
675                    (g.children.clone(), w, h)
676                }
677                None => {
678                    log::warn!("group '{name}' not found");
679                    (Vec::new(), display.width, display.height)
680                }
681            },
682            None => (display.widgets.clone(), display.width, display.height),
683        };
684        let lang = if display.legacy {
685            Lang::JavaScript
686        } else {
687            Lang::Python
688        };
689        let nodes = self.nodes(&widgets, (0.0, 0.0), &macros, base.as_ref(), lang, depth);
690        // Phoebus shows widgets beyond the declared display size, so grow to
691        // fit them. BOY clipped them, and EDM conversions leave stray widgets
692        // far outside the display, so an `.opi` keeps its declared size.
693        let (width, height) = nodes
694            .iter()
695            .filter(|n| !display.legacy && n.w.bool("visible") != Some(false))
696            .fold((width, height), |(w, h), n| {
697                (w.max(n.rect[0] + n.rect[2]), h.max(n.rect[1] + n.rect[3]))
698            });
699        let title = title(&display, &macros);
700        Screen {
701            display,
702            nodes,
703            width,
704            height,
705            title,
706        }
707    }

709    fn nodes(
710        &self,
711        widgets: &[Widget],
712        origin: (f64, f64),
713        macros: &Macros,
714        base: Option<&Source>,
715        lang: Lang,
716        depth: u32,
717    ) -> Vec<Node> {
718        let mut nodes = Vec::new();
719        for w in widgets {
720            // Widgets past the budget are left out, after one that says so.
721            let over = !Budget::take(&self.budget.widgets);
722            let mut node = self.node(w, origin, macros, base, lang, depth);
723            if over {
724                log::warn!("too many widgets; the rest are left out");
725                node.error = Some("too many widgets; the rest are left out".into());
726                nodes.push(node);
727                break;
728            }
729            nodes.push(node);
730        }
731        nodes
732    }

734    fn node(
735        &self,
736        w: &Widget,
737        origin: (f64, f64),
738        parent: &Macros,
739        base: Option<&Source>,
740        lang: Lang,
741        depth: u32,
742    ) -> Node {
743        let macros = parent.with_budget(&w.macros(), &self.budget.macros);
744        let [x, y, width, height] = geometry(w);
745        let rect = [origin.0 + x, origin.1 + y, width, height];
746        let ex = |t: &str| self.expand_charged(w, &macros, base, t);
747        let pv_name = w.text("pv_name").map(ex).unwrap_or_default();
748        let pv = (!pv_name.is_empty()).then(|| self.pvs.subscribe(&pv_name));
749        let (all_rules, all_actions) = (w.rules(), w.actions());
750        if all_rules.len() > MAX_PER_WIDGET || all_actions.len() > MAX_PER_WIDGET {
751            log::warn!("over {MAX_PER_WIDGET} rules or actions in a widget; the rest are ignored");
752        }
753        let rules = all_rules
754            .iter()
755            .take(MAX_PER_WIDGET)
756            .map(|r| CompiledRule::new(r, lang, &ex, self.pvs))
757            .collect();
758
759        let link = crate::group_link::Link::new(w, &ex, self.pvs);
760        let mut action_pvs: Vec<Pv> = all_actions
761            .iter()
762            .take(MAX_PER_WIDGET)
763            .filter_map(|a| match a {
764                bob::Action::WritePv { pv_name, .. } => Some(self.pvs.subscribe(&ex(pv_name))),
765                _ => None,
766            })
767            .collect();
768        // A menu mux's targets, which a pick writes, and the values of its
769        // initial item, which BOY writes as the display opens.
770        if let Some(mux) = w.menu_mux() {
771            let sets = w
772                .get("menu_mux")
773                .map_or(0, |m| m.children.iter().filter(|c| c.name == "set").count());
774            if sets > bob::MAX_MUX_SETS {
775                log::warn!(
776                    "over {} sets in a menu mux; the rest are ignored",
777                    bob::MAX_MUX_SETS
778                );
779            }
780            for set in &mux.sets {
781                let target = self.pvs.subscribe(&ex(&set.target));
782                if let Some(value) = mux.initial.and_then(|i| set.values.get(i)) {
783                    target.write(ex(value));
784                }
785                action_pvs.push(target);
786            }
787            if let (true, Some(i), Some(pv)) = (mux.index, mux.initial, &pv) {
788                pv.write(i.to_string());
789            }
790        }
791
792        let mut node = Node {
793            // Without the children, which the nodes hold, so that nested
794            // groups don't each keep a copy of everything below them.
795            w: Widget {
796                kind: w.kind.clone(),
797                props: w.props.clone(),
798                children: Vec::new(),
799            },
800            rect,
801            macros,
802            pv,
803            pv_name,
804            children: Vec::new(),
805            embedded: None,
806            tabs: Vec::new(),
807            active_tab: 0,
808            rules,
809            orig_props: None,
810            rule_outcomes: Vec::new(),
811            rule_versions: None,
812            action_pvs,
813            edit: None,
814            error: None,
815            loading: false,
816            #[cfg(not(target_arch = "wasm32"))]
817            building: None,
818            base: base.cloned(),
819            image: None,
820            symbol: Default::default(),
821            traces: Vec::new(),
822            series: Vec::new(),
823            link,
824            budget: self.budget.clone(),
825        };
826
827        match w.kind.as_str() {
828            "group" => {
829                let (ix, iy) = group_inset(w);
830                node.children = self.nodes(
831                    &w.children,
832                    (rect[0] + ix, rect[1] + iy),
833                    &node.macros,
834                    base,
835                    lang,
836                    depth,
837                );
838            }
839            "tabs" => {
840                let (ix, iy) = tabs_inset(w);
841                let content = (rect[0] + ix, rect[1] + iy);
842                node.active_tab = w.i64("active_tab").unwrap_or(0).max(0) as usize;
843                node.tabs = tab_elements(w)
844                    .iter()
845                    .map(|tab| {
846                        let children: Vec<Widget> = tab
847                            .child("children")
848                            .map(|c| {
849                                c.children
850                                    .iter()
851                                    .filter(|e| e.name == "widget")
852                                    .map(|e| Widget::from_element(e.clone()))
853                                    .collect()
854                            })
855                            .unwrap_or_default();
856                        self.nodes(&children, content, &node.macros, base, lang, depth)
857                    })
858                    .collect();
859            }
860            "embedded" => {
861                let file = node.prop("file");
862                let group = node.prop("group_name");
863                let macros = node.macros.clone();
864                self.embed(&mut node, &file, &macros, &group, depth);
865            }
866            "xyplot" => {
867                node.traces = crate::plot::subscribe(
868                    w,
869                    &|t| self.expand_charged(w, &node.macros, node.base.as_ref(), t),
870                    self.pvs,
871                );
872            }
873            "stripchart" => {
874                node.traces = crate::plot::subscribe(
875                    w,
876                    &|t| self.expand_charged(w, &node.macros, node.base.as_ref(), t),
877                    self.pvs,
878                );
879                node.series = node
880                    .traces
881                    .iter()
882                    .map(|t| t.y.clone().map(crate::stripchart::Series::new))
883                    .collect();
884            }
885            "navtabs" => {
886                node.active_tab = w.i64("active_tab").unwrap_or(0).max(0) as usize;
887                let active = node.active_tab;
888                self.select_navtab(&mut node, active, depth);
889            }
890            _ => {}
891        }
892        node
893    }

895    /// Load `file` into `node.embedded`, or record why it failed. On the UI
896    /// thread ([`Builder::ui`]) the display is read and built on a thread,
897    /// and the node shows it as loading until [`Node::poll_building`] finds
898    /// it built.
899    pub fn embed(&self, node: &mut Node, file: &str, macros: &Macros, group: &str, depth: u32) {
900        // Embedded again once the web build has fetched it: counted already.
901        let counted = std::mem::take(&mut node.loading);
902        node.embedded = None;
903        node.error = None;
904        // Dropping a build still under way discards it when it finishes.
905        #[cfg(not(target_arch = "wasm32"))]
906        {
907            node.building = None;
908        }
909        if file.is_empty() {
910            return;
911        }
912        if depth >= MAX_DEPTH {
913            node.error = Some("embedded displays nested too deeply".into());
914            return;
915        }
916        // The node's budget, not this builder's, which may be a fresh one:
917        // the display is part of the load that made the node.
918        let this = Builder::sharing(self.pvs, node.budget.clone(), self.waits());
919        if !counted && !Budget::take(&this.budget.screens) {
920            node.error = Some("too many embedded displays".into());
921            return;
922        }
923        let source = node.resolve(file);
924        #[cfg(not(target_arch = "wasm32"))]
925        if !this.wait {
926            let (tx, rx) = std::sync::mpsc::channel();
927            let (pvs, src, budget) = (self.pvs.clone(), source.clone(), this.budget.clone());
928            let (macros, group) = (macros.clone(), group.to_string());
929            let in_flight = InFlight::new();
930            let spawned = std::thread::Builder::new()
931                .name("display-builder".into())
932                .stack_size(BUILD_STACK)
933                .spawn(move || {
934                    // Still the same load, so on the same budget: a display
935                    // that embeds itself can't get a fresh one per thread.
936                    let built =
937                        Builder::sharing(&pvs, budget, true).build(&src, &macros, &group, depth);
938                    // Unsent (the node moved on), the result and its count drop here.
939                    tx.send((built, in_flight)).ok();
940                });
941            if let Err(e) = spawned {
942                let why = format!("cannot start a thread to build it: {e}");
943                node.place(&source, Err(refused(&source, &why)));
944                return;
945            }
946            node.error = Some(format!("Loading {source}"));
947            node.building = Some(Building { source, rx });
948            node.poll_building(QUICK);
949            return;
950        }
951        let built = this.build(&source, macros, group, depth);
952        node.place(&source, built);
953    }

955    /// Read and build the display at `source`, embedded `depth` deep, if
956    /// the load has enough of its budget for embedded files left.
957    fn build(
958        &self,
959        source: &Source,
960        macros: &Macros,
961        group: &str,
962        depth: u32,
963    ) -> Result<Screen, bob::Error> {
964        let d = crate::fetch::load(source)?;
965        if !Budget::take_bytes(&self.budget.files, d.size) {
966            return Err(refused(source, "embedded displays too large in total"));
967        }
968        Ok(self.screen(d, macros, Some(group), depth + 1))
969    }

1678    fn held_server() -> (String, std::sync::mpsc::Sender<()>) {
1679        use std::io::{BufRead, BufReader, Write};
1680        let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
1681        let base = format!("http://{}", listener.local_addr().unwrap());
1682        let (release, held) = std::sync::mpsc::channel::<()>();
1683        std::thread::spawn(move || {
1684            for mut stream in listener.incoming().flatten() {
1685                let mut reader = BufReader::new(stream.try_clone().unwrap());
1686                let mut first = String::new();
1687                reader.read_line(&mut first).unwrap();
1688                let mut line = String::new();
1689                while reader.read_line(&mut line).unwrap() > 2 {
1690                    line.clear();
1691                }
1692                if held.recv().is_err() {
1693                    return;
1694                }
1695                let body = r#"<display version="2.0.0"><name>Tab</name></display>"#;
1696                let (status, body) = if first.starts_with("GET /missing.bob ") {
1697                    ("404 Not Found", "")
1698                } else {
1699                    ("200 OK", body)
1700                };
1701                write!(
1702                    stream,
1703                    "HTTP/1.1 {status}\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
1704                    body.len()
1705                )
1706                .unwrap();
1707            }
1708        });
1709        (base, release)
1710    }
