28/// Samples kept by a concatenating trace, at most.
29const MAX_BUFFER: usize = 100_000;
30
31/// Which updates add to a concatenating trace, as BOY's `update_mode`:
32/// either PV's, both PVs' (once each has updated), the x PV's or the y PV's.
33#[derive(Clone, Copy, Debug, PartialEq)]
34enum UpdateMode {
35    XOrY,
36    XAndY,
37    X,
38    Y,
39}
40
41/// A concatenating trace's last `size` points, as BOY plots with
42/// `concatenate_data`: each update appends the y PV's values, paired with
43/// the x PV's or, without one, plotted against their index.
44#[derive(Debug)]
45struct Buffer {
46    size: usize,
47    mode: UpdateMode,
48    has_x: bool,
49    /// The PVs' latest values, and whether each has updated since the last
50    /// append.
51    x: Option<Vec<f64>>,
52    y: Option<Vec<f64>>,
53    x_new: bool,
54    y_new: bool,
55    /// Each PV's last time stamp and value, while connected: a PV's state
56    /// also changes without a new value (CA's metadata read, access rights,
57    /// a write's result), which appends nothing.
58    x_seen: Option<(Option<SystemTime>, Value)>,
59    y_seen: Option<(Option<SystemTime>, Value)>,
60    /// (x, y), x unused without an x PV.
61    points: VecDeque<(f64, f64)>,
62}
63
64impl Buffer {
65    fn new(size: usize, mode: UpdateMode, has_x: bool) -> Self {
66        Self {
67            size: size.clamp(1, MAX_BUFFER),
68            mode,
69            has_x,
70            x: None,
71            y: None,
72            x_new: false,
73            y_new: false,
74            x_seen: None,
75            y_seen: None,
76            points: VecDeque::new(),
77        }
78    }
79
80    /// Take in a change of the x PV's state (`is_x`) or the y PV's: an
81    /// update if it has a new time stamp or value, or has disconnected.
82    fn changed(&mut self, is_x: bool, s: &PvState) {
83        let seen = if is_x {
84            &mut self.x_seen
85        } else {
86            &mut self.y_seen
87        };
88        let now = s.connected.then(|| (s.timestamp, s.value.clone()));
89        // NaN is not equal to itself, but a NaN still standing is no update.
90        let nan = |v: &Value| matches!(v, Value::Double(d) if d.is_nan());
91        let same = match (&*seen, &now) {
92            (Some((t0, v0)), Some((t1, v1))) => t0 == t1 && (v0 == v1 || nan(v0) && nan(v1)),
93            _ => false,
94        };
95        if same {
96            return;
97        }
98        *seen = now;
99        self.update(is_x, numbers(s));
100    }
101
102    /// Take in an update of the x PV (`is_x`) or the y PV: its numbers, or
103    /// `None` when it has none, which appends nothing.
104    fn update(&mut self, is_x: bool, v: Option<Vec<f64>>) {
105        if is_x {
106            (self.x, self.x_new) = (v, true);
107        } else {
108            (self.y, self.y_new) = (v, true);
109        }
110        let append = !self.has_x
111            || match self.mode {
112                UpdateMode::XOrY => true,
113                UpdateMode::XAndY => self.x_new && self.y_new,
114                UpdateMode::X => is_x,
115                UpdateMode::Y => !is_x,
116            };
117        let (Some(y), true) = (&self.y, append) else {
118            return;
119        };
120        if self.has_x {
121            let Some(x) = &self.x else { return };
122            self.points.extend(x.iter().copied().zip(y.iter().copied()));
123        } else {
124            self.points.extend(y.iter().map(|v| (0.0, *v)));
125        }
126        (self.x_new, self.y_new) = (false, false);
127        let extra = self.points.len().saturating_sub(self.size);
128        self.points.drain(..extra);
129    }
130
131    fn points(&self) -> Vec<(f64, f64)> {
132        if self.has_x {
133            self.points.iter().copied().collect()
134        } else {
135            let y: Vec<f64> = self.points.iter().map(|p| p.1).collect();
136            points(None, &y)
137        }
138    }
139}

141/// A concatenating trace's buffer, filled by its PVs' listeners.
142struct History {
143    /// Locked by the listeners while a PV's state is locked, so never read
144    /// a PV's state while holding this lock.
145    buffer: Arc<Mutex<Buffer>>,
146    _listeners: Vec<Listener>,
147}
148
149impl History {
150    /// The trace's history, if it concatenates (`concatenate_data`, from a
151    /// BOY XY graph) and has a y PV.
152    fn new(t: &Element, x: Option<&Pv>, y: Option<&Pv>) -> Option<Self> {
153        let t = Widget::from_element(t.clone());
154        let y = y.filter(|_| t.bool("concatenate_data") == Some(true))?;
155        let mode = match t.i64("update_mode") {
156            Some(1) => UpdateMode::XAndY,
157            Some(2) => UpdateMode::X,
158            Some(3) => UpdateMode::Y,
159            // X or Y is BOY's default; a trigger PV is not supported.
160            _ => UpdateMode::XOrY,
161        };
162        // BOY's default buffer size is 100.
163        let size = t.f64("buffer_size").unwrap_or(100.0).max(0.0) as usize;
164        let buffer = Arc::new(Mutex::new(Buffer::new(size, mode, x.is_some())));
165        let listen = |pv: &Pv, is_x: bool| {
166            let into = buffer.clone();
167            let listener: Listener = Arc::new(move |s: &PvState| into.lock().changed(is_x, s));
168            pv.listen(&listener);
169            listener
170        };
171        let mut listeners = vec![listen(y, false)];
172        listeners.extend(x.map(|x| listen(x, true)));
173        Some(Self {
174            buffer,
175            _listeners: listeners,
176        })
177    }
178}

186/// Subscribe to each trace's x and y PVs. `expand` expands macros.
187pub fn subscribe(w: &Widget, expand: &dyn Fn(&str) -> String, pvs: &PvManager) -> Vec<TracePvs> {
188    trace_elements(w)
189        .iter()
190        .map(|t| {
191            let pv = |k: &str| {
192                t.child_text(k)
193                    .map(expand)
194                    .filter(|n| !n.trim().is_empty())
195                    .map(|n| pvs.subscribe(&n))
196            };
197            let (x, y) = (pv("x_pv"), pv("y_pv"));
198            let history = History::new(t, x.as_ref(), y.as_ref());
199            TracePvs { x, y, history }
200        })
201        .collect()
202}

204/// `traces[0].y_pv` and the like: a property of one trace, as Phoebus lets
205/// trace names refer to them.
206fn trace_prop(w: &Widget, path: &str) -> Option<String> {
207    let rest = path.strip_prefix("traces[")?;
208    let (index, prop) = rest.split_once("].")?;
209    let t = *trace_elements(w).get(index.parse::<usize>().ok()?)?;
210    t.child_text(prop).map(str::to_string)
211}

213/// Expand macros in `text`, including `$(traces[N].prop)`; a trace's PV
214/// (`$(traces[0].y_pv)`, the default trace name) is its plain name, without a
215/// local PV's type and initial value.
216fn expand(node: &Node, text: &str) -> String {
217    bob::expand(text, &|name| {
218        node.macros
219            .get(name)
220            .map(str::to_string)
221            .or_else(|| {
222                trace_prop(&node.w, name).map(|raw| {
223                    let value = node.expand(&raw);
224                    if name.ends_with("_pv") {
225                        plain_pv_name(&value)
226                    } else {
227                        value
228                    }
229                })
230            })
231            .or_else(|| {
232                node.w
233                    .text(name)
234                    .filter(|t| !t.contains('$'))
235                    .map(str::to_string)
236            })
237    })
238}

240/// The numbers in a PV value, if it has any.
241fn numbers(s: &PvState) -> Option<Vec<f64>> {
242    if !s.connected {
243        return None;
244    }
245    match &s.value {
246        Value::DoubleArray(a) => Some(a.to_vec()),
247        Value::LongArray(a) => Some(a.iter().map(|v| *v as f64).collect()),
248        v => v.as_f64().map(|v| vec![v]),
249    }
250}

252/// Points of a trace: y against x, or against the index when there is no x PV.
253fn points(x: Option<&[f64]>, y: &[f64]) -> Vec<(f64, f64)> {
254    match x {
255        Some(x) => x.iter().copied().zip(y.iter().copied()).collect(),
256        None => y.iter().enumerate().map(|(i, v)| (i as f64, *v)).collect(),
257    }
258}

274pub(crate) fn axis_cfg(el: Option<&Element>, title: &str, node: &Node, z: f32) -> AxisCfg {
275    let w = Widget::from_element(el.cloned().unwrap_or_default());
276    let font = |k: &str, bold: bool| {
277        let f = w.font(k).unwrap_or_else(|| bob::Font {
278            style: if bold {
279                bob::FontStyle::Bold
280            } else {
281                bob::FontStyle::Regular
282            },
283            ..Default::default()
284        });
285        style::font_id(&f, z)
286    };
287    AxisCfg {
288        title: w
289            .text("title")
290            .map(|t| expand(node, t))
291            .unwrap_or_else(|| title.to_string()),
292        autoscale: w.bool("autoscale").unwrap_or(false),
293        log: w.bool("log_scale").unwrap_or(false),
294        range: (
295            w.f64("minimum").unwrap_or(0.0),
296            w.f64("maximum").unwrap_or(100.0),
297        ),
298        grid: w.bool("show_grid").unwrap_or(false),
299        visible: w.bool("visible").unwrap_or(true),
300        on_right: w.bool("on_right").unwrap_or(false),
301        color: w
302            .color("color")
303            .map(|c| color(&c))
304            .unwrap_or(Color32::BLACK),
305        title_font: font("title_font", true),
306        scale_font: font("scale_font", false),
307    }
308}

310/// The range an axis shows: its own limits, or the data's when autoscaling.
311fn axis_range(cfg: &AxisCfg, data: impl Iterator<Item = f64>, nice: bool) -> (f64, f64) {
312    if !cfg.autoscale {
313        return cfg.range;
314    }
315    autoscale(cfg.log, data, nice).unwrap_or(cfg.range)
316}
317
318/// The range that shows `data`: whole decades on a log scale, else rounded
319/// out to whole ticks when `nice`. `None` when there is nothing to show.
320pub(crate) fn autoscale(
321    log: bool,
322    data: impl Iterator<Item = f64>,
323    nice: bool,
324) -> Option<(f64, f64)> {
325    let ok = |v: &f64| v.is_finite() && (!log || *v > 0.0);
326    let (lo, hi) = data
327        .filter(ok)
328        .fold((f64::INFINITY, f64::NEG_INFINITY), |(a, b), v| {
329            (a.min(v), b.max(v))
330        });
331    if lo > hi {
332        return None;
333    }
334    Some(if log {
335        let (a, b) = (
336            lo.log10().floor(),
337            hi.log10().ceil().max(lo.log10().floor() + 1.0),
338        );
339        (10f64.powf(a), 10f64.powf(b))
340    } else if nice || lo == hi {
341        axis::nice_range(lo, hi, 8)
342    } else {
343        (lo, hi)
344    })
345}

360/// Each trace's settings, and its points from its PVs' arrays.
361pub(crate) fn traces(node: &Node, z: f32) -> Vec<Trace> {
362    trace_elements(&node.w)
363        .iter()
364        .enumerate()
365        .map(|(i, el)| {
366            let t = Widget::from_element((*el).clone());
367            let pvs = node.traces.get(i);
368            let data = |pv: Option<&Pv>| pv.and_then(|p| numbers(&p.state()));
369            let points = if let Some(h) = pvs.and_then(|p| p.history.as_ref()) {
370                h.buffer.lock().points()
371            } else {
372                let y = data(pvs.and_then(|p| p.y.as_ref()));
373                let x = data(pvs.and_then(|p| p.x.as_ref()));
374                let has_x = pvs.is_some_and(|p| p.x.is_some());
375                match (y, has_x) {
376                    (Some(y), false) => points(None, &y),
377                    (Some(y), true) => x.map(|x| points(Some(&x), &y)).unwrap_or_default(),
378                    (None, _) => Vec::new(),
379                }
380            };
381            Trace {
382                name: t.text("name").map(|n| expand(node, n)).unwrap_or_default(),
383                axis: t.i64("axis").unwrap_or(0).max(0) as usize,
384                kind: t.i64("trace_type").unwrap_or(1),
385                color: t
386                    .color("color")
387                    .map(|c| color(&c))
388                    .unwrap_or(Color32::from_rgb(0, 0, 255)),
389                width: (t.f64("line_width").unwrap_or(1.0) as f32 * z).max(0.5),
390                line_style: t.i64("line_style").unwrap_or(0),
391                point: t.i64("point_type").unwrap_or(0),
392                point_size: t.f64("point_size").unwrap_or(10.0) as f32 * z,
393                visible: t.bool("visible").unwrap_or(true),
394                points,
395            }
396        })
397        .collect()
398}

400/// The worst state of the plot's PVs, for the alarm border.
401pub(crate) fn worst_state(node: &Node) -> Option<PvState> {
402    let states: Vec<PvState> = node
403        .traces
404        .iter()
405        .flat_map(|t| t.x.iter().chain(&t.y))
406        .map(|p| p.state())
407        .collect();
408    states
409        .iter()
410        .find(|s| !s.connected)
411        .or_else(|| states.iter().max_by_key(|s| s.severity))
412        .cloned()
413}

421/// Ticks along a horizontal scale, spaced so their labels do not overlap.
422fn fit_ticks(
423    painter: &egui::Painter,
424    scale: &Scale,
425    font: &FontId,
426    gap: f32,
427) -> (Vec<(f64, String)>, f64) {
428    let mut spacing = text_size(painter, "0.00", font).x + gap;
429    let labelled = |spacing: f32| {
430        let (ticks, step) = scale.ticks(spacing);
431        let labels: Vec<(f64, String)> = ticks
432            .into_iter()
433            .map(|v| (v, axis::label(v, step)))
434            .collect();
435        let widest = labels
436            .iter()
437            .map(|(_, l)| text_size(painter, l, font).x)
438            .fold(0.0, f32::max);
439        (labels, step, widest + gap)
440    };
441    for _ in 0..3 {
442        let (labels, step, needed) = labelled(spacing);
443        if needed <= spacing {
444            return (labels, step);
445        }
446        spacing = needed;
447    }
448    let (labels, step, _) = labelled(spacing);
449    (labels, step)
450}

452/// Text turned to read upwards, centred on `center`.
453pub(crate) fn vertical_text(
454    painter: &egui::Painter,
455    center: Pos2,
456    text: &str,
457    font: FontId,
458    c: Color32,
459) {
460    let galley = painter.layout_no_wrap(text.to_string(), font, c);
461    let size = galley.size();
462    let pos = Pos2::new(center.x - size.y / 2.0, center.y + size.x / 2.0);
463    painter.add(epaint::TextShape::new(pos, galley, c).with_angle(-FRAC_PI_2));
464}

466/// The plot's title across the top, if it has one; returns where the rest
467/// of the plot starts.
468pub(crate) fn title(painter: &egui::Painter, node: &Node, rect: Rect, fg: Color32, z: f32) -> f32 {
469    let pad = 4.0 * z;
470    let top = rect.top() + pad;
471    let title = node
472        .w
473        .text("title")
474        .map(|t| expand(node, t))
475        .unwrap_or_default();
476    if title.is_empty() {
477        return top;
478    }
479    let f = node.w.font("title_font").unwrap_or(bob::Font {
480        style: bob::FontStyle::Bold,
481        size: 18.0,
482        ..Default::default()
483    });
484    let f = style::font_id(&f, z);
485    let h = text_size(painter, &title, &f).y;
486    painter.text(
487        Pos2::new(rect.center().x, top),
488        Align2::CENTER_TOP,
489        &title,
490        f,
491        fg,
492    );
493    top + h + pad
494}

496pub fn draw(outer: &egui::Painter, node: &Node, rect: Rect, z: f32) {
497    let w = &node.w;
498    let painter = &outer.with_clip_rect(outer.clip_rect().intersect(rect));
499    let back = w
500        .color("background_color")
501        .map(|c| color(&c))
502        .unwrap_or(Color32::WHITE);
503    fill(painter, rect, back);
504    let fg = w
505        .color("foreground_color")
506        .map(|c| color(&c))
507        .unwrap_or(Color32::BLACK);
508    let pad = 4.0 * z;
509    let tick = 5.0 * z;
510    let traces = traces(node, z);
511
512    let top = title(painter, node, rect, fg, z);
513
514    // Legend along the bottom.
515    let legend_font = style::font_id(&w.font("legend_font").unwrap_or_default(), z);
516    let shown: Vec<&Trace> = traces.iter().filter(|t| t.visible).collect();
517    let legend_h = if w.bool("show_legend").unwrap_or(true) && !shown.is_empty() {
518        text_size(painter, "Xg", &legend_font).y + pad
519    } else {
520        0.0
521    };
522    let mut bottom = rect.bottom() - pad - legend_h;
523
524    // The x axis below the plot.
525    let x_cfg = axis_cfg(w.get("x_axis"), "X", node, z);
526    let x_range = axis_range(
527        &x_cfg,
528        shown.iter().flat_map(|t| t.points.iter().map(|p| p.0)),
529        false,
530    );
531    let x_label_h = text_size(painter, "0", &x_cfg.scale_font).y;
532    if x_cfg.visible {
533        bottom -= tick + x_label_h;
534        if !x_cfg.title.is_empty() {
535            bottom -= text_size(painter, &x_cfg.title, &x_cfg.title_font).y;
536        }
537    }
538
539    // The y axes beside it, with their widths.
540    let y_els: Vec<Element> = w
541        .get("y_axes")
542        .map(|e| {
543            e.children
544                .iter()
545                .filter(|c| c.name == "y_axis")
546                .cloned()
547                .collect()
548        })
549        .unwrap_or_default();
550    let y_els = if y_els.is_empty() {
551        vec![Element::default()]
552    } else {
553        y_els
554    };
555    // The top tick label is centred on the top edge, so leave room for its upper half.
556    let half_label = y_els
557        .iter()
558        .map(|el| text_size(painter, "0", &axis_cfg(Some(el), "Y", node, z).scale_font).y)
559        .fold(0.0, f32::max)
560        / 2.0;
561    let top = top + (half_label - pad).max(0.0);
562    let ys: Vec<(AxisCfg, Scale, Vec<f64>, f64, f32)> = y_els
563        .iter()
564        .enumerate()
565        .map(|(i, el)| {
566            let cfg = axis_cfg(Some(el), "Y", node, z);
567            let data = shown
568                .iter()
569                .filter(|t| t.axis == i)
570                .flat_map(|t| t.points.iter().map(|p| p.1));
571            let (lo, hi) = axis_range(&cfg, data, true);
572            let scale = Scale {
573                lo,
574                hi,
575                log: cfg.log,
576                from: bottom,
577                to: top,
578            };
579            let line_h = text_size(painter, "0", &cfg.scale_font).y;
580            let (ticks, step) = scale.ticks(line_h * 2.5);
581            let label_w = ticks
582                .iter()
583                .map(|v| text_size(painter, &axis::label(*v, step), &cfg.scale_font).x)
584                .fold(0.0, f32::max);
585            let mut width = tick + pad + label_w;
586            if !cfg.title.is_empty() {
587                width += text_size(painter, &cfg.title, &cfg.title_font).y + pad;
588            }
589            (cfg, scale, ticks, step, width)
590        })
591        .collect();
592    let side = |right: bool| {
593        ys.iter()
594            .filter(|y| y.0.visible && y.0.on_right == right)
595            .map(|y| y.4)
596            .sum::<f32>()
597    };
598    let left = rect.left() + pad + side(false);
599    let x_scale = |right: f32| Scale {
600        lo: x_range.0,
601        hi: x_range.1,
602        log: x_cfg.log,
603        from: left,
604        to: right,
605    };
606    // Leave room for half of the last x label when there is no right axis.
607    let rough = x_scale(rect.right() - pad - side(true));
608    let x_end_room = fit_ticks(painter, &rough, &x_cfg.scale_font, 20.0 * z)
609        .0
610        .last()
611        .map_or(0.0, |(_, l)| {
612            text_size(painter, l, &x_cfg.scale_font).x / 2.0
613        });
614    let right = rect.right() - pad - side(true).max(x_end_room);
615    let plot = Rect::from_min_max(Pos2::new(left, top), Pos2::new(right, bottom));
616    if plot.width() < 4.0 || plot.height() < 4.0 {
617        return;
618    }
619    let xs = x_scale(plot.right());
620    let (x_ticks, _) = fit_ticks(painter, &xs, &x_cfg.scale_font, 20.0 * z);
621
622    // Grid lines first, so traces go over them.
623    let grid = Stroke::new(1.0, named("Grid").gamma_multiply(0.4));
624    if x_cfg.grid {
625        for (v, _) in &x_ticks {
626            let x = xs.pos(*v);
627            painter.line_segment(
628                [Pos2::new(x, plot.top()), Pos2::new(x, plot.bottom())],
629                grid,
630            );
631        }
632    }
633    for (cfg, scale, ticks, _, _) in &ys {
634        if cfg.grid {
635            for v in ticks {
636                let y = scale.pos(*v);
637                painter.line_segment(
638                    [Pos2::new(plot.left(), y), Pos2::new(plot.right(), y)],
639                    grid,
640                );
641            }
642        }
643    }
644
645    // Traces, clipped to the plot area.
646    for t in &shown {
647        let Some((_, ys_scale, ..)) = ys.get(t.axis).or(ys.first()) else {
648            continue;
649        };
650        draw_trace(painter, t, &xs, ys_scale, plot);
651    }
652
653    // Axes over the plot edges.
654    if x_cfg.visible {
655        let stroke = Stroke::new(z.max(1.0), x_cfg.color);
656        painter.line_segment([plot.left_bottom(), plot.right_bottom()], stroke);
657        for (v, label) in &x_ticks {
658            let x = xs.pos(*v);
659            painter.line_segment(
660                [
661                    Pos2::new(x, plot.bottom()),
662                    Pos2::new(x, plot.bottom() + tick),
663                ],
664                stroke,
665            );
666            painter.text(
667                Pos2::new(x, plot.bottom() + tick),
668                Align2::CENTER_TOP,
669                label,
670                x_cfg.scale_font.clone(),
671                x_cfg.color,
672            );
673        }
674        if !x_cfg.title.is_empty() {
675            painter.text(
676                Pos2::new(plot.center().x, plot.bottom() + tick + x_label_h),
677                Align2::CENTER_TOP,
678                &x_cfg.title,
679                x_cfg.title_font.clone(),
680                x_cfg.color,
681            );
682        }
683    }
684    let (mut left_x, mut right_x) = (plot.left(), plot.right());
685    for (cfg, scale, ticks, step, width) in ys.iter().filter(|y| y.0.visible) {
686        let stroke = Stroke::new(z.max(1.0), cfg.color);
687        let (x, dir) = if cfg.on_right {
688            (right_x, 1.0)
689        } else {
690            (left_x, -1.0)
691        };
692        painter.line_segment(
693            [Pos2::new(x, plot.top()), Pos2::new(x, plot.bottom())],
694            stroke,
695        );
696        for v in ticks {
697            let y = scale.pos(*v);
698            painter.line_segment([Pos2::new(x, y), Pos2::new(x + dir * tick, y)], stroke);
699            painter.text(
700                Pos2::new(x + dir * (tick + pad / 2.0), y),
701                if cfg.on_right {
702                    Align2::LEFT_CENTER
703                } else {
704                    Align2::RIGHT_CENTER
705                },
706                axis::label(*v, *step),
707                cfg.scale_font.clone(),
708                cfg.color,
709            );
710        }
711        if !cfg.title.is_empty() {
712            let h = text_size(painter, &cfg.title, &cfg.title_font).y;
713            let cx = x + dir * (width - h / 2.0);
714            vertical_text(
715                painter,
716                Pos2::new(cx, plot.center().y),
717                &cfg.title,
718                cfg.title_font.clone(),
719                cfg.color,
720            );
721        }
722        if cfg.on_right {
723            right_x += width;
724        } else {
725            left_x -= width;
726        }
727    }
728
729    if legend_h > 0.0 {
730        legend(painter, &shown, rect, legend_h, pad, &legend_font, fg, z);
731    }
732
733    if let Some(s) = worst_state(node) {
734        alarm_border(outer, w, &s, rect, z);
735    }
736}

738/// Trace names with a sample of their line, centred along the bottom.
739#[allow(clippy::too_many_arguments)]
740pub(crate) fn legend(
741    painter: &egui::Painter,
742    shown: &[&Trace],
743    rect: Rect,
744    height: f32,
745    pad: f32,
746    font: &FontId,
747    fg: Color32,
748    z: f32,
749) {
750    let sample = 20.0 * z;
751    let widths: Vec<f32> = shown
752        .iter()
753        .map(|t| sample + pad + text_size(painter, &t.name, font).x)
754        .collect();
755    let total = widths.iter().sum::<f32>() + 3.0 * pad * (widths.len() as f32 - 1.0);
756    let mut x = (rect.center().x - total / 2.0).max(rect.left() + pad);
757    let y = rect.bottom() - pad - height / 2.0 + pad / 2.0;
758    for (t, w) in shown.iter().zip(widths) {
759        painter.line_segment(
760            [Pos2::new(x, y), Pos2::new(x + sample, y)],
761            Stroke::new(t.width.max(2.0 * z), t.color),
762        );
763        painter.text(
764            Pos2::new(x + sample + pad, y),
765            Align2::LEFT_CENTER,
766            &t.name,
767            font.clone(),
768            fg,
769        );
770        x += w + 3.0 * pad;
771    }
772}

774/// How much egui's anti-aliasing widens a line, in points.
775const FEATHER_WIDTH: f32 = 0.4;
776/// Thinnest a trace's line is drawn, so a 1 pixel line stays solid enough.
777const MIN_TRACE_WIDTH: f32 = 0.6;
778
779fn draw_trace(outer: &egui::Painter, t: &Trace, xs: &Scale, ys: &Scale, plot: Rect) {
780    let painter = &outer.with_clip_rect(outer.clip_rect().intersect(plot));
781    // Markers may poke out of the plot by half their size.
782    let markers =
783        outer.with_clip_rect(outer.clip_rect().intersect(plot.expand(t.point_size / 2.0)));
784    // Far-off points are pulled in so the maths stays finite.
785    let bounds = plot.expand(plot.width().max(plot.height()) * 4.0);
786    let screen = |(x, y): (f64, f64)| {
787        (x.is_finite() && y.is_finite()).then(|| {
788            Pos2::new(
789                xs.pos(x).clamp(bounds.left(), bounds.right()),
790                ys.pos(y).clamp(bounds.top(), bounds.bottom()),
791            )
792        })
793    };
794    let pts: Vec<Option<Pos2>> = t.points.iter().map(|p| screen(*p)).collect();
795    // egui's anti-aliasing widens a line by about a pixel, which makes
796    // thin traces look heavier than the aliased ones of CS-Studio.
797    let stroke = Stroke::new((t.width - FEATHER_WIDTH).max(MIN_TRACE_WIDTH), t.color);
798
799    // Runs of finite points; a NaN breaks the line.
800    let runs = pts
801        .split(Option::is_none)
802        .filter(|r| !r.is_empty())
803        .map(|r| {
804            r.iter()
805                .map(|p| p.unwrap_or_default())
806                .collect::<Vec<Pos2>>()
807        });
808    match t.kind {
809        1 | 4 => runs.for_each(|run| line(painter, decimate(run), stroke, t.line_style)),
810        2 => runs.for_each(|run| {
811            let mut steps = Vec::with_capacity(run.len() * 2);
812            for (i, p) in run.iter().enumerate() {
813                if i > 0 {
814                    steps.push(Pos2::new(p.x, run[i - 1].y));
815                }
816                steps.push(*p);
817            }
818            line(painter, steps, stroke, t.line_style)
819        }),
820        5 => {
821            let base = ys.pos(if ys.log {
822                ys.lo
823            } else {
824                0.0f64.clamp(ys.lo, ys.hi)
825            });
826            let n = pts.iter().flatten().count().max(1) as f32;
827            // A bar is a line from the axis, as in CS-Studio.
828            let half = (t.width / 2.0).max(0.5).min(plot.width() / n * 0.4);
829            for p in pts.iter().flatten() {
830                let bar =
831                    Rect::from_x_y_ranges(p.x - half..=p.x + half, p.y.min(base)..=p.y.max(base));
832                painter.rect_filled(bar, 0.0, t.color);
833            }
834        }
835        _ => {}
836    }
837    if t.point > 0 {
838        for p in pts.iter().flatten() {
839            marker(&markers, *p, t.point, t.point_size, t.color);
840        }
841    }
842}

844/// Thin a line with many points per pixel column to the first, lowest,
845/// highest and last point of each column, which draws the same.
846pub(crate) fn decimate(run: Vec<Pos2>) -> Vec<Pos2> {
847    if run.len() < 4096 {
848        return run;
849    }
850    let mut out = Vec::new();
851    for column in run.chunk_by(|a, b| a.x.floor() == b.x.floor()) {
852        let lowest = column.iter().max_by(|a, b| a.y.total_cmp(&b.y));
853        let highest = column.iter().min_by(|a, b| a.y.total_cmp(&b.y));
854        let (first, last) = (column.first(), column.last());
855        let mut keep: Vec<Pos2> = [first, lowest, highest, last]
856            .into_iter()
857            .flatten()
858            .copied()
859            .collect();
860        keep.dedup();
861        out.extend(keep);
862    }
863    out
864}

866pub(crate) fn line(painter: &egui::Painter, mut pts: Vec<Pos2>, stroke: Stroke, style: i64) {
867    // egui's tessellator turns a zero-length segment into NaN vertices,
868    // which can paint one triangle over the whole display.
869    pts.dedup_by(|b, a| a.distance_sq(*b) < 1e-6);
870    if pts.len() < 2 {
871        if let Some(p) = pts.first() {
872            painter.circle_filled(*p, stroke.width, stroke.color);
873        }
874        return;
875    }
876    let w = stroke.width.max(1.0);
877    let (dash, gap) = match style {
878        1 => (6.0 * w, 4.0 * w),
879        2 => (w, 2.0 * w),
880        3 | 4 => (6.0 * w, 2.0 * w),
881        _ => {
882            painter.add(Shape::line(pts, stroke));
883            return;
884        }
885    };
886    painter.extend(Shape::dashed_line(&pts, stroke, dash, gap));
887}
888
889fn marker(painter: &egui::Painter, p: Pos2, kind: i64, size: f32, c: Color32) {
890    let r = size / 2.0;
891    match kind {
892        1 => {
893            painter.rect_filled(Rect::from_center_size(p, Vec2::splat(size)), 0.0, c);
894        }
895        2 => {
896            painter.circle_filled(p, r, c);
897        }
898        3 => {
899            painter.add(Shape::convex_polygon(
900                vec![
901                    p + Vec2::new(0.0, -r),
902                    p + Vec2::new(r, 0.0),
903                    p + Vec2::new(0.0, r),
904                    p + Vec2::new(-r, 0.0),
905                ],
906                c,
907                Stroke::NONE,
908            ));
909        }
910        4 => {
911            let s = Stroke::new((size / 6.0).max(1.0), c);
912            painter.line_segment([p + Vec2::new(-r, -r), p + Vec2::new(r, r)], s);
913            painter.line_segment([p + Vec2::new(-r, r), p + Vec2::new(r, -r)], s);
914        }
915        _ => {
916            painter.add(Shape::convex_polygon(
917                vec![
918                    p + Vec2::new(0.0, -r),
919                    p + Vec2::new(r, r),
920                    p + Vec2::new(-r, r),
921                ],
922                c,
923                Stroke::NONE,
924            ));
925        }
926    }
927}
