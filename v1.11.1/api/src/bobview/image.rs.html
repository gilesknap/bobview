64/// The part of the image shown, in fractions of its size (y down), with
65/// undo and redo history.
66#[derive(Debug)]
67struct Zoom {
68    shown: Rect,
69    undo: Vec<Rect>,
70    redo: Vec<Rect>,
71}
72
73fn whole() -> Rect {
74    Rect::from_min_max(Pos2::ZERO, Pos2::new(1.0, 1.0))
75}
76
77impl Default for Zoom {
78    fn default() -> Self {
79        Self {
80            shown: whole(),
81            undo: Vec::new(),
82            redo: Vec::new(),
83        }
84    }
85}

87impl Zoom {
88    /// Show `r`, kept inside the image and no larger than it.
89    fn set(&mut self, r: Rect) {
90        let size = r.size().min(Vec2::splat(1.0)).max(Vec2::splat(1e-4));
91        let min = r.min.clamp(Pos2::ZERO, (Vec2::splat(1.0) - size).to_pos2());
92        let r = Rect::from_min_size(min, size);
93        if r != self.shown {
94            self.undo.push(self.shown);
95            if self.undo.len() > 100 {
96                self.undo.remove(0);
97            }
98            self.redo.clear();
99            self.shown = r;
100        }
101    }
102
103    /// Scale the view by `factor` (below 1 zooms in) about `at`, in image fractions.
104    fn scale_about(&mut self, at: Pos2, factor: f32) {
105        let s = self.shown;
106        let min = at + (s.min - at) * factor;
107        self.set(Rect::from_min_size(min, s.size() * factor));
108    }
109
110    fn pan(&mut self, by: Vec2) {
111        let s = self.shown;
112        let size = s.size();
113        let min = (s.min - by).clamp(Pos2::ZERO, (Vec2::splat(1.0) - size).to_pos2());
114        // Panning moves continuously, so it is not recorded step by step.
115        self.shown = Rect::from_min_size(min, size);
116    }
117
118    fn undo(&mut self) {
119        if let Some(r) = self.undo.pop() {
120            self.redo.push(self.shown);
121            self.shown = r;
122        }
123    }
124
125    fn redo(&mut self) {
126        if let Some(r) = self.redo.pop() {
127            self.undo.push(self.shown);
128            self.shown = r;
129        }
130    }
131
132    /// Image fraction under screen point `p` of `area`.
133    fn to_image(&self, area: Rect, p: Pos2) -> Pos2 {
134        let t = (p - area.min) / area.size();
135        self.shown.min + t * self.shown.size()
136    }
137
138    /// Screen point of image fraction `f` in `area`.
139    fn to_screen(&self, area: Rect, f: Pos2) -> Pos2 {
140        area.min + (f - self.shown.min) / self.shown.size() * area.size()
141    }
142}
143/// 256 colours from the widget's `color_map` (default VIRIDIS).
144fn lut(w: &Widget) -> [Color32; 256] {
145    let el = w.get("color_map");
146    let custom: Vec<[u8; 4]> = el
147        .map(|e| {
148            e.children
149                .iter()
150                .filter(|c| c.name == "section")
151                .filter_map(|c| {
152                    let a = |k: &str| c.attr(k)?.trim().parse::<u8>().ok();
153                    Some([a("value")?, a("red")?, a("green")?, a("blue")?])
154                })
155                .collect()
156        })
157        .unwrap_or_default();
158    let name = el.and_then(|e| e.child_text("name")).unwrap_or("VIRIDIS");
159    let sections: &[[u8; 4]] = if custom.len() >= 2 {
160        &custom
161    } else {
162        PREDEFINED
163            .iter()
164            .find(|(n, _)| n.eq_ignore_ascii_case(name))
165            .unwrap_or(&PREDEFINED[0])
166            .1
167    };
168    let mut out = [Color32::BLACK; 256];
169    for (i, c) in out.iter_mut().enumerate() {
170        let i = i as f32;
171        let hi = sections
172            .iter()
173            .position(|s| s[0] as f32 >= i)
174            .unwrap_or(sections.len() - 1);
175        let lo = hi.saturating_sub(1);
176        let (a, b) = (sections[lo], sections[hi]);
177        let t = if b[0] > a[0] {
178            (i - a[0] as f32) / (b[0] - a[0]) as f32
179        } else {
180            0.0
181        };
182        let mix = |k: usize| {
183            (a[k] as f32 + (b[k] as f32 - a[k] as f32) * t.clamp(0.0, 1.0)).round() as u8
184        };
185        *c = Color32::from_rgb(mix(1), mix(2), mix(3));
186    }
187    out
188}

190/// The image in `s`: an NTNDArray, or a numeric array shaped by the widget.
191fn image_of(w: &Widget, s: &PvState) -> Option<Arc<Image>> {
192    let shaped = |data: ImageData| {
193        let width = w.i64("data_width").unwrap_or(100).max(1) as usize;
194        let height = w.i64("data_height").unwrap_or(100).max(1) as usize;
195        width
196            .checked_mul(height)
197            .is_some_and(|n| data.len() >= n)
198            .then(|| {
199                Arc::new(Image {
200                    width,
201                    height,
202                    rgb: false,
203                    data,
204                    unique_id: 0,
205                })
206            })
207    };
208    match &s.value {
209        Value::Image(img) => Some(img.clone()),
210        Value::DoubleArray(a) => shaped(ImageData::F64(a.clone())),
211        Value::LongArray(a) => shaped(ImageData::I64(a.clone())),
212        _ => None,
213    }
214}

216/// Identifies what a texture shows: the frame's unique id, its data and the
217/// display settings. The data is held, not just its address: a CA array is
218/// freed when the next one arrives, and an allocator may give the next one
219/// the same address, which would leave the old picture on screen.
220struct TexKey {
221    unique_id: i64,
222    data: ImageData,
223    settings: String,
224}
225
226impl TexKey {
227    fn matches(&self, unique_id: i64, data: &ImageData, settings: &str) -> bool {
228        self.unique_id == unique_id && same_data(&self.data, data) && self.settings == settings
229    }
230}

232/// Whether two images' pixels are the same allocation.
233fn same_data(a: &ImageData, b: &ImageData) -> bool {
234    use ImageData as D;
235    match (a, b) {
236        (D::I8(a), D::I8(b)) => Arc::ptr_eq(a, b),
237        (D::U8(a), D::U8(b)) => Arc::ptr_eq(a, b),
238        (D::I16(a), D::I16(b)) => Arc::ptr_eq(a, b),
239        (D::U16(a), D::U16(b)) => Arc::ptr_eq(a, b),
240        (D::I32(a), D::I32(b)) => Arc::ptr_eq(a, b),
241        (D::U32(a), D::U32(b)) => Arc::ptr_eq(a, b),
242        (D::I64(a), D::I64(b)) => Arc::ptr_eq(a, b),
243        (D::U64(a), D::U64(b)) => Arc::ptr_eq(a, b),
244        (D::F32(a), D::F32(b)) => Arc::ptr_eq(a, b),
245        (D::F64(a), D::F64(b)) => Arc::ptr_eq(a, b),
246        _ => false,
247    }
248}

250/// Colour the image. Returns the pixels and the data range used.
251fn colorize(
252    img: &Image,
253    unsigned: bool,
254    range: Option<(f64, f64)>,
255    lut: &[Color32; 256],
256) -> (ColorImage, (f64, f64)) {
257    if let Some(fast) = colorize_bytes(img, unsigned, range, lut) {
258        return fast;
259    }
260    let step = img.width.max(img.height).div_ceil(MAX_SIDE).max(1);
261    let (w, h) = (img.width.div_ceil(step), img.height.div_ceil(step));
262    let channels = if img.rgb { 3 } else { 1 };
263    let sample = |x: usize, y: usize, c: usize| {
264        img.data
265            .get((y * step * img.width + x * step) * channels + c, unsigned)
266    };
267
268    let (lo, hi) = range.unwrap_or_else(|| {
269        let mut r = (f64::INFINITY, f64::NEG_INFINITY);
270        for y in 0..h {
271            for x in 0..w {
272                for c in 0..channels {
273                    let v = sample(x, y, c);
274                    if v.is_finite() {
275                        r = (r.0.min(v), r.1.max(v));
276                    }
277                }
278            }
279        }
280        if r.0 > r.1 { (0.0, 1.0) } else { r }
281    });
282    let scale = if hi > lo { 255.0 / (hi - lo) } else { 0.0 };
283    let level = |v: f64| ((v - lo) * scale).clamp(0.0, 255.0) as u8;
284
285    let mut pixels = Vec::with_capacity(w * h);
286    for y in 0..h {
287        for x in 0..w {
288            pixels.push(if img.rgb {
289                Color32::from_rgb(
290                    level(sample(x, y, 0)),
291                    level(sample(x, y, 1)),
292                    level(sample(x, y, 2)),
293                )
294            } else {
295                lut[level(sample(x, y, 0)) as usize]
296            });
297        }
298    }
299    (
300        ColorImage {
301            size: [w, h],
302            source_size: Vec2::new(w as f32, h as f32),
303            pixels,
304        },
305        (lo, hi),
306    )
307}

309/// Fast path for 8-bit mono images: colour every byte value once.
310fn colorize_bytes(
311    img: &Image,
312    unsigned: bool,
313    range: Option<(f64, f64)>,
314    lut: &[Color32; 256],
315) -> Option<(ColorImage, (f64, f64))> {
316    let n = img.width * img.height;
317    if img.rgb || img.width > MAX_SIDE || img.height > MAX_SIDE {
318        return None;
319    }
320    let (bytes, signed): (&[u8], bool) = match &img.data {
321        ImageData::U8(a) => (&a[..n], false),
322        // Reinterpreting i8 as u8 keeps the bit patterns; the table maps them back.
323        ImageData::I8(a) => (
324            // SAFETY: i8 and u8 have the same size and alignment.
325            unsafe { std::slice::from_raw_parts(a.as_ptr().cast::<u8>(), n) },
326            !unsigned,
327        ),
328        _ => return None,
329    };
330    let value = |b: u8| if signed { b as i8 as f64 } else { b as f64 };
331    let (lo, hi) = range.unwrap_or_else(|| {
332        let mut seen = [false; 256];
333        for &b in bytes {
334            seen[b as usize] = true;
335        }
336        let vals = (0..=255u8).filter(|b| seen[*b as usize]).map(value);
337        vals.fold((f64::INFINITY, f64::NEG_INFINITY), |r, v| {
338            (r.0.min(v), r.1.max(v))
339        })
340    });
341    let scale = if hi > lo { 255.0 / (hi - lo) } else { 0.0 };
342    let table: [Color32; 256] =
343        std::array::from_fn(|b| lut[((value(b as u8) - lo) * scale).clamp(0.0, 255.0) as usize]);
344    let pixels = bytes.iter().map(|b| table[*b as usize]).collect();
345    Some((
346        ColorImage {
347            size: [img.width, img.height],
348            source_size: Vec2::new(img.width as f32, img.height as f32),
349            pixels,
350        },
351        (lo, hi),
352    ))
353}

355#[allow(clippy::too_many_arguments)]
356pub fn draw(
357    ui: &mut egui::Ui,
358    node: &mut Node,
359    state: Option<&PvState>,
360    rect: Rect,
361    clip: Rect,
362    id: Id,
363    z: f32,
364) -> Response {
365    let w = node.w.clone();
366    let clip = clip.intersect(ui.clip_rect());
367    let painter = ui.painter().with_clip_rect(clip.intersect(rect));
368    fill(&painter, rect, bg(&w, "Background"));
369    let cache = node.image.get_or_insert_with(ImageCache::default);
370    let crosshair = *cache
371        .crosshair
372        .get_or_insert_with(|| w.bool("cursor_crosshair").unwrap_or(false));
373
374    let mut body = rect;
375    if w.bool("show_toolbar") == Some(true) {
376        let bar = Rect::from_min_size(rect.min, Vec2::new(rect.width(), TOOLBAR * z));
377        body.min.y = bar.bottom();
378        toolbar(ui, &painter, cache, bar, clip, id, z);
379    }
380
381    let bar = w.get("color_bar");
382    let bar_visible = bar
383        .and_then(|b| b.child_text("visible"))
384        .and_then(bob::parse_bool)
385        .unwrap_or(true);
386    let bar_size = bar
387        .and_then(|b| b.child_text("bar_size"))
388        .and_then(|v| v.trim().parse::<f32>().ok())
389        .unwrap_or(40.0)
390        * z;
391    let (area, bar_rect) = if bar_visible && body.width() > 3.0 * bar_size {
392        let gap = 4.0 * z;
393        let area = Rect::from_min_max(
394            body.min,
395            Pos2::new(body.right() - bar_size - gap, body.bottom()),
396        );
397        let bar_rect = Rect::from_min_max(Pos2::new(body.right() - bar_size, body.top()), body.max);
398        (area, Some(bar_rect))
399    } else {
400        (body, None)
401    };
402
403    // Only the zoom-in and pan modes take drags, so that otherwise a drag on
404    // the image scrolls the display, as it must on a touch screen.
405    let sense = match cache.mode {
406        Mode::ZoomIn | Mode::Pan => Sense::click_and_drag(),
407        Mode::Pointer | Mode::ZoomOut => Sense::click(),
408    };
409    let resp = ui.interact(area.intersect(clip), id, sense);
410    mouse(ui, cache, &resp, area, crosshair);
411
412    let img = state.filter(|s| s.connected).and_then(|s| image_of(&w, s));
413    let Some(img) = img else {
414        let msg = match state {
415            Some(s) if s.connected => "no image yet".to_string(),
416            Some(s) => s
417                .error
418                .clone()
419                .unwrap_or_else(|| format!("<{}>", node.pv_name)),
420            None => "image".to_string(),
421        };
422        placeholder(&painter, area, &msg, z);
423        return resp;
424    };
425
426    let lut = lut(&w);
427    let unsigned = w.bool("unsigned").unwrap_or(false);
428    let fixed = (w.bool("autoscale") == Some(false)).then(|| {
429        (
430            w.f64("minimum").unwrap_or(0.0),
431            w.f64("maximum").unwrap_or(255.0),
432        )
433    });
434    let settings = format!("{:?}{unsigned}{fixed:?}", w.get("color_map"));
435    if !cache
436        .tex
437        .as_ref()
438        .is_some_and(|c| c.key.matches(img.unique_id, &img.data, &settings))
439    {
440        let key = TexKey {
441            unique_id: img.unique_id,
442            data: img.data.clone(),
443            settings,
444        };
445        let (pixels, range) = colorize(&img, unsigned, fixed, &lut);
446        match cache.tex.as_mut() {
447            Some(tex) => {
448                tex.texture.set(pixels, PIXELS);
449                tex.key = key;
450                tex.range = range;
451            }
452            None => {
453                let texture = ui.ctx().load_texture("bobview-image", pixels, PIXELS);
454                cache.tex = Some(Texture {
455                    key,
456                    texture,
457                    range,
458                });
459            }
460        }
461    }
462    let Some(tex) = &cache.tex else { return resp };
463    // The whole image, placed so the zoomed part fills the area.
464    let full = Rect::from_min_max(
465        cache.view.to_screen(area, Pos2::ZERO),
466        cache.view.to_screen(area, Pos2::new(1.0, 1.0)),
467    );
468    let inside = painter.with_clip_rect(painter.clip_rect().intersect(area));
469    inside.image(tex.texture.id(), full, whole(), Color32::WHITE);
470
471    if let Some(from) = cache.band_from
472        && let Some(to) = resp.interact_pointer_pos()
473    {
474        inside.rect_stroke(
475            Rect::from_two_pos(from, to),
476            0.0,
477            Stroke::new(1.0, Color32::WHITE),
478            StrokeKind::Inside,
479        );
480    }
481    if crosshair && let Some(at) = cache.cross_at {
482        draw_crosshair(
483            &inside,
484            &img,
485            unsigned,
486            cache.view.to_screen(area, at),
487            at,
488            area,
489            z,
490        );
491    }
492
493    if let Some(bar_rect) = bar_rect {
494        colour_bar(&painter, &w, bar_rect, &lut, tex.range, z);
495    }
496    resp
497}

499/// Zoom, pan and crosshair placement with the mouse, per the toolbar mode.
500fn mouse(ui: &egui::Ui, cache: &mut ImageCache, resp: &Response, area: Rect, crosshair: bool) {
501    let view = &mut cache.view;
502    match cache.mode {
503        Mode::ZoomIn => {
504            if resp.drag_started() {
505                cache.band_from = resp.interact_pointer_pos();
506            }
507            if resp.drag_stopped()
508                && let (Some(a), Some(b)) = (cache.band_from.take(), resp.interact_pointer_pos())
509            {
510                let band = Rect::from_two_pos(a, b);
511                if band.width() > 4.0 && band.height() > 4.0 {
512                    let r = Rect::from_two_pos(view.to_image(area, a), view.to_image(area, b));
513                    view.set(r);
514                }
515            } else if resp.clicked()
516                && let Some(p) = resp.interact_pointer_pos()
517            {
518                view.scale_about(view.to_image(area, p), 0.5);
519            }
520        }
521        Mode::ZoomOut => {
522            if resp.clicked()
523                && let Some(p) = resp.interact_pointer_pos()
524            {
525                view.scale_about(view.to_image(area, p), 2.0);
526            }
527        }
528        Mode::Pan => {
529            if resp.drag_started() {
530                // One undo step per drag.
531                let shown = view.shown;
532                view.undo.push(shown);
533                view.redo.clear();
534            }
535            if resp.dragged() {
536                let d = resp.drag_delta() / area.size() * view.shown.size();
537                view.pan(d);
538            }
539        }
540        Mode::Pointer => {}
541    }
542    if crosshair
543        && cache.mode == Mode::Pointer
544        && resp.clicked()
545        && let Some(p) = resp.interact_pointer_pos()
546    {
547        cache.cross_at = Some(cache.view.to_image(area, p));
548    }
549    if !resp.dragged() && !resp.drag_stopped() {
550        cache.band_from = None;
551    }
552    if resp.hovered() {
553        let icon = match cache.mode {
554            Mode::Pan => egui::CursorIcon::Grab,
555            Mode::ZoomIn => egui::CursorIcon::ZoomIn,
556            Mode::ZoomOut => egui::CursorIcon::ZoomOut,
557            Mode::Pointer if crosshair => egui::CursorIcon::Crosshair,
558            Mode::Pointer => egui::CursorIcon::Default,
559        };
560        ui.ctx().set_cursor_icon(icon);
561    }
562}

564/// Crosshair lines through `p`, with the pixel and its value.
565fn draw_crosshair(
566    painter: &egui::Painter,
567    img: &Image,
568    unsigned: bool,
569    p: Pos2,
570    at: Pos2,
571    area: Rect,
572    z: f32,
573) {
574    for (c, width) in [(Color32::BLACK, 3.0), (Color32::WHITE, 1.0)] {
575        let s = Stroke::new(width * z.max(1.0), c);
576        painter.line_segment(
577            [Pos2::new(area.left(), p.y), Pos2::new(area.right(), p.y)],
578            s,
579        );
580        painter.line_segment(
581            [Pos2::new(p.x, area.top()), Pos2::new(p.x, area.bottom())],
582            s,
583        );
584    }
585    let x = ((at.x * img.width as f32) as usize).min(img.width.saturating_sub(1));
586    let y = ((at.y * img.height as f32) as usize).min(img.height.saturating_sub(1));
587    let channels = if img.rgb { 3 } else { 1 };
588    let i = (y * img.width + x) * channels;
589    let values: Vec<String> = (0..channels)
590        .filter(|c| i + c < img.data.len())
591        .map(|c| {
592            let v = img.data.get(i + c, unsigned);
593            if v.fract() == 0.0 {
594                format!("{v}")
595            } else {
596                format!("{v:.3}")
597            }
598        })
599        .collect();
600    let text = format!("{x}, {y}: {}", values.join(", "));
601    let font = FontId::proportional(12.0 * z.max(0.6));
602    let galley = painter.layout_no_wrap(text, font, Color32::WHITE);
603    let at = area.min + Vec2::splat(4.0 * z);
604    let back = Rect::from_min_size(at, galley.size()).expand(2.0 * z);
605    painter.rect_filled(back, 2.0, Color32::from_black_alpha(160));
606    painter.galley(at, galley, Color32::WHITE);
607}

617    ("Redo zoom", Button::Redo),

623    Mode(Mode),
624    Undo,

628fn toolbar(
629    ui: &mut egui::Ui,
630    painter: &egui::Painter,
631    cache: &mut ImageCache,
632    bar: Rect,
633    clip: Rect,
634    id: Id,
635    z: f32,
636) {
637    fill(painter, bar, Color32::from_gray(236));
638    painter.line_segment(
639        [bar.left_bottom(), bar.right_bottom()],
640        Stroke::new(1.0, Color32::from_gray(190)),
641    );
642    let size = bar.height() - 4.0 * z;
643    let mut x = bar.left() + 2.0 * z;
644    for (i, (tip, button)) in BUTTONS.iter().enumerate() {
645        // Separators after the crosshair and after the mouse modes.
646        if i == 1 || i == 5 {
647            x += 6.0 * z;
648        }
649        let r = Rect::from_min_size(Pos2::new(x, bar.top() + 2.0 * z), Vec2::splat(size));
650        x += size + 2.0 * z;
651        let resp = ui.interact(r.intersect(clip), id.with(("toolbar", i)), Sense::click());
652        let (on, enabled) = match button {
653            Button::Crosshair => (cache.crosshair == Some(true), true),
654            Button::Mode(m) => (cache.mode == *m, true),
655            Button::Undo => (false, !cache.view.undo.is_empty()),
656            Button::Redo => (false, !cache.view.redo.is_empty()),
657        };
658        button_box(
659            painter,
660            r,
661            if on {
662                Color32::from_gray(200)
663            } else {
664                Color32::from_gray(236)
665            },
666            on || resp.is_pointer_button_down_on(),
667            z,
668        );
669        let ink = if enabled {
670            Color32::from_gray(40)
671        } else {
672            Color32::from_gray(160)
673        };
674        icon(painter, *button, r.shrink(5.0 * z), ink, z);
675        if resp.clicked() && enabled {
676            match button {
677                Button::Crosshair => cache.crosshair = Some(cache.crosshair != Some(true)),
678                Button::Mode(m) => cache.mode = *m,
679                Button::Undo => cache.view.undo(),
680                Button::Redo => cache.view.redo(),
681            }
682        }
683        resp.on_hover_text(*tip);
684    }
685}

687fn icon(painter: &egui::Painter, button: Button, r: Rect, c: Color32, z: f32) {
688    let s = Stroke::new(1.5 * z.max(0.7), c);
689    let lens = |sign: Option<bool>| {
690        let centre = r.min + r.size() * 0.4;
691        let rad = r.width() * 0.33;
692        painter.circle_stroke(centre, rad, s);
693        let d = Vec2::splat(rad * std::f32::consts::FRAC_1_SQRT_2);
694        painter.line_segment([centre + d, r.max], Stroke::new(s.width * 1.6, c));
695        if let Some(plus) = sign {
696            let h = rad * 0.55;
697            painter.line_segment([centre - Vec2::new(h, 0.0), centre + Vec2::new(h, 0.0)], s);
698            if plus {
699                painter.line_segment([centre - Vec2::new(0.0, h), centre + Vec2::new(0.0, h)], s);
700            }
701        }
702    };
703    let arrow_head = |tip: Pos2, dir: Vec2| {
704        let n = Vec2::new(-dir.y, dir.x);
705        let l = r.width() * 0.25;
706        painter.add(Shape::convex_polygon(
707            vec![
708                tip,
709                tip - dir * l + n * l * 0.6,
710                tip - dir * l - n * l * 0.6,
711            ],
712            c,
713            Stroke::NONE,
714        ));
715    };
716    match button {
717        Button::Crosshair => {
718            painter.circle_stroke(r.center(), r.width() * 0.3, s);
719            painter.line_segment([r.left_center(), r.right_center()], s);
720            painter.line_segment([r.center_top(), r.center_bottom()], s);
721        }
722        Button::Mode(Mode::ZoomIn) => lens(Some(true)),
723        Button::Mode(Mode::ZoomOut) => lens(Some(false)),
724        Button::Mode(Mode::Pan) => {
725            painter.line_segment([r.left_center(), r.right_center()], s);
726            painter.line_segment([r.center_top(), r.center_bottom()], s);
727            arrow_head(r.left_center(), Vec2::new(-1.0, 0.0));
728            arrow_head(r.right_center(), Vec2::new(1.0, 0.0));
729            arrow_head(r.center_top(), Vec2::new(0.0, -1.0));
730            arrow_head(r.center_bottom(), Vec2::new(0.0, 1.0));
731        }
732        Button::Mode(Mode::Pointer) => {
733            let p = |x: f32, y: f32| r.min + r.size() * Vec2::new(x, y);
734            painter.add(Shape::convex_polygon(
735                vec![p(0.2, 0.0), p(0.85, 0.6), p(0.5, 0.62), p(0.2, 0.95)],
736                c,
737                Stroke::NONE,
738            ));
739            painter.line_segment([p(0.5, 0.6), p(0.7, 1.0)], Stroke::new(s.width * 1.5, c));
740        }
741        Button::Undo | Button::Redo => {
742            let redo = matches!(button, Button::Redo);
743            let flip = |p: Pos2| {
744                if redo {
745                    Pos2::new(r.left() + r.right() - p.x, p.y)
746                } else {
747                    p
748                }
749            };
750            let centre = r.center() + Vec2::new(0.0, r.height() * 0.15);
751            let rad = r.width() * 0.38;
752            // An arc over the top, ending in a head pointing down-left.
753            let pts: Vec<Pos2> = (0..=12)
754                .map(|i| {
755                    let a = std::f32::consts::PI * (i as f32 / 12.0);
756                    flip(centre + Vec2::new(rad * a.cos(), -rad * a.sin()))
757                })
758                .collect();
759            let tip = *pts.last().unwrap_or(&centre);
760            painter.add(Shape::line(pts, s));
761            arrow_head(tip + Vec2::new(0.0, r.height() * 0.15), Vec2::new(0.0, 1.0));
762        }
763    }
764}

766/// A vertical gradient with a tick scale of the data range beside it.
767fn colour_bar(
768    painter: &egui::Painter,
769    w: &Widget,
770    rect: Rect,
771    lut: &[Color32; 256],
772    range: (f64, f64),
773    z: f32,
774) {
775    let font = FontId::proportional(11.0 * z.max(0.6));
776    let label_h = painter
777        .layout_no_wrap("0".into(), font.clone(), Color32::BLACK)
778        .size()
779        .y;
780    let strip = Rect::from_min_max(
781        rect.min + Vec2::new(0.0, label_h / 2.0),
782        Pos2::new(
783            rect.left() + (rect.width() * 0.4).max(4.0),
784            rect.bottom() - label_h / 2.0,
785        ),
786    );
787    let mut mesh = egui::Mesh::default();
788    for (i, c) in lut
789        .iter()
790        .enumerate()
791        .step_by(4)
792        .chain(std::iter::once((255, &lut[255])))
793    {
794        let y = strip.bottom() - strip.height() * i as f32 / 255.0;
795        let idx = mesh.vertices.len() as u32;
796        mesh.colored_vertex(Pos2::new(strip.left(), y), *c);
797        mesh.colored_vertex(Pos2::new(strip.right(), y), *c);
798        if idx >= 2 {
799            mesh.add_triangle(idx - 2, idx - 1, idx);
800            mesh.add_triangle(idx - 1, idx, idx + 1);
801        }
802    }
803    painter.add(mesh);
804    outline(painter, strip, 1.0, Color32::GRAY);
805
806    let scale = Scale {
807        lo: range.0,
808        hi: range.1,
809        log: false,
810        from: strip.bottom(),
811        to: strip.top(),
812    };
813    let (ticks, step) = scale.ticks(label_h * 2.5);
814    let tick = 3.0 * z;
815    let stroke = Stroke::new(1.0, fg(w));
816    for v in ticks {
817        let y = scale.pos(v);
818        painter.line_segment(
819            [
820                Pos2::new(strip.right(), y),
821                Pos2::new(strip.right() + tick, y),
822            ],
823            stroke,
824        );
825        painter.text(
826            Pos2::new(strip.right() + tick + 1.0, y),
827            Align2::LEFT_CENTER,
828            axis::label(v, step),
829            font.clone(),
830            fg(w),
831        );
832    }
833}

839    #[test]
840    fn a_new_frame_in_a_reused_allocation_is_redrawn() {
841        // CA arrays all have unique id 0, so only the data tells frames apart.
842        let first: Arc<[f64]> = Arc::from(vec![0.0; 64]);
843        let key = TexKey {
844            unique_id: 0,
845            data: ImageData::F64(first.clone()),
846            settings: "viridis".into(),
847        };
848        let addr = first.as_ptr();
849        // The PV drops the first frame when the second arrives, which is the
850        // same size: the allocator would happily hand back the same block.
851        drop(first);
852        let second: Arc<[f64]> = Arc::from(vec![1.0; 64]);
853        assert_ne!(second.as_ptr(), addr, "the key keeps the first frame alive");
854        assert!(!key.matches(0, &ImageData::F64(second), "viridis"));
855        // The same frame, cloned as each paint does, matches; new settings do not.
856        assert!(key.matches(0, &key.data.clone(), "viridis"));
857        assert!(!key.matches(0, &key.data, "gray"));
858        assert!(!key.matches(1, &key.data, "viridis"));
859    }
